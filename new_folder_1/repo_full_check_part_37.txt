"""Mobile Crash & Battery Death Recovery Manager.
Executes automatically on app startup to recover unfinished sessions, finalize interrupted segments,
and re-enqueue pending uploads without requiring any user intervention.
"""
import logging
from pathlib import Path
from typing import Dict, Any, List, Optional

from dispatch_mobile.db import get_mobile_db, calculate_sha256, MOBILE_STORAGE_DIR

logger = logging.getLogger("dispatch_mobile.recovery")


def recover_mobile_state(storage_dir: Optional[Path] = None) -> Dict[str, Any]:
    """Scan and recover unfinished recording sessions after phone reboot or crash."""
    recordings_dir = storage_dir or (MOBILE_STORAGE_DIR / "recordings")
    recovered_segments = []
    quarantined_segments = []
    sealed_sessions = []

    with get_mobile_db() as conn:
        cursor = conn.cursor()

        # 1. Find sessions interrupted while RECORDING
        cursor.execute("SELECT session_id FROM mobile_sessions WHERE status = 'RECORDING'")
        interrupted_sessions = [row["session_id"] for row in cursor.fetchall()]

        for sess_id in interrupted_sessions:
            logger.info("Found interrupted session: %s. Performing crash recovery.", sess_id)

            # Find segments left in RECORDING status
            cursor.execute("""
                SELECT segment_id, sequence_number FROM mobile_segments
                WHERE session_id = ? AND status = 'RECORDING'
            """, (sess_id,))
            pending_segments = cursor.fetchall()

            for seg in pending_segments:
                seg_id = seg["segment_id"]
                tmp_file = recordings_dir / f"{seg_id}.tmp"
                mp4_file = recordings_dir / f"{seg_id}.mp4"

                if tmp_file.exists() and tmp_file.stat().st_size > 0:
                    # File has data: finalize it as a valid partial segment
                    file_size = tmp_file.stat().st_size
                    sha256 = calculate_sha256(tmp_file)
                    tmp_file.rename(mp4_file)

                    cursor.execute("""
                        UPDATE mobile_segments
                        SET status = 'QUEUED_FOR_UPLOAD',
                            file_size_bytes = ?,
                            sha256_hash = ?,
                            finalized_at = CURRENT_TIMESTAMP
                        WHERE segment_id = ?
                    """, (file_size, sha256, seg_id))

                    cursor.execute("""
                        INSERT OR REPLACE INTO mobile_outbox (
                            segment_id, session_id, remote_offset, status, updated_at
                        ) VALUES (?, ?, 0, 'QUEUED_FOR_UPLOAD', CURRENT_TIMESTAMP)
                    """, (seg_id, sess_id))

                    recovered_segments.append(seg_id)
                    logger.info("Successfully recovered segment %s (%d bytes)", seg_id, file_size)

                elif tmp_file.exists() and tmp_file.stat().st_size == 0:
                    # Empty file: safely delete
                    tmp_file.unlink()
                    cursor.execute("DELETE FROM mobile_segments WHERE segment_id = ?", (seg_id,))
                    quarantined_segments.append(seg_id)
                else:
                    cursor.execute("DELETE FROM mobile_segments WHERE segment_id = ?", (seg_id,))
                    quarantined_segments.append(seg_id)

            # Seal the session
            cursor.execute("""
                UPDATE mobile_sessions 
                SET status = 'FINALIZED',
                    total_segments = (SELECT COUNT(*) FROM mobile_segments WHERE session_id = ? AND status != 'RECORDING')
                WHERE session_id = ?
            """, (sess_id, sess_id))
            sealed_sessions.append(sess_id)

        # 2. Check for any finalized segments missing from outbox
        cursor.execute("""
            SELECT s.segment_id, s.session_id FROM mobile_segments s
            LEFT JOIN mobile_outbox o ON s.segment_id = o.segment_id
            WHERE s.status IN ('FINALIZED', 'QUEUED_FOR_UPLOAD') AND o.segment_id IS NULL
        """)
        orphaned_outbox = cursor.fetchall()
        for o in orphaned_outbox:
            cursor.execute("""
                INSERT OR REPLACE INTO mobile_outbox (
                    segment_id, session_id, remote_offset, status, updated_at
                ) VALUES (?, ?, 0, 'QUEUED_FOR_UPLOAD', CURRENT_TIMESTAMP)
            """, (o["segment_id"], o["session_id"]))
            logger.info("Re-enqueued orphaned segment %s into outbox", o["segment_id"])

    logger.info("Mobile crash recovery complete. Recovered %d segments across %d sessions.",
                len(recovered_segments), len(sealed_sessions))

    return {
        "recovered_segments": recovered_segments,
        "quarantined_segments": quarantined_segments,
        "sealed_sessions": sealed_sessions
    }

"""Mobile storage retention and cleanup manager.
Enforces strict cleanup policies: never deletes a phone file merely on an HTTP 200.
Requires verified byte count and SHA-256 confirmation from the laptop before deleting local media.
"""
import logging
from pathlib import Path
from typing import Dict, Any, List

from dispatch_mobile.db import get_mobile_db

logger = logging.getLogger("dispatch_mobile.retention")


def cleanup_verified_segments() -> int:
    """Delete local video files for segments whose copies have been strictly verified by the laptop.
    
    Returns:
        Number of cleaned-up local files.
    """
    cleaned_count = 0

    with get_mobile_db() as conn:
        cursor = conn.cursor()

        # Check policy
        cursor.execute("SELECT value FROM mobile_settings WHERE key = 'cleanup_policy'")
        row = cursor.fetchone()
        policy = row["value"] if row else "DELETE_AFTER_VERIFIED_COPY"

        if policy == "NEVER_DELETE":
            logger.debug("Cleanup policy is NEVER_DELETE; skipping phone storage cleanup.")
            return 0

        # Find segments confirmed VERIFIED_BY_LAPTOP that haven't been deleted yet
        cursor.execute("""
            SELECT segment_id, filepath FROM mobile_segments
            WHERE status = 'VERIFIED_BY_LAPTOP'
        """)
        eligible = cursor.fetchall()

        for item in eligible:
            seg_id = item["segment_id"]
            file_path = Path(item["filepath"])

            if file_path.exists():
                try:
                    file_path.unlink()
                    logger.info("Safely deleted local segment %s (%s) after verified laptop receipt.",
                                seg_id, file_path.name)
                    cleaned_count += 1
                except Exception as e:
                    logger.warning("Failed to delete local segment file %s: %s", file_path.name, e)

            cursor.execute("""
                UPDATE mobile_segments
                SET status = 'DELETED_FROM_PHONE'
                WHERE segment_id = ?
            """, (seg_id,))

            # Remove from outbox
            cursor.execute("DELETE FROM mobile_outbox WHERE segment_id = ?", (seg_id,))

    return cleaned_count

"""Mobile Recording Segmenter.
Splits continuous recording into atomic, durable segments to guarantee crash and power safety.
If phone dies or crashes, at most the currently-active segment is affected; all previous segments survive intact.
"""
import uuid
import logging
from pathlib import Path
from datetime import datetime
from typing import Optional, Dict, Any

from dispatch_mobile.db import get_mobile_db, calculate_sha256, MOBILE_STORAGE_DIR

logger = logging.getLogger("dispatch_mobile.segmenter")


class RecordingSessionManager:
    """Manages active mobile recording sessions and rolling segment creation."""

    def __init__(self, storage_dir: Optional[Path] = None, segment_duration_seconds: int = 600):
        self.storage_dir = storage_dir or (MOBILE_STORAGE_DIR / "recordings")
        self.storage_dir.mkdir(parents=True, exist_ok=True)
        self.segment_duration_seconds = segment_duration_seconds  # Default 10 minutes (600s)
        self.current_session_id: Optional[str] = None
        self.current_segment_id: Optional[str] = None
        self.current_sequence: int = 0
        self.current_tmp_path: Optional[Path] = None

    def start_session(self, notes: str = "") -> str:
        """Start a new recording session."""
        self.current_session_id = f"sess_{datetime.now().strftime('%Y%m%d_%H%M%S')}_{uuid.uuid4().hex[:6]}"
        self.current_sequence = 0

        with get_mobile_db() as conn:
            conn.execute("""
                INSERT INTO mobile_sessions (session_id, status, total_segments, notes)
                VALUES (?, 'RECORDING', 0, ?)
            """, (self.current_session_id, notes))

        logger.info("Started mobile recording session %s", self.current_session_id)
        # Immediately start first segment
        self.start_next_segment()
        return self.current_session_id

    def start_next_segment(self) -> Dict[str, Any]:
        """Start a new rolling segment (.tmp) under the active session."""
        if not self.current_session_id:
            raise RuntimeError("Cannot start segment without an active session")

        # If there's an ongoing segment, finalize it first
        if self.current_segment_id and self.current_tmp_path and self.current_tmp_path.exists():
            self.finalize_current_segment()

        self.current_sequence += 1
        self.current_segment_id = f"{self.current_session_id}_seg_{self.current_sequence:04d}"
        self.current_tmp_path = self.storage_dir / f"{self.current_segment_id}.tmp"

        # Touch/create the temporary file
        self.current_tmp_path.touch()

        with get_mobile_db() as conn:
            conn.execute("""
                INSERT INTO mobile_segments (
                    segment_id, session_id, sequence_number, filename, filepath, status
                ) VALUES (?, ?, ?, ?, ?, 'RECORDING')
            """, (
                self.current_segment_id,
                self.current_session_id,
                self.current_sequence,
                f"{self.current_segment_id}.mp4",
                str(self.storage_dir / f"{self.current_segment_id}.mp4")
            ))

        logger.info("Started recording segment %s -> %s", self.current_segment_id, self.current_tmp_path.name)
        return {
            "session_id": self.current_session_id,
            "segment_id": self.current_segment_id,
            "tmp_path": self.current_tmp_path
        }

    def finalize_current_segment(self) -> Optional[Dict[str, Any]]:
        """Atomically finalize current temporary segment (.tmp -> .mp4) and enqueue for upload."""
        if not self.current_segment_id or not self.current_tmp_path:
            return None

        final_mp4_path = self.storage_dir / f"{self.current_segment_id}.mp4"

        if not self.current_tmp_path.exists() or self.current_tmp_path.stat().st_size == 0:
            logger.warning("Segment %s temp file is missing or empty, removing record", self.current_segment_id)
            with get_mobile_db() as conn:
                conn.execute("DELETE FROM mobile_segments WHERE segment_id = ?", (self.current_segment_id,))
            return None

        # 1. Compute checksum and file size
        file_size = self.current_tmp_path.stat().st_size
        sha256 = calculate_sha256(self.current_tmp_path)

        # 2. Atomic rename from .tmp to .mp4
        self.current_tmp_path.rename(final_mp4_path)

        # 3. Update SQLite status to FINALIZED & QUEUED_FOR_UPLOAD
        with get_mobile_db() as conn:
            conn.execute("""
                UPDATE mobile_segments
                SET status = 'QUEUED_FOR_UPLOAD',
                    file_size_bytes = ?,
                    sha256_hash = ?,
                    finalized_at = CURRENT_TIMESTAMP
                WHERE segment_id = ?
            """, (file_size, sha256, self.current_segment_id))

            # Enqueue to mobile outbox
            conn.execute("""
                INSERT OR REPLACE INTO mobile_outbox (
                    segment_id, session_id, remote_offset, status, updated_at
                ) VALUES (?, ?, 0, 'QUEUED_FOR_UPLOAD', CURRENT_TIMESTAMP)
            """, (self.current_segment_id, self.current_session_id))

            # Update session total segments
            conn.execute("""
                UPDATE mobile_sessions 
                SET total_segments = total_segments + 1 
                WHERE session_id = ?
            """, (self.current_session_id,))

        logger.info("Finalized segment %s (%d bytes, SHA256: %s)",
                    self.current_segment_id, file_size, sha256[:12])

        info = {
            "segment_id": self.current_segment_id,
            "session_id": self.current_session_id,
            "filepath": final_mp4_path,
            "size_bytes": file_size,
            "sha256": sha256
        }

        self.current_segment_id = None
        self.current_tmp_path = None
        return info

    def stop_session(self) -> Dict[str, Any]:
        """Stop recording session, finalize the last segment, and seal session as FINALIZED."""
        if not self.current_session_id:
            raise RuntimeError("No active recording session to stop")

        # Finalize the last active segment
        if self.current_segment_id and self.current_tmp_path and self.current_tmp_path.exists():
            self.finalize_current_segment()

        session_id = self.current_session_id
        with get_mobile_db() as conn:
            conn.execute("""
                UPDATE mobile_sessions 
                SET status = 'FINALIZED' 
                WHERE session_id = ?
            """, (session_id,))

            cursor = conn.cursor()
            cursor.execute("SELECT total_segments FROM mobile_sessions WHERE session_id = ?", (session_id,))
            total = cursor.fetchone()[0]

        logger.info("Recording session %s stopped and sealed (total %d segments)", session_id, total)

        self.current_session_id = None
        self.current_sequence = 0

        return {
            "session_id": session_id,
            "total_segments": total,
            "status": "FINALIZED"
        }

## 7. Architecture Decisions

1. **Single Canonical Cloud Inbox:** YouTube Private upload from phone -> PC Poller/Catcher. No fragile LAN video streaming or UDP discovery for heavy media.

4. **Single Source of Truth for Subtitles:** `CaptionTrack` with `export_to_ass` in `dispatch/captions/exporter.py` is the single canonical ASS subtitle generator, eliminating subtitle stripping bugs.
5. **Raw Creator Media Preservation:** Raw source videos are never deleted during pipeline finalization; only temporary chunks and successfully published mobile files are pruned after verifiable durable storage proof.
6. **Hardware Touch Targets:** Scrubber and interactive controls strictly maintain 44px hitboxes per `AGENTS.md`.

# ADR-001 — Dispatch Canonical Multi-User Architecture (PROPOSED — awaiting owner approval)

Status: **APPROVED by owner (2026-10-24, MCQ):** Decision 2 = Bound per-worker token (owner-issued single-use enrollment code → user-bound worker credential; shared token becomes maintenance-only); Decision 1 = review lives in the **Android app only** for the alpha (no hosted web review app); Decision 4 = **server-side encrypted token vault on the PC backend** (no tokens on devices). Decision 3 boundaries stand as written. Workstream implementation may now start in the gap-map order.

## Decision 1 — What runs where
- **Android app (primary user interface):** email/password sign-in via Supabase Auth;
  on-device YouTube OAuth initiation; recording; review of own clips; publishing approval.
- **Hosted cloud (source of truth for user-owned state):** Supabase Postgres (profiles,
  sessions/clips metadata, review status, per-user OAuth token vault) + R2 (clip media) +
  Cloudflare Worker (authenticated control plane). Cloud review state **overrides** local.
- **Windows PC (owner's worker, no login UI):** downloads/transcribes/selects/renders/publishes.
  Removed: PC sign-in, PC YouTube login, all credential transfer.
- **Landing page:** stays a static public download page.

## Decision 2 — Worker identity and enrollment (the key question)
**The PC never receives a shared token that can act as any user.** Enrollment:

1. The owner signs in to Dispatch (email/password) on the PC's local dashboard — a one-time
   local action on the owner's own machine, not a cloud login surface for friends.
2. In the app/PC dashboard the owner opens **"Link processing PC"**; the backend issues a
   short-lived, single-use **enrollment code bound to that owner's user_id** (same
   digest-and-rate-limit pattern the old pairing codes used — reused only as a
   worker-enrollment mechanism, never for YouTube credentials).
3. The PC exchanges the code (over HTTPS) for a **worker credential: a per-worker token
   stored in the PC env, cryptographically bound to that one user_id** (backend stores
   token→user binding; every worker API call resolves the user from the token server-side —
   the client can never claim another user).
4. Revoking the binding in the admin panel invalidates the worker token; the PC loses access
   to that user's jobs until re-enrolled.

Consequently: friends' PCs (if ever enrolled by those friends) would each hold a token bound
ONLY to their own user. The shared `DISPATCH_WORKER_TOKEN` becomes backend-maintenance-only
and is deleted from user-facing paths. Adversarial tests: worker-A token cannot read/claim
user B's jobs, media, or tokens; replayed/expired enrollment codes fail; revoked tokens fail.

## Decision 3 — Source-of-truth boundaries
- **Cloud-owned:** identity, sessions/clips metadata + review state, per-user OAuth tokens
  (encrypted at rest), quota, publication history.
- **Local-owned (PC):** processing leases, local file paths, checkpoints, temporary renders.
- **Phone-owned:** recordings until uploaded (unchanged), Room outbox (unchanged).
- **Conflict rule:** remote approval/rejection is durable in the cloud; the PC applies it on
  next sync and never destructively changes local state on sync failure (existing rule).
- **Legacy local data migration:** existing PC-side records stay owned by the PC's enrolled
  worker identity (the owner's own user). Never reassigned to "whoever logs in first"; no
  recordings deleted.

## Decision 4 — YouTube OAuth (per Google web-server flow)
User initiates in-app → Custom Tab → offline-access consent → one-time code → **PC backend
exchanges it server-side** (Web-server client secret in PC env) → refresh token stored
**server-side, encrypted per user** (never on device, never in PC's old token file) →
refresh/revoke/status handled by backend → truthful UI states. The Android OAuth client ID
is public (no secret). No token in URLs, logs, or browser storage.

## Consequences / risks
- Requires the owner to create the Web-server OAuth client + consent screen.
- Workers depend on cloud reachability for job claiming; offline behavior = pause, not
  data loss (existing local checkpointing preserved).
- The old single-user flows keep working for the owner's own processing during migration.

## Explicitly rejected alternatives
- Shared worker token with user-ID header (current prototype): cross-tenant risk — rejected.
- Refresh tokens stored on device: contradicts Google guidance for offline access — rejected.
- Any new pairing code for YouTube credentials: banned by owner directive.
- Automatic LAN discovery as PC identity: unauthenticated — rejected.

**Owner approval needed on:** D1–D4 as written, or amendments in chat.

## Amendment 2 (2026-10-24) — Owner constraints: law of least power + Pareto
- **Alpha worker topology simplified:** in the alpha there is exactly ONE processing worker — the owner's PC, holding an **admin-worker credential** (bound to the admin role, server-side authorized to process ALL users' jobs). Friends never receive worker tokens; their phones are plain authenticated clients. Friends' self-enrollment PCs are deferred until actually needed. This keeps the per-worker security model AND matches the product rule that friends don't need their own PCs.
- **Least-power choices:** auth = Supabase Auth as-is (no custom auth backend); identity boundary = verify the Supabase JWT on the existing FastAPI routes (no new service); token vault = one encrypted store on the PC backend; review UI = the existing app screens; no hosted web app, no new frameworks, no speculative abstractions. Build only what an acceptance test needs.
- **Build order per Pareto:** auth+isolation → admin minimal → on-device OAuth; each phase is demo-able and tested before the next.

# Dispatch Cloud Migration — Architecture & Migration Plan

Status: **Plan approved as working baseline (2026-10-23). Implementation in progress.**
Owner brief: Principal Architect Implementation Brief (Steps 0–9). This doc is the authoritative decision record + migration checklist. Current-state index: root `AGENT_HANDOFF.md`.

**Principle:** extend Dispatch into a small multi-user cloud-connected app while ALL video processing stays on the owner's Windows PC. No cloud rendering. Originals stay on YouTube. Free-tier only.

## 1. Baseline findings (Step 0 — verified from code at `9db3054`)

**Components & state ownership**
- Android (`android/`, Kotlin + Compose + WebView): CameraX recording, 10-min segmentation, direct-to-YouTube upload worker. Room SQLite on phone. Does not talk to the PC.
- PC daemon (`dispatch/main.py`): YouTube poller (`youtube_inbox/poller.py`+`oauth.py`), catcher/validator (`catcher.py`, `ingestion/validator.py`), transcription (`transcription/worker.py`, `gemini_engine.py`, `audio.py`), highlight worker (`ai_clips/worker.py`, `highlight_finder.py`), renderer (`video_engine/renderer.py`, `reframer.py`, captions/), publishing outbox (`publisher/outbox.py` + per-platform adapters), governor, orchestrator (leases/recovery). SQLite `storage/database/dispatch.db` (WAL).
- Web/API: FastAPI `dispatch/web/app.py` serves the React studio (built into `android/app/src/main/assets/web` and the same code served at `127.0.0.1:8000`). `/download/dispatch.apk` serves the debug APK.

**Exact rejection behavior** (`dispatch/db.py:729` `reject_clip`): immediately `unlink()`s `video_path` + `thumbnail_path` from disk, inserts a row into `rejections` (id, clip_id, title, transcript_snippet, reason) for negative preference learning, sets `status='rejected'`, NULLs both paths. Failure to unlink is only logged (orphan risk; DB still clears paths) — **cloud version must add a deletion-pending/retry state instead**.

**Exact 7-day cleanup** (`dispatch/db.py:765` `cleanup_expired_published_clips`, invoked from `web/app.py:703`): purges local MP4 for `status IN ('published','approved')` where `published_at/approved_at/created_at` older than 7 days; **preserves `thumbnail_path`, `remote_url`, and all metadata**; sets `video_path=NULL`. Not a background scheduler — runs from the web layer.

**Auth boundaries:** local FastAPI has **no auth** (grep confirms no Depends/auth/token checks); its only boundary is the loopback bind (`config.py:99` warns if non-loopback). Phone holds its own YouTube OAuth refresh token; PC holds `youtube_token.json`. No multi-user concept anywhere.

**Local-only UI/API:** everything — dashboard, clips API, trim/caption/render/approve/reject endpoints, APK download. Remote access today would require exposing this server (forbidden by the brief).

**Clip metadata & range-revision semantics (must be preserved):** `clips` table carries `start_time/end_time`, `ai_suggested_start_time/ai_suggested_end_time` (immutable), `range_revision` (monotonic), `caption_range_revision/start/end`, `caption_timing_warning`, `render_stale`, `layout_mode/rendered_layout_mode`, `video_path/thumbnail_path`, `published_at/remote_url`. Optimistic concurrency: `update_clip_trim`, `update_clip_media`, `save_clip_caption_data`, render finalization and approval all take/compare `expected_range_revision` (HTTP 409 on mismatch). Approval validates ffprobe + duration ±1 s and rejects phrase-timing captions (422).

**Test baseline:** focused suite **80 tests OK (74.1 s)** at `79444b7` (frontend + Android results in `AGENT_HANDOFF.md` §8). "380/380" is historical (2026-10-09), not re-verified.

**Migration risks:** single-account YouTube credentials (blocks friends' private uploads — D10); local absolute Windows paths stored in SQLite; rejection deletion has no retry state; unauthenticated local API must never be exposed; legacy `dispatch/sync/`/`transport/` still imported by legacy tests; SQLite↔Postgres identifier namespaces differ (need local↔cloud mapping); Supabase Free pauses after inactivity and has no managed backups.

## 2. Decisions

**D1 — System boundaries.** Android records/uploads to YouTube (unchanged). YouTube = canonical originals (never uploaded to R2). PC downloads/transcribes/selects/renders/publishes (unchanged). Supabase Auth = identity. Supabase Postgres = cloud truth for identity, clip metadata/review state, media-object records, quota policy, publishing history. Cloudflare R2 (private bucket) = clip video + thumbnail bytes only. Cloudflare Worker = control plane: presigned URL issuance, upload completion, signed GET URLs, deletion, quota enforcement, scheduled cleanup. Free static hosting serves the existing frontend for remote review. **Rejected:** VPS/cloud rendering; Worker as video proxy (Free request-body limits); public bucket; exposing FastAPI.

**D2 — Source of truth.** Local-only (SQLite): pipeline state, leases, local paths, checkpoints, processing temps. Cloud-owned (Postgres): `user_id` ownership, cloud clip metadata + review status, media-object records, quota, publication history. Originals: YouTube. Clip media bytes: R2. Mapping: `clips_cloud.local_clip_id` + `unique(user_id, local_clip_id)` bridges namespaces (local clip IDs are not globally unique).

**D3 — Conflict resolution.** Every cloud clip row carries `range_revision` + `status` + `local_updated_at`; the PC pushes new revisions monotonically. Remote approve/reject apply only against the revision the client saw (stale → 409, client reloads). Approval while PC offline: durable `approved` row; PC picks it up on next sync, renders if needed, publishes — no destructive local transition on failure (matches existing stale-render semantics). Upload retries: client-generated reservation token, idempotent completion. Approval arriving during a local run: PC re-checks `range_revision` before finalizing (existing 409 path already handles this locally).

**D4 — Cloud database (reproducible SQL migrations, `supabase/migrations/`).** Tables: `profiles` (user_id PK → `auth.users`, `storage_quota_bytes` default 2,000,000,000 B, max 10,000,000,000 B per alpha brief), `source_sessions` (owner + `youtube_video_id`, unique per user), `clips_cloud` (minimal review metadata + `ai_suggested_*`, `range_revision`, `status` CHECK, `render_stale`), `media_objects` (kind CHECK clip_video/thumbnail, status CHECK pending/active/deletion_pending/deleted, `staging_key`/`final_key` UNIQUE, `byte_count`, `expires_at`), `upload_reservations` (client_token UNIQUE per user, expected_bytes, expires_at), `publications` (platform, remote_id/remote_url, published_at), `rejections` (lightweight audit). JSONB only for caption snapshots/versioned AI output. RLS on every user-owned table (`user_id = auth.uid()`), explicit grants, status constraints, indexes on (user_id,status), (expires_at), (final_key). Quota reservations enforced inside one Postgres transaction (count pending+active bytes + insert reservation) so concurrent uploads cannot overshoot.

**D4 weight — the exact atomic reservation contract (Phase D implements; tables alone do NOT enforce anything).** One transaction per `/reserve` takes BOTH serialization points before any accounting statement, always in the same fixed order on every path — first the global budget lock `pg_advisory_xact_lock(hashtextextended('dispatch:global-quota-budget', 0))` (serializes reservations ACROSS users; per-user row locks alone cannot serialize the global budget between users), then the per-user row lock `select storage_quota_bytes from public.profiles where user_id = $1 for update` (serializes the SAME user's parallel reservations); usage = `sum(byte_count)` over the user's media_objects with status `active`/`deletion_pending` (pending rows always hold `byte_count=0`, enforced by CHECK) plus `sum(expected_bytes)` over the user's reservations with status `pending` and `expires_at > now()` (expired rows never hold headroom); insert only if user sum + expected ≤ quota, global sums + expected ≤ budget, expected ≤ kind cap, and the outstanding limit is STRICT: the user's current pending count must be < 3, so the insert that would create a fourth outstanding reservation is refused (matches D6 "≤3 outstanding reservations/user" and the quota kernel's refuse when outstanding ≥ max); commit or roll back whole. `/complete` is one transaction: HEAD-verify the real R2 object, promote media to `active` with real `byte_count` and flip the reservation to `completed` together, re-validate the sums, refuse + cancel on oversize; completion never raises the accounted total (real bytes ≤ reservation bound), so it takes no advisory lock and merely flips reservation+media atomically. Browser clients hold no INSERT grant on these tables, so the browser cannot race or bypass the kernel. `tests/rls_two_user_checklist.sql` §8 exercises the accounting query once a project exists.

**D5 — Media contract.** Private bucket; no public URLs, ever. Direct client→R2 PUT via short-lived presigned URL (600 s) from a unique staging key; Worker validates ownership, MIME allowlist, declared size bound, outstanding-reservation limit, and atomically enforces user+global budgets before issuing. Completion verifies **actual** R2 object size/type (declared size is not proof), promotes staging→immutable final key, records real byte count, releases reservation; idempotent on repeat. Download: ownership + active status → 600 s signed GET. Delete: ownership + valid state → idempotent; R2 failure → `deletion_pending` + retry by scheduled cleanup; quota not released until deletion confirmed. **Rejected:** uploading bytes through Worker/Postgres; permanent URLs.

**D6 — Quotas & limits (all server-enforced; updated 2026-10-23 per alpha brief — decimal bytes).** Default 2,000,000,000 B/user (admin-adjustable; ceiling = the 10,000,000,000 B hard cap); global active+pending hard cap 10,000,000,000 B (an application cap is NOT a provider billing guarantee); max individual clip object 512 MiB, thumbnail 8 MiB; ≤3 outstanding reservations/user; rate limits on reservation/completion/deletion endpoints; explicit quota-exceeded errors. Decision kernel in `dispatch/cloud/quota.py`; authoritative enforcement = Postgres transaction + Worker (D4).

**D7 — Retention (mirrors verified local behavior).** Rejected: delete video+thumbnail from R2 (deletion-pending on failure, retryable), preserve rejection record + lightweight metadata for preference learning; never report success before confirmation. Published: set `expires_at = published_at + 7 days` for the video object; delete at expiry; **keep thumbnail + publication history + remote URLs** (matches `cleanup_expired_published_clips`). Failed publishing: media retained for retry, no expiry; explicit abandon required. Orphans: scheduled Worker cleanup expires staging objects/abandoned reservations; R2 lifecycle rule on `staging/` prefix as safety net only; all cleanup idempotent.

**D8 — Sync layer.** One interface (`dispatch/cloud/media_store.py`) for all cloud media ops; PC never touches R2/Supabase SDKs directly — it calls the Worker with `DISPATCH_WORKER_URL` + `DISPATCH_WORKER_TOKEN` (shared secret, server-side both ends). PC uploads only completed reviewable clip media + thumbnails after render, pushes minimal metadata, polls for remote review actions. Cloud outage = sync pauses; local pipeline unaffected; no destructive local transitions. Idempotent resume after restart (queue table keyed by local_clip_id + revision). Local file storage remains the fallback for all existing local workflows/tests.

**D9 — Remote review.** Reuse the existing frontend (no redesign): Supabase Auth login, RLS-scoped clip metadata, signed-URL media playback, approve/reject with revision+status checks. Local studio controls stay distinct (local-only endpoints never exposed). No new editing features in this migration.

**D10 — Multi-user YouTube processing (BLOCKER, recorded not solved).** The pipeline is bound to one PC-side YouTube account/token. Friends' private uploads require per-user authorization design (per-user tokens, consent, secure storage). **Not doing:** copying refresh tokens, plaintext token storage, shared OAuth secrets in clients, privacy changes without consent, or claiming multi-user processing works on the owner's account. Cloud storage/review foundations proceed independently.

**D11 — PC↔Worker authentication.** PC = trusted server-side actor holding the owner's Worker token; end-user clients authenticate with Supabase JWTs which the Worker verifies and maps to `user_id`. Worker secrets (R2 binding, service-role key) never leave Cloudflare. R2 credentials exist only as a Worker binding — not on the PC, not in clients.

## 1. Overall System Pipeline

   ↓

   ↓

   ↓ (status: DOWNLOADED)
PILLAR 2: Dual-Stage Timestamped Transcription
   ↓ (status: TRANSCRIBED)
PILLAR 3: Gemini Semantic Highlight Selection
   ↓ (status: CLIPS_DEFINED)
PILLAR 4: FFmpeg Adaptive 9:16 Reframer & Subtitle Renderer
   ↓ (status: CLIPS_CREATED)
PILLAR 5: PC Local Review & Control Dashboard
   ↓ (user action: APPROVED / REJECTED)
PILLAR 6: Multi-Platform Publishing Outbox
   ↓ (status: PUBLISHED)
PILLAR 7: Master Daemon & Coordinator (surrounds and coordinates PC system)

| # | Pillar | Core Responsibility | Input → Output Contract | Status |
|---|---|---|---|---|
| **1** | **YouTube Ingestion & Media Validation** | Discover uploads via authenticated API, download media, validate file with ffprobe + full frame decode, create durable state | YouTube → `DISCOVERED` → `DOWNLOADING` → `VALIDATING` → **`DOWNLOADED`** | ✅ **Verified Complete** |
| **2** | **Transcription Engine** | Obtain reliable timestamped transcript with asynchronous YouTube caption polling and explicit Whisper fallback | `DOWNLOADED` → `WAITING_FOR_TRANSCRIPT` → **`TRANSCRIBED`** | ✅ **Verified Complete** |

## 3. Dedicated Track: Phone (POCO C65)

The phone application is a standalone client track, **not** one of the 7 PC pillars.

### Contract:
$$\text{CameraX Recording} \longrightarrow \text{Local Room SQLite} \longrightarrow \text{WorkManager Upload} \longrightarrow \text{YouTube (Unlisted)}$$

### Core Requirements:
1. **Zero Phone↔PC Networking:** No LAN sync, no HTTP receivers, no UDP beacons, no Tailscale handoffs. YouTube is the **only** handoff.
2. **Transport Visibility:** Videos are uploaded as **`unlisted`** with title prefix `[DISPATCH]` and description marker `dispatch_id: <id>`.
3. **10-Minute Rolling Segments:** Continuous background capture split cleanly into rolling segments.
4. **Foreground Service & WakeLocks:** Recording survives screen dim, lock screen, and system Doze.
5. **Durable Local Storage:** Room database tracks segments; files pruned only after upload confirmation.

---

## 4. Pillar 2 Detailed Specification (Transcription Engine)

Pillar 2 governs the transition from `DOWNLOADED` to `TRANSCRIBED`.

```text
DOWNLOADED
    ↓
Check YouTube Captions Available?
    ├── YES ──→ Parse VTT ──→ Normalize Segments ──→ TRANSCRIBED (source='youtube')
    └── NO  ──→ WAITING_FOR_TRANSCRIPT (asynchronous poll window)
                     ↓ (timeout window exceeded)
                Fallback to Local Whisper (Hinglish prompt)
                     ↓
                Normalize Segments ──→ TRANSCRIBED (source='whisper')
```

### Key Decisions & Rules:
1. **Primary Source:** YouTube captions (auto-generated or creator-provided).
2. **Asynchronous Handling:** YouTube captions take minutes to process; the system must transition to `WAITING_FOR_TRANSCRIPT` and check periodically without blocking or failing prematurely.
3. **Explicit Fallback:** If captions remain unavailable after the defined wait timeout window, explicitly transition to local `faster-whisper`.
4. **Source Tracking:** The resulting record must record `transcript_source: 'youtube'` or `transcript_source: 'whisper'`. No hidden fallback.
5. **Unified Output Contract:** Whether from YouTube VTT or Whisper, output must normalize into identical JSON segment structure:
   ```json
   [
     {
       "id": 0,
       "start": 0.00,
       "end": 4.52,
       "text": "Yeh step automate ho chuka hai.",
       "words": [{"word": "Yeh", "start": 0.00, "end": 0.35}]
     }
   ]
   ```
6. **Stop Condition:** Pillar 2 stops strictly at **`TRANSCRIBED`**. It does not call Gemini or FFmpeg.

---

## 5. Architectural Invariants

- **Idempotency:** Any process restart at any stage must recognize existing state and resume without duplicating DB rows or corrupting files.
- **Durable Progress:** State in SQLite advances only *after* the required artifact exists and passes validation.
- **Failures Surface:** If any stage fails, record `status = 'FAILED'` with a descriptive `last_error`; never swallow errors or fake success.
