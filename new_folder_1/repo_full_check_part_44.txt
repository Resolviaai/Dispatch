function applyColorOpacity(color: string, opacity: number = 1.0): string {
  if (opacity >= 1.0) return color;
  if (color.startsWith('#')) {
    const hex = color.slice(1);
    let r = 0, g = 0, b = 0;
    if (hex.length === 3) {
      r = parseInt(hex[0] + hex[0], 16);
      g = parseInt(hex[1] + hex[1], 16);
      b = parseInt(hex[2] + hex[2], 16);
    } else if (hex.length >= 6) {
      r = parseInt(hex.slice(0, 2), 16);
      g = parseInt(hex.slice(2, 4), 16);
      b = parseInt(hex.slice(4, 6), 16);
    }
    return `rgba(${r}, ${g}, ${b}, ${opacity})`;
  }
  return color;
}

interface CanvasGroupItemProps {
  group: CaptionGroup;
  defaultStyle: StyleConfig;
  defaultLayout: LayoutConfig;
  defaultAnimation: any;
  isSelected: boolean;
  currentTime: number;
  containerRef: React.RefObject<HTMLDivElement>;
  onSelectGroup: (groupId: string | null) => void;
  onOpenTextEditor?: (groupId: string) => void;
  onUpdateGroup: (groupId: string, patch: Partial<CaptionGroup>, commitToHistory?: boolean) => void;
  onApplyLayoutToAll?: (groupId: string) => void;
  layoutMode: string;
  isInteractive: boolean;
}

const CanvasGroupItem: React.FC<CanvasGroupItemProps> = React.memo(({
  group,
  defaultStyle,
  defaultLayout,
  defaultAnimation,
  isSelected,
  currentTime,
  containerRef,
  onSelectGroup,
  onOpenTextEditor,
  onUpdateGroup,
  onApplyLayoutToAll,
  layoutMode,
  isInteractive
}) => {
  const [isDragging, setIsDragging] = useState(false);
  const [isResizing, setIsResizing] = useState(false);

  const style: StyleConfig = group.style || defaultStyle;
  const layout: LayoutConfig = group.layout || defaultLayout;
  const anim = group.animation || defaultAnimation;

  // Layer type awareness
  const isCaption = (group.type || 'caption') === 'caption';
  const highlightEnabled = isCaption && (style.highlight_enabled !== false);

  let posX = layout.position_x ?? 0.5;
  let posY = layout.position_y ?? 0.78;
  const scale = layout.scale ?? 1.0;

  if (isCaption && layoutMode === 'landscape') {
    posY = Math.max(0.78, posY);
  }

  const dragStartRef = useRef<{ 
    x: number; 
    y: number; 
    initialX: number; 
    initialY: number; 
    containerWidth: number; 
    containerHeight: number; 
  }>({ x: 0, y: 0, initialX: 0.5, initialY: 0.78, containerWidth: 1, containerHeight: 1 });

  const resizeStartRef = useRef<{ 
    x: number; 
    initialScale: number; 
    containerWidth: number; 
  }>({ x: 0, initialScale: 1.0, containerWidth: 1 });

  const latestPatchRef = useRef<{ groupId: string; patch: Partial<CaptionGroup> } | null>(null);

  const fontFam = style.font_family || 'Montserrat';
  const fontSize = Math.max(10, Math.round((style.font_size || 44) * 0.44));
  const fontWeight = style.font_weight || 'bold';
  const letterSpacing = style.letter_spacing || 0;

  const visualLines = useMemo(() => {
    return partitionWordsIntoVisualLines(
      group.words || [],
      fontFam,
      fontSize,
      fontWeight,
      letterSpacing,
      360 * (layout.max_width || 0.88),
      style.text_transform
    );
  }, [group.words, fontFam, fontSize, fontWeight, letterSpacing, layout.max_width, style.text_transform]);

  const handlePointerDownDrag = (e: React.PointerEvent) => {
    if (!isInteractive) return;
    e.stopPropagation();
    onSelectGroup(group.id);

    const rect = containerRef.current?.getBoundingClientRect() || { width: 393, height: 700 };
    dragStartRef.current = {
      x: e.clientX,
      y: e.clientY,
      initialX: posX,
      initialY: posY,
      containerWidth: Math.max(1, rect.width),
      containerHeight: Math.max(1, rect.height),
    };
    latestPatchRef.current = null;
    setIsDragging(true);
    (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
  };

  const handlePointerMove = (e: React.PointerEvent) => {
    if (isDragging) {
      const deltaX = (e.clientX - dragStartRef.current.x) / dragStartRef.current.containerWidth;
      const deltaY = (e.clientY - dragStartRef.current.y) / dragStartRef.current.containerHeight;

      const newX = Math.max(0.05, Math.min(0.95, dragStartRef.current.initialX + deltaX));
      const newY = Math.max(0.05, Math.min(0.95, dragStartRef.current.initialY + deltaY));

      const patch: Partial<CaptionGroup> = {
        layout: {
          ...layout,
          position_x: Number(newX.toFixed(3)),
          position_y: Number(newY.toFixed(3))
        }
      };

      latestPatchRef.current = { groupId: group.id, patch };
      onUpdateGroup(group.id, patch, false);
    } else if (isResizing) {
      const deltaX = (e.clientX - resizeStartRef.current.x) / (resizeStartRef.current.containerWidth * 0.4);
      const newScale = Math.max(0.6, Math.min(2.4, resizeStartRef.current.initialScale + deltaX));

      const patch: Partial<CaptionGroup> = {
        layout: {
          ...layout,
          scale: Number(newScale.toFixed(2))
        }
      };

      latestPatchRef.current = { groupId: group.id, patch };
      onUpdateGroup(group.id, patch, false);
    }
  };

  const handlePointerUp = (e: React.PointerEvent) => {
    if (latestPatchRef.current) {
      onUpdateGroup(latestPatchRef.current.groupId, latestPatchRef.current.patch, true);
      latestPatchRef.current = null;
    }
    if (isDragging) setIsDragging(false);
    if (isResizing) setIsResizing(false);
    try {
      (e.target as HTMLElement).releasePointerCapture?.(e.pointerId);
    } catch {}
  };

  const handleResizePointerDown = (e: React.PointerEvent) => {
    e.stopPropagation();
    const rect = containerRef.current?.getBoundingClientRect() || { width: 393, height: 700 };
    resizeStartRef.current = {
      x: e.clientX,
      initialScale: scale,
      containerWidth: Math.max(1, rect.width),
    };
    latestPatchRef.current = null;
    setIsResizing(true);
    (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
  };

  // Deterministic 3-layer animation evaluation (Enter / Exit / Loop)
  const timeSinceStart = Math.max(0, currentTime - group.start);
  const timeUntilEnd = Math.max(0, group.end - currentTime);
  
  let enterOpacity = 1.0;
  let enterTransform = '';
  const enterType = anim.enter?.type || (anim as any)?.enter_type || 'none';
  const enterDur = (anim.enter?.duration_ms || (anim as any)?.enter_duration_ms || 200) / 1000;
  if (enterType !== 'none' && timeSinceStart < enterDur) {
    const p = Math.max(0, Math.min(1, timeSinceStart / enterDur));
    if (enterType === 'fade') enterOpacity = p;
    else if (enterType === 'pop') {
      enterOpacity = Math.min(1, p * 2);
      enterTransform = `scale(${0.5 + 0.5 * Math.sin(p * Math.PI / 2)})`;
    } else if (enterType === 'slide_up') {
      enterOpacity = p;
      enterTransform = `translateY(${(1 - p) * 20}px)`;
    }
  }

  let exitOpacity = 1.0;
  let exitTransform = '';
  const exitType = anim.exit?.type || (anim as any)?.exit_type || 'none';
  const exitDur = (anim.exit?.duration_ms || (anim as any)?.exit_duration_ms || 150) / 1000;
  if (exitType !== 'none' && timeUntilEnd < exitDur) {
    const p = Math.max(0, Math.min(1, timeUntilEnd / exitDur));
    if (exitType === 'fade') exitOpacity = p;
    else if (exitType === 'scale_down') {
      exitOpacity = p;
      exitTransform = `scale(${0.5 + 0.5 * p})`;
    } else if (exitType === 'slide_down') {
      exitOpacity = p;
      exitTransform = `translateY(${(1 - p) * 20}px)`;
    }
  }

  let loopTransform = '';
  const loopType = anim.loop?.type || (anim as any)?.loop_type || 'none';
  if (loopType === 'pulse') {
    const speed = (anim.loop?.speed_ms || (anim as any)?.loop_speed_ms || 1000) / 1000;
    const loopTime = timeSinceStart / speed;
    loopTransform = `scale(${1.0 + 0.04 * Math.sin(loopTime * 2 * Math.PI)})`;
  } else if (loopType === 'wiggle') {
    const speed = (anim.loop?.speed_ms || (anim as any)?.loop_speed_ms || 1000) / 1000;
    const loopTime = timeSinceStart / speed;
    loopTransform = `rotate(${2.0 * Math.sin(loopTime * 2 * Math.PI)}deg)`;
  }

  const computedOpacity = (style.opacity ?? 1.0) * enterOpacity * exitOpacity;
  const extraAnimTransform = [enterTransform, exitTransform, loopTransform].filter(Boolean).join(' ');

  const bgEnabled = style.background_enabled !== false && style.background_mode !== 'none';
  const rawBgMode = bgEnabled ? (style.background_mode || 'block') : 'none';
  const isBgBlock = rawBgMode === 'block' || (rawBgMode as string) === 'full_box';
  const isBgLine = rawBgMode === 'line' || (rawBgMode as string) === 'per_line';
  const effectiveBgColor = applyColorOpacity(style.background_color || '#000000', style.background_opacity ?? 1.0);
  const bgOffsetX = (isBgBlock || isBgLine) ? (style.background_offset_x || 0) : 0;
  const bgOffsetY = (isBgBlock || isBgLine) ? (style.background_offset_y || 0) : 0;
  const bgOffsetTransform = (bgOffsetX !== 0 || bgOffsetY !== 0) ? `translate(${bgOffsetX}px, ${bgOffsetY}px)` : '';

  const rotation = layout.rotation ?? 0;

  return (
    <div
      onPointerMove={handlePointerMove}
      onPointerUp={handlePointerUp}
      onPointerCancel={handlePointerUp}
      style={{
        position: 'absolute',
        left: `${posX * 100}%`,
        top: `${posY * 100}%`,
        transform: `translate(-50%, -50%) rotate(${rotation}deg) scale(${scale}) ${extraAnimTransform}`.trim(),
        width: 'max-content',
        maxWidth: containerRef.current?.clientWidth
          ? `${Math.round(containerRef.current.clientWidth * (layout.max_width ?? 0.88))}px`
          : `${(layout.max_width ?? 0.88) * 100}%`,
        boxSizing: 'border-box',
        opacity: computedOpacity,
        cursor: isInteractive ? (isDragging ? 'grabbing' : 'grab') : 'default',
        touchAction: 'none'
      }}
      onPointerDown={handlePointerDownDrag}
      onClick={(e) => {
        if (!isInteractive) return;
        e.stopPropagation();
        onSelectGroup(group.id);
      }}
      onDoubleClick={(e) => {
        if (!isInteractive) return;
        e.stopPropagation();
        onOpenTextEditor?.(group.id);
      }}
      className={`relative transition-shadow group ${
        isSelected && isInteractive ? 'ring-2 ring-primary/80 shadow-[0_0_15px_rgba(37,99,235,0.35)]' : ''
      }`}
    >
      {/* CAPCUT SELECTION OVERLAY & TRANSFORM HANDLES */}
      {isSelected && isInteractive && (
        <div className="absolute -inset-2.5 border border-dashed border-primary rounded-lg pointer-events-none z-30">
          {/* Top-Left: Close / Deselect handle */}
          <div 
            onClick={(e) => { e.stopPropagation(); onSelectGroup(null); }}
            className="absolute -top-5 -left-5 w-11 h-11 flex items-center justify-center pointer-events-auto cursor-pointer"
            title="Deselect"
          >
            <div className="w-5 h-5 rounded-full bg-surface-200 border border-border flex items-center justify-center hover:bg-red-500 transition-colors shadow-md">
              <X size={10} className="text-white" />
            </div>
          </div>

          {/* Bottom-Right: Scale / Resize handle */}
          <div 
            onPointerDown={handleResizePointerDown}
            className="absolute -bottom-5 -right-5 w-11 h-11 flex items-center justify-center pointer-events-auto cursor-nwse-resize"
            title="Drag to resize text scale"
          >
            <div className="w-5 h-5 rounded-full bg-primary border border-white flex items-center justify-center hover:scale-110 transition-transform shadow-md">
              <Maximize2 size={10} className="text-white" />
            </div>
          </div>

          {/* Bottom-Center: Move Anchor handle */}
          <div 
            className="absolute -bottom-2.5 left-1/2 -translate-x-1/2 w-4 h-4 rounded-full bg-surface-200 border border-primary flex items-center justify-center pointer-events-none"
          >
            <Move size={8} className="text-primary" />
          </div>

          {/* Quick Apply to All Captions Action Pill (Captions only) */}
          {onApplyLayoutToAll && isCaption && (
            <div className="absolute -bottom-10 left-1/2 -translate-x-1/2 pointer-events-auto z-40">
              <button

                onClick={(e) => {
                  e.stopPropagation();
                  onApplyLayoutToAll(group.id);
                }}
                className="h-6 px-2.5 rounded-full bg-[#1C1F26]/95 border border-[#2E3239] hover:border-primary text-text-secondary hover:text-white text-[10px] font-semibold flex items-center gap-1.5 shadow-lg active:scale-95 transition-all whitespace-nowrap"
                title="Apply this caption's position & size to all spoken captions"

          )}
        </div>
      )}

      {/* RENDERED TEXT AND WORD-LEVEL POP ANIMATION */}
      <div
        className={`w-full flex flex-col ${
          layout.alignment === 'left' ? 'items-start text-left' : layout.alignment === 'right' ? 'items-end text-right' : 'items-center text-center'
        } p-1.5 rounded-lg transition-all`}
        style={{
          transform: bgOffsetTransform || undefined,
          ...(isBgBlock && {
            backgroundColor: effectiveBgColor,
            padding: `${style.background_padding_y || 8}px ${style.background_padding_x || 14}px`,
            borderRadius: `${style.background_corner_radius || 12}px`,
          })
        }}
      >
        {visualLines.map((lineWords, lIdx) => {
          const lineContent = lineWords.map((w, idx) => {
            const hasWordTiming = isCaption && (group as any).has_word_timing !== false && (group as any).timing_mode !== 'phrase' && w.is_exact !== false;
            const isSpoken = hasWordTiming && currentTime >= w.start && currentTime <= w.end;
            const wordOverride = w.style_override;

            // Tactile scale pop calculation (Spring Punch) ONLY when layer is caption
            let wordScale = 1.0;
            const activeAnimType = anim.active_word?.type || anim.type;
            const punchScale = anim.active_word?.active_scale || anim.active_scale || 1.16;
            const animDurationMs = anim.active_word?.duration_ms || anim.duration_ms || 100;

              const elapsed = Math.max(0, currentTime - w.start);
              if (elapsed < durationSec) {
                const progress = elapsed / durationSec;
                const punch = punchScale - 1.0;
                wordScale = 1.0 + punch * Math.cos(progress * (Math.PI / 2));
              }
            }

            // Color resolution: Normal text strictly preserves primary_color; Spoken highlight only applies to captions
            let textColor = style.primary_color;
            if (isCaption && isSpoken && highlightEnabled) {
              textColor = wordOverride?.highlight_color || style.highlight_color;
            } else if (wordOverride?.color) {
              textColor = wordOverride.color;
            }

            const strokeWidth = style.outline_enabled ? `${Math.max(1.5, (style.outline_width || 4.5) * 0.6)}px` : '0px';

            const filters: string[] = [];
            if (style.shadow_enabled) {
              filters.push(`drop-shadow(${style.shadow_offset_x || 0}px ${style.shadow_offset_y || 2}px ${style.shadow_blur || 4}px ${style.shadow_color || 'rgba(0,0,0,0.85)'})`);
            }
            if (style.glow_enabled) {
              filters.push(`drop-shadow(0px 0px ${style.glow_blur || 14}px ${style.glow_color || '#00F0FF'})`);
            }

            return (
              <React.Fragment key={w.id || idx}>
                <span
                  style={{
                    fontFamily: fontFam,
                    fontSize: `${fontSize}px`,
                    fontWeight: w.style_override?.font_weight || style.font_weight || 'bold',
                    textTransform: (style.text_transform as any) || 'uppercase',
                    letterSpacing: `${style.letter_spacing || 0}px`,
                    color: textColor,
                    WebkitTextStroke: style.outline_enabled ? `${strokeWidth} ${style.outline_color}` : undefined,
                    paintOrder: 'stroke fill',
                    filter: filters.length > 0 ? filters.join(' ') : undefined,
                    transform: `scale(${wordScale})`,
                    display: 'inline-block',
                    transformOrigin: 'center bottom',
                    transition: 'transform 0.04s cubic-bezier(0.16, 1, 0.3, 1), color 0.05s ease'
                  }}
                >
                  {w.text}
                </span>
                {idx < lineWords.length - 1 ? ' ' : null}
              </React.Fragment>
            );
          });

          if (isBgLine) {
            return (
              <div key={lIdx} className="my-0.5 inline-block text-center w-auto">
                <span
                  style={{
                    display: 'inline-block',
                    backgroundColor: effectiveBgColor,
                    padding: `${Math.round((style.background_padding_y || 6) * 0.45)}px ${Math.round((style.background_padding_x || 12) * 0.45)}px`,
                    borderRadius: `${Math.round((style.background_corner_radius || 8) * 0.6)}px`,
                  }}
                >
                  {lineContent}
                </span>
              </div>
            );
          }

          return (
            <div key={lIdx} className="my-0.5 inline-block text-center w-auto">
              {lineContent}
            </div>
          );
        })}
      </div>
    </div>
  );
});

export const InteractiveTextCanvas: React.FC<InteractiveTextCanvasProps> = ({
  track,
  currentTime,
  selectedGroupId,
  onSelectGroup,
  onOpenTextEditor,
  onUpdateGroup,
  onApplyLayoutToAll,
  layoutMode = 'fit_black',
  isInteractive = true
}) => {
  const containerRef = useRef<HTMLDivElement>(null);

  if (!track || !track.groups || track.groups.length === 0) {
    return null;
  }

  // Find all active visible groups for current playback time (timeline-aware visibility)
  const activeGroups = track.groups.filter(
    grp => grp.type !== 'audio' && grp.type !== 'voiceover' && !grp.is_hidden && currentTime >= grp.start && currentTime <= grp.end
  );

  if (activeGroups.length === 0) return null;

  return (
    <div
      ref={containerRef}
      className="absolute inset-0 z-20 overflow-hidden select-none"
      style={{
        paddingLeft: '4%',
        paddingRight: '4%',
      }}
    >
      {activeGroups.map((grp) => (
        <CanvasGroupItem
          key={grp.id}
          group={grp}
          defaultStyle={track.default_style}
          defaultLayout={track.default_layout}
          defaultAnimation={track.default_animation}
          isSelected={selectedGroupId === grp.id}
          currentTime={currentTime}
          containerRef={containerRef}
          onSelectGroup={onSelectGroup}
          onOpenTextEditor={onOpenTextEditor}
          onUpdateGroup={onUpdateGroup}
          onApplyLayoutToAll={onApplyLayoutToAll}
          layoutMode={layoutMode}
          isInteractive={isInteractive}
        />
      ))}
    </div>
  );
};

/**
 * linePartition.ts
 * Deterministic text and caption line layout engine.
 * Calculates visual line boundaries based on font family, font size, font weight,
 * letter spacing, and available container width using an offscreen canvas context.
 */

export interface WordLike {
  id?: string;
  text?: string;
  word?: string;
  start?: number;
  end?: number;
  style_override?: any;
  [key: string]: any;
}

let cachedCanvas: HTMLCanvasElement | null = null;
let cachedCtx: CanvasRenderingContext2D | null = null;

function getSharedMeasureContext(): CanvasRenderingContext2D | null {
  if (typeof document === 'undefined') return null;
  if (!cachedCanvas) {
    try {
      cachedCanvas = document.createElement('canvas');
      cachedCtx = cachedCanvas.getContext('2d');
    } catch {
      cachedCtx = null;
    }
  }
  return cachedCtx;
}

/**
 * Deterministically partitions an array of words into visual lines matching
 * actual rendered layout constraints.
 */
export function partitionWordsIntoVisualLines<T extends WordLike>(
  words: T[],
  fontFamily: string,
  fontSize: number,
  fontWeight: string | number,
  letterSpacing: number,
  maxWidthPx: number,
  textTransform?: string
): T[][] {
  if (!words || words.length === 0) return [];

  const ctx = getSharedMeasureContext();
  if (ctx) {
    ctx.font = `${fontWeight || 'bold'} ${Math.round(fontSize)}px "${fontFamily || 'Montserrat'}", sans-serif`;
  }

  const spaceWidth = ctx ? ctx.measureText(' ').width : fontSize * 0.3;
  const lines: T[][] = [];
  let currentLine: T[] = [];
  let currentLineWidth = 0;

  for (const word of words) {
    let rawText = (word.text || word.word || '').trim();
    if (textTransform === 'uppercase') rawText = rawText.toUpperCase();
    else if (textTransform === 'lowercase') rawText = rawText.toLowerCase();

    // Support explicit newlines inside caption text
    if (rawText.includes('\n')) {
      const parts = rawText.split('\n');
      for (let i = 0; i < parts.length; i++) {
        const partText = parts[i].trim();
        if (i > 0 && currentLine.length > 0) {
          lines.push(currentLine);
          currentLine = [];
          currentLineWidth = 0;
        }
        if (partText) {
          const wWidth = ctx
            ? ctx.measureText(partText).width + (letterSpacing || 0) * partText.length
            : partText.length * fontSize * 0.55;
          currentLine.push({ ...word, text: partText });
          currentLineWidth += wWidth + spaceWidth;
        }
      }
      continue;
    }

    const wordWidth = ctx
      ? ctx.measureText(rawText).width + (letterSpacing || 0) * rawText.length
      : rawText.length * fontSize * 0.55;

    // Wrap to a new line when the word exceeds maximum content width
    if (currentLine.length > 0 && currentLineWidth + wordWidth > maxWidthPx) {
      lines.push(currentLine);
      currentLine = [word];
      currentLineWidth = wordWidth + spaceWidth;
    } else {
      currentLine.push(word);
      currentLineWidth += wordWidth + spaceWidth;
    }
  }

  if (currentLine.length > 0) {
    lines.push(currentLine);
  }

  return lines;
}

import { AnimationConfig, CaptionGroup, CaptionTrack, LayerType, LayoutConfig, StyleConfig, Word } from './types';

export function normalizeCaptionTrack(raw: unknown, clipId: string, duration: number): CaptionTrack {
  const source = raw && typeof raw === 'object' ? raw as Record<string, any> : {};
  const rawStyle = source.default_style || {};
  const rawLayout = source.default_layout || {};
  const rawAnimation = source.default_animation || {};

  const default_style: StyleConfig = {
    font_family: rawStyle.font_family || 'Montserrat',
    font_size: rawStyle.font_size || 44,
    font_weight: rawStyle.font_weight || '900',
    text_transform: rawStyle.text_transform || 'none',
    letter_spacing: rawStyle.letter_spacing || 0,
    line_height: rawStyle.line_height || 1.25,
    primary_color: rawStyle.primary_color || rawStyle.color || '#FFFF00',
    highlight_color: rawStyle.highlight_color || '#FFFFFF',
    outline_enabled: rawStyle.outline_enabled ?? Boolean(rawStyle.stroke_width),
    outline_color: rawStyle.outline_color || rawStyle.stroke_color || '#000000',
    outline_width: rawStyle.outline_width || rawStyle.stroke_width || 3,
    shadow_enabled: rawStyle.shadow_enabled ?? rawStyle.has_shadow ?? true,
    shadow_color: rawStyle.shadow_color || '#000000',
    shadow_offset_x: rawStyle.shadow_offset_x || 0,
    shadow_offset_y: rawStyle.shadow_offset_y || 2,

import { CaptionTrack } from './types';

/**
 * Format seconds into SubRip (SRT) timestamp: HH:MM:SS,mmm
 */
function formatSrtTimestamp(seconds: number): string {
  const safe = Math.max(0, Number.isFinite(seconds) ? seconds : 0);
  const hrs = Math.floor(safe / 3600);
  const mins = Math.floor((safe % 3600) / 60);
  const secs = Math.floor(safe % 60);
  const millis = Math.floor((safe % 1) * 1000);

  const hh = hrs.toString().padStart(2, '0');
  const mm = mins.toString().padStart(2, '0');
  const ss = secs.toString().padStart(2, '0');
  const mmm = millis.toString().padStart(3, '0');

  return `${hh}:${mm}:${ss},${mmm}`;
}

/**
 * Generate standard SubRip (.srt) subtitle content from authentic caption track.
 */
export function generateSrtFromTrack(track: CaptionTrack | null): string {
  if (!track || !track.groups || track.groups.length === 0) {
    return '';
  }

  // Filter only spoken caption or text groups, sort by start time
  const validGroups = track.groups
    .filter((g) => (!g.type || g.type === 'caption' || g.type === 'text') && g.words && g.words.length > 0)
    .sort((a, b) => a.start - b.start);

  if (validGroups.length === 0) {
    return '';
  }

  const srtBlocks: string[] = [];

  validGroups.forEach((grp, idx) => {
    const startStr = formatSrtTimestamp(grp.start);
    const endStr = formatSrtTimestamp(grp.end);
    const text = (grp.words || []).map((w) => w.text).join(' ');

    if (!text.trim()) return;

    srtBlocks.push(`${idx + 1}\n${startStr} --> ${endStr}\n${text.trim()}`);
  });

  return srtBlocks.join('\n\n') + '\n';
}

/**
 * Trigger client-side file download of .srt subtitles
 */
export function downloadSrtFile(track: CaptionTrack | null, clipTitle: string = 'captions') {
  const content = generateSrtFromTrack(track);
  if (!content) {
    alert('No caption text available to export.');
    return;
  }

  const cleanFilename = (clipTitle || 'dispatch_captions')
    .toLowerCase()
    .replace(/[^a-z0-9_-]+/g, '_')
    .slice(0, 32);

  const blob = new Blob([content], { type: 'text/plain;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  const anchor = document.createElement('a');
  anchor.href = url;
  anchor.download = `${cleanFilename}.srt`;
  document.body.appendChild(anchor);
  anchor.click();
  document.body.removeChild(anchor);
  URL.revokeObjectURL(url);
}

export function extendSourceRange(

export function sourceContextRange(
  range: SourceRange,
  side: 'before' | 'after',
  sourceDuration: number,
  seconds = 4,
): SourceRange | null {
  if (!Number.isFinite(sourceDuration) || sourceDuration <= 0 || seconds <= 0) return null;
  const context = side === 'before'
    ? { start: Math.max(0, range.start - seconds), end: range.start }
    : { start: range.end, end: Math.min(sourceDuration, range.end + seconds) };
  return context.end > context.start ? context : null;
}

  return Number.isFinite(currentTime) && Number.isFinite(selectedEnd) && currentTime >= selectedEnd;
}

export function beginTrimPointer(
  side: 'start' | 'end',
  pointerId: number,
  clientX: number,
  range: SourceRange,
): TrimPointerState {
  return { side, pointerId, startX: clientX, initialRange: { ...range } };
}

/** Apply a real pointer delta to the active source-time selection. */
export function moveTrimPointer(
  pointer: TrimPointerState,
  pointerId: number,
  clientX: number,
  pixelsPerSecond: number,
  sourceDuration: number,
): SourceRange {
  if (pointer.pointerId !== pointerId || pixelsPerSecond <= 0 || !Number.isFinite(clientX)) {
    return pointer.initialRange;
  }
  const sourceEnd = Number.isFinite(sourceDuration) && sourceDuration > 0 ? sourceDuration : 0;
  const delta = (clientX - pointer.startX) / pixelsPerSecond;
  const { start, end } = pointer.initialRange;
  const minLength = Math.min(MIN_CLIP_SELECTION_SECONDS, sourceEnd);
  const maxLength = Math.min(MAX_CLIP_SELECTION_SECONDS, sourceEnd);

  if (pointer.side === 'start') {
    const minStart = Math.max(0, end - maxLength);
    const maxStart = Math.max(minStart, end - minLength);
    return { start: clamp(start + delta, minStart, maxStart), end };
  }

  const minEnd = Math.min(sourceEnd, start + minLength);
  const maxEnd = Math.min(sourceEnd, start + maxLength);
  return { start, end: clamp(end + delta, minEnd, maxEnd) };
}

function clamp(value: number, min: number, max: number): number {
  return Math.min(max, Math.max(min, value));
}

import { CaptionGroup, Word } from './types';

/**
 * Intelligent word-timing preservation engine.
 * 
 * Invariants:
 * 1. Single word edits strictly preserve every other word's ID, start/end timestamps, and style overrides.
 * 2. Multi-word edits find unchanged anchor words via alignment and only interpolate timing for the modified span.
 * 3. Never silently distribute words evenly across a segment when authentic Whisper timing exists.
 */

export function preserveWordTimingsOnEdit(
  originalWords: Word[] = [],
  newText: string,
  groupStart: number,
  groupEnd: number,
  groupId: string = 'grp'
): Word[] {
  const trimmed = newText.trim();
  if (!trimmed) {
    return [];
  }

  const rawTokens = trimmed.split(/\s+/).filter(Boolean);
  if (rawTokens.length === 0) {
    return [];
  }

  const totalDuration = Math.max(0.2, groupEnd - groupStart);

  // Case 0: No prior words existed — construct baseline with estimated flags
  if (!originalWords || originalWords.length === 0) {
    const step = totalDuration / rawTokens.length;
    return rawTokens.map((text, idx) => ({
      id: `${groupId}-w${idx}-${Date.now()}`,
      text,
      start: Number((groupStart + idx * step).toFixed(2)),
      end: Number((groupStart + (idx + 1) * step).toFixed(2)),
      confidence: 0.8,
    }));
  }

  // Case 1: Word count is identical — 1:1 mapping preserving all unmodified words and timestamps
  if (originalWords.length === rawTokens.length) {
    return rawTokens.map((token, idx) => {
      const orig = originalWords[idx];
      return {
        ...orig,
        text: token,
        // Preserve original id and timings exactly
        id: orig.id || `${groupId}-w${idx}`,
        start: orig.start,
        end: orig.end,
        style_override: orig.style_override,
      };
    });
  }

  // Case 2: Multi-word insertion or deletion — find prefix and suffix matches (anchor points)
  let prefixCount = 0;
  while (
    prefixCount < originalWords.length &&
    prefixCount < rawTokens.length &&
    originalWords[prefixCount].text.toLowerCase() === rawTokens[prefixCount].toLowerCase()
  ) {
    prefixCount++;
  }

  let suffixCount = 0;
  while (
    suffixCount < (originalWords.length - prefixCount) &&
    suffixCount < (rawTokens.length - prefixCount) &&
    originalWords[originalWords.length - 1 - suffixCount].text.toLowerCase() ===
      rawTokens[rawTokens.length - 1 - suffixCount].toLowerCase()
  ) {
    suffixCount++;
  }

  const result: Word[] = [];

  // 1. Add matched prefix words with untouched timestamps and IDs
  for (let i = 0; i < prefixCount; i++) {
    result.push({
      ...originalWords[i],
      text: rawTokens[i], // Keep casing of new text
    });
  }

  // 2. Identify the middle span to retime
  const changedTokens = rawTokens.slice(prefixCount, rawTokens.length - suffixCount);
  
  if (changedTokens.length > 0) {
    const spanStart = prefixCount > 0 ? originalWords[prefixCount - 1].end : groupStart;
    const spanEnd = suffixCount > 0 
      ? originalWords[originalWords.length - suffixCount].start 
      : groupEnd;
    
    const spanDuration = Math.max(0.1, spanEnd - spanStart);
    const step = spanDuration / changedTokens.length;

    changedTokens.forEach((token, idx) => {
      result.push({
        id: `${groupId}-mod-${prefixCount + idx}-${Date.now()}`,
        text: token,
        start: Number((spanStart + idx * step).toFixed(2)),
        end: Number((spanStart + (idx + 1) * step).toFixed(2)),
        confidence: 0.8,
      });
    });
  }

  // 3. Add matched suffix words with untouched timestamps and IDs
  const origSuffixStartIndex = originalWords.length - suffixCount;
  const newSuffixStartIndex = rawTokens.length - suffixCount;

  for (let i = 0; i < suffixCount; i++) {
    const origWord = originalWords[origSuffixStartIndex + i];
    result.push({
      ...origWord,
      text: rawTokens[newSuffixStartIndex + i],
    });
  }

  return result;
}

/**
 * Split a caption group at splitTime while preserving authentic word timestamps.
 */
export function splitCaptionGroupAtTime(
  group: CaptionGroup,
  splitTime: number
): [CaptionGroup, CaptionGroup] | null {
  if (splitTime <= group.start + 0.1 || splitTime >= group.end - 0.1) {
    return null;
  }

  const words = group.words || [];
  const leftWords: Word[] = [];
  const rightWords: Word[] = [];

  words.forEach((w) => {
    const mid = (w.start + w.end) / 2;
    if (mid <= splitTime) {
      leftWords.push({
        ...w,
        end: Math.min(w.end, splitTime),
      });
    } else {
      rightWords.push({
        ...w,
        start: Math.max(w.start, splitTime),
      });
    }
  });

  const leftGroup: CaptionGroup = {
    ...group,
    id: group.id,
    start: group.start,
    end: splitTime,
    words: leftWords,
  };

  const rightGroup: CaptionGroup = {
    ...group,
    id: `grp_${Date.now()}_split`,
    start: splitTime,
    end: group.end,
    words: rightWords,
  };

  return [leftGroup, rightGroup];
}

/**
 * Merge two adjacent caption groups into one, preserving all authentic word timestamps.
 */
export function mergeAdjacentCaptionGroups(
  groupA: CaptionGroup,
  groupB: CaptionGroup
): CaptionGroup {
  const first = groupA.start <= groupB.start ? groupA : groupB;
  const second = groupA.start <= groupB.start ? groupB : groupA;

  const combinedWords = [...(first.words || []), ...(second.words || [])].sort(
    (a, b) => a.start - b.start
  );

  return {
    ...first,
    id: first.id,
    start: Math.min(first.start, second.start),
    end: Math.max(first.end, second.end),
    words: combinedWords,
  };
}

import { useState, useCallback, useRef } from 'react';
import { CaptionTrack } from './types';

/**
 * High-performance undo/redo history hook for caption tracks.
 * 
 * Supports:
 * - Transient updates during active gestures (drag, resize, trim) with zero cloning overhead.
 * - Single immutable commit boundary on gesture completion (`pointerup`).
 * - Reactive canUndo and canRedo state tracking.
 * - History isolation bounded strictly to 50 snapshots.
 */
export function useCaptionHistory(initialTrack: CaptionTrack | null) {
  const [track, setInternalTrack] = useState<CaptionTrack | null>(initialTrack);
  const historyRef = useRef<CaptionTrack[]>(initialTrack ? [JSON.parse(JSON.stringify(initialTrack))] : []);
  const indexRef = useRef<number>(0);

  const [canUndo, setCanUndo] = useState<boolean>(false);
  const [canRedo, setCanRedo] = useState<boolean>(false);

  const syncFlags = useCallback(() => {
    setCanUndo(indexRef.current > 0);
    setCanRedo(indexRef.current < historyRef.current.length - 1);
  }, []);

  /**
   * Set caption track.
   * @param newTrack The updated caption track.
   * @param commitToHistory If false, updates track for fluid preview without appending a history snapshot.
   */
  const setTrack = useCallback((newTrack: CaptionTrack, commitToHistory: boolean = true) => {
    if (!commitToHistory) {
      // Fluid preview update: update state directly without JSON cloning or history mutation
      setInternalTrack(newTrack);
      return;
    }

    const cloned = JSON.parse(JSON.stringify(newTrack));
    const newHistory = historyRef.current.slice(0, indexRef.current + 1);
    newHistory.push(cloned);
    if (newHistory.length > 50) {
      newHistory.shift();
    }
    historyRef.current = newHistory;
    indexRef.current = newHistory.length - 1;
    setInternalTrack(cloned);
    syncFlags();
  }, [syncFlags]);

  /**
   * Explicit commit at gesture end.
   */
  const commitTrack = useCallback((finalTrack: CaptionTrack) => {
    setTrack(finalTrack, true);
  }, [setTrack]);

  /**
   * Explicit transient preview during active gesture.
   */
  const setTransientTrack = useCallback((previewTrack: CaptionTrack) => {
    setTrack(previewTrack, false);
  }, [setTrack]);

  const undo = useCallback((): CaptionTrack | null => {
    if (indexRef.current > 0) {
      indexRef.current -= 1;
      const prev = JSON.parse(JSON.stringify(historyRef.current[indexRef.current]));
      setInternalTrack(prev);
      syncFlags();
      return prev;
    }
    return null;
  }, [syncFlags]);

  const redo = useCallback((): CaptionTrack | null => {
    if (indexRef.current < historyRef.current.length - 1) {
      indexRef.current += 1;
      const next = JSON.parse(JSON.stringify(historyRef.current[indexRef.current]));
      setInternalTrack(next);
      syncFlags();
      return next;
    }
    return null;
  }, [syncFlags]);

  const resetHistory = useCallback((freshTrack: CaptionTrack) => {
    const cloned = JSON.parse(JSON.stringify(freshTrack));
    historyRef.current = [cloned];
    indexRef.current = 0;
    setInternalTrack(cloned);
    setCanUndo(false);
    setCanRedo(false);
  }, []);

  return {
    track,
    setTrack,
    commitTrack,
    setTransientTrack,
    undo,
    redo,
    canUndo,
    canRedo,
    resetHistory,
  };
}

// Color math helpers
function clamp(val: number, min: number, max: number): number {
  return Math.max(min, Math.min(max, val));
}

function parseHex(hexStr: string): { r: number; g: number; b: number; a: number } {
  let hex = hexStr.trim().replace(/^#/, '');
  if (hex.length === 3) {
    hex = hex.split('').map((c) => c + c).join('') + 'ff';
  } else if (hex.length === 4) {
    hex = hex.split('').map((c) => c + c).join('');
  } else if (hex.length === 6) {
    hex = hex + 'ff';
  } else if (hex.length !== 8) {
    return { r: 255, g: 255, b: 255, a: 1 };
  }

  const r = parseInt(hex.substring(0, 2), 16) || 0;
  const g = parseInt(hex.substring(2, 4), 16) || 0;
  const b = parseInt(hex.substring(4, 6), 16) || 0;
  const a = parseInt(hex.substring(6, 8), 16) / 255;

  return { r, g, b, a: isNaN(a) ? 1 : a };
}

function rgbToHex(r: number, g: number, b: number, a: number = 1, includeAlpha: boolean = false): string {
  const toHex = (n: number) => clamp(Math.round(n), 0, 255).toString(16).padStart(2, '0').toUpperCase();
  const hex = `#${toHex(r)}${toHex(g)}${toHex(b)}`;
  if (includeAlpha && a < 0.999) {
    return `${hex}${toHex(a * 255)}`;
  }
  return hex;
}

function rgbToHsv(r: number, g: number, b: number): { h: number; s: number; v: number } {
  const rn = r / 255;
  const gn = g / 255;
  const bn = b / 255;
  const max = Math.max(rn, gn, bn);
  const min = Math.min(rn, gn, bn);
  const delta = max - min;

  let h = 0;
  if (delta !== 0) {
    if (max === rn) {
      h = ((gn - bn) / delta) % 6;
    } else if (max === gn) {
      h = (bn - rn) / delta + 2;
    } else {
      h = (rn - gn) / delta + 4;
    }
    h = Math.round(h * 60);
    if (h < 0) h += 360;
  }

  const s = max === 0 ? 0 : delta / max;
  const v = max;

  return { h, s, v };
}

function hsvToRgb(h: number, s: number, v: number): { r: number; g: number; b: number } {
  const c = v * s;
  const x = c * (1 - Math.abs(((h / 60) % 2) - 1));
  const m = v - c;

  let rPrime = 0;
  let gPrime = 0;
  let bPrime = 0;

  if (h >= 0 && h < 60) {
    rPrime = c; gPrime = x; bPrime = 0;
  } else if (h >= 60 && h < 120) {
    rPrime = x; gPrime = c; bPrime = 0;
  } else if (h >= 120 && h < 180) {
    rPrime = 0; gPrime = c; bPrime = x;
  } else if (h >= 180 && h < 240) {
    rPrime = 0; gPrime = x; bPrime = c;
  } else if (h >= 240 && h < 300) {
    rPrime = x; gPrime = 0; bPrime = c;
  } else {
    rPrime = c; gPrime = 0; bPrime = x;
  }

  return {
    r: Math.round((rPrime + m) * 255),
    g: Math.round((gPrime + m) * 255),
    b: Math.round((bPrime + m) * 255),
  };
}

  const handleReset = () => {
    onChange(initialColor);
    const { r, g, b, a } = parseHex(initialColor);
    const newHsv = rgbToHsv(r, g, b);
    setHsv(newHsv);
    setAlpha(a);
    setHexInput(rgbToHex(r, g, b, a, enableAlpha));
    setIsHexValid(true);
  };

  if (!isOpen) return null;

  const currentRgb = hsvToRgb(hsv.h, hsv.s, hsv.v);
  const currentHex = rgbToHex(currentRgb.r, currentRgb.g, currentRgb.b, alpha, enableAlpha);
  const pureHueRgb = hsvToRgb(hsv.h, 1, 1);
  const pureHueHex = rgbToHex(pureHueRgb.r, pureHueRgb.g, pureHueRgb.b);

  return (
    <div

      onClick={onClose}

      <div

              title="Reset to original color"
              aria-label="Reset color"
            >
              <RotateCcw size={13} />
            </button>
            <button
              type="button"
              onClick={onClose}
              className="w-8 h-8 rounded-lg bg-[#232323] hover:bg-[#282828] border border-[#2E2E2E] flex items-center justify-center text-text-secondary hover:text-text-main active:scale-95 transition-all"
              title="Close"
              aria-label="Close"
            >
              <X size={15} />

            backgroundColor: pureHueHex,

              <span className="font-mono tabular-nums">{hsv.h}°</span>
            </div>
            <div className="relative h-6 rounded-lg overflow-hidden border border-[#2E2E2E] flex items-center">
              <input
                type="range"

                className="w-full h-full opacity-0 cursor-pointer z-10"

              <div
                className="absolute inset-0 pointer-events-none"
                style={{
                  background: 'linear-gradient(to right, #ff0000 0%, #ffff00 17%, #00ff00 33%, #00ffff 50%, #0000ff 67%, #ff00ff 83%, #ff0000 100%)',

                    max={255}

                    className="w-full bg-transparent text-[10px] font-mono text-right text-text-main outline-none [appearance:textfield] [&::-webkit-outer-spin-button]:appearance-none [&::-webkit-inner-spin-button]:appearance-none"
                  />
                </div>
              ))}
            </div>
          </div>
        </div>

        {/* Recent Colors Shelf */}
        <div className="space-y-1.5">
          <span className="text-[11px] font-medium text-text-secondary">Recent Colors</span>
          <div className="flex items-center gap-1.5 overflow-x-auto no-scrollbar py-0.5">
            {recentColors.map((c, i) => (
              <button
                key={`${c}-${i}`}
                type="button"
                onClick={() => {
                  const { r, g, b, a } = parseHex(c);
                  const newHsv = rgbToHsv(r, g, b);
                  setHsv(newHsv);
                  setAlpha(a);
                  setHexInput(c);
                  setIsHexValid(true);
                  onChange(c);
                }}
                className={`w-7 h-7 rounded-lg shrink-0 border-2 transition-transform active:scale-90 ${
                  currentHex.toUpperCase() === c.toUpperCase() ? 'border-primary scale-110 shadow-hero-sm' : 'border-[#2E2E2E]'
                }`}
                style={{ backgroundColor: c }}
                title={c}
              />
            ))}
          </div>
        </div>

        {/* Done Button (Law of Single Hero CTA) */}
        <div className="pt-1">
          <button
            type="button"
            onClick={handleDone}
            className="w-full h-10 rounded-xl bg-primary hover:bg-primary-hover text-white text-[12px] font-semibold flex items-center justify-center gap-1.5 active:scale-[0.98] transition-all shadow-hero-glow"
          >
            <Check size={14} />
            <span>Apply Custom Color</span>
          </button>
        </div>
      </div>
    </div>
  );
};

import React from 'react';
import { AlertCircle, RotateCw } from 'lucide-react';

interface ScreenErrorBoundaryProps {
  children: React.ReactNode;
  title: string;
  resetKey?: string | number;
}

interface ScreenErrorBoundaryState {
  error: Error | null;
  retryCount: number;
}

export class ScreenErrorBoundary extends React.Component<ScreenErrorBoundaryProps, ScreenErrorBoundaryState> {
  state: ScreenErrorBoundaryState = { error: null, retryCount: 0 };

  static getDerivedStateFromError(error: Error): Partial<ScreenErrorBoundaryState> {
    return { error };
  }

  componentDidCatch(error: Error, info: React.ErrorInfo) {
    console.error(`[Dispatch] ${this.props.title}`, error, info.componentStack);
  }

  componentDidUpdate(previousProps: ScreenErrorBoundaryProps) {
    if (previousProps.resetKey !== this.props.resetKey && this.state.error) {
      this.setState({ error: null });
    }
  }

  private retry = () => this.setState((state) => ({ error: null, retryCount: state.retryCount + 1 }));

  render() {
    const { error, retryCount } = this.state;
    if (!error) {
      return <React.Fragment key={retryCount}>{this.props.children}</React.Fragment>;
    }

    return (
      <main role="alert" className="h-full w-full flex flex-col items-center justify-center gap-3 bg-studio px-6 text-center text-text-main">
        <div className="flex h-11 w-11 items-center justify-center rounded-xl border border-danger/30 bg-surface-100 text-danger">
          <AlertCircle size={21} />
        </div>
        <h2 className="text-sm font-semibold">{this.props.title}</h2>
        <p className="max-w-xl break-words text-xs text-text-secondary">
          {error.message || 'An unexpected rendering error occurred.'}
        </p>
        <button
          type="button"
          onClick={this.retry}
          className="min-h-11 px-4 rounded-lg bg-primary text-white text-xs font-semibold shadow-hero-glow hover:bg-primary-hover active:scale-[0.98] transition-all flex items-center gap-2"
        >
          <RotateCw size={13} />
          <span>Retry screen</span>
        </button>
      </main>
    );
  }
}
