def format_into_paragraphs(text, target_words=70):
    text = " ".join(text.split())
    sentences = re.split(r"(?<=[.!?])\s+", text)
    paragraphs = []
    current_para = []
    current_count = 0
    for s in sentences:
        s = s.strip()
        if not s:
            continue
        current_para.append(s)
        current_count += len(s.split())
        if current_count >= target_words:
            paragraphs.append(" ".join(current_para))
            current_para = []
            current_count = 0
    if current_para:
        paragraphs.append(" ".join(current_para))
    return "\n\n".join(paragraphs)

def fetch_oembed_title(vid):
    try:
        url = f"https://www.youtube.com/watch?v={vid}"
        oembed_url = f"https://www.youtube.com/oembed?url={url}&format=json"
        req = urllib.request.Request(oembed_url, headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(req, timeout=10) as resp:
            data = json.loads(resp.read().decode("utf-8"))
            return data.get("title")
    except Exception:
        return None


def format_into_paragraphs(text, target_words=70):
    text = " ".join(text.split())
    # Clean replacement characters
    text = text.replace("\ufffd", "'")
    sentences = re.split(r"(?<=[.!?])\s+", text)
    words = text.split()

    # If text has few/no sentence delimiters (common in auto captions), chunk words directly
    if len(sentences) <= 3 and len(words) > 100:
        paragraphs = []
        for i in range(0, len(words), target_words):
            chunk = " ".join(words[i : i + target_words])
            paragraphs.append(chunk)
        return "\n\n".join(paragraphs)

    paragraphs = []
    current_para = []
    current_count = 0
    for s in sentences:
        s = s.strip()
        if not s:
            continue
        current_para.append(s)
        current_count += len(s.split())
        if current_count >= target_words:
            paragraphs.append(" ".join(current_para))
            current_para = []
            current_count = 0
    if current_para:
        paragraphs.append(" ".join(current_para))
    return "\n\n".join(paragraphs)


def sanitize_filename(name):
    clean = re.sub(r'[\\/*?:"<>|]', "", name)
    clean = clean.strip().replace(" ", "_")
    return clean[:90]


def find_existing_transcript_on_disk(idx):
    prefix = f"{idx:02d}_"
    for fname in os.listdir(OUTPUT_DIR):
        if fname.startswith(prefix) and fname.endswith(".txt"):
            fpath = os.path.join(OUTPUT_DIR, fname)
            try:
                with open(fpath, "r", encoding="utf-8", errors="replace") as f:
                    content = f.read()
                parts = content.split("\n\n", 1)
                body = parts[1].strip() if len(parts) > 1 else content.strip()
                if body and "[Transcript could not be retrieved]" not in body and len(body.split()) > 30:
                    return body
            except Exception:
                pass
    return None

def transcribe_via_whisper(vid):
    audio_pattern = os.path.join(OUTPUT_DIR, f"temp_{vid}.*")
    # Clean any stale temp files first
    for f in glob.glob(audio_pattern):
        try:
            os.remove(f)
        except Exception:
            pass

    out_tmpl = os.path.join(OUTPUT_DIR, f"temp_{vid}.%(ext)s")
    print(f"  [Whisper] Downloading audio stream with yt-dlp for {vid}...")
    cmd = [
        "yt-dlp",
        "-f", "bestaudio",
        f"https://www.youtube.com/watch?v={vid}",
        "-o", out_tmpl
    ]
    subprocess.run(cmd, check=True)

    matches = glob.glob(audio_pattern)
    if not matches:
        print(f"  [Whisper] Could not find downloaded audio file for {vid}.")
        return None

    audio_file = matches[0]
    try:
        model = get_whisper_model()
        print(f"  [Whisper] Transcribing {os.path.basename(audio_file)}...")
        segments, _ = model.transcribe(audio_file, beam_size=5)
        raw_text_parts = [s.text.strip() for s in segments if s.text.strip()]
        return " ".join(raw_text_parts)
    finally:
        # Clean up temp audio file
        try:
            if os.path.exists(audio_file):
                os.remove(audio_file)
        except Exception:
            pass
