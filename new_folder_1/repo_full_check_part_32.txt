"""Pillar 2: Autonomous Transcription Worker for YouTube Inbox.
Consumes items in status 'DOWNLOADED' and 'WAITING_FOR_TRANSCRIPT',
implements an explicit asynchronous wait for YouTube captions,
and falls back to local faster-whisper only after the deadline.
"""
import os
import json
import time
import logging
import threading
from datetime import datetime, timezone, timedelta
from pathlib import Path
from typing import Dict, Any, List, Optional, Tuple

from dispatch import db
from dispatch.youtube_inbox.catcher import YouTubeInboxCatcher
from dispatch.transcription.transcriber import transcribe_video

logger = logging.getLogger("dispatch.transcription.worker")


def _parse_iso_or_sqlite_timestamp(ts_str: Optional[str]) -> Optional[datetime]:
    """Parse timestamp string from SQLite (YYYY-MM-DD HH:MM:SS or ISO-8601)."""
    if not ts_str:
        return None
    # Replace space with T if needed
    clean_ts = ts_str.strip().replace(" ", "T")
    try:
        dt = datetime.fromisoformat(clean_ts)
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        return dt
    except Exception:
        pass
    # Fallback formats
    for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%dT%H:%M:%S.%f", "%Y-%m-%d %H:%M:%S", "%Y-%m-%d %H:%M:%S.%f"):
        try:
            dt = datetime.strptime(ts_str.strip(), fmt)
            return dt.replace(tzinfo=timezone.utc)
        except ValueError:
            continue
    return None


class TranscriptionWorker:
    """Standalone worker for Pillar 2 transcription contract.
    
    States:
      DOWNLOADED -> WAITING_FOR_TRANSCRIPT -> TRANSCRIBED (source: 'youtube' or 'whisper')
    """

    def __init__(
        self,
        poll_interval_seconds: Optional[int] = None,
        caption_wait_seconds: Optional[int] = None,
        caption_reprobe_seconds: Optional[int] = None,
        catcher: Optional[YouTubeInboxCatcher] = None
    ):
        # Poll interval: check DB for downloadable/transcribable items every 15s by default
        env_interval = os.getenv("DISPATCH_TRANSCRIPTION_POLL_INTERVAL")
        self.poll_interval = poll_interval_seconds or (int(env_interval) if env_interval and env_interval.isdigit() else 15)

        # Wait window for YouTube captions before Whisper fallback (default 4 hours = 14400s)
        env_wait = os.getenv("DISPATCH_TRANSCRIPT_WAIT_SECONDS")
        self.caption_wait_seconds = caption_wait_seconds or (int(env_wait) if env_wait and env_wait.isdigit() else 14400)

        # Reprobe interval for checking YouTube captions (default 5 minutes = 300s)
        env_reprobe = os.getenv("DISPATCH_CAPTION_REPROBE_SECONDS")
        self.caption_reprobe_seconds = caption_reprobe_seconds or (int(env_reprobe) if env_reprobe and env_reprobe.isdigit() else 300)

        self.catcher = catcher or YouTubeInboxCatcher()
        self._stop_event = threading.Event()
        self._thread: Optional[threading.Thread] = None
        self._lock = threading.Lock()

    def start(self):
        """Start the background transcription worker thread."""
        if self._thread and self._thread.is_alive():
            logger.warning("Transcription worker is already running.")
            return

        self._stop_event.clear()
        self._thread = threading.Thread(target=self._run_loop, name="TranscriptionWorkerThread", daemon=True)
        self._thread.start()
        logger.info(
            "Transcription worker started (Poll interval: %ds, Caption wait window: %ds / %.1fh, Reprobe interval: %ds)",
            self.poll_interval, self.caption_wait_seconds, self.caption_wait_seconds / 3600.0, self.caption_reprobe_seconds
        )

    def stop(self):
        """Signal worker to stop."""
        self._stop_event.set()
        if self._thread:
            self._thread.join(timeout=5)
            logger.info("Transcription worker stopped.")

    def _run_loop(self):
        """Periodic loop scanning for DOWNLOADED and WAITING_FOR_TRANSCRIPT records."""
        # Initial run on boot
        try:
            self.process_pending()
        except Exception as e:
            logger.error("Initial transcription scan error: %s", e)

        while not self._stop_event.is_set():
            if self._stop_event.wait(timeout=self.poll_interval):
                break
            try:
                self.process_pending()
            except Exception as e:
                logger.error("Error in transcription worker cycle: %s", e)

    def process_pending(self) -> List[Dict[str, Any]]:
        """Scan DB for items requiring transcription processing and step their states."""
        with self._lock:
            processed = []

            # 0. FIFO Rolling Backlog: drain any queued transcriptions from yesterday / available quota
            try:
                from dispatch.transcription.gemini_engine import drain_queued_transcriptions
                drained = drain_queued_transcriptions(max_batch=5)
                if drained > 0:
                    logger.info("Drained %d queued transcription clips from backlog.", drained)
            except Exception as e:
                logger.error("Error running drain_queued_transcriptions: %s", e)

            # 1. Promote DOWNLOADED items to WAITING_FOR_TRANSCRIPT
            downloaded_items = db.list_youtube_inbox(status="DOWNLOADED", limit=20)
            for item in downloaded_items:
                res = self.process_downloaded_item(item)
                if res:
                    processed.append(res)

            # 2. Check WAITING_FOR_TRANSCRIPT items
            waiting_items = db.list_youtube_inbox(status="WAITING_FOR_TRANSCRIPT", limit=20)
            for item in waiting_items:
                res = self.process_waiting_item(item)
                if res:
                    processed.append(res)

            return processed

    def process_downloaded_item(self, item: Dict[str, Any]) -> Optional[Dict[str, Any]]:
        """Transition DOWNLOADED item into WAITING_FOR_TRANSCRIPT and probe captions immediately."""
        video_id = item["video_id"]
        now_dt = datetime.now(timezone.utc)
        deadline_dt = now_dt + timedelta(seconds=self.caption_wait_seconds)
        next_probe_dt = now_dt + timedelta(seconds=self.caption_reprobe_seconds)

        now_str = now_dt.strftime("%Y-%m-%d %H:%M:%S")
        deadline_str = deadline_dt.strftime("%Y-%m-%d %H:%M:%S")
        next_probe_str = next_probe_dt.strftime("%Y-%m-%d %H:%M:%S")

        logger.info(
            "Video %s: Entering WAITING_FOR_TRANSCRIPT (Deadline: %s, Wait window: %ds, Reprobe: %ds)",
            video_id, deadline_str, self.caption_wait_seconds, self.caption_reprobe_seconds
        )

        # Stage 1 Fast Funnel: If local video exists and Gemini API key is configured,
        # transcribe immediately with gemini-3.5-flash-lite in ~3s without waiting for YouTube captions!
        local_path_str = item.get("local_video_path")
        if local_path_str and Path(local_path_str).exists():
            from dispatch.config import GEMINI_API_KEY
            api_key = os.getenv("GEMINI_API_KEY") or GEMINI_API_KEY
            if api_key:
                try:
                    logger.info("Stage 1 Fast Ingestion: Transcribing %s with gemini-3.5-flash-lite...", video_id)
                    from dispatch.transcription.gemini_engine import transcribe_stage1_flash_lite
                    stage1_segments = transcribe_stage1_flash_lite(Path(local_path_str))
                    if stage1_segments:
                        logger.info("Stage 1 Gemini transcription succeeded for %s (%d segments). Finalizing immediately.",
                                    video_id, len(stage1_segments))
                        return self._finalize_transcription(
                            video_id=video_id,
                            local_video_path=local_path_str,
                            duration=item.get("duration", 0.0),
                            source="gemini_stage1",
                            segments=stage1_segments
                        )
                except Exception as e:
                    logger.warning("Gemini Stage 1 transcription failed for %s: %s. Falling back to caption wait / Whisper.", video_id, e)

        db.update_youtube_video(
            video_id=video_id,
            status="WAITING_FOR_TRANSCRIPT",
            transcript_wait_started_at=now_str,
            transcript_wait_deadline=deadline_str,
            next_caption_probe_at=next_probe_str
        )

        # Check YouTube captions immediately once
        return self._try_fetch_captions_or_wait(
            video_id=video_id,
            local_video_path=item.get("local_video_path"),
            duration=item.get("duration", 0.0),
            deadline_dt=deadline_dt,
            is_probe_due=True
        )

    def process_waiting_item(self, item: Dict[str, Any]) -> Optional[Dict[str, Any]]:
        """Check captions for WAITING_FOR_TRANSCRIPT item, or fall back to Whisper if deadline passed."""
        video_id = item["video_id"]
        deadline_str = item.get("transcript_wait_deadline")
        deadline_dt = _parse_iso_or_sqlite_timestamp(deadline_str)

        now_dt = datetime.now(timezone.utc)
        if deadline_dt is None:
            # If no deadline was set, calculate from started_at or set to now + wait window
            deadline_dt = now_dt + timedelta(seconds=self.caption_wait_seconds)
            db.update_youtube_video(
                video_id=video_id,
                status="WAITING_FOR_TRANSCRIPT",
                transcript_wait_deadline=deadline_dt.strftime("%Y-%m-%d %H:%M:%S")
            )

        # Check if caption re-probe is due based on persisted next_caption_probe_at
        next_probe_str = item.get("next_caption_probe_at")
        next_probe_dt = _parse_iso_or_sqlite_timestamp(next_probe_str)
        is_probe_due = (next_probe_dt is None or now_dt >= next_probe_dt)

        return self._try_fetch_captions_or_wait(
            video_id=video_id,
            local_video_path=item.get("local_video_path"),
            duration=item.get("duration", 0.0),
            deadline_dt=deadline_dt,
            is_probe_due=is_probe_due
        )

    def _try_fetch_captions_or_wait(
        self,
        video_id: str,
        local_video_path: Optional[str],
        duration: float,
        deadline_dt: datetime,
        is_probe_due: bool = True
    ) -> Optional[Dict[str, Any]]:
        """Attempt to fetch YouTube captions if probe is due. If unavailable and past deadline, run Whisper."""
        now_dt = datetime.now(timezone.utc)
        remaining = (deadline_dt - now_dt).total_seconds()

        # 1. Attempt to fetch YouTube captions only if probe is due
        if is_probe_due:
            logger.info("Caption probe is due for %s. Probing YouTube captions...", video_id)
            segments = None
            try:
                segments = self.catcher.fetch_youtube_captions(video_id)
            except Exception as e:
                logger.warning("Error probing YouTube captions for %s: %s", video_id, e)
                segments = None

            # Schedule the next caption probe timestamp durably
            next_probe_dt = now_dt + timedelta(seconds=self.caption_reprobe_seconds)
            db.update_youtube_video(
                video_id=video_id,
                status="WAITING_FOR_TRANSCRIPT",
                next_caption_probe_at=next_probe_dt.strftime("%Y-%m-%d %H:%M:%S")
            )

            if segments is not None:
                logger.info("Found YouTube captions for %s (%d segments). Transitioning to TRANSCRIBED.",
                            video_id, len(segments))
                return self._finalize_transcription(
                    video_id=video_id,
                    local_video_path=local_video_path,
                    duration=duration,
                    source="youtube",
                    segments=segments
                )
        else:
            logger.debug("Caption probe is NOT yet due for %s (rate-limiting YouTube checks).", video_id)

        # 2. Captions unavailable. Check deadline.
        if remaining > 0:
            logger.info(
                "YouTube captions not yet available for %s. Waiting (%.1f minutes / %.0fs remaining).",
                video_id, remaining / 60.0, remaining
            )
            return None

        # 3. Deadline passed → fall back to local Whisper transcription
        logger.info(
            "Caption wait deadline passed for %s. Falling back to local Whisper transcription.",
            video_id
        )
        return self._run_whisper_fallback(
            video_id=video_id,
            local_video_path=local_video_path,
            duration=duration
        )

    def _run_whisper_fallback(
        self,
        video_id: str,
        local_video_path: Optional[str],
        duration: float
    ) -> Dict[str, Any]:
        """Execute local faster-whisper transcription on the downloaded video."""
        if not local_video_path:
            err = f"YouTube captions unavailable and cannot run Whisper fallback for {video_id}: local_video_path is missing"
            logger.warning(err)
            db.update_youtube_video(video_id=video_id, status="CAPTIONS_UNAVAILABLE", last_error=err)
            return {
                "video_id": video_id,
                "status": "CAPTIONS_UNAVAILABLE",
                "source": "none",
                "segments": []
            }

        video_path = Path(local_video_path)
        if not video_path.exists() or video_path.stat().st_size == 0:
            err = f"YouTube captions unavailable and cannot run Whisper fallback for {video_id}: local file not found at {video_path}"
            logger.warning(err)
            db.update_youtube_video(video_id=video_id, status="CAPTIONS_UNAVAILABLE", last_error=err)
            return {
                "video_id": video_id,
                "status": "CAPTIONS_UNAVAILABLE",
                "source": "none",
                "segments": []
            }

        try:
            logger.info("Running local faster-whisper on %s (%.2f MB)...",
                        video_path.name, video_path.stat().st_size / (1024 * 1024))
            whisper_result = transcribe_video(video_path=video_path)
            segments = whisper_result.get("segments", [])
            logger.info("Whisper finished for %s with %d segments.", video_id, len(segments))

            return self._finalize_transcription(
                video_id=video_id,
                local_video_path=local_video_path,
                duration=duration,
                source="whisper",
                segments=segments
            )
        except Exception as e:
            err = f"Whisper transcription failed for {video_id}: {e}"
            logger.error(err)
            db.update_youtube_video(video_id=video_id, status="FAILED", last_error=err)
            raise RuntimeError(err) from e

    def _resolve_source_media_metadata(
        self,
        local_video_path: Optional[str],
        fallback_duration: float
    ) -> Tuple[float, int, int, str]:
        """Extract REAL source media dimensions and aspect ratio. Never invent 1080x1920."""
        from dispatch.ingestion.validator import probe_video

        width = 0
        height = 0
        aspect_ratio = "unknown"
        duration = fallback_duration

        if local_video_path:
            vpath = Path(local_video_path)
            if vpath.exists() and vpath.stat().st_size > 0:
                is_valid, meta, _ = probe_video(vpath)
                if is_valid and meta:
                    duration = meta.get("duration", fallback_duration)
                    width = meta.get("width", 0)
                    height = meta.get("height", 0)
                    aspect_ratio = meta.get("aspect_ratio", "unknown")

        return duration, width, height, aspect_ratio

    def _finalize_transcription(
        self,
        video_id: str,
        local_video_path: Optional[str],
        duration: float,
        source: str,
        segments: List[Dict[str, Any]]
    ) -> Dict[str, Any]:
        """P0 & P1: Resolve authoritative chunk, persist transcript, then update status = TRANSCRIBED.
        
        Strict order:
          1. Resolve / create authoritative chunk preserving real media metadata.
          2. Save authoritative transcript idempotently.
          3. Confirm transcript persistence succeeded.
          4. Only then update youtube_inbox to status = 'TRANSCRIBED'.
        """
        full_text = " ".join(seg.get("text", "").strip() for seg in segments if seg.get("text")).strip()

        # Resolve session_id from youtube_inbox if present
        yt_record = db.get_youtube_video(video_id)
        session_id = yt_record.get("session_id") if yt_record else None

        # 1. Resolve or create chunk record preserving REAL source media metadata
        file_hash = f"yt_{video_id}"
        chunk = db.get_chunk_by_file_hash(file_hash)
        if not chunk and local_video_path:
            vpath = Path(local_video_path)
            real_duration, real_width, real_height, real_aspect = self._resolve_source_media_metadata(
                local_video_path, duration
            )
            chunk_id = db.register_chunk(
                session_id=session_id,
                filename=vpath.name,
                filepath=str(vpath),
                file_hash=file_hash
            )
            db.update_chunk_metadata(
                chunk_id=chunk_id,
                duration=real_duration,
                width=real_width,
                height=real_height,
                aspect_ratio=real_aspect,
                status="verified"
            )
        elif chunk:
            chunk_id = chunk["id"]
            if session_id and not chunk.get("session_id"):
                with db.get_db_connection() as conn:
                    conn.cursor().execute("UPDATE chunks SET session_id = ? WHERE id = ?", (session_id, chunk_id))
                    conn.commit()
        else:
            chunk_id = f"chk_yt_{video_id}"

        # 2. Persist transcript into SQLite first
        try:
            transcript_id = db.save_transcript(
                chunk_id=chunk_id,
                session_id=session_id,
                full_text=full_text,
                segments=segments
            )
        except Exception as e:
            err = f"Failed to persist transcript for {video_id} (chunk {chunk_id}): {e}"
            logger.error(err)
            db.update_youtube_video(video_id=video_id, status="FAILED", last_error=err)
            raise RuntimeError(err) from e

        # 3. Confirm transcript persistence succeeded
        if not transcript_id:
            err = f"db.save_transcript returned empty ID for {video_id} (chunk {chunk_id})"
            logger.error(err)
            db.update_youtube_video(video_id=video_id, status="FAILED", last_error=err)
            raise RuntimeError(err)

        # 4. Only after successful persistence, update youtube_inbox to TRANSCRIBED
        db.update_youtube_video(
            video_id=video_id,
            status="TRANSCRIBED",
            transcript_source=source,
            segments_json=json.dumps(segments, ensure_ascii=False)
        )

        logger.info(
            "Pillar 2 Transcription SUCCESS for %s -> TRANSCRIBED (Source: %s, Segments: %d, Words: %d, TranscriptID: %s)",
            video_id, source, len(segments), len(full_text.split()), transcript_id
        )

        return {
            "video_id": video_id,
            "status": "TRANSCRIBED",
            "source": source,
            "segments_count": len(segments),
            "words_count": len(full_text.split()),
            "transcript_id": transcript_id,
            "chunk_id": chunk_id
        }

"""Autonomous UDP Zero-Configuration Beacon & Discovery Service for Dispatch.
Enables instant pairing between mobile phone and PC without manual IP configuration.

Functions:
- Responds to 'DISPATCH_DISCOVER' UDP requests with PC LAN URL and pairing credentials.
- Periodically broadcasts discovery beacons on local subnet (port 8765).
"""
import json
import time
import socket
import logging
import threading
from typing import Optional

from dispatch.config import WEB_PORT
from dispatch.sync.receiver import get_auth_token

logger = logging.getLogger("dispatch.discovery")

DISCOVERY_PORT = 8765
DISCOVER_MAGIC = b"DISPATCH_DISCOVER"
BEACON_MAGIC = "DISPATCH_ANNOUNCE"
SERVICE_VERSION = "1.0.0"


def get_local_lan_ip() -> str:
    """Detect the active LAN IPv4 address of this machine."""
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        # Doesn't actually send data, just connects to determine outbound interface
        s.connect(("8.8.8.8", 80))
        ip = s.getsockname()[0]
        s.close()
        return ip
    except Exception:
        return "127.0.0.1"


class DiscoveryBeaconServer:
    """UDP listener and broadcaster for zero-config phone-to-PC pairing."""

    def __init__(self, port: int = DISCOVERY_PORT):
        self.port = port
        self.running = False
        self._thread: Optional[threading.Thread] = None
        self._beacon_thread: Optional[threading.Thread] = None

    def start(self):
        """Start UDP discovery listener and background beacon broadcaster."""
        if self.running:
            return
        self.running = True

        self._thread = threading.Thread(target=self._listen_loop, daemon=True, name="UDP-Discovery-Listener")
        self._thread.start()

        self._beacon_thread = threading.Thread(target=self._beacon_loop, daemon=True, name="UDP-Beacon-Broadcaster")
        self._beacon_thread.start()

        logger.info("Autonomous UDP Discovery Service active on port %d", self.port)

    def stop(self):
        """Stop UDP discovery."""
        self.running = False

    def _get_payload(self) -> dict:
        """Construct UDP discovery announcement payload.
        SECURITY REQUIREMENT: Must NEVER broadcast plaintext auth_token, PIN, or credentials.
        Advertises server presence so Android discovery can identify the server without intercepting secret tokens.
        """
        lan_ip = get_local_lan_ip()
        return {
            "magic": BEACON_MAGIC,
            "service": "dispatch",
            "version": SERVICE_VERSION,
            "lan_url": f"http://{lan_ip}:{WEB_PORT}",
            "ip": lan_ip,
            "port": WEB_PORT,
            "hostname": socket.gethostname(),
            "pairing_required": True,
        }

    def _listen_loop(self):
        """Listen for incoming DISPATCH_DISCOVER UDP datagrams and reply directly."""
        sock = None
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
            sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
            sock.bind(("0.0.0.0", self.port))
            sock.settimeout(1.0)

            while self.running:
                try:
                    data, addr = sock.recvfrom(1024)
                    if DISCOVER_MAGIC in data:
                        payload = self._get_payload()
                        resp_bytes = json.dumps(payload).encode("utf-8")
                        sock.sendto(resp_bytes, addr)
                        logger.info("Answered UDP discovery query from %s:%d -> %s", addr[0], addr[1], payload["lan_url"])
                except socket.timeout:
                    continue
                except Exception as e:
                    if self.running:
                        logger.debug("UDP listen error: %s", e)
        except Exception as e:
            logger.error("Failed to bind UDP discovery listener on port %d: %s", self.port, e)
        finally:
            if sock:
                try:
                    sock.close()
                except Exception:
                    pass

    def _beacon_loop(self):
        """Periodically broadcast UDP announcement beacon across LAN."""
        sock = None
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
            sock.setsockopt(socket.SOL_SOCKET, socket.SO_BROADCAST, 1)

            while self.running:
                try:
                    payload = self._get_payload()
                    msg = json.dumps(payload).encode("utf-8")

                    # Broadcast to global broadcast
                    sock.sendto(msg, ("255.255.255.255", self.port))

                    # Also broadcast to subnet broadcast (e.g. 192.168.0.255)
                    lan_ip = payload.get("ip", "")
                    if "." in lan_ip:
                        parts = lan_ip.split(".")
                        if len(parts) == 4:
                            subnet_bcast = f"{parts[0]}.{parts[1]}.{parts[2]}.255"
                            sock.sendto(msg, (subnet_bcast, self.port))
                except Exception as e:
                    logger.debug("UDP beacon send error: %s", e)

                # Sleep 2.5 seconds between announcement beacons
                for _ in range(25):
                    if not self.running:
                        break
                    time.sleep(0.1)
        except Exception as e:
            logger.error("UDP beacon broadcaster error: %s", e)
        finally:
            if sock:
                try:
                    sock.close()
                except Exception:
                    pass

"""Transport Abstraction & Remote Connectivity Manager.
Enables seamless switching between local LAN Wi-Fi, Tailscale private network,
and custom remote endpoints with automatic latency probing, failover, and policy enforcement.
"""
import time
import logging
from typing import List, Dict, Optional, Tuple, Any
from dataclasses import dataclass, field
import requests

logger = logging.getLogger("dispatch.transport")


@dataclass
class EndpointCandidate:
    name: str              # "lan", "tailscale", "custom"
    url: str               # "http://192.168.1.15:8765" or "http://laptop.tailnet.ts.net:8765"
    priority: int = 1      # 1 (LAN, fastest direct path), 2 (Tailscale), 3 (Custom)
    is_reachable: bool = False
    latency_ms: float = 9999.0
    last_probed: float = 0.0


@dataclass
class NetworkPolicy:
    wifi_only: bool = True
    allow_cellular: bool = False
    pause_on_roaming: bool = True
    max_cellular_mb_per_session: float = 200.0


class TransportManager:
    """Manages multi-path connectivity between mobile phone and home laptop."""

    def __init__(
        self,
        auth_token: str,
        lan_url: Optional[str] = None,
        tailscale_url: Optional[str] = None,
        custom_url: Optional[str] = None,
        policy: Optional[NetworkPolicy] = None
    ):
        self.auth_token = auth_token
        self.policy = policy or NetworkPolicy()
        self.candidates: List[EndpointCandidate] = []

        if lan_url:
            self.candidates.append(EndpointCandidate(name="lan", url=lan_url.rstrip("/"), priority=1))
        if tailscale_url:
            self.candidates.append(EndpointCandidate(name="tailscale", url=tailscale_url.rstrip("/"), priority=2))
        if custom_url:
            self.candidates.append(EndpointCandidate(name="custom", url=custom_url.rstrip("/"), priority=3))

    def add_candidate(self, name: str, url: str, priority: int = 2):
        self.candidates.append(EndpointCandidate(name=name, url=url.rstrip("/"), priority=priority))

    def probe_single_endpoint(self, candidate: EndpointCandidate, timeout_seconds: float = 2.0) -> bool:
        """Pings a single endpoint and measures network roundtrip latency."""
        ping_url = f"{candidate.url}/api/sync/ping"
        start = time.perf_counter()
        try:
            resp = requests.get(ping_url, timeout=timeout_seconds)
            elapsed_ms = (time.perf_counter() - start) * 1000.0
            if resp.status_code == 200 and resp.json().get("status") == "online":
                candidate.is_reachable = True
                candidate.latency_ms = round(elapsed_ms, 1)
                candidate.last_probed = time.time()
                logger.debug("Candidate %s (%s) ONLINE: %.1fms", candidate.name, candidate.url, elapsed_ms)
                return True
        except Exception as e:
            logger.debug("Candidate %s (%s) UNREACHABLE: %s", candidate.name, candidate.url, e)

        candidate.is_reachable = False
        candidate.latency_ms = 9999.0
        candidate.last_probed = time.time()
        return False

    def probe_all_endpoints(self, timeout_seconds: float = 2.0) -> List[EndpointCandidate]:
        """Probes all configured endpoint candidates and updates reachability."""
        for c in self.candidates:
            self.probe_single_endpoint(c, timeout_seconds=timeout_seconds)
        return self.candidates

    def select_best_endpoint(self, auto_probe: bool = True) -> Optional[EndpointCandidate]:
        """Selects the optimal reachable endpoint, prioritizing LAN over Tailscale, then lowest latency."""
        if auto_probe:
            self.probe_all_endpoints()

        reachable = [c for c in self.candidates if c.is_reachable]
        if not reachable:
            return None

        # Sort by priority first (1=LAN, 2=Tailscale), then by latency
        reachable.sort(key=lambda c: (c.priority, c.latency_ms))
        best = reachable[0]
        logger.info("Selected optimal route: %s (%s, %.1fms)", best.name, best.url, best.latency_ms)
        return best

    def evaluate_network_policy(
        self,
        network_type: str,  # "wifi", "cellular", "ethernet", "none"
        is_roaming: bool = False
    ) -> Tuple[bool, str]:
        """Evaluates whether current phone network conditions permit syncing under policy.
        
        Returns:
            (is_allowed: bool, reason: str)
        """
        net = network_type.lower()

        if net in ("none", "disconnected", "offline"):
            return False, "No active network interface on device"

        if is_roaming and self.policy.pause_on_roaming:
            return False, "Data sync paused while device is roaming"

        if net in ("wifi", "ethernet"):
            return True, f"High-speed unmetered connection ({net}) allowed"

        if net in ("cellular", "mobile", "4g", "5g"):
            if not self.policy.allow_cellular:
                return False, "Mobile cellular sync is disabled by policy (Wi-Fi only)"
            return True, "Cellular connection permitted under policy"

        return False, f"Unknown network interface type: {network_type}"

    def get_auth_headers(self) -> Dict[str, str]:
        """Constructs secure device authentication headers."""
        return {
            "X-Dispatch-Device-Token": self.auth_token,
            "X-Dispatch-Client": "DispatchMobile/1.0"
        }

"""Adaptive video reframing filter generator.
Converts between portrait (9:16) and landscape (16:9) formats based on
original session input orientation.
"""
from pathlib import Path
from typing import Optional, Tuple
from dispatch.config import TARGET_WIDTH, TARGET_HEIGHT, FONTS_DIR

# Output resolution standards
PORTRAIT_WIDTH = TARGET_WIDTH     # 1080
PORTRAIT_HEIGHT = TARGET_HEIGHT   # 1920
WIDE_WIDTH = 1920
WIDE_HEIGHT = 1080


def escape_ffmpeg_filter_path(filepath: Path) -> str:
    """Escape Windows file paths for FFmpeg filter arguments (subtitles filter)."""
    # Use forward slashes and escape the colon in Windows drive letter (e.g. C\: -> C\:/)
    path_str = str(filepath.resolve()).replace("\\", "/")
    if len(path_str) > 1 and path_str[1] == ":":
        path_str = path_str[0] + "\\:" + path_str[2:]
    return path_str


def build_filter_complex(
    aspect_ratio: str,
    layout_mode: str = "fit_blur",
    ass_subtitle_path: Optional[Path] = None,
    input_v_label: str = "[0:v]"
) -> Tuple[str, str]:
    """Build FFmpeg filter_complex string for orientation-aware conversion and subtitle burning.
    
    Args:
        aspect_ratio: Source video aspect ratio: '9:16' / 'portrait' or '16:9' / 'landscape'.
        layout_mode:
            For 9:16 Portrait source:
                - 'fit_black': 16:9 widescreen output with black left/right pillarbox bars.
                - 'fit_blur': 16:9 widescreen output with blurred left/right pillarbox bars.
                - 'native_916' (or 'native_portrait'): 9:16 vertical output, 100% full original video (no bars, no crop).
            For 16:9 Landscape source:
                - 'crop_916' (or 'crop_follow'): 9:16 vertical output, centered vertical crop.
                - 'fit_blur': 9:16 vertical output with blurred top/bottom letterbox bars.
                - 'fit_black': 9:16 vertical output with black top/bottom letterbox bars.
                - 'native_169' (or 'landscape'): 16:9 widescreen output, 100% full original video (no bars, no crop).
        ass_subtitle_path: Optional path to .ass subtitle file.
        input_v_label: Input video label (default '[0:v]').
        
    Returns:
        (filter_complex_string, output_video_label)
    """
    filters = []
    mode = (layout_mode or "").strip().lower()
    is_portrait = str(aspect_ratio).strip().lower() in ("9:16", "portrait", "vertical")
    in_v = input_v_label

    if is_portrait:
        # ============================================================
        # CASE A — ORIGINAL INPUT IS VERTICAL / 9:16
        # ============================================================
        if mode == "fit_black":
            # OPTION 1 — LONG FORM / 16:9 + BLACK
            # Output: 16:9. Source fully visible in center with black pillarbox bars.
            filters.append(
                f"{in_v}scale={WIDE_WIDTH}:{WIDE_HEIGHT}:force_original_aspect_ratio=decrease,"
                f"pad={WIDE_WIDTH}:{WIDE_HEIGHT}:(ow-iw)/2:(oh-ih)/2:black[base]"
            )
        elif mode == "fit_blur":
            # OPTION 2 — LONG FORM / 16:9 + BLUR
            # Output: 16:9. Source fully visible in center with blurred & darkened pillarbox bars.
            # Uses boxblur=30:3 and colorchannelmixer=rr=0.75:gg=0.75:bb=0.75 for exact parity with CSS blur-2xl brightness-75
            filters.append(
                f"{in_v}split=2[bg_in][fg_in];"
                f"[bg_in]scale={WIDE_WIDTH}:{WIDE_HEIGHT}:force_original_aspect_ratio=increase,"
                f"crop={WIDE_WIDTH}:{WIDE_HEIGHT},boxblur=30:3,colorchannelmixer=rr=0.75:gg=0.75:bb=0.75[bg];"
                f"[fg_in]scale=-2:{WIDE_HEIGHT}[fg];"
                f"[bg][fg]overlay=(W-w)/2:(H-h)/2[base]"
            )
        else:
            # OPTION 3 — NATIVE 9:16 ('native_916', 'native_portrait', default)
            # Output: 9:16. 100% full original vertical video. No crop, no blur, no bars.
            filters.append(
                f"{in_v}scale={PORTRAIT_WIDTH}:{PORTRAIT_HEIGHT}:force_original_aspect_ratio=decrease,"
                f"pad={PORTRAIT_WIDTH}:{PORTRAIT_HEIGHT}:(ow-iw)/2:(oh-ih)/2:black[base]"
            )
    else:
        # ============================================================
        # CASE B — ORIGINAL INPUT IS HORIZONTAL / 16:9
        # ============================================================
        if mode in ("crop_916", "crop_follow"):
            # OPTION 1 — CROP 9:16
            # Output: 9:16. 16:9 source cropped vertically to fill 9:16 frame.
            filters.append(
                f"{in_v}crop=w=trunc(ih*9/16/2)*2:h=ih:x=trunc((iw-ow)/2):y=0,scale={PORTRAIT_WIDTH}:{PORTRAIT_HEIGHT}[base]"
            )
        elif mode == "fit_blur":
            # OPTION 2 — FIT + BLUR
            # Output: 9:16. 16:9 source in center with blurred & darkened top/bottom letterbox.
            # Uses boxblur=30:3 and colorchannelmixer=rr=0.75:gg=0.75:bb=0.75 for exact parity with CSS blur-2xl brightness-75
            filters.append(
                f"{in_v}split=2[bg_in][fg_in];"
                f"[bg_in]scale={PORTRAIT_WIDTH}:{PORTRAIT_HEIGHT}:force_original_aspect_ratio=increase,"
                f"crop={PORTRAIT_WIDTH}:{PORTRAIT_HEIGHT},boxblur=30:3,colorchannelmixer=rr=0.75:gg=0.75:bb=0.75[bg];"
                f"[fg_in]scale={PORTRAIT_WIDTH}:-2[fg];"
                f"[bg][fg]overlay=(W-w)/2:(H-h)/2[base]"
            )
        elif mode == "fit_black":
            # OPTION 3 — FIT + BLACK
            # Output: 9:16. 16:9 source in center with black top/bottom letterbox bars.
            filters.append(
                f"{in_v}scale={PORTRAIT_WIDTH}:{PORTRAIT_HEIGHT}:force_original_aspect_ratio=decrease,"
                f"pad={PORTRAIT_WIDTH}:{PORTRAIT_HEIGHT}:(ow-iw)/2:(oh-ih)/2:black[base]"
            )
        else:
            # OPTION 4 — NATIVE 16:9 ('native_169', 'landscape', default)
            # Output: 16:9. 100% full original horizontal video. No crop, no blur, no bars.
            filters.append(
                f"{in_v}scale={WIDE_WIDTH}:{WIDE_HEIGHT}:force_original_aspect_ratio=decrease,"
                f"pad={WIDE_WIDTH}:{WIDE_HEIGHT}:(ow-iw)/2:(oh-ih)/2:black[base]"
            )

    current_label = "[base]"

    if ass_subtitle_path and ass_subtitle_path.exists():
        escaped_sub = escape_ffmpeg_filter_path(ass_subtitle_path)
        sub_filter = f"subtitles='{escaped_sub}'"
        if FONTS_DIR and FONTS_DIR.exists():
            escaped_fonts = escape_ffmpeg_filter_path(FONTS_DIR)
            sub_filter += f":fontsdir='{escaped_fonts}'"
        filters.append(f"{current_label}{sub_filter}[outv]")
        final_label = "[outv]"
    else:
        final_label = current_label

    filter_complex_str = ";".join(filters)
    return filter_complex_str, final_label

"""FFmpeg Video Renderer for Dispatch.
Cuts exact time segments, applies dead-air silence removal with frame-accurate
subtitle synchronization, applies adaptive framing, burns animated subtitles,
normalizes audio, generates preview thumbnails, and enforces strict duration limits.
"""
import subprocess
import logging
from pathlib import Path
from typing import List, Dict, Any, Optional
from dispatch.config import CLIPS_DIR, PROCESSING_DIR, MAX_CLIP_DURATION, MIN_CLIP_DURATION
from dispatch.video_engine.subtitle_generator import generate_ass_subtitles
from dispatch.video_engine.reframer import build_filter_complex
from dispatch.video_engine.silence import (
    analyze_silence_gaps,
    remap_timestamps_for_silence_cuts,
    build_silence_keep_intervals
)
from dispatch import db

logger = logging.getLogger("dispatch.renderer")


def render_clip(
    source_video: Path,
    clip_id: str,
    start_time: float,
    end_time: float,
    aspect_ratio: Optional[str] = None,
    layout_mode: str = "fit_blur",
    segments: Optional[List[Dict[str, Any]]] = None,
    subtitle_options: Optional[Dict[str, Any]] = None,
    expected_range_revision: Optional[int] = None,
) -> Dict[str, Any]:
    """Render a single candidate short-form video clip.
    
    Invariants:
      1. Minimum final duration: 5.0 seconds.
      2. Maximum final duration: 180.0 seconds (inclusive).
      3. Automatically removes dead-air silences (>= 3.0s) while preserving conversational
         cadence, thinking pauses, dramatic tension, and comedic timing.
      4. Subtitle timestamps remain frame-accurate and strictly synchronized after silence removal.
      5. Final output file is probed; if duration is outside [5.0s, 180.0s], the clip is
         rejected and unlinked before downstream processing or publishing.
    """
    if not source_video.exists():
        raise FileNotFoundError(f"Source video not found: {source_video}")

    clip_at_start = db.get_clip_by_id(clip_id)
    render_range_revision = expected_range_revision
    if render_range_revision is None and clip_at_start:
        render_range_revision = int(clip_at_start.get("range_revision") or 0)

    from dispatch.ingestion.validator import inspect_video_orientation
    info = inspect_video_orientation(source_video)
    if not aspect_ratio or aspect_ratio == "auto":
        if info.get("valid"):
            aspect_ratio = "9:16" if info.get("orientation") == "portrait" else "16:9"
        else:
            aspect_ratio = "16:9"

    has_audio = info.get("has_audio", True)

    # Candidate/editor renders preserve source timing so the browser caption overlay and
    # approved render share the same clip-relative timeline.
    sub_opts = subtitle_options or {}
    if sub_opts.get("preserve_timing"):
        eff_duration, cuts = max(0.0, end_time - start_time), []
    else:
        eff_duration, cuts = analyze_silence_gaps(start_time, end_time, segments)
    has_silence_cuts = bool(cuts)

    output_video_path = CLIPS_DIR / f"{clip_id}.mp4"
    output_thumb_path = CLIPS_DIR / f"{clip_id}.jpg"
    ass_path = PROCESSING_DIR / f"{clip_id}.ass"
    temp_video_path = CLIPS_DIR / f"{clip_id}.tmp.mp4"
    temp_thumb_path = CLIPS_DIR / f"{clip_id}.tmp.jpg"

    # 2. Subtitle generation with synchronized timeline
    if sub_opts.get("enabled", True):
        captured_track = sub_opts.get("caption_track")
        existing_cap_json = db.get_clip_caption_data(clip_id) if captured_track is None else None
        if captured_track is not None and not has_silence_cuts:
            try:
                from dispatch.captions.model import CaptionTrack
                from dispatch.captions.exporter import export_to_ass
                track = CaptionTrack.from_dict(captured_track)
                export_to_ass(track, ass_path)
            except Exception as e:
                logger.warning("Failed to export captured CaptionTrack for clip %s: %s", clip_id, e)
                ass_path.unlink(missing_ok=True)
                raise RuntimeError(f"Final caption export failed: {e}") from e
        elif existing_cap_json and not has_silence_cuts:
            try:
                import json
                from dispatch.captions.model import CaptionTrack
                from dispatch.captions.exporter import export_to_ass
                track = CaptionTrack.from_dict(json.loads(existing_cap_json))
                export_to_ass(track, ass_path)
            except Exception as e:
                logger.warning("Failed to export existing CaptionTrack for clip %s: %s", clip_id, e)
                ass_path = None
        elif segments:
            # Remap subtitle timestamps if silence cuts occurred
            active_segments = remap_timestamps_for_silence_cuts(segments, start_time, end_time, cuts) if has_silence_cuts else segments
            try:
                import json
                from dispatch.captions import group_words_into_captions, export_to_ass
                from dispatch.captions.presets import get_default_preset_id
                track = group_words_into_captions(
                    raw_segments=active_segments,
                    clip_start=start_time,
                    clip_end=start_time + eff_duration if has_silence_cuts else end_time,
                    clip_id=clip_id,
                    preset_id=sub_opts.get("style_preset") or get_default_preset_id()
                )
                export_to_ass(track, ass_path)
                saved = db.save_clip_caption_data(
                    clip_id, json.dumps(track.to_dict()),
                    expected_range_revision=render_range_revision,
                    source_range_start=start_time,
                    source_range_end=end_time,
                )
                if render_range_revision is not None and not saved:
                    raise RuntimeError("Clip range changed while captions were being rendered")
            except Exception as e:
                logger.warning("Failed to generate subtitles via CaptionTrack: %s", e)
                ass_path = None
        else:
            ass_path = None
    else:
        ass_path = None

    # 3. Construct FFmpeg command
    keep_intervals = build_silence_keep_intervals(start_time, end_time, cuts) if has_silence_cuts else []

    if has_silence_cuts and len(keep_intervals) > 1:
        # Complex filter: trim and concatenate speech blocks, then apply framing and subtitles
        trim_filters = []
        concat_in = []
        for idx, (k_s, k_e) in enumerate(keep_intervals):
            trim_filters.append(f"[0:v]trim=start={k_s:.3f}:end={k_e:.3f},setpts=PTS-STARTPTS[v_trim_{idx}]")
            if has_audio:
                trim_filters.append(f"[0:a]atrim=start={k_s:.3f}:end={k_e:.3f},asetpts=PTS-STARTPTS[a_trim_{idx}]")
                concat_in.append(f"[v_trim_{idx}][a_trim_{idx}]")
            else:
                concat_in.append(f"[v_trim_{idx}]")

        a_cnt = 1 if has_audio else 0
        concat_filter = f"{''.join(concat_in)}concat=n={len(keep_intervals)}:v=1:a={a_cnt}[concat_v]" + ("[concat_a]" if has_audio else "")
        trim_filters.append(concat_filter)

        framing_filter, final_v_label = build_filter_complex(
            aspect_ratio=aspect_ratio,
            layout_mode=layout_mode,
            ass_subtitle_path=ass_path if (ass_path and ass_path.exists()) else None,
            input_v_label="[concat_v]"
        )

        full_filter_complex = ";".join(trim_filters) + ";" + framing_filter

        cmd = [
            "ffmpeg", "-y",
            "-i", str(source_video),
            "-filter_complex", full_filter_complex,
            "-map", final_v_label,
        ]
        if has_audio:
            cmd.extend([
                "-map", "[concat_a]",
                "-af", "loudnorm=I=-14:LRA=11:TP=-1.5",
                "-c:a", "aac",
                "-b:a", "192k",
            ])
        cmd.extend([
            "-c:v", "libx264",
            "-preset", "fast",
            "-crf", "20",
            "-pix_fmt", "yuv420p",
            "-threads", "2",
            "-movflags", "+faststart",
            str(temp_video_path)
        ])
    else:
        # Standard contiguous segment render
        filter_complex_str, final_v_label = build_filter_complex(
            aspect_ratio=aspect_ratio,
            layout_mode=layout_mode,
            ass_subtitle_path=ass_path if (ass_path and ass_path.exists()) else None,
            input_v_label="[0:v]"
        )

        cmd = [
            "ffmpeg", "-y",
            "-ss", str(round(start_time, 3)),
            "-to", str(round(end_time, 3)),
            "-i", str(source_video),
            "-filter_complex", filter_complex_str,
            "-map", final_v_label,
            "-map", "0:a?",
            "-c:v", "libx264",
            "-preset", "fast",
            "-crf", "20",
            "-pix_fmt", "yuv420p",
            "-threads", "2",
            "-af", "loudnorm=I=-14:LRA=11:TP=-1.5",
            "-c:a", "aac",
            "-b:a", "192k",
            "-movflags", "+faststart",
            str(temp_video_path)
        ]

    logger.info("Executing FFmpeg atomic render for clip %s (%.1fs - %.1fs, effective: %.1fs)",
                clip_id, start_time, end_time, eff_duration)
    result = subprocess.run(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.PIPE, text=True, timeout=300)

    if result.returncode != 0:
        error_msg = result.stderr.strip()
        logger.error("FFmpeg render failed for %s: %s", clip_id, error_msg)
        if temp_video_path.exists():
            temp_video_path.unlink()
        if ass_path and ass_path.exists():
            ass_path.unlink()
        raise RuntimeError(f"FFmpeg render failed: {error_msg}")

    # 4. Mandatory Duration Verification on Final Output Video
    from dispatch.ingestion.validator import probe_video
    is_valid, meta, validation_error = probe_video(temp_video_path)
    if not is_valid or not meta or not meta.get("duration"):
        temp_video_path.unlink(missing_ok=True)
        if ass_path and ass_path.exists():
            ass_path.unlink()
        raise RuntimeError(f"Rendered clip failed media validation: {validation_error or 'invalid or unreadable output'}")
    actual_duration = float(meta["duration"])

    if actual_duration > MAX_CLIP_DURATION:
        logger.error("Clip %s rendered duration (%.2fs) exceeds MAX_CLIP_DURATION (%.1fs). Rejecting.",
                     clip_id, actual_duration, MAX_CLIP_DURATION)
        if temp_video_path.exists():
            temp_video_path.unlink()
        if ass_path and ass_path.exists():
            ass_path.unlink()
        raise ValueError(f"Clip {clip_id} exceeds hard duration limit ({actual_duration:.2f}s > {MAX_CLIP_DURATION}s)")

    if actual_duration < MIN_CLIP_DURATION:
        logger.error("Clip %s rendered duration (%.2fs) below MIN_CLIP_DURATION (%.1fs). Rejecting.",
                     clip_id, actual_duration, MIN_CLIP_DURATION)
        if temp_video_path.exists():
            temp_video_path.unlink()
        if ass_path and ass_path.exists():
            ass_path.unlink()
        raise ValueError(f"Clip {clip_id} below minimum duration limit ({actual_duration:.2f}s < {MIN_CLIP_DURATION}s)")

    # Atomic rename: guarantees system never exposes an unplayable partial video
    temp_video_path.replace(output_video_path)

    # 5. Generate preview thumbnail atomically
    thumb_time = max(0.0, min(1.0, actual_duration / 2.0))
    thumb_cmd = [
        "ffmpeg", "-y",
        "-ss", str(round(thumb_time, 3)),
        "-i", str(output_video_path),
        "-vframes", "1",
        "-q:v", "2",
        str(temp_thumb_path)
    ]
    thumb_res = subprocess.run(thumb_cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, timeout=30)
    if thumb_res.returncode == 0 and temp_thumb_path.exists():
        temp_thumb_path.replace(output_thumb_path)
    elif temp_thumb_path.exists():
        temp_thumb_path.unlink()

    # Clean up temporary ASS file
    if ass_path and ass_path.exists():
        try:
            ass_path.unlink()
        except Exception:
            pass

    # 6. Update DB with verified rendered layout
    media_updated = db.update_clip_media(
        clip_id=clip_id,
        video_path=str(output_video_path),
        thumbnail_path=str(output_thumb_path) if output_thumb_path.exists() else None,
        rendered_layout_mode=layout_mode,
        expected_range_revision=render_range_revision,
    )
    if render_range_revision is not None and not media_updated:
        raise RuntimeError("Clip range changed during rendering; obsolete media was not installed")

    logger.info("Successfully rendered clip %s -> %s (duration: %.2fs)", clip_id, output_video_path.name, actual_duration)

    return {
        "clip_id": clip_id,
        "video_path": output_video_path,
        "thumbnail_path": output_thumb_path if output_thumb_path.exists() else None,
        "duration": actual_duration
    }

"""Silence Detection, Removal, and Timeline Synchronization Module.
Provides intelligent silence analysis for candidate clips, allowing longer source intervals
to qualify if genuine dead-air silences are removed, while strictly preserving meaningful pauses,
comedic timing, dramatic tension, and subtitle synchronization.
"""
import copy
import logging
from typing import List, Dict, Any, Tuple, Optional

logger = logging.getLogger("dispatch.video_engine.silence")

# Threshold above which a pause is considered dead-air / prolonged silence (seconds)
DEFAULT_SILENCE_THRESHOLD = 3.0

# Natural pause duration to retain across a trimmed dead-air gap (seconds)
DEFAULT_TARGET_PAUSE = 0.8


def analyze_silence_gaps(
    start_time: float,
    end_time: float,
    segments: Optional[List[Dict[str, Any]]],
    silence_threshold: float = DEFAULT_SILENCE_THRESHOLD,
    target_pause: float = DEFAULT_TARGET_PAUSE
) -> Tuple[float, List[Tuple[float, float, float]]]:
    """Analyze silence gaps within [start_time, end_time] and compute effective duration.

    Invariants:
      1. Conversational pauses (< 2.0s) and thinking/dramatic pauses (< silence_threshold)
         are fully preserved to respect speaker cadence and intentional delivery.
      2. Dead air gaps (>= silence_threshold) are trimmed down to target_pause, never
         cut abruptly to 0.0s, to prevent jarring audio jumps.
      3. Returns (effective_duration, cuts) where each cut is:
         (cut_start, cut_end, removed_duration).
    """
    raw_duration = max(0.0, end_time - start_time)
    if not segments or raw_duration <= 0.0:
        return raw_duration, []

    # Extract all active speech intervals within [start_time, end_time]
    speech_intervals: List[Tuple[float, float]] = []
    for s in segments:
        s_start = float(s.get("start", 0.0))
        s_end = float(s.get("end", 0.0))
        if s_end <= start_time or s_start >= end_time:
            continue

        words = s.get("words", [])
        text_words = s.get("text", "").split()
        if words and len(words) >= max(1, int(len(text_words) * 0.75)):
            for w in words:
                w_start = float(w.get("start", 0.0))
                w_end = float(w.get("end", 0.0))
                if w_end > start_time and w_start < end_time:
                    clamped_s = max(start_time, w_start)
                    clamped_e = min(end_time, w_end)
                    if clamped_e > clamped_s:
                        speech_intervals.append((clamped_s, clamped_e))
        else:
            clamped_s = max(start_time, s_start)
            clamped_e = min(end_time, s_end)
            if clamped_e > clamped_s:
                speech_intervals.append((clamped_s, clamped_e))

    if not speech_intervals:
        return raw_duration, []

    # Sort and merge overlapping or tightly adjacent speech intervals (< 0.2s)
    speech_intervals.sort(key=lambda x: x[0])
    merged_speech: List[Tuple[float, float]] = []
    cur_s, cur_e = speech_intervals[0]
    for nxt_s, nxt_e in speech_intervals[1:]:
        if nxt_s <= cur_e + 0.20:
            cur_e = max(cur_e, nxt_e)
        else:
            merged_speech.append((cur_s, cur_e))
            cur_s, cur_e = nxt_s, nxt_e
    merged_speech.append((cur_s, cur_e))

    cuts: List[Tuple[float, float, float]] = []
    total_removed = 0.0

    # 1. Check leading silence before the first speech
    first_speech_start = merged_speech[0][0]
    leading_gap = first_speech_start - start_time
    if leading_gap >= silence_threshold:
        rem_dur = leading_gap - target_pause
        cut_start = start_time
        cut_end = start_time + rem_dur
        cuts.append((cut_start, cut_end, rem_dur))
        total_removed += rem_dur

    # 2. Check gaps between consecutive speech blocks
    for i in range(len(merged_speech) - 1):
        gap_start = merged_speech[i][1]
        gap_end = merged_speech[i + 1][0]
        gap_duration = gap_end - gap_start
        if gap_duration >= silence_threshold:
            rem_dur = gap_duration - target_pause
            # Leave target_pause / 2 after speech and target_pause / 2 before next speech
            margin = target_pause / 2.0
            cut_start = gap_start + margin
            cut_end = gap_end - margin
            if cut_end > cut_start:
                actual_cut_dur = cut_end - cut_start
                cuts.append((cut_start, cut_end, actual_cut_dur))
                total_removed += actual_cut_dur

    # 3. Check trailing silence after the last speech
    last_speech_end = merged_speech[-1][1]
    trailing_gap = end_time - last_speech_end
    if trailing_gap >= silence_threshold:
        rem_dur = trailing_gap - target_pause
        cut_start = end_time - rem_dur
        cut_end = end_time
        cuts.append((cut_start, cut_end, rem_dur))
        total_removed += rem_dur

    effective_duration = max(0.0, round(raw_duration - total_removed, 2))
    return effective_duration, cuts


def remap_timestamps_for_silence_cuts(
    segments: List[Dict[str, Any]],
    clip_start: float,
    clip_end: float,
    cuts: List[Tuple[float, float, float]]
) -> List[Dict[str, Any]]:
    """Remap transcript segment and word timestamps to match the silence-trimmed timeline.

    Ensures subtitles remain frame-accurate and strictly synchronized after dead-air cuts.
    """
    if not segments:
        return []

    if not cuts:
        return copy.deepcopy(segments)

    def _remap_point(t: float) -> float:
        shift = 0.0
        for cut_start, cut_end, cut_dur in cuts:
            if t >= cut_end:
                shift += cut_dur
            elif t > cut_start:
                # Inside the cut: smoothly compress
                offset_in_cut = t - cut_start
                cut_span = cut_end - cut_start
                ratio = offset_in_cut / cut_span if cut_span > 0 else 0.0
                shift += (cut_dur * ratio)
                break
            else:
                break
        return max(0.0, t - shift)

    remapped_segments = []
    for s in segments:
        s_start = float(s.get("start", 0.0))
        s_end = float(s.get("end", 0.0))
        if s_end <= clip_start or s_start >= clip_end:
            continue

        new_s = copy.deepcopy(s)
        new_s["start"] = _remap_point(s_start)
        new_s["end"] = _remap_point(s_end)

        if "words" in new_s and new_s["words"]:
            for w in new_s["words"]:
                w_start = float(w.get("start", 0.0))
                w_end = float(w.get("end", 0.0))
                w["start"] = _remap_point(w_start)
                w["end"] = _remap_point(w_end)

        remapped_segments.append(new_s)

    return remapped_segments


def build_silence_keep_intervals(
    start_time: float,
    end_time: float,
    cuts: List[Tuple[float, float, float]]
) -> List[Tuple[float, float]]:
    """Convert silence cuts into a list of keep intervals [(k_start, k_end), ...] for FFmpeg."""
    if not cuts:
        return [(start_time, end_time)]

    keep_intervals: List[Tuple[float, float]] = []
    cur_pos = start_time

    for cut_start, cut_end, _ in sorted(cuts, key=lambda c: c[0]):
        if cut_start > cur_pos:
            keep_intervals.append((cur_pos, cut_start))
        cur_pos = max(cur_pos, cut_end)

    if cur_pos < end_time:
        keep_intervals.append((cur_pos, end_time))

    return [k for k in keep_intervals if k[1] > k[0] + 0.05]


def find_natural_subsection_within_limit(
    start_time: float,
    end_time: float,
    segments: List[Dict[str, Any]],
    max_duration: float = 180.0,
    min_duration: float = 5.0
) -> Optional[Tuple[float, float]]:
    """Find a natural, complete thought subsection that fits within [min_duration, max_duration].

    Used when a candidate thought exceeds 180s even after legitimate silence removal.
    Looks for complete sentence boundaries within the candidate to find a standalone section.
    """
    if not segments:
        return None

    # Filter segments within candidate range
    candidate_segs = [
        s for s in segments
        if float(s.get("end", 0.0)) > start_time and float(s.get("start", 0.0)) < end_time
    ]
    if len(candidate_segs) < 2:
        return None

    # Try natural sentence/segment endings from start_time
    base_start = start_time
    for s in reversed(candidate_segs):
        cand_end = float(s.get("end", 0.0))
        eff_dur, _ = analyze_silence_gaps(base_start, cand_end, candidate_segs)
        if min_duration <= eff_dur <= max_duration:
            # Check punctuation indicating complete sentence (. ! ? or pause)
            text = s.get("text", "").strip()
            if text and text[-1] in (".", "!", "?") or (s is not candidate_segs[-1]):
                return base_start, cand_end

    return None

def format_ass_timestamp(seconds: float) -> str:
    """Format seconds into ASS timestamp format: H:MM:SS.cs"""
    hrs = int(seconds // 3600)
    mins = int((seconds % 3600) // 60)
    secs = int(seconds % 60)
    centis = int(round((seconds - int(seconds)) * 100))
    if centis >= 100:
        centis = 99
    return f"{hrs}:{mins:02d}:{secs:02d}.{centis:02d}"


def generate_ass_subtitles(
    segments: List[Dict[str, Any]],
    output_path: Path,
    clip_start_time: float = 0.0,
    clip_end_time: float = float("inf"),
    font_name: str = "Arial Black",
    font_size: int = 54,
    style_preset: str = "yellow_pop",
    position: str = "bottom",
    enable_pop_bounce: bool = True,
    words_per_card: int = 3
) -> Path:
    """Generate high-retention, CapCut-quality ASS subtitles with word-by-word pop bounce.

    Args:
        segments: Transcript segments with word-level timestamps.
        output_path: Destination path for the .ass subtitle file.
        clip_start_time: Clip start relative to source video in seconds.
        clip_end_time: Clip end relative to source video in seconds.
        font_name: Typography font family (e.g. 'Impact', 'Arial Black', 'Montserrat').
        font_size: Subtitle font size in points.
        style_preset: Style preset ('yellow_pop', 'hormozi_green', 'electric_cyan', 'boxed_minimal').
        position: Vertical placement ('bottom' for Shorts safe-zone, 'middle' for center).
        enable_pop_bounce: Whether to apply the tactile scale punch animation.
        words_per_card: How many words appear simultaneously on screen (2-3 recommended).

    Returns:
        Path to generated .ass file.
    """
    preset = STYLE_PRESETS.get(style_preset, STYLE_PRESETS["yellow_pop"])
    primary_col = preset["primary"]
    highlight_col = preset["highlight"]
    outline_col = preset["outline"]
    border_style = preset["border_style"]
    outline_w = preset["outline_width"]
    shadow_val = preset["shadow"]

    # Safe margin calculation
    # Vertical 9:16 (1080x1920) safe zone: bottom margin 420px prevents overlap with Shorts/Reels UI
    if position == "middle":
        margin_v = 960
        alignment = 5  # Center Middle
    else:
        margin_v = 420
        alignment = 2  # Bottom Center

    font_family = FONT_PRESETS.get(font_name.lower().replace(" ", "_"), font_name)

    ass_header = f"""[Script Info]
ScriptType: v4.00+
PlayResX: {TARGET_WIDTH}
PlayResY: {TARGET_HEIGHT}
WrapStyle: 0
ScaledBorderAndShadow: yes

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding
Style: Default,{font_family},{font_size},{primary_col},&H000000FF,{outline_col},&H80000000,-1,0,0,0,100,100,0.5,0,{border_style},{outline_w},{shadow_val},{alignment},40,40,{margin_v},1
Style: Highlight,{font_family},{font_size},{highlight_col},&H000000FF,{outline_col},&H80000000,-1,0,0,0,100,100,0.5,0,{border_style},{outline_w},{shadow_val},{alignment},40,40,{margin_v},1

[Events]
Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text
"""

    events = []

    for seg in segments:
        seg_start = seg.get("start", 0.0)
        seg_end = seg.get("end", 0.0)

        if seg_end <= clip_start_time or seg_start >= clip_end_time:
            continue

        words = seg.get("words", [])

        if words:
            # Chunk words into rapid 2-3 word retention cards
            chunk_size = max(1, min(4, words_per_card))
            for i in range(0, len(words), chunk_size):
                chunk = words[i:i + chunk_size]
                chunk_start = max(clip_start_time, chunk[0]["start"])
                chunk_end = min(clip_end_time, chunk[-1]["end"])

                if chunk_end <= chunk_start or chunk_end <= clip_start_time or chunk_start >= clip_end_time:
                    continue

                # Generate karaoke illumination for each word within this card
                for j, target_word in enumerate(chunk):
                    w_start = max(clip_start_time, target_word["start"])
                    if j < len(chunk) - 1:
                        w_end = min(clip_end_time, max(w_start + 0.12, chunk[j + 1]["start"]))
                    else:
                        w_end = min(clip_end_time, max(w_start + 0.12, target_word["end"]))

                    if w_end <= w_start:
                        continue

                    rel_start = format_ass_timestamp(max(0.0, w_start - clip_start_time))
                    rel_end = format_ass_timestamp(max(0.0, w_end - clip_start_time))

                    card_words = []
                    for k, w in enumerate(chunk):
                        cleaned = w["word"].strip().upper()
                        if not cleaned:
                            continue

                        if k == j:
                            # Active spoken word with tactile pop bounce animation
                            if enable_pop_bounce:
                                # Punch to 118% scale, ease smoothly back to 100% in 90ms
                                anim_tag = f"{{\\c{highlight_col}&\\fscx118\\fscy118\\t(0,90,\\fscx100\\fscy100)}}"
                            else:
                                anim_tag = f"{{\\c{highlight_col}&}}"

                            card_words.append(f"{anim_tag}{cleaned}{{\\c{primary_col}&\\fscx100\\fscy100}}")
                        else:
                            # Inactive words in the card
                            card_words.append(cleaned)

                    text_display = " ".join(card_words)
                    events.append(f"Dialogue: 0,{rel_start},{rel_end},Default,,0,0,0,,{text_display}")

        else:
            # Fallback for plain transcript segments without word timestamps
            rel_start = format_ass_timestamp(max(0.0, seg_start - clip_start_time))
            rel_end = format_ass_timestamp(max(0.0, min(clip_end_time, seg_end) - clip_start_time))
            cleaned_text = seg.get("text", "").strip().upper()
            if cleaned_text:
                events.append(f"Dialogue: 0,{rel_start},{rel_end},Default,,0,0,0,,{cleaned_text}")

    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, "w", encoding="utf-8") as f:
        f.write(ass_header)
        f.write("\n".join(events))
        f.write("\n")

    return output_path

"""dispatch.video_engine.thumbnail
Production-grade video thumbnail generation, freshness validation, and cache invalidation.

Guarantees:
- Atomicity: Uses temporary files with atomic rename (os.replace) to prevent partially written images.
- Freshness: High-resolution nanosecond timestamps (st_mtime_ns), file size, and fast header/footer
  content fingerprinting (<0.2ms) via sidecar metadata.
- True File Integrity: Eliminates artificial filesystem timestamp distortion (e.g. mtime + 1.0).
- Automatic Regeneration: Automatically regenerates stale or missing thumbnails when videos change.
- Concurrency Protection: Thread-safe locking per clip_id prevents duplicate parallel FFmpeg extractions.
- Robust Versioning: Generates collision-free version tokens for query strings (?v={token}) ensuring
  instant cache busting across mobile WebViews even on rapid same-second replacements.
"""
import hashlib
import json
import logging
import os
import subprocess
import threading
import time
import uuid
from pathlib import Path
from typing import Optional, Dict, Any

from dispatch.config import CLIPS_DIR, PROCESSING_DIR, INCOMING_DIR
from dispatch import db

logger = logging.getLogger("dispatch.video_engine.thumbnail")

# Concurrency locks per clip_id to prevent parallel duplicate extractions
_thumbnail_locks: Dict[str, threading.Lock] = {}
_thumbnail_locks_guard = threading.Lock()


def get_clip_lock(clip_id: str) -> threading.Lock:
    """Retrieve or create a thread lock dedicated to a specific clip ID."""
    with _thumbnail_locks_guard:
        if clip_id not in _thumbnail_locks:
            _thumbnail_locks[clip_id] = threading.Lock()
        return _thumbnail_locks[clip_id]


def get_file_fingerprint(path: Path) -> str:
    """Compute a lightweight, deterministic file fingerprint (<0.2ms) without full file read.
    Combines st_mtime_ns, st_size, and a 64KB sample of header and footer bytes.
    Guarantees collision-free detection of file updates even if replaced within the same second
    or if mtimes are preserved by an external copying tool.
    """
    if not path.exists() or not path.is_file():
        return ""
    try:
        st = path.stat()
        size = st.st_size
        mtime_ns = getattr(st, "st_mtime_ns", int(st.st_mtime * 1e9))

        h = hashlib.sha256()
        h.update(f"{mtime_ns}:{size}:".encode("utf-8"))

        with open(path, "rb") as f:
            head = f.read(32768)
            h.update(head)
            if size > 65536:
                f.seek(max(0, size - 32768))
                tail = f.read(32768)
                h.update(tail)

        return h.hexdigest()[:16]
    except Exception as e:
        logger.debug("Error computing fingerprint for %s: %s", path, e)
        return ""


def resolve_clip_source_path(clip: Dict[str, Any]) -> Optional[Path]:
    """Safely resolve the original raw source video chunk for a clip."""
    # 1. Resolve through chunk_id in chunks table
    chunk_id = clip.get("chunk_id")
    if chunk_id:
        with db.get_db_connection() as conn:
            row = conn.execute("SELECT filepath FROM chunks WHERE id = ?", (chunk_id,)).fetchone()
            if row and row["filepath"]:
                cand = Path(row["filepath"])
                if cand.exists() and cand.is_file() and cand.stat().st_size > 0:
                    return cand

    # 2. Check storage/incoming and storage/processing using session_id or chunk_id
    for search_dir in (PROCESSING_DIR, INCOMING_DIR):
        if clip.get("session_id"):
            sess = clip["session_id"]
            cand = search_dir / f"{sess}.mp4"
            if cand.exists() and cand.stat().st_size > 0:
                return cand
        if chunk_id:
            cand = search_dir / f"{chunk_id}.mp4"
            if cand.exists() and cand.stat().st_size > 0:
                return cand

    # 3. Check YouTube inbox
    clip_id_str = clip.get("id", "")
    if clip_id_str.startswith("yt_") or (clip.get("chunk_id") and str(clip["chunk_id"]).startswith("yt_")):
        yt_id = clip.get("chunk_id") or clip_id_str
        clean_yt_id = str(yt_id).replace("yt_", "").strip()
        yt_row = db.get_youtube_video(clean_yt_id)
        if yt_row and yt_row.get("local_video_path"):
            cand = Path(yt_row["local_video_path"])
            if cand.exists() and cand.stat().st_size > 0:
                return cand

    return None


def resolve_active_clip_video(clip: Dict[str, Any]) -> Optional[Path]:

def generate_thumbnail_from_video(
    video_path: Path,
    output_thumb_path: Path,
    timestamp_sec: float = 0.5
) -> bool:
    """Atomically extract a single high-quality frame from video using FFmpeg.
    Uses temporary files with atomic rename to avoid exposing partially written images.
    Records high-resolution video metadata in sidecar .jpg.meta to guarantee freshness
    without artificial filesystem timestamp alteration.
    """
    if not video_path.exists() or video_path.stat().st_size == 0:
        logger.warning("Cannot generate thumbnail: source video does not exist or is empty (%s)", video_path)
        return False

    output_thumb_path.parent.mkdir(parents=True, exist_ok=True)
    nonce = uuid.uuid4().hex[:8]
    temp_thumb_path = output_thumb_path.with_name(f".tmp_{nonce}_{output_thumb_path.name}")
    temp_meta_path = output_thumb_path.with_name(f".tmp_{nonce}_{output_thumb_path.name}.meta")

    cmd = [
        "ffmpeg", "-y",
        "-ss", f"{max(0.0, timestamp_sec):.3f}",
        "-i", str(video_path),
        "-vframes", "1",
        "-q:v", "2",
        str(temp_thumb_path)
    ]

    try:
        res = subprocess.run(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, timeout=25)
        if res.returncode == 0 and temp_thumb_path.exists() and temp_thumb_path.stat().st_size > 0:
            # Generate deterministic sidecar metadata recording video fingerprint, size, and nanosecond timestamp
            try:
                vid_st = video_path.stat()
                vid_mtime_ns = getattr(vid_st, "st_mtime_ns", int(vid_st.st_mtime * 1e9))
                meta = {
                    "video_path": str(video_path.resolve()),
                    "video_mtime_ns": vid_mtime_ns,
                    "video_mtime": vid_st.st_mtime,
                    "video_size": vid_st.st_size,
                    "video_fingerprint": get_file_fingerprint(video_path),
                    "created_at": time.time(),
                    "timestamp_sec": timestamp_sec,
                }
                temp_meta_path.write_text(json.dumps(meta), encoding="utf-8")
            except Exception as e:
                logger.debug("Failed writing thumbnail meta for %s: %s", output_thumb_path, e)

            # Atomic rename into final location
            os.replace(str(temp_thumb_path), str(output_thumb_path))
            if temp_meta_path.exists():
                meta_dest = output_thumb_path.with_suffix(".jpg.meta")
                os.replace(str(temp_meta_path), str(meta_dest))

            return True
        else:
            logger.warning("FFmpeg thumbnail extraction returned code %d for %s", res.returncode, video_path)
    except Exception as e:
        logger.error("FFmpeg thumbnail extraction exception for %s: %s", video_path, e)
    finally:
        if temp_thumb_path.exists():
            try:
                temp_thumb_path.unlink()
            except Exception:
                pass
        if temp_meta_path.exists():
            try:
                temp_meta_path.unlink()
            except Exception:
                pass

    return False


def is_thumbnail_stale(video_path: Path, thumb_path: Path) -> bool:
    """Return True if the thumbnail is missing, empty, or older/different than the underlying video.
    Supports sub-second precision, file size changes, and content fingerprinting.
    """
    if not thumb_path.exists() or thumb_path.stat().st_size == 0:
        return True
    if not video_path.exists() or video_path.stat().st_size == 0:
        return False  # Cannot regenerate if video is absent; retain existing thumbnail

    # 1. Check sidecar metadata if available
    meta_path = thumb_path.with_suffix(".jpg.meta")
    if meta_path.exists():
        try:
            meta = json.loads(meta_path.read_text(encoding="utf-8"))
            vid_st = video_path.stat()
            curr_mtime_ns = getattr(vid_st, "st_mtime_ns", int(vid_st.st_mtime * 1e9))

            # Video modified after thumbnail generation
            if curr_mtime_ns > meta.get("video_mtime_ns", 0):
                return True
            # Video size changed
            if vid_st.st_size != meta.get("video_size"):
                return True
            # Content fingerprint changed (e.g. preserved timestamp replacement)
            curr_fp = get_file_fingerprint(video_path)
            if curr_fp and curr_fp != meta.get("video_fingerprint"):
                return True

            return False
        except Exception as e:
            logger.debug("Failed reading thumbnail meta %s: %s", meta_path, e)

    # 2. Fallback timestamp comparison (sub-second nanoseconds when available)
    try:
        v_st = video_path.stat()
        t_st = thumb_path.stat()
        v_mtime_ns = getattr(v_st, "st_mtime_ns", int(v_st.st_mtime * 1e9))
        t_mtime_ns = getattr(t_st, "st_mtime_ns", int(t_st.st_mtime * 1e9))
        if t_mtime_ns < v_mtime_ns:
            return True
        if t_st.st_mtime < v_st.st_mtime:
            return True
    except Exception:
        return True

    return False


def ensure_clip_thumbnail(clip_or_id: Any, force: bool = False) -> Optional[Path]:
    """Ensure a clip has a fresh, valid thumbnail on disk.
    If the video on disk was updated, replaced, or modified after the thumbnail was generated,
    this function regenerates the thumbnail immediately and updates database records.
    Thread-safe: serializes generation per clip_id with double-checked locking.
    """
    if isinstance(clip_or_id, dict):
        clip = clip_or_id
    elif isinstance(clip_or_id, str):
        clip = db.get_clip_by_id(clip_or_id)
        if not clip:
            # Check if clip_or_id is a chunk
            chunk = db.get_chunk_by_id(clip_or_id)
            if chunk:
                clip = {
                    "id": clip_or_id,
                    "chunk_id": clip_or_id,
                    "duration": chunk.get("duration", 10.0),
                    "video_path": chunk.get("filepath")
                }
            else:
                return None
    else:
        return None

    clip_id = clip.get("id")
    if not clip_id:
        return None

    # Determine target thumbnail path
    existing_thumb_str = clip.get("thumbnail_path")
    if existing_thumb_str:
        candidate_path = Path(existing_thumb_str)
        if candidate_path.parent == CLIPS_DIR and candidate_path.suffix.lower() in (".jpg", ".jpeg", ".png"):
            target_thumb_path = candidate_path
        else:
            target_thumb_path = CLIPS_DIR / f"{clip_id}.jpg"
    else:
        target_thumb_path = CLIPS_DIR / f"{clip_id}.jpg"

    clip_lock = get_clip_lock(clip_id)
    with clip_lock:
        active_video = resolve_active_clip_video(clip)

        # If no active video is on disk (e.g. cloud-only retention or deleted source):
        if not active_video or not active_video.exists() or active_video.stat().st_size == 0:
            if target_thumb_path.exists() and target_thumb_path.stat().st_size > 0:
                return target_thumb_path
            if existing_thumb_str and Path(existing_thumb_str).exists() and Path(existing_thumb_str).stat().st_size > 0:
                return Path(existing_thumb_str)
            return None

        # Double-check freshness under lock
        if not force and not is_thumbnail_stale(active_video, target_thumb_path):
            return target_thumb_path

        # Thumbnail is missing or stale -> regenerate!
        duration = float(clip.get("duration") or 2.0)
        is_rendered = bool(clip.get("video_path") and Path(clip["video_path"]).exists() and Path(clip["video_path"]) == active_video)

        if is_rendered:
            thumb_time = max(0.0, min(1.0, duration / 2.0))
        else:
            start_time = float(clip.get("start_time") or 0.0)
            thumb_time = max(0.0, start_time + min(1.0, duration / 2.0))

        logger.info("Regenerating thumbnail for clip %s from %s (time: %.3fs)", clip_id, active_video.name, thumb_time)
        success = generate_thumbnail_from_video(active_video, target_thumb_path, timestamp_sec=thumb_time)
        if success and target_thumb_path.exists():
            if clip.get("thumbnail_path") != str(target_thumb_path):
                try:
                    db.update_clip_media(
                        clip_id=clip_id,
                        video_path=clip.get("video_path") or str(active_video),
                        thumbnail_path=str(target_thumb_path)
                    )
                except Exception as e:
                    logger.warning("Could not update thumbnail_path in DB for %s: %s", clip_id, e)
            return target_thumb_path

        # Fallback to existing if generation failed
        if target_thumb_path.exists() and target_thumb_path.stat().st_size > 0:
            return target_thumb_path

        return None


def get_thumbnail_version_token(thumb_path: Optional[Path]) -> Optional[str]:
    """Return a collision-free version token string for cache busting.
    Uses high-resolution mtimes and content fingerprints to avoid integer second collisions.
    """
    if not thumb_path or not thumb_path.exists() or thumb_path.stat().st_size == 0:
        return None

    meta_path = thumb_path.with_suffix(".jpg.meta")
    if meta_path.exists():
        try:
            meta = json.loads(meta_path.read_text(encoding="utf-8"))
            mtime_ns = meta.get("video_mtime_ns")
            fp = meta.get("video_fingerprint", "")
            if mtime_ns:
                return f"{mtime_ns}_{fp[:8]}" if fp else f"{mtime_ns}"
        except Exception:
            pass

    try:
        st = thumb_path.stat()
        mtime_ns = getattr(st, "st_mtime_ns", int(st.st_mtime * 1e9))
        return f"{mtime_ns}"
    except Exception:
        return None


def get_thumbnail_version(thumb_path: Optional[Path]) -> Optional[Any]:
    """Return version representation for backwards compatibility."""
    token = get_thumbnail_version_token(thumb_path)
    if token:
        if token.isdigit():
            return int(token)
        return token
    return None


def populate_clip_media_urls(clip: Dict[str, Any]) -> Dict[str, Any]:
    """Ensure thumbnail freshness and populate versioned media URLs on clip dictionary.
    
    Guarantees:
    - Freshness: Ensures the thumbnail on disk is up to date with the underlying video.
    - Versioning: Injects collision-free ?v={token} into media URLs so browsers/WebViews
      cache aggressively while instantly busting cache when a video or thumbnail changes.
    - Backward-compatibility: Sets standard fields (video_url, thumbnail_url, source_video_url).
    """
    clip_id = clip.get("id")
    if not clip_id:
        return clip

    # 1. Fresh thumbnail validation & generation
    thumb_path = ensure_clip_thumbnail(clip)
    if thumb_path and thumb_path.exists() and thumb_path.stat().st_size > 0:
        v_token = get_thumbnail_version_token(thumb_path) or str(int(thumb_path.stat().st_mtime))
        clip["thumbnail_path"] = str(thumb_path)
        clip["thumbnail_url"] = f"/clips/{thumb_path.name}?v={v_token}"
        clip["thumbnail_version"] = v_token
    else:
        ex_thumb = clip.get("thumbnail_path")
        if ex_thumb and Path(ex_thumb).exists() and Path(ex_thumb).stat().st_size > 0:
            tp = Path(ex_thumb)
            v_token = get_thumbnail_version_token(tp) or str(int(tp.stat().st_mtime))
            clip["thumbnail_url"] = f"/clips/{tp.name}?v={v_token}"
            clip["thumbnail_version"] = v_token
        else:
            clip["thumbnail_url"] = None

    # 2. Rendered video URL
    vpath = clip.get("video_path")
    if vpath:
        vp = Path(vpath)
        if vp.exists() and vp.stat().st_size > 0:
            vm_ns = getattr(vp.stat(), "st_mtime_ns", int(vp.stat().st_mtime * 1e9))
            clip["video_url"] = f"/clips/{vp.name}?v={vm_ns}"
            clip["video_version"] = vm_ns
            clip["is_cloud_only"] = False
        else:
            clip["video_url"] = None
            clip["is_cloud_only"] = True
    else:
        clip["video_url"] = None

    # 3. Source video URL with version timestamp
    active_source = resolve_clip_source_path(clip)
    if active_source and active_source.exists():
        sm_ns = getattr(active_source.stat(), "st_mtime_ns", int(active_source.stat().st_mtime * 1e9))
        clip["source_video_url"] = f"/api/clips/{clip_id}/source?v={sm_ns}"
    else:
        clip["source_video_url"] = f"/api/clips/{clip_id}/source"

    return clip
