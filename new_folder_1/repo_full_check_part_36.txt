"""YouTube Inbox Catcher & Video Processing Pipeline.
Polls or accepts YouTube video URLs/IDs, downloads media via yt-dlp, extracts timestamped transcripts
(fetching YouTube auto-captions with an automatic fallback to local faster-whisper), runs semantic highlight
detection, renders vertical 9:16 clips, and populates the Review Dashboard idempotently.
"""
import re
import os
import json
import logging
import subprocess
import threading
from pathlib import Path
from typing import Dict, Any, List, Optional, Tuple

import yt_dlp

from dispatch.config import YOUTUBE_INBOX_DIR, CLIPS_DIR, DEFAULT_PUBLISH_MODE
from dispatch import db
from dispatch.youtube_inbox.vtt_parser import parse_vtt_file, parse_vtt_content
from dispatch.ai_clips.highlight_finder import identify_and_save_highlights
from dispatch.video_engine.renderer import render_clip
from dispatch.transcription.transcriber import transcribe_video

logger = logging.getLogger("dispatch.youtube_inbox.catcher")

# Regex to extract 11-character YouTube video ID
YOUTUBE_ID_REGEX = re.compile(
    r"(?:v=|\/|youtu\.be\/|embed\/|shorts\/)([a-zA-Z0-9_-]{11})(?:\?|&|$)"
)


def extract_youtube_video_id(url_or_id: str) -> Optional[str]:
    """Extract standard 11-character YouTube video ID from URL or return raw ID if valid."""
    cleaned = url_or_id.strip()
    if len(cleaned) == 11 and re.match(r"^[a-zA-Z0-9_-]{11}$", cleaned):
        return cleaned

    match = YOUTUBE_ID_REGEX.search(cleaned)
    if match:
        return match.group(1)
    return None


def _apply_auth_opts(ydl_opts: Dict[str, Any]) -> None:
    """Apply cookie authentication options if configured via environment variables.

    Cookies are opt-in (off by default) and CANNOT bypass the privacy gate: private
    sources are rejected before the download step based on the Data API verdict.
    """
    cookies_file = os.getenv("DISPATCH_YTDLP_COOKIES_FILE")
    cookies_browser = os.getenv("DISPATCH_YTDLP_COOKIES_BROWSER")
    if cookies_file and Path(cookies_file).exists():
        ydl_opts["cookiefile"] = str(Path(cookies_file).resolve())
    elif cookies_browser:
        ydl_opts["cookiesfrombrowser"] = (cookies_browser,)


# Single-flight ingestion state: video IDs with an ingestion run currently active
# in this process. Shared by all callers (dashboard endpoint threads + poller) so a
# duplicate submission can never start a competing yt-dlp download for the same video.
# NOTE: process-local only. The single-process dispatch daemon is the only writer;
# multi-process deployments would additionally need a database-level claim.
_INGEST_INFLIGHT_LOCK = threading.Lock()
_INGEST_INFLIGHT: set = set()


def is_ingestion_in_flight(video_id: str) -> bool:
    """True if an ingestion run for this video is currently active in this process."""
    with _INGEST_INFLIGHT_LOCK:
        return video_id in _INGEST_INFLIGHT


class YouTubeInboxCatcher:
    """Manages downloading, transcript extraction, and end-to-end processing of YouTube videos."""

    def __init__(self, storage_dir: Optional[Path] = None):
        self.storage_dir = storage_dir or YOUTUBE_INBOX_DIR
        self.storage_dir.mkdir(parents=True, exist_ok=True)

    def fetch_video_info(self, video_id: str) -> Dict[str, Any]:
        """Fetch video metadata without downloading video stream."""
        url = f"https://www.youtube.com/watch?v={video_id}"
        ydl_opts = {
            "quiet": True,
            "no_warnings": True,
            "skip_download": True,
        }
        _apply_auth_opts(ydl_opts)
        with yt_dlp.YoutubeDL(ydl_opts) as ydl:
            info = ydl.extract_info(url, download=False)
            return {
                "id": info.get("id", video_id),
                "title": info.get("title", f"YouTube Video {video_id}"),
                "channel_id": info.get("channel_id", ""),
                "channel": info.get("uploader", ""),
                "duration": float(info.get("duration", 0.0) or 0.0),
                "upload_date": info.get("upload_date", ""),
                "description": info.get("description", "")
            }

    def fetch_youtube_captions(self, video_id: str) -> Optional[List[Dict[str, Any]]]:
        """Attempt to fetch existing YouTube captions or auto-generated VTT subtitles."""
        url = f"https://www.youtube.com/watch?v={video_id}"
        temp_sub_prefix = self.storage_dir / f"sub_{video_id}"
        
        ydl_opts = {
            "skip_download": True,
            "writeautomaticsub": True,
            "writesubtitles": True,
            "subtitleslangs": ["en", "hi", "en-orig", "hi-orig"],
            "subtitlesformat": "vtt",
            "outtmpl": f"{temp_sub_prefix}.%(ext)s",
            "quiet": True,
            "no_warnings": True,
            "ignoreerrors": True,
        }
        _apply_auth_opts(ydl_opts)

        try:
            with yt_dlp.YoutubeDL(ydl_opts) as ydl:
                ydl.download([url])
        except Exception as e:
            logger.warning("yt-dlp subtitle download note for %s: %s", video_id, e)

        # Check for written .vtt files (can be present even if a secondary language returned an error)
        vtt_files = list(self.storage_dir.glob(f"sub_{video_id}*.vtt"))
        if vtt_files:
            # Choose preferred language: 1. English, 2. Hindi, 3. First available
            target_file = None
            for f in vtt_files:
                fname = f.name.lower()
                if ".en" in fname or "en-orig" in fname:
                    target_file = f
                    break
            if not target_file:
                for f in vtt_files:
                    fname = f.name.lower()
                    if ".hi" in fname or "hi-orig" in fname:
                        target_file = f
                        break
            if not target_file:
                target_file = vtt_files[0]

            logger.info("Found YouTube caption file %s for video %s", target_file.name, video_id)
            segments = parse_vtt_file(target_file)

            # Cleanup sub files
            for f in vtt_files:
                try:
                    f.unlink()
                except Exception:
                    pass

            if segments:
                return segments

        # YouTube Data API v3 fallback for authorized channels
        try:
            from dispatch.youtube_inbox.oauth import get_youtube_service
            service = get_youtube_service()
            if service:
                captions_resp = service.captions().list(part="snippet", videoId=video_id).execute()
                items = captions_resp.get("items", [])
                if items:
                    target_cid = None
                    for it in items:
                        snip = it.get("snippet", {})
                        lang = snip.get("language", "").lower()
                        if lang.startswith("en") or lang.startswith("hi"):
                            target_cid = it.get("id")
                            break
                    if not target_cid and items:
                        target_cid = items[0].get("id")
                    if target_cid:
                        caption_content = service.captions().download(id=target_cid, tfmt="vtt").execute()
                        cap_str = caption_content.decode("utf-8", errors="replace") if isinstance(caption_content, bytes) else str(caption_content)
                        parsed = parse_vtt_content(cap_str)
                        if parsed:
                            logger.info("Retrieved %d segments via YouTube Data API caption track for %s", len(parsed), video_id)
                            return parsed
        except Exception as api_err:
            logger.debug("YouTube Data API caption retrieval fallback failed for %s: %s", video_id, api_err)

        return None

    def download_video(self, video_id: str) -> Path:
        """Download video file locally via yt-dlp, limiting resolution to 1080p MP4."""
        target_path = self.storage_dir / f"{video_id}.mp4"
        if target_path.exists() and target_path.stat().st_size > 1024:
            logger.info("Video %s already downloaded at %s", video_id, target_path)
            return target_path

        url = f"https://www.youtube.com/watch?v={video_id}"
        outtmpl = str(self.storage_dir / f"{video_id}.%(ext)s")

        ydl_opts = {
            "format": "bestvideo[vcodec^=avc1][height<=1080]+bestaudio[ext=m4a]/bestvideo[height<=1080][ext=mp4]+bestaudio[ext=m4a]/best[height<=1080][ext=mp4]/best",
            "outtmpl": outtmpl,
            "merge_output_format": "mp4",
            "quiet": True,
            "no_warnings": True,
        }
        _apply_auth_opts(ydl_opts)

        logger.info("Starting yt-dlp download for video %s", video_id)
        with yt_dlp.YoutubeDL(ydl_opts) as ydl:
            ydl.download([url])

        if not target_path.exists():
            # Check if downloaded with different container and convert/rename
            # Exclude incomplete downloads, transcripts, and audio-only/video-only DASH splits
            ignored_patterns = re.compile(r"(\.part$|\.ytdl$|\.vtt$|\.aria2$|\.tmp$|\.temp$|\.f\d+\.)", re.IGNORECASE)
            candidates = [
                f for f in self.storage_dir.glob(f"{video_id}.*")
                if f != target_path and not ignored_patterns.search(f.name) and f.stat().st_size > 1024
            ]
            if candidates:
                # Pick the largest candidate file
                candidates.sort(key=lambda f: f.stat().st_size, reverse=True)
                cand = candidates[0]
                if cand.suffix.lower() != ".mp4":
                    # Convert/remux to MP4
                    proc = subprocess.run(
                        ["ffmpeg", "-y", "-i", str(cand), "-c", "copy", str(target_path)],
                        stdout=subprocess.DEVNULL,
                        stderr=subprocess.DEVNULL,
                        check=False
                    )
                    if proc.returncode == 0 and target_path.exists() and target_path.stat().st_size > 1024:
                        cand.unlink(missing_ok=True)
                    else:
                        target_path.unlink(missing_ok=True)
                        raise RuntimeError(f"FFmpeg remux of candidate {cand.name} to MP4 failed with exit code {proc.returncode}")
                else:
                    cand.rename(target_path)

        if not target_path.exists():
            raise FileNotFoundError(f"Failed to download video {video_id}")

        logger.info("Downloaded %s successfully (%.2f MB)", video_id, target_path.stat().st_size / (1024 * 1024))
        return target_path

    def ingest_video(
        self,
        url_or_id: str,
        dispatch_id: Optional[str] = None,
        session_id: Optional[str] = None,
        sequence_number: Optional[int] = None
    ) -> Dict[str, Any]:
        """Pillar 1: Dedicated YouTube Ingestion & Media Validation.

        Contract:
          1. Discover & register video record in youtube_inbox idempotently (status: DISCOVERED).
          2. Transition to DOWNLOADING and download video via yt-dlp.
          3. Transition to VALIDATING and run real ffprobe validation (stream, decodable, duration, dimensions).
          4. Persist verified metadata and local path, then transition to DOWNLOADED.
          5. STOP. (Downstream stages consume DOWNLOADED records).

        Single-flight: only one ingestion run per video may be active in this process
        at a time, regardless of caller (dashboard endpoint threads or the poller).
        """
        from dispatch.ingestion.validator import probe_video

        video_id = extract_youtube_video_id(url_or_id)
        if not video_id:
            raise ValueError(f"Invalid YouTube URL or ID: {url_or_id}")

        existing = db.get_youtube_video(video_id)
        # Idempotency check: If already DOWNLOADED or completed downstream, return existing record
        if existing and existing["status"] in (
            "DOWNLOADED", "WAITING_FOR_TRANSCRIPT", "TRANSCRIBED",
            "TRANSCRIPT_FETCHED", "ANALYZING", "CLIPS_DEFINED",
            "CLIPS_CREATED", "COMPLETED"
        ):
            local_path = Path(existing["local_video_path"]) if existing.get("local_video_path") else None
            if local_path and local_path.exists() and local_path.stat().st_size > 0:
                is_valid, _, _ = probe_video(local_path)
                if is_valid:
                    logger.info("Video %s is already downloaded and verified. Skipping duplicate ingest.", video_id)
                    return {
                        "video_id": video_id,
                        "status": existing["status"],
                        "local_video_path": str(local_path),
                        "verified": True
                    }

        with _INGEST_INFLIGHT_LOCK:
            if video_id in _INGEST_INFLIGHT:
                logger.info("Ingestion already running for video %s. Skipping duplicate run.", video_id)
                return {
                    "video_id": video_id,
                    "status": "ALREADY_INGESTING",
                    "verified": False,
                    "duplicate": True
                }
            _INGEST_INFLIGHT.add(video_id)
        try:
            return self._ingest_video_locked(
                video_id=video_id,
                dispatch_id=dispatch_id,
                session_id=session_id,
                sequence_number=sequence_number
            )
        finally:
            with _INGEST_INFLIGHT_LOCK:
                _INGEST_INFLIGHT.discard(video_id)

    def _ingest_video_locked(
        self,
        video_id: str,
        dispatch_id: Optional[str] = None,
        session_id: Optional[str] = None,
        sequence_number: Optional[int] = None
    ) -> Dict[str, Any]:
        """Pillar 1 run body. Caller must hold the per-video in-flight slot from ingest_video()."""
        from dispatch.ingestion.validator import probe_video
        from dispatch.youtube_inbox.oauth import extract_dispatch_id, extract_session_metadata

        logger.info("Starting Pillar 1 Ingestion for YouTube video: %s", video_id)

        # 1. Fetch metadata & register DISCOVERED
        info = {}
        s_sha = None
        s_sz = None
        try:
            info = self.fetch_video_info(video_id)
            desc = info.get("description", "")
            if not dispatch_id:
                dispatch_id = extract_dispatch_id(desc)
            s_sess, s_seg, s_seq = extract_session_metadata(desc)
            from dispatch.youtube_inbox.oauth import extract_source_hash_and_size
            s_sha, s_sz = extract_source_hash_and_size(desc)
            if not session_id:
                session_id = s_sess
            if sequence_number is None:
                sequence_number = s_seq
            if not dispatch_id and s_seg:
                dispatch_id = s_seg

            db.register_youtube_video(
                video_id=video_id,
                title=info.get("title", f"YouTube Video {video_id}"),
                channel_id=info.get("channel_id", ""),
                upload_time=info.get("upload_date"),
                duration=info.get("duration", 0.0),
                dispatch_id=dispatch_id,
                session_id=session_id,
                sequence_number=sequence_number,
                source_sha256=s_sha,
                source_file_size=s_sz
            )
        except Exception as e:
            logger.warning("Could not fetch remote info for video %s (%s). Registering placeholder.", video_id, e)
            db.register_youtube_video(
                video_id=video_id,
                title=f"YouTube Video {video_id}",
                dispatch_id=dispatch_id,
                session_id=session_id,
                sequence_number=sequence_number
            )

        db.update_youtube_video(
            video_id,
            status="DISCOVERED",
            dispatch_id=dispatch_id,
            session_id=session_id,
            sequence_number=sequence_number,
            source_sha256=s_sha,
            source_file_size=s_sz
        )

        # 1.5 Authoritative YouTube Data API status check before download attempt
        from dispatch.youtube_inbox.oauth import get_youtube_processing_status
        status_info = None
        try:
            status_map = get_youtube_processing_status([video_id])
            status_info = status_map.get(video_id)
        except Exception as e:
            logger.debug("Authoritative status check error for %s: %s", video_id, e)

        if status_info:
            p_status = status_info.get("processing_status")
            u_status = status_info.get("upload_status")
            fail_reason = status_info.get("failure_reason")

            if p_status == "processing" or (u_status == "uploaded" and p_status != "succeeded"):
                logger.info(
                    "Video %s is still processing on YouTube (processingStatus=%s, uploadStatus=%s). Deferring download.",
                    video_id, p_status, u_status
                )
                db.update_youtube_video(
                    video_id,
                    status="WAITING_FOR_YOUTUBE_PROCESSING",
                    last_error=f"Processing on YouTube (processingStatus={p_status}, uploadStatus={u_status})..."
                )
                return {
                    "video_id": video_id,
                    "status": "WAITING_FOR_YOUTUBE_PROCESSING",
                    "verified": False,
                    "processing": True
                }

            if p_status in ("failed", "terminated") or u_status in ("failed", "rejected"):
                err_msg = f"YouTube processing failed: {fail_reason or 'rejected'}"
                logger.error("Video %s failed processing on YouTube: %s", video_id, err_msg)
                db.update_youtube_video(video_id, status="FAILED", last_error=err_msg)
                return {
                    "video_id": video_id,
                    "status": "FAILED",
                    "verified": False,
                    "error": err_msg
                }

        # 1.5 PRIVACY GATE (Friends-ready Alpha contract): only public/unlisted
        # sources may be ingested. The authoritative visibility comes from the
        # YouTube Data API, and if it cannot be verified the ingestion is refused -
        # authenticated yt-dlp cookies must never silently bypass this restriction.
        from dispatch.youtube_inbox.oauth import get_youtube_processing_status
        try:
            privacy_info = get_youtube_processing_status([video_id]).get(video_id, {})
        except Exception as status_err:
            # Fail-closed policy (alpha brief B1): a transient visibility-API failure
            # must NEVER fall through to the download path. Defer with a durable,
            # non-terminal status; the next ingest/poll attempt re-verifies.
            err_msg = (
                "Visibility check unavailable (network/API error); download deferred. "
                "Only verified public/unlisted sources are ingested."
            )
            logger.warning("Deferring %s: visibility check unavailable (%s)", video_id, status_err)
            db.update_youtube_video(video_id, status="WAITING_FOR_VISIBILITY_CHECK", last_error=err_msg)
            return {"video_id": video_id, "status": "WAITING_FOR_VISIBILITY_CHECK", "verified": False, "error": err_msg}
        privacy_status = (privacy_info.get("privacy_status") or "").lower()
        if privacy_status == "private":
            err_msg = "Private videos are not supported as sources."
            logger.error("Rejecting private source video %s", video_id)
            db.update_youtube_video(video_id, status="FAILED", last_error=err_msg)
            return {"video_id": video_id, "status": "FAILED", "verified": False, "error": err_msg}
        if privacy_status not in ("public", "unlisted"):
            err_msg = (
                f"Source visibility could not be verified (got '{privacy_status or 'unknown'}'). "
                "Only public or unlisted videos can be ingested."
            )
            logger.error("Rejecting unverifiable source video %s: %s", video_id, privacy_status)
            db.update_youtube_video(video_id, status="FAILED", last_error=err_msg)
            return {"video_id": video_id, "status": "FAILED", "verified": False, "error": err_msg}

        # 2. Transition to DOWNLOADING & download media
        db.update_youtube_video(video_id, status="DOWNLOADING")
        try:
            local_video_path = self.download_video(video_id)
        except Exception as e:
            err = f"Download failed: {e}"
            err_str = str(e).lower()
            if "processing" in err_str or "check back later" in err_str:
                logger.info("Video %s is still processing on YouTube. Transitioning to WAITING_FOR_YOUTUBE_PROCESSING.", video_id)
                db.update_youtube_video(video_id, status="WAITING_FOR_YOUTUBE_PROCESSING", last_error=err)
                return {
                    "video_id": video_id,
                    "status": "WAITING_FOR_YOUTUBE_PROCESSING",
                    "verified": False,
                    "processing": True
                }
            logger.error("Download failed for video %s: %s", video_id, err)
            db.update_youtube_video(video_id, status="FAILED", last_error=err)
            raise RuntimeError(err) from e

        # 3. Transition to VALIDATING & run real ffprobe verification
        db.update_youtube_video(video_id, status="VALIDATING", local_video_path=str(local_video_path))
        is_valid, meta, err = probe_video(local_video_path)
        if not is_valid:
            error_msg = f"Media validation failed for {local_video_path.name}: {err}"
            logger.error(error_msg)
            # Remove corrupt or zero-byte file
            try:
                local_video_path.unlink(missing_ok=True)
            except Exception:
                pass
            db.update_youtube_video(video_id, status="FAILED", last_error=error_msg)
            raise ValueError(error_msg)

        # 4. Verified: Persist metadata & set status = DOWNLOADED
        total_duration = meta.get("duration", 0.0)
        db.update_youtube_video(
            video_id=video_id,
            status="DOWNLOADED",
            local_video_path=str(local_video_path)
        )

        with db.get_db_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(
                "UPDATE youtube_inbox SET duration = ? WHERE video_id = ?",
                (total_duration, video_id)
            )
            conn.commit()

        logger.info("Pillar 1 Ingestion SUCCESS for %s -> DOWNLOADED (Duration: %.1fs, Dimensions: %dx%d)",
                    video_id, total_duration, meta.get("width", 0), meta.get("height", 0))

        return {
            "video_id": video_id,
            "status": "DOWNLOADED",
            "local_video_path": str(local_video_path),
            "metadata": meta,
            "verified": True
        }

    def process_video(
        self,
        url_or_id: str,
        session_id: Optional[str] = None,
        dispatch_id: Optional[str] = None,
        force_whisper: bool = False
    ) -> Dict[str, Any]:
        """[LEGACY DOWNSTREAM DEMO] Full end-to-end processing pipeline for a single YouTube video.
        WARNING: Do NOT use this in the active ingestion path! Active ingestion is handled
        strictly by `ingest_video()`, which stops at `DOWNLOADED` per the Pillar 1 contract.
        This function is preserved only for downstream testing until Pillars 2-4 take over.

        1. Extract and validate video ID.
        2. Idempotent check in SQLite DB.
        3. Fetch metadata and register in youtube_inbox with dispatch_id.
        4. Download video media (DOWNLOADING).
        5. Fetch transcript (TRANSCRIBING: YouTube captions / VTT with Whisper fallback).
        6. Detect semantic highlights (ANALYZING: Gemini Flash / heuristic).
        7. Render vertical 9:16 candidate clips (RENDERING: FFmpeg + ASS).
        8. Mark completed (CLIPS_CREATED -> ready_review) and return result summary.
        """
        video_id = extract_youtube_video_id(url_or_id)
        if not video_id:
            raise ValueError(f"Invalid YouTube URL or ID: {url_or_id}")

        # Check idempotency: If already completed and clips created, return cached record
        existing = db.get_youtube_video(video_id)
        if existing and existing["status"] in ("CLIPS_CREATED", "COMPLETED"):
            logger.info("Video %s has already been completely processed. Skipping duplicate run.", video_id)
            return {
                "video_id": video_id,
                "status": "ALREADY_PROCESSED",
                "clips_created": []
            }

        logger.info("Processing YouTube Inbox item: %s", video_id)

        # 1. Fetch metadata
        try:
            info = self.fetch_video_info(video_id)
            desc = info.get("description", "")
            if not dispatch_id:
                from dispatch.youtube_inbox.oauth import extract_dispatch_id
                dispatch_id = extract_dispatch_id(desc)
            if not session_id:
                from dispatch.youtube_inbox.oauth import extract_session_metadata
                s_sess, s_seg, s_seq = extract_session_metadata(desc)
                if s_sess:
                    session_id = s_sess
                if not dispatch_id and s_seg:
                    dispatch_id = s_seg

            from dispatch.youtube_inbox.oauth import extract_source_hash_and_size
            s_sha, s_sz = extract_source_hash_and_size(desc)

            db.register_youtube_video(
                video_id=video_id,
                title=info["title"],
                channel_id=info["channel_id"],
                upload_time=info["upload_date"],
                duration=info["duration"],
                dispatch_id=dispatch_id,
                session_id=session_id,
                source_sha256=s_sha,
                source_file_size=s_sz
            )
        except Exception as e:
            logger.error("Failed to fetch info for video %s: %s", video_id, e)
            db.register_youtube_video(
                video_id=video_id,
                title=f"YouTube Video {video_id}",
                dispatch_id=dispatch_id,
                session_id=session_id
            )
            info = {"title": f"YouTube Video {video_id}", "duration": 0.0}

        db.update_youtube_video(
            video_id,
            status="DOWNLOADING",
            dispatch_id=dispatch_id,
            session_id=session_id,
            source_sha256=s_sha if 's_sha' in locals() else None,
            source_file_size=s_sz if 's_sz' in locals() else None
        )

        # 2. Download media
        try:
            local_video_path = self.download_video(video_id)
            db.update_youtube_video(video_id, status="TRANSCRIBING", local_video_path=str(local_video_path))
        except Exception as e:
            error_msg = f"Failed to download video: {e}"
            logger.error(error_msg)
            db.update_youtube_video(video_id, status="FAILED", last_error=error_msg)
            raise

        # 3. Fetch Transcript (YouTube captions first -> Whisper fallback)
        segments: Optional[List[Dict[str, Any]]] = None
        transcript_source = "none"

        if not force_whisper:
            logger.info("Attempting to retrieve YouTube auto/manual captions for %s", video_id)
            segments = self.fetch_youtube_captions(video_id)
            if segments:
                transcript_source = "youtube"
                logger.info("Successfully loaded %d segments from YouTube captions", len(segments))

        if not segments:
            if force_whisper:
                logger.info("force_whisper requested. Running local faster-whisper on %s", local_video_path.name)
                whisper_result = transcribe_video(local_video_path)
                segments = whisper_result.get("segments", [])
                transcript_source = "whisper"
                logger.info("Local faster-whisper generated %d segments", len(segments))
            else:
                logger.info("YouTube captions unavailable for %s and automatic Whisper fallback is disabled.", video_id)
                db.update_youtube_video(
                    video_id,
                    status="CAPTIONS_UNAVAILABLE",
                    transcript_source="none",
                    last_error="YouTube captions unavailable",
                    segments_json="[]"
                )
                return {
                    "video_id": video_id,
                    "status": "CAPTIONS_UNAVAILABLE",
                    "error": "YouTube captions unavailable"
                }

        db.update_youtube_video(
            video_id,
            status="TRANSCRIPT_FETCHED",
            transcript_source=transcript_source,
            segments_json=json.dumps(segments, ensure_ascii=False)
        )

        # 4. Probe video duration & aspect ratio if info["duration"] was 0
        total_duration = info["duration"]
        if total_duration <= 0.0:
            from dispatch.ingestion.validator import probe_video
            valid, meta, error = probe_video(local_video_path)
            if not valid:
                raise RuntimeError(f"Downloaded YouTube video is not readable: {error}")
            total_duration = meta.get("duration", 60.0)

        # Also register a chunk in the DB so clips have a valid chunk_id reference
        file_hash = f"yt_{video_id}"
        chunk_id = db.register_chunk(
            session_id=session_id,
            filename=local_video_path.name,
            filepath=str(local_video_path),
            file_hash=file_hash
        )
        db.update_chunk_metadata(
            chunk_id=chunk_id,
            duration=total_duration,
            width=1920,
            height=1080,
            aspect_ratio="16:9",
            status="transcribed"
        )

        full_text = " ".join(s["text"] for s in segments)
        db.save_transcript(
            chunk_id=chunk_id,
            session_id=session_id,
            full_text=full_text,
            segments=segments
        )

        # 5. Extract highlights (ANALYZING)
        db.update_youtube_video(video_id, status="ANALYZING")
        logger.info("Detecting candidate highlights for %s (Duration: %.1fs)", video_id, total_duration)
        publish_mode = db.get_setting("publish_mode", DEFAULT_PUBLISH_MODE) or "private"
        clip_ids = identify_and_save_highlights(
            chunk_id=chunk_id,
            session_id=session_id,
            segments=segments,
            total_duration=total_duration,
            publish_mode=publish_mode
        )

        logger.info("Identified %d candidate clips for YouTube video %s", len(clip_ids), video_id)
        if not clip_ids:
            raise RuntimeError(f"Gemini returned no renderable highlights for YouTube video {video_id}")

        # 6. Render candidate clips (RENDERING)
        db.update_youtube_video(video_id, status="RENDERING")
        rendered_clips = []
        for cid in clip_ids:
            with db.get_db_connection() as conn:
                cursor = conn.cursor()
                cursor.execute("SELECT * FROM clips WHERE id = ?", (cid,))
                c_row = cursor.fetchone()

            start_t = c_row["start_time"] if c_row else 0.0
            end_t = c_row["end_time"] if c_row else total_duration
            l_mode = (c_row["layout_mode"] if c_row else None) or "fit_blur"

            try:
                render_result = render_clip(
                    source_video=local_video_path,
                    clip_id=cid,
                    start_time=start_t,
                    end_time=end_t,
                    aspect_ratio="16:9",
                    layout_mode=l_mode,
                    segments=segments
                )
                db.update_clip_media(
                    clip_id=cid,
                    video_path=str(render_result.get("video_path", "")),
                    thumbnail_path=str(render_result.get("thumbnail_path", ""))
                )
                rendered_clips.append(cid)
            except Exception as e:
                logger.exception("Failed to render clip %s", cid)
                db.update_youtube_video(video_id, status="FAILED", last_error=f"FFmpeg render failed: {e}")
                raise

        if not rendered_clips:
            raise RuntimeError(f"FFmpeg did not produce an output clip for YouTube video {video_id}")

        db.update_youtube_video(video_id, status="CLIPS_CREATED")
        logger.info("YouTube Inbox processing complete for %s. %d clips ready for review.", video_id, len(rendered_clips))

        return {
            "video_id": video_id,
            "title": info["title"],
            "status": "CLIPS_CREATED",
            "transcript_source": transcript_source,
            "clips_count": len(rendered_clips),
            "clip_ids": rendered_clips
        }


def scan_channel_for_dispatch_uploads(
    channel_id: Optional[str] = None,
    marker: str = "[DISPATCH]",
    limit: int = 10
) -> List[Dict[str, Any]]:
    """Scan channel uploads using yt-dlp for videos matching marker prefix (e.g. [DISPATCH]).
    Returns list of discovered videos that have not yet been processed.
    """
    if not channel_id:
        return []

    # Use the YouTube uploads playlist ID: replacing 'UC' prefix with 'UU'
    if channel_id.startswith("UC"):
        target_url = f"https://www.youtube.com/playlist?list=UU{channel_id[2:]}"
    else:
        target_url = f"https://www.youtube.com/channel/{channel_id}/videos"

    logger.debug("Scanning YouTube uploads playlist %s for marker '%s'", target_url, marker)
    ydl_opts = {
        "extract_flat": "in_playlist",
        "playlistend": limit,
        "quiet": True,
        "no_warnings": True,
        "ignoreerrors": True,
    }

    discovered = []
    try:
        with yt_dlp.YoutubeDL(ydl_opts) as ydl:
            res = ydl.extract_info(target_url, download=False)
            if not res:
                return []
            entries = res.get("entries", [])
            for item in entries:
                if not item:
                    continue
                title = item.get("title", "")
                v_id = item.get("id")
                if not v_id:
                    continue

                # Check if marker matches (if marker specified)
                if marker and marker.lower() not in title.lower():
                    continue

                if not db.is_youtube_video_processed(v_id):
                    discovered.append({
                        "video_id": v_id,
                        "title": title,
                        "duration": float(item.get("duration", 0.0) or 0.0),
                    })
    except Exception as e:
        logger.debug("Uploads playlist scan encounter note: %s", e)

    return discovered

# Regex to extract dispatch_id from YouTube description or tags
DISPATCH_ID_REGEX = re.compile(r"dispatch_id[:=]\s*([a-zA-Z0-9_-]+)", re.IGNORECASE)
SESSION_ID_REGEX = re.compile(r"session_id[:=]\s*([a-zA-Z0-9_-]+)", re.IGNORECASE)
SEGMENT_ID_REGEX = re.compile(r"segment_id[:=]\s*([a-zA-Z0-9_-]+)", re.IGNORECASE)
SEQUENCE_NUM_REGEX = re.compile(r"sequence_number[:=]\s*(\d+)", re.IGNORECASE)
SOURCE_SHA256_REGEX = re.compile(r"source_sha256[:=]\s*([a-fA-F0-9]{64})", re.IGNORECASE)
SOURCE_SIZE_REGEX = re.compile(r"source_file_size[:=]\s*(\d+)", re.IGNORECASE)


def extract_source_hash_and_size(description: str, tags: Optional[List[str]] = None) -> Tuple[Optional[str], Optional[int]]:
    """Extract source_sha256 and source_file_size from video description or tags."""
    sha256 = None
    size = None
    if description:
        m_sha = SOURCE_SHA256_REGEX.search(description)
        if m_sha:
            sha256 = m_sha.group(1).strip().lower()
        m_sz = SOURCE_SIZE_REGEX.search(description)
        if m_sz:
            try:
                size = int(m_sz.group(1).strip())
            except ValueError:
                pass

    if tags and not sha256:
        for t in tags:
            t_clean = t.strip()
            if t_clean.lower().startswith("sha256_"):
                candidate = t_clean[7:].strip()
                if len(candidate) == 64:
                    sha256 = candidate.lower()
                    break

    return sha256, size


def extract_session_metadata(description: str, tags: Optional[List[str]] = None) -> Tuple[Optional[str], Optional[str], Optional[int]]:
    """Extract session_id, segment_id, sequence_number from description or tags."""
    session_id = None
    segment_id = None
    sequence_number = None

    if description:
        m_sess = SESSION_ID_REGEX.search(description)
        if m_sess:
            session_id = m_sess.group(1).strip()
        m_seg = SEGMENT_ID_REGEX.search(description)
        if m_seg:
            segment_id = m_seg.group(1).strip()
        m_seq = SEQUENCE_NUM_REGEX.search(description)
        if m_seq:
            try:
                sequence_number = int(m_seq.group(1).strip())
            except ValueError:
                pass

    if tags:
        for t in tags:
            t_clean = t.strip()
            if t_clean.lower().startswith("dispatch_session_"):
                if not session_id:
                    session_id = t_clean[len("dispatch_session_"):].strip()
            elif t_clean.lower().startswith("dispatch_seg_"):
                if sequence_number is None:
                    try:
                        sequence_number = int(t_clean[len("dispatch_seg_"):].strip())
                    except ValueError:
                        pass
            elif t_clean.lower().startswith("dispatch_id_"):
                if not segment_id:
                    segment_id = t_clean[len("dispatch_id_"):].strip()

    return session_id, segment_id, sequence_number


def extract_dispatch_id(description: str, tags: Optional[List[str]] = None) -> Optional[str]:
    """Extract dispatch_id from description or tags if present."""
    if description:
        match = DISPATCH_ID_REGEX.search(description)
        if match:
            return match.group(1).strip()

    if tags:
        for t in tags:
            if t.lower().startswith("dispatch_id_"):
                return t[len("dispatch_id_"):].strip()
            match = DISPATCH_ID_REGEX.search(t)
            if match:
                return match.group(1).strip()

    return None


from dispatch.config import ROOT_DIR, YOUTUBE_TOKEN_PATH

def get_youtube_service(token_path: Optional[Path] = None):
    """Return authenticated YouTube Data API v3 service resource if credentials exist."""
    candidates = [
        token_path,
        Path(os.getenv("DISPATCH_YOUTUBE_TOKEN_PATH", str(YOUTUBE_TOKEN_PATH))),
        YOUTUBE_TOKEN_PATH,
        ROOT_DIR / "youtube_token.json",
        Path("youtube_token.json"),
    ]

    token_file = None
    for c in candidates:
        if c and c.exists():
            token_file = c
            break

    try:
        from googleapiclient.discovery import build
        from google.oauth2.credentials import Credentials
        from google.auth.transport.requests import Request as GoogleAuthRequest

        if token_file and token_file.exists():
            creds = Credentials.from_authorized_user_file(str(token_file))
            if creds.expired and creds.refresh_token:
                try:
                    creds.refresh(GoogleAuthRequest())
                    if hasattr(creds, "to_json"):
                        with open(token_file, "w", encoding="utf-8") as f:
                            f.write(creds.to_json())
                        logger.info("Refreshed and persisted YouTube OAuth token to %s", token_file)
                except Exception as refresh_err:
                    logger.warning("Failed to refresh YouTube OAuth token: %s", refresh_err)
            return build("youtube", "v3", credentials=creds)

        # Check environment variables
        access_token = os.getenv("YOUTUBE_ACCESS_TOKEN")
        refresh_token = os.getenv("YOUTUBE_REFRESH_TOKEN")
        client_id = os.getenv("YOUTUBE_CLIENT_ID")
        client_secret = os.getenv("YOUTUBE_CLIENT_SECRET")

        if refresh_token and client_id and client_secret:
            creds = Credentials(
                token=access_token,
                refresh_token=refresh_token,
                token_uri="https://oauth2.googleapis.com/token",
                client_id=client_id,
                client_secret=client_secret
            )
            if creds.expired and creds.refresh_token:
                try:
                    creds.refresh(GoogleAuthRequest())
                except Exception as refresh_err:
                    logger.warning("Failed to refresh env-based YouTube OAuth token: %s", refresh_err)
            return build("youtube", "v3", credentials=creds)

    except Exception as e:
        logger.debug("YouTube API service not initialized: %s", e)

    return None


def list_authenticated_user_uploads(
    max_results_per_page: int = 50,
    stop_on_known_page: bool = True
) -> List[Dict[str, Any]]:
    """Query YouTube Data API v3 for the authenticated user's uploads (including Unlisted and Private).
    Paginates dynamically until nextPageToken is exhausted OR until an entire page consists of
    already-ingested videos (early-stop to avoid wasteful API quota consumption).
    Returns list of discovered videos with metadata and extracted dispatch_id.
    """
    from dispatch import db

    service = get_youtube_service()
    if not service:
        return []

    discovered = []
    try:
        # 1. Fetch channel's uploads playlist ID
        channel_resp = service.channels().list(mine=True, part="contentDetails").execute()
        items = channel_resp.get("items", [])
        if not items:
            return []

        uploads_playlist_id = items[0]["contentDetails"]["relatedPlaylists"]["uploads"]

        # 2. Query uploads playlist with pageToken pagination
        page_token = None
        per_page = min(max_results_per_page, 50)

        while True:
            kwargs = {
                "playlistId": uploads_playlist_id,
                "part": "snippet,status",
                "maxResults": per_page
            }
            if page_token:
                kwargs["pageToken"] = page_token

            playlist_resp = service.playlistItems().list(**kwargs).execute()
            page_items = playlist_resp.get("items", [])
            if not page_items:
                break

            page_has_new_item = False

            for item in page_items:
                snippet = item.get("snippet", {})
                video_id = snippet.get("resourceId", {}).get("videoId")
                if not video_id:
                    continue

                title = snippet.get("title", "")
                description = snippet.get("description", "")
                tags = snippet.get("tags", [])
                published_at = snippet.get("publishedAt", "")
                privacy = item.get("status", {}).get("privacyStatus", "unlisted")

                dispatch_id = extract_dispatch_id(description, tags)
                session_id, segment_id, sequence_number = extract_session_metadata(description, tags)
                if not dispatch_id and segment_id:
                    dispatch_id = segment_id
                if not dispatch_id and session_id:
                    seq_str = f"{sequence_number:04d}" if sequence_number is not None else "0001"
                    dispatch_id = f"{session_id}_seg_{seq_str}"

                # Check if this is a Dispatch upload:
                # Matches if dispatch_id or session_id is present, OR "[DISPATCH]" is in title, OR "dispatch" in tags
                is_dispatch = (
                    dispatch_id is not None
                    or session_id is not None
                    or "[dispatch]" in title.lower()
                    or any("dispatch" in t.lower() for t in tags)
                )

                if is_dispatch:
                    if not dispatch_id:
                        dispatch_id = f"dsp_{published_at[:10]}_{video_id[:6]}"

                    source_sha, source_sz = extract_source_hash_and_size(description, tags)
                    discovered.append({
                        "video_id": video_id,
                        "dispatch_id": dispatch_id,
                        "session_id": session_id,
                        "sequence_number": sequence_number,
                        "title": title,
                        "description": description,
                        "published_at": published_at,
                        "privacy_status": privacy,
                        "source_sha256": source_sha,
                        "source_file_size": source_sz
                    })

                    # Check if this video is already ingested/completed in the database
                    if not db.is_youtube_video_processed(video_id):
                        page_has_new_item = True

            # Early-stop optimization: If a full page was inspected and contained ONLY already-processed
            # videos, stop paging further back in history.
            if stop_on_known_page and not page_has_new_item and len(page_items) >= per_page:
                logger.debug("Early-stop: Page contained only already-ingested videos. Halting pagination.")
                break

            page_token = playlist_resp.get("nextPageToken")
            if not page_token:
                break

    except Exception as e:
        logger.warning("Error querying authenticated YouTube uploads: %s", e)

    return discovered


def get_youtube_processing_status(video_ids: List[str]) -> Dict[str, Dict[str, Any]]:
    """Query authoritative YouTube Data API v3 for uploadStatus, processingStatus,
    failure reasons, and content duration for a list of video IDs.

    Batches up to 50 video IDs per request to respect quota and network efficiency.
    Returns a dict mapping video_id to structured metadata:
        {
            "video_id": str,
            "upload_status": str,       # 'uploaded', 'processed', 'failed', 'rejected', 'deleted'
            "processing_status": str,   # 'processing', 'succeeded', 'failed', 'terminated', or None
            "failure_reason": str|None, # 'transcodingFailed', 'uploadFailed', 'other', or None
            "duration": str,            # ISO 8601 string e.g. 'PT56S', 'P0D'
            "title": str,
            "published_at": str,
            "privacy_status": str
        }
    """
    if not video_ids:
        return {}

    service = get_youtube_service()
    if not service:
        logger.debug("YouTube API service unavailable; skipping authoritative status query.")
        return {}

    results: Dict[str, Dict[str, Any]] = {}
    batch_size = 50

    for i in range(0, len(video_ids), batch_size):
        chunk = video_ids[i:i + batch_size]
        try:
            resp = service.videos().list(
                part="snippet,status,processingDetails,contentDetails",
                id=",".join(chunk)
            ).execute()

            for item in resp.get("items", []):
                vid = item.get("id")
                if not vid:
                    continue
                st = item.get("status", {})
                pr = item.get("processingDetails", {})
                cd = item.get("contentDetails", {})
                snip = item.get("snippet", {})

                results[vid] = {
                    "video_id": vid,
                    "upload_status": st.get("uploadStatus"),
                    "processing_status": pr.get("processingStatus"),
                    "failure_reason": pr.get("processingFailureReason") or st.get("rejectionReason"),
                    "duration": cd.get("duration"),
                    "title": snip.get("title", ""),
                    "published_at": snip.get("publishedAt", ""),
                    "privacy_status": st.get("privacyStatus", "unlisted")
                }
        except Exception as e:
            logger.warning("Error querying authoritative YouTube video status for batch %s: %s", chunk, e)

    return results

def parse_iso_datetime(ts_val: Any) -> Optional[datetime]:
    """Parse ISO-8601 timestamps supporting optional timezone and fractional seconds."""
    if not ts_val:
        return None
    s = str(ts_val).strip()
    if s.endswith("Z"):
        s = s[:-1] + "+00:00"
    try:
        dt = datetime.fromisoformat(s)
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        return dt
    except Exception:
        pass
    for fmt in ("%Y-%m-%d %H:%M:%S", "%Y-%m-%d %H:%M", "%Y-%m-%d"):
        try:
            return datetime.strptime(s, fmt).replace(tzinfo=timezone.utc)
        except ValueError:
            pass
    return None


class YouTubeInboxPoller:
    """Background daemon that monitors YouTube for Dispatch phone recordings."""

    def __init__(
        self,
        poll_interval_seconds: Optional[int] = None,
        catcher: Optional[YouTubeInboxCatcher] = None
    ):
        # Default 10 minutes (600 seconds)
        env_interval = os.getenv("DISPATCH_YOUTUBE_POLL_INTERVAL")
        default_val = int(env_interval) if env_interval and env_interval.isdigit() else 600
        self.poll_interval = poll_interval_seconds or default_val

        self.catcher = catcher or YouTubeInboxCatcher()
        self._stop_event = threading.Event()
        self._worker_thread: Optional[threading.Thread] = None
        self._lock = threading.Lock()
        self.last_poll_time: float = 0.0

    def start(self):
        """Start the background polling thread."""
        if self._worker_thread and self._worker_thread.is_alive():
            logger.warning("YouTube inbox poller is already running.")
            return

        self._stop_event.clear()
        self._worker_thread = threading.Thread(target=self._run_loop, name="YouTubePollerThread", daemon=True)
        self._worker_thread.start()
        logger.info("YouTube Cloud Inbox autonomous poller started (Poll interval: %ds / %.1f min)",
                    self.poll_interval, self.poll_interval / 60.0)

    def stop(self):
        """Signal poller to stop."""
        self._stop_event.set()
        if self._worker_thread:
            self._worker_thread.join(timeout=5)
            logger.info("YouTube Cloud Inbox poller stopped.")

    def _run_loop(self):
        """Main daemon loop. Runs immediately on boot, then waits poll_interval with backoff on failure."""
        # 1. IMMEDIATE RUN ON BOOT: Catch recordings uploaded while laptop was off
        logger.info("Laptop boot discovery: Checking YouTube for recordings created while laptop was asleep/off...")
        try:
            self.poll_once()
        except Exception as e:
            logger.error("Initial YouTube boot scan error: %s", e)

        # 2. PERIODIC RUN: Wake up every poll_interval with error backoff
        consecutive_errors = 0
        while not self._stop_event.is_set():
            sleep_duration = self.poll_interval
            if consecutive_errors > 0:
                backoff_extra = min(300, 30 * (2 ** min(consecutive_errors - 1, 4)))
                sleep_duration += backoff_extra
                logger.info("Poller backoff active (%d consecutive errors): waiting %ds before next poll",
                            consecutive_errors, sleep_duration)

            if self._stop_event.wait(timeout=sleep_duration):
                break

            try:
                self.poll_once()
                consecutive_errors = 0
            except Exception as e:
                consecutive_errors += 1
                logger.error("Error during scheduled YouTube inbox poll (failures=%d): %s",
                             consecutive_errors, e)

    def poll_once(self) -> List[Dict[str, Any]]:
        """Execute a single check against YouTube and process newly discovered or ready videos."""
        from datetime import datetime, timezone
        from dispatch.youtube_inbox.oauth import get_youtube_processing_status

        with self._lock:
            self.last_poll_time = time.time()
            discovered_videos: List[Dict[str, Any]] = []

            # 1. Fetch recent authenticated uploads
            try:
                auth_uploads = list_authenticated_user_uploads(max_results_per_page=15)
                for item in auth_uploads:
                    discovered_videos.append(item)
            except Exception as e:
                logger.warning("Authenticated YouTube upload check failed: %s", e)

            # 2. Recheck waiting and stalled videos so a delayed YouTube job can recover without re-uploading.
            try:
                waiting_in_db = []
                for status in ("WAITING_FOR_YOUTUBE_PROCESSING", "STALLED_YOUTUBE_PROCESSING"):
                    waiting_in_db.extend(db.list_youtube_inbox(status=status, limit=50))
                known_ids = {v["video_id"] for v in discovered_videos}
                for w in waiting_in_db:
                    vid = w["video_id"]
                    if vid not in known_ids:
                        discovered_videos.append({
                            "video_id": vid,
                            "title": w.get("title", f"Dispatch Video {vid}"),
                            "dispatch_id": w.get("dispatch_id"),
                            "session_id": w.get("session_id"),
                            "sequence_number": w.get("sequence_number"),
                            "published_at": w.get("upload_time") or w.get("discovered_at")
                        })
            except Exception as e:
                logger.warning("Error fetching waiting YouTube videos from DB: %s", e)

            if not discovered_videos:
                logger.debug("YouTube poll complete: 0 new uploads found.")
                return []

            # Filter candidates that are not already finished downstream
            candidate_items = [
                item for item in discovered_videos
                if not db.is_youtube_video_processed(item["video_id"])
            ]

            if not candidate_items:
                logger.debug("YouTube poll complete: All %d discovered videos already processed.", len(discovered_videos))
                return []

            logger.info("YouTube poll found %d candidate video(s) to verify status.", len(candidate_items))

            # 3. Batch query authoritative YouTube processing status
            candidate_ids = [item["video_id"] for item in candidate_items]
            status_map: Dict[str, Dict[str, Any]] = {}
            status_query_error: Optional[str] = None
            try:
                status_map = get_youtube_processing_status(candidate_ids)
            except Exception as e:
                status_query_error = str(e)
                logger.warning("Batch YouTube processing status check failed: %s", e)

                video_id = item["video_id"]
                title = item.get("title", f"Dispatch Video {video_id}")
                dispatch_id = item.get("dispatch_id")
                session_id = item.get("session_id")
                sequence_number = item.get("sequence_number")
                pub_time = item.get("published_at")

                existing = db.get_youtube_video(video_id)
                if not existing:
                    logger.info("Discovered NEW phone upload on YouTube: ID=%s, DispatchID=%s, SessionID=%s, Seq=%s, Title='%s'",
                                video_id, dispatch_id, session_id, sequence_number, title)
                    db.register_youtube_video(
                        video_id=video_id,
                        title=title,
                        upload_time=pub_time,
                        dispatch_id=dispatch_id,
                        session_id=session_id,
                        sequence_number=sequence_number
                    )
                    existing = db.get_youtube_video(video_id)

                st_info = status_map.get(video_id)

                if not st_info:
                    # A failed status query is not evidence that YouTube is still processing or
                    # that the source should be downloaded again. Keep the current durable state.
                    if existing and existing.get("status") != "FAILED":
                        query_detail = status_query_error or "YouTube returned no status for this video ID"
                        db.update_youtube_video(
                            video_id,
                            status=existing.get("status") or "DISCOVERED",
                            last_error=f"[YOUTUBE_STATUS_UNAVAILABLE] {query_detail}; will check again without re-uploading."
                        )
                    continue

                p_status = st_info.get("processing_status")
                u_status = st_info.get("upload_status")
                fail_reason = st_info.get("failure_reason")

                # Check terminal states before the broad uploaded/in-progress condition below.
                if p_status in ("failed", "terminated") or u_status in ("failed", "rejected", "deleted"):
                    err_msg = (
                        "[YOUTUBE_PROCESSING_FAILED] "
                        f"upload_status={u_status or 'unknown'}; processing_status={p_status or 'unknown'}; "
                        f"failure_reason={fail_reason or 'not provided'}; duration={st_info.get('duration') or 'unknown'}"
                    )
                    logger.error("Video %s failed processing on YouTube: %s", video_id, err_msg)
                    if not existing or existing.get("status") != "FAILED" or existing.get("last_error") != err_msg:
                        db.update_youtube_video(video_id, status="FAILED", last_error=err_msg)
                    continue

                if existing and existing.get("status") == "FAILED" and not (
                    p_status == "succeeded" or u_status == "processed"
                ):
                    # Do not turn a durable terminal failure into a generic waiting state
                    # when the API later omits or weakens its processing status.
                    continue

                # Case A: Video is still processing on YouTube
                if p_status == "processing" or (u_status == "uploaded" and p_status != "succeeded"):
                    # Calculate elapsed processing time
                    start_time_str = pub_time or (existing.get("discovered_at") if existing else None)
                    elapsed_min = 0.0
                    if start_time_str:
                        dt = parse_iso_datetime(start_time_str)
                        if dt:
                            now_utc = datetime.now(timezone.utc)
                            elapsed_sec = (now_utc - dt).total_seconds()
                            elapsed_min = max(0.0, elapsed_sec / 60.0)

                    if elapsed_min >= timeout_minutes:
                        warn_msg = (
                            f"[STALLED_YOUTUBE_PROCESSING] Processing exceeded {timeout_minutes}m "
                            f"({elapsed_min:.0f}m elapsed); upload_status={u_status or 'unknown'}, "
                            f"processing_status={p_status or 'unknown'}, failure_reason={fail_reason or 'not provided'}. "
                            "Check YouTube Studio or sync directly over LAN."
                        )
                        logger.warning(
                            "[STALLED] YouTube processing for %s (%s) has exceeded %d minutes (elapsed: %.1f min).",
                            video_id, title, timeout_minutes, elapsed_min
                        )
                        db.update_youtube_video(video_id, status="STALLED_YOUTUBE_PROCESSING", last_error=warn_msg)
                    else:
                        logger.info(
                            "Video %s is still processing on YouTube (%.1fm elapsed). Waiting for completion without downloading.",
                            video_id, elapsed_min
                        )
                        db.update_youtube_video(
                            video_id,
                            status="WAITING_FOR_YOUTUBE_PROCESSING",
                            last_error=(
                                f"[WAITING_FOR_YOUTUBE_PROCESSING] upload_status={u_status or 'unknown'}; "
                                f"processing_status={p_status or 'unknown'}; elapsed={elapsed_min:.0f}m; "
                                f"failure_reason={fail_reason or 'not provided'}"
                            )
                        )
                    continue

                # Case B: Video processing confirmed SUCCEEDED on YouTube
                if p_status == "succeeded" or u_status == "processed":
                    logger.info(
                        "YouTube video %s processing confirmed SUCCEEDED (Duration: %s). Proceeding to Pillar 1 media download.",
                        video_id, st_info.get("duration")
                    )
                    if existing:
                        db.update_youtube_video(video_id, status=existing.get("status") or "DISCOVERED", last_error="")
                else:
                    # Unknown API states are not permission to attempt a media download.
                    state = existing.get("status") if existing else "DISCOVERED"
                    db.update_youtube_video(
                        video_id,
                        status=state or "DISCOVERED",
                        last_error=(
                            f"[YOUTUBE_STATUS_UNAVAILABLE] Unrecognized upload_status={u_status or 'unknown'}; "
                            f"processing_status={p_status or 'unknown'}; retrying status check without re-uploading."
                        )
                    )
                    continue

                # Ingestion: Discover -> Download -> Validate -> DOWNLOADED
                try:
                    logger.info("Starting Pillar 1 Ingestion for YouTube video %s (%s, session %s)", video_id, dispatch_id, session_id)
                    result = self.catcher.ingest_video(
                        url_or_id=video_id,
                        dispatch_id=dispatch_id,
                        session_id=session_id,
                        sequence_number=sequence_number
                    )
                    processed_items.append(result)
                except Exception as e:
                    logger.error("Failed to ingest YouTube video %s: %s", video_id, e)
                    db.update_youtube_video(video_id, status="FAILED", last_error=str(e))

            return processed_items

"""WebVTT subtitle parser and normalizer for YouTube transcripts.
Converts raw .vtt caption files downloaded from YouTube into structured segment
dictionaries with start/end timestamps and clean text, stripping VTT markup and deduplicating
rolling cues.
"""
import re
import html
import logging
from pathlib import Path
from typing import List, Dict, Any, Optional

logger = logging.getLogger("dispatch.youtube_inbox.vtt")

# Regex patterns for VTT parsing
TIMESTAMP_PATTERN = re.compile(
    r"(?:(\d{1,2}):)?(\d{2}):(\d{2})[.,](\d{3})\s+-->\s+(?:(\d{1,2}):)?(\d{2}):(\d{2})[.,](\d{3})"
)
WORD_TIME_PATTERN = re.compile(r"<(\d{2}):(\d{2}):(\d{2})[.,](\d{3})>([^<]*)")
TAG_STRIP_PATTERN = re.compile(r"<[^>]+>")


def parse_timestamp_to_seconds(hours: Optional[str], minutes: str, seconds: str, millis: str) -> float:
    """Convert timestamp components into total seconds as a float."""
    h = int(hours) if hours else 0
    m = int(minutes)
    s = int(seconds)
    ms = int(millis)
    return round(h * 3600 + m * 60 + s + ms / 1000.0, 3)


def clean_vtt_text(text: str) -> str:
    """Clean cue text by stripping tags, unescaping HTML entities, and normalizing spaces."""
    cleaned = TAG_STRIP_PATTERN.sub("", text)
    cleaned = html.unescape(cleaned)
    # Remove cue positioning artifacts
    cleaned = re.sub(r"align:\S+|position:\S+|line:\S+|size:\S+", "", cleaned)
    cleaned = re.sub(r"\s+", " ", cleaned).strip()
    return cleaned


def parse_inline_words(raw_text: str, cue_start: float, cue_end: float) -> List[Dict[str, Any]]:
    """Parse exact inline word timestamps like `<00:00:01.200><c> word</c>` if present."""
    pattern = re.compile(r"<(\d{1,2}):(\d{2}):(\d{2})[.,](\d{3})>")
    parts = pattern.split(raw_text)
    if len(parts) <= 1:
        return []

    time_points = [cue_start]
    texts = [parts[0]]
    i = 1
    while i < len(parts):
        h, m, s, ms = parts[i:i+4]
        t = int(h) * 3600 + int(m) * 60 + int(s) + int(ms) / 1000.0
        time_points.append(round(t, 3))
        texts.append(parts[i+4])
        i += 5
    time_points.append(cue_end)

    words = []
    for idx, txt in enumerate(texts):
        cleaned = clean_vtt_text(txt)
        if not cleaned:
            continue
        w_start = time_points[idx]
        w_end = max(w_start + 0.05, time_points[idx+1])
        tokens = cleaned.split()
        if not tokens:
            continue
        dt = max(0.01, w_end - w_start) / len(tokens)
        for sub_i, tok in enumerate(tokens):
            words.append({
                "word": tok,
                "start": round(w_start + sub_i * dt, 3),
                "end": round(w_start + (sub_i + 1) * dt, 3),
                "probability": 1.0,
                "is_exact": True
            })
    return words


def parse_vtt_content(vtt_content: str) -> List[Dict[str, Any]]:
    """Parse WebVTT string content into structured segment dictionaries.
    
    Returns:
        List of segments: [{"start": float, "end": float, "text": str, "words": [...]}]
    """
    lines = vtt_content.splitlines()
    raw_cues = []
    current_start = None
    current_end = None
    current_text_lines = []

    for line in lines:
        line_str = line.strip()
        if not line_str or line_str.startswith("WEBVTT") or line_str.startswith("NOTE") or line_str.startswith("Kind:") or line_str.startswith("Language:"):
            continue

        # Check for timestamp line
        match = TIMESTAMP_PATTERN.search(line_str)
        if match:
            # If we already had a cue being collected, save it
            if current_start is not None and current_text_lines:
                raw_text = " ".join(current_text_lines)
                cleaned = clean_vtt_text(raw_text)
                if cleaned:
                    inline_words = parse_inline_words(raw_text, current_start, current_end)
                    raw_cues.append({
                        "start": current_start,
                        "end": current_end,
                        "text": cleaned,
                        "raw": raw_text,
                        "words": inline_words
                    })
                current_text_lines = []

            h1, m1, s1, ms1, h2, m2, s2, ms2 = match.groups()
            current_start = parse_timestamp_to_seconds(h1, m1, s1, ms1)
            current_end = parse_timestamp_to_seconds(h2, m2, s2, ms2)
        elif current_start is not None:
            current_text_lines.append(line_str)

    # Flush the last cue
    if current_start is not None and current_text_lines:
        raw_text = " ".join(current_text_lines)
        cleaned = clean_vtt_text(raw_text)
        if cleaned:
            inline_words = parse_inline_words(raw_text, current_start, current_end)
            raw_cues.append({
                "start": current_start,
                "end": current_end,
                "text": cleaned,
                "raw": raw_text,
                "words": inline_words
            })

    if not raw_cues:
        return []

    # Deduplicate rolling captions (YouTube auto-captions frequently repeat the previous line)
    deduped_cues: List[Dict[str, Any]] = []
    last_text = ""
    for cue in raw_cues:
        text = cue["text"]
        # Skip if exact duplicate
        if text == last_text:
            continue

        # If cue starts with previous cue text, keep only new words
        if last_text and text.startswith(last_text):
            remainder = text[len(last_text):].strip()
            if remainder:
                prefix_word_count = len(last_text.split())
                cue_words = cue.get("words", [])
                if len(cue_words) > prefix_word_count:
                    rem_words = cue_words[prefix_word_count:]
                else:
                    rem_words = [w for w in cue_words if w.get("word") in remainder.split()]
                if rem_words:
                    start_time = rem_words[0]["start"]
                else:
                    # In rolling cues without inline word timestamps, the remainder was spoken
                    # after the previous cue's end. Avoid setting start_time to cue["start"]
                    # which introduces a multi-second constant timing offset.
                    prev_end = deduped_cues[-1]["end"] if deduped_cues else cue["start"]
                    start_time = min(cue["end"] - 0.1, max(cue["start"], prev_end))
                deduped_cues.append({
                    "start": start_time,
                    "end": cue["end"],
                    "text": remainder,
                    "words": rem_words
                })
                last_text = text
            continue

        deduped_cues.append({
            "start": cue["start"],
            "end": cue["end"],
            "text": text,
            "words": cue["words"]
        })
        last_text = text

    # Merge cues: only merge if cues contain word timestamps.
    # For phrase-only cues without inline word timestamps, keep authentic cue boundaries intact.
    has_any_inline_words = any(bool(c.get("words")) for c in deduped_cues)
    merged_segments: List[Dict[str, Any]] = []

    if has_any_inline_words:
        current_seg: Optional[Dict[str, Any]] = None
        for cue in deduped_cues:
            if current_seg is None:
                current_seg = {
                    "start": cue["start"],
                    "end": cue["end"],
                    "text": cue["text"],
                    "words": list(cue.get("words", []))
                }
                continue

            gap = cue["start"] - current_seg["end"]
            duration = current_seg["end"] - current_seg["start"]

            # Merge if gap is small and cumulative duration is under 5 seconds
            if gap < 1.5 and duration < 5.0 and not current_seg["text"].endswith((".", "?", "!")):
                current_seg["end"] = cue["end"]
                current_seg["text"] = f"{current_seg['text']} {cue['text']}".strip()
                current_seg["words"].extend(cue.get("words", []))
            else:
                merged_segments.append(current_seg)
                current_seg = {
                    "start": cue["start"],
                    "end": cue["end"],
                    "text": cue["text"],
                    "words": list(cue.get("words", []))
                }

        if current_seg is not None:
            merged_segments.append(current_seg)
    else:
        # Preserve authentic phrase boundaries directly from deduped cues
        for cue in deduped_cues:
            merged_segments.append({
                "start": cue["start"],
                "end": cue["end"],
                "text": cue["text"],
                "words": []
            })

    # Build word boundaries for segments: keep exact words if authentic; do NOT fabricate linear words!
    for seg in merged_segments:
        text_words = seg["text"].split()
        if not text_words:
            seg["words"] = []
            seg["has_word_timestamps"] = False
            continue

        existing_words = seg.get("words", [])
        # If text words have corresponding exact words from inline VTT tags, preserve them
        if existing_words and all(w.get("is_exact") for w in existing_words):
            seg["has_word_timestamps"] = True
            seg["words"] = existing_words
        else:
            # Phrase-level only: do NOT divide segment duration evenly across words.
            # Preserve honest phrase boundaries with is_exact=False.
            seg["has_word_timestamps"] = False
            word_objs = []
            for w in text_words:
                word_objs.append({
                    "word": w,
                    "start": seg["start"],
                    "end": seg["end"],
                    "probability": 1.0,
                    "is_exact": False
                })
            seg["words"] = word_objs

    logger.info("Parsed %d raw VTT cues into %d clean speech segments", len(raw_cues), len(merged_segments))
    return merged_segments


def parse_vtt_file(vtt_path: Path) -> List[Dict[str, Any]]:
    """Read and parse a .vtt subtitle file from disk."""
    if not vtt_path.exists():
        raise FileNotFoundError(f"VTT file not found: {vtt_path}")

    with open(vtt_path, "r", encoding="utf-8", errors="replace") as f:
        content = f.read()

    return parse_vtt_content(content)

## Dispatch defaults — already decided; do not re-ask

1. Multi-tenant ownership: explicitly store user_id on each critical user-owned record. Back it with foreign keys, Supabase RLS, and server-side ownership checks. Explicit ownership helps avoid complicated inference and accidental cross-user leaks.

6. Processing: the owner's Windows PC does the expensive rendering for all invited users in the alpha. Jobs are durable and leased with heartbeat/retry/idempotency; jobs resume after PC restart/offline time. Never put Supabase service-role credentials on the PC.
7. Admin priority: a simple drag-and-drop user order is stored in the database. The PC chooses the highest-priority user with eligible work and processes that user's FIFO jobs. 8. YouTube-first: implement and validate YouTube publishing before secondary platforms. Preserve private workspace boundaries. Instagram, LinkedIn, X, and Facebook must not delay the first usable alpha.
9. Source ingestion: private URL ingestion is explicitly out of scope. Public/unlisted URLs do not automatically authorize downloading. Infrastructure boundary: no Docker start/stop/pulls, Supabase CLI initialization/link/reset/migration execution, live SQL/schema/RLS changes, Worker deployments, billing/resource creation, or Git push without explicit owner approval. MOBILE_STORAGE_DIR = Path(__file__).resolve().parent.parent / "storage" / "mobile"
MOBILE_STORAGE_DIR.mkdir(parents=True, exist_ok=True)
MOBILE_DB_PATH = MOBILE_STORAGE_DIR / "dispatch_mobile.db"


@contextlib.contextmanager
def get_mobile_db():
    """Context manager yielding a thread-safe connection to the mobile database with auto-commit."""
    conn = sqlite3.connect(str(MOBILE_DB_PATH), timeout=30.0)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode = WAL;")
    conn.execute("PRAGMA foreign_keys = ON;")
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def init_mobile_db():
    """Initialize mobile schema for sessions, segments, and outbox."""
    with get_mobile_db() as conn:
        cursor = conn.cursor()

        # Sessions Table
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS mobile_sessions (
                session_id TEXT PRIMARY KEY,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                status TEXT NOT NULL DEFAULT 'RECORDING', -- RECORDING, FINALIZED, ABANDONED
                total_segments INTEGER DEFAULT 0,
                notes TEXT
            );
        """)

        # Segments Table
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS mobile_segments (
                segment_id TEXT PRIMARY KEY,
                session_id TEXT NOT NULL,
                sequence_number INTEGER NOT NULL,
                filename TEXT NOT NULL,
                filepath TEXT NOT NULL,
                file_size_bytes INTEGER DEFAULT 0,
                sha256_hash TEXT,
                status TEXT NOT NULL DEFAULT 'RECORDING',
                -- Status states: RECORDING, FINALIZED, QUEUED_FOR_UPLOAD, UPLOADING, UPLOADED, VERIFIED_BY_LAPTOP, DELETED_FROM_PHONE
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                finalized_at TIMESTAMP,
                verified_at TIMESTAMP,
                FOREIGN KEY (session_id) REFERENCES mobile_sessions (session_id) ON DELETE CASCADE
            );
        """)

        # Upload Outbox Table
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS mobile_outbox (
                segment_id TEXT PRIMARY KEY,
                session_id TEXT NOT NULL,
                remote_offset INTEGER DEFAULT 0,
                attempt_count INTEGER DEFAULT 0,
                next_retry_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                last_error TEXT,
                status TEXT NOT NULL DEFAULT 'QUEUED_FOR_UPLOAD',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (segment_id) REFERENCES mobile_segments (segment_id) ON DELETE CASCADE
            );
        """)

        # Settings Table
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS mobile_settings (
                key TEXT PRIMARY KEY,
                value TEXT NOT NULL
            );
        """)

        cursor.execute("""
            INSERT OR IGNORE INTO mobile_settings (key, value) VALUES
            ('cleanup_policy', 'DELETE_AFTER_VERIFIED_COPY'),
            ('network_policy', 'ANY_NETWORK'),
            ('laptop_url', 'http://127.0.0.1:8000'),
            ('auth_token', 'dispatch_paired_secret_default');
        """)


def calculate_sha256(filepath: Path, block_size: int = 65536) -> str:
    """Calculate SHA-256 hash for strict end-to-end media verification."""
    hasher = hashlib.sha256()
    with open(filepath, "rb") as f:
        for chunk in iter(lambda: f.read(block_size), b""):
            hasher.update(chunk)
    return hasher.hexdigest()
