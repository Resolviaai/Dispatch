"""Canonical Caption Exporter Engine.
Generates preview/render-parity outputs from the Canonical CaptionTrack:
- ASS (Advanced SubStation Alpha) with frame-accurate active-word pop bounce
- Standard SRT
- WebVTT
- JSON
"""
import json
import math
from pathlib import Path
from typing import Dict, Any, List, Optional
from dispatch.captions.model import CaptionTrack, CaptionGroup, Word
from dispatch.config import TARGET_WIDTH, TARGET_HEIGHT


def float_to_ass_alpha(opacity: Any) -> str:
    """Convert float opacity [0.0 - 1.0] (1.0=opaque, 0.0=transparent) to ASS 2-digit hex alpha (00=opaque, FF=transparent)."""
    if opacity is None:
        return "00"
    try:
        val = float(opacity)
    except (ValueError, TypeError):
        return "00"
    clamped = max(0.0, min(1.0, val))
    ass_alpha_val = int(round((1.0 - clamped) * 255))
    return f"{ass_alpha_val:02X}"


def apply_case_transform(text: str, mode: str) -> str:
    """Apply non-destructive case transform to text for preview and export parity."""
    if not text:
        return ""
    if mode == "uppercase":
        return text.upper()
    if mode == "lowercase":
        return text.lower()
    if mode == "title":
        return " ".join(part.capitalize() for part in text.split(" "))
    return text


def hex_to_ass_color(hex_color: str, alpha: str = "00") -> str:
    """Convert standard Hex color (#RRGGBB) to ASS color format (&HAABBGGRR&)."""
    clean_hex = hex_color.lstrip("#")
    if len(clean_hex) == 6:
        r = clean_hex[0:2]
        g = clean_hex[2:4]
        b = clean_hex[4:6]
        return f"&H{alpha}{b}{g}{r}&"
    return f"&H{alpha}FFFFFF&"


def format_ass_time(seconds: float) -> str:
    """Format seconds into ASS timestamp: H:MM:SS.cs"""
    hrs = int(seconds // 3600)
    mins = int((seconds % 3600) // 60)
    secs = int(seconds % 60)
    centis = int(round((seconds - int(seconds)) * 100))
    if centis >= 100:
        centis = 99
    return f"{hrs}:{mins:02d}:{secs:02d}.{centis:02d}"


def format_srt_time(seconds: float) -> str:
    """Format seconds into SRT timestamp: HH:MM:SS,mmm"""
    hrs = int(seconds // 3600)
    mins = int((seconds % 3600) // 60)
    secs = int(seconds % 60)
    millis = int(round((seconds - int(seconds)) * 1000))
    if millis >= 1000:
        millis = 999
    return f"{hrs:02d}:{mins:02d}:{secs:02d},{millis:03d}"


def get_ass_weight_tag(weight: Any) -> str:
    """Map font weight names or numbers to ASS \\b tag."""
    if weight is None:
        return r"\b1"
    s = str(weight).strip().lower()
    if s in ("bold", "700"):
        return r"\b700"
    if s in ("bolder", "800", "heavy", "extrabold"):
        return r"\b800"
    if s in ("black", "900"):
        return r"\b900"
    if s in ("semibold", "600"):
        return r"\b600"
    if s in ("medium", "500"):
        return r"\b500"
    if s in ("normal", "regular", "400"):
        return r"\b400"
    if s in ("light", "300"):
        return r"\b300"
    if s.isdigit():
        return f"\\b{s}"
    return r"\b1"


def export_to_ass(track: CaptionTrack, output_path: Path) -> Path:
    """Export canonical CaptionTrack to ASS with exact parity for fonts, colors, and pop animations."""
    style = track.default_style
    layout = track.default_layout
    anim = track.default_animation

    primary_alpha = float_to_ass_alpha(getattr(style, "text_opacity", getattr(style, "opacity", 1.0)))
    outline_alpha = float_to_ass_alpha(getattr(style, "outline_opacity", 1.0))
    shadow_alpha = float_to_ass_alpha(getattr(style, "shadow_opacity", 1.0))
    bg_alpha = float_to_ass_alpha(getattr(style, "background_opacity", 1.0))

    primary_ass = hex_to_ass_color(style.primary_color, primary_alpha)
    highlight_ass = hex_to_ass_color(style.highlight_color, primary_alpha)
    outline_ass = hex_to_ass_color(style.outline_color, outline_alpha)
    bg_ass = hex_to_ass_color(getattr(style, "background_color", "#000000") or "#000000", bg_alpha)
    shadow_ass = hex_to_ass_color(getattr(style, "shadow_color", "#000000") or "#000000", shadow_alpha)

    # Resolution-independent to 1080x1920 vertical coordinate space
    margin_v = int(round((1.0 - layout.position_y) * TARGET_HEIGHT))
    alignment = 2 if layout.alignment == "center" else (1 if layout.alignment == "left" else 3)

    outline_w = style.outline_width if style.outline_enabled else 0.0
    shadow_val = style.shadow_offset_y if style.shadow_enabled else 0.0
    bg_pad = max(4.0, float(getattr(style, "background_padding_y", 8.0) or 8.0))

    header = f"""[Script Info]
ScriptType: v4.00+
PlayResX: {TARGET_WIDTH}
PlayResY: {TARGET_HEIGHT}
WrapStyle: 0
ScaledBorderAndShadow: yes

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding
Style: Default,{style.font_family},{style.font_size},{primary_ass},&H000000FF,{outline_ass},{shadow_ass},-1,0,0,0,100,100,{style.letter_spacing},0,1,{outline_w},{shadow_val},{alignment},40,40,{margin_v},1
Style: Boxed,{style.font_family},{style.font_size},{primary_ass},&H000000FF,{bg_ass},{shadow_ass},-1,0,0,0,100,100,{style.letter_spacing},0,3,{bg_pad},{shadow_val},{alignment},40,40,{margin_v},1

[Events]
Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text
"""

    events = []
    for grp in track.groups:
        grp_style = grp.style or style
        grp_anim = grp.animation or anim
        grp_layout = grp.layout or layout
        words = grp.words

        grp_type = getattr(grp, "type", "caption") or "caption"
        if grp_type in ("audio", "voiceover", "overlay"):
            continue
        if getattr(grp, "is_hidden", False):
            continue

        if not words:
            continue

        bg_mode = getattr(grp_style, "background_mode", None)
        if bg_mode is None:
            is_boxed = bool(getattr(grp_style, "background_enabled", False))
            bg_mode = "block" if is_boxed else "none"
        else:
            is_boxed = bg_mode in ("line", "block") and bool(getattr(grp_style, "background_enabled", True))

        style_name = "Boxed" if is_boxed else "Default"

        # Position and transform tags for exact canvas parity
        pos_x = int(round((grp_layout.position_x or 0.5) * TARGET_WIDTH))
        pos_y = int(round((grp_layout.position_y or 0.78) * TARGET_HEIGHT))
        base_scale = int(round((grp_layout.scale or 1.0) * 100))
        an_tag = r"\an5" if grp_layout.alignment == "center" else (r"\an4" if grp_layout.alignment == "left" else r"\an6")

        # Apply background offsets if boxed
        bg_off_x = int(round(float(getattr(grp_style, "background_offset_x", 0) or 0)))
        bg_off_y = int(round(float(getattr(grp_style, "background_offset_y", 0) or 0)))
        if is_boxed and (bg_off_x != 0 or bg_off_y != 0):
            pos_x += bg_off_x
            pos_y += bg_off_y

        grp_text_alpha = float_to_ass_alpha(getattr(grp_style, "text_opacity", getattr(grp_style, "opacity", 1.0)))
        grp_outline_alpha = float_to_ass_alpha(getattr(grp_style, "outline_opacity", 1.0))
        grp_shadow_alpha = float_to_ass_alpha(getattr(grp_style, "shadow_opacity", 1.0))
        grp_bg_alpha = float_to_ass_alpha(getattr(grp_style, "background_opacity", 1.0))

        grp_primary_ass = hex_to_ass_color(grp_style.primary_color, grp_text_alpha)
        grp_highlight_ass = hex_to_ass_color(grp_style.highlight_color, grp_text_alpha)
        grp_outline_ass = hex_to_ass_color(grp_style.outline_color or "#000000", grp_outline_alpha)
        grp_outline_w = grp_style.outline_width if grp_style.outline_enabled else 0.0
        grp_shadow_ass = hex_to_ass_color(grp_style.shadow_color or "#000000", grp_shadow_alpha)

        # Shadow distance and angle calculations
        shadow_dist = getattr(grp_style, "shadow_distance", None)
        shadow_angle = getattr(grp_style, "shadow_angle", None)
        if shadow_dist is not None and shadow_angle is not None:
            rad = math.radians(float(shadow_angle))
            shadow_x = round(float(shadow_dist) * math.cos(rad), 1)
            shadow_y = round(float(shadow_dist) * math.sin(rad), 1)
        else:
            shadow_x = float(getattr(grp_style, "shadow_offset_x", 2.0) or 2.0)
            shadow_y = float(getattr(grp_style, "shadow_offset_y", 2.0) or 2.0)

        shad_tag = f"\\xshad{shadow_x}\\yshad{shadow_y}" if grp_style.shadow_enabled else "\\shad0"

        # Explicit font styling tags matching preview canvas
        font_fn = f"\\fn{grp_style.font_family}" if grp_style.font_family else ""
        font_fs = f"\\fs{int(round(grp_style.font_size or 54))}" if grp_style.font_size else ""
        font_b = get_ass_weight_tag(grp_style.font_weight)
        font_i = r"\i1" if getattr(grp_style, "font_style", "normal") == "italic" else r"\i0"
        font_u = r"\u1" if getattr(grp_style, "underline", False) else r"\u0"
        font_fsp = f"\\fsp{grp_style.letter_spacing}" if grp_style.letter_spacing is not None else ""
        font_tags = f"{font_fn}{font_fs}{font_b}{font_i}{font_u}{font_fsp}"

        # Animation evaluation for ASS tags
        enter_obj = getattr(grp_anim, "enter", None)
        enter_type = enter_obj.get("type", "none") if isinstance(enter_obj, dict) else getattr(enter_obj, "type", getattr(grp_anim, "enter_type", "none"))
        enter_dur_ms = int(enter_obj.get("duration_ms", 200)) if isinstance(enter_obj, dict) else int(getattr(enter_obj, "duration_ms", getattr(grp_anim, "enter_duration_ms", 200)))

        exit_obj = getattr(grp_anim, "exit", None)
        exit_type = exit_obj.get("type", "none") if isinstance(exit_obj, dict) else getattr(exit_obj, "type", getattr(grp_anim, "exit_type", "none"))
        exit_dur_ms = int(exit_obj.get("duration_ms", 150)) if isinstance(exit_obj, dict) else int(getattr(exit_obj, "duration_ms", getattr(grp_anim, "exit_duration_ms", 150)))

        loop_obj = getattr(grp_anim, "loop", None)
        loop_type = loop_obj.get("type", "none") if isinstance(loop_obj, dict) else getattr(loop_obj, "type", "none")
        loop_speed_ms = int(loop_obj.get("speed_ms", 1000)) if isinstance(loop_obj, dict) else int(getattr(loop_obj, "speed_ms", 1000))

        total_card_dur_ms = max(100, int(round((grp.end - grp.start) * 1000)))

        # Build ASS animation tags
        anim_tags = ""
        pos_tag = f"\\pos({pos_x},{pos_y})"

        # 1. Entrance Animation
        fad_in = enter_dur_ms if enter_type == "fade" else 0
        fad_out = exit_dur_ms if exit_type == "fade" else 0
        if fad_in > 0 or fad_out > 0:
            anim_tags += f"\\fad({fad_in},{fad_out})"

        if enter_type == "pop":
            pop_dur = max(30, enter_dur_ms)
            half_in = max(15, pop_dur // 2)
            pop_start_scale = max(10, int(round(base_scale * 0.4)))
            pop_peak_scale = int(round(base_scale * 1.15))
            anim_tags += f"\\fscx{pop_start_scale}\\fscy{pop_start_scale}\\alpha&HFF&\\t(0,{half_in},\\alpha&H00&\\fscx{pop_peak_scale}\\fscy{pop_peak_scale})\\t({half_in},{pop_dur},\\fscx{base_scale}\\fscy{base_scale})"
        elif enter_type == "slide_up":
            slide_dur = max(30, enter_dur_ms)
            slide_offset = 35
            pos_tag = f"\\move({pos_x},{pos_y + slide_offset},{pos_x},{pos_y},0,{slide_dur})"
            if fad_in == 0:
                anim_tags += f"\\fad({slide_dur},0)"

        # 2. Exit Animation
        exit_start_ms = max(0, total_card_dur_ms - exit_dur_ms)
        if exit_type == "scale_down":
            anim_tags += f"\\t({exit_start_ms},{total_card_dur_ms},\\fscx{int(base_scale * 0.4)}\\fscy{int(base_scale * 0.4)}\\alpha&HFF&)"
        elif exit_type == "slide_down":
            if enter_type != "slide_up":
                pos_tag = f"\\move({pos_x},{pos_y},{pos_x},{pos_y + 35},{exit_start_ms},{total_card_dur_ms})"
            if fad_out == 0:
                anim_tags += f"\\fad(0,{exit_dur_ms})"

        # 3. Loop Animation
        if loop_type == "pulse":
            p_scale = int(round(base_scale * 1.15))
            h_ms = max(50, loop_speed_ms // 2)
            anim_tags += f"\\t(0,{h_ms},\\fscx{p_scale}\\fscy{p_scale})\\t({h_ms},{loop_speed_ms},\\fscx{base_scale}\\fscy{base_scale})"
        elif loop_type == "wiggle":
            w_ms = max(30, loop_speed_ms // 4)
            anim_tags += f"\\t(0,{w_ms},\\frz-6)\\t({w_ms},{3 * w_ms},\\frz6)\\t({3 * w_ms},{loop_speed_ms},\\frz0)"

        if is_boxed:
            grp_bg_ass = hex_to_ass_color(getattr(grp_style, "background_color", "#000000") or "#000000", grp_bg_alpha)
            pad_y = float(getattr(grp_style, "background_padding_y", 8.0) or 8.0)
            pad_x = float(getattr(grp_style, "background_padding_x", 12.0) or 12.0)
            grp_bg_pad = max(2.0, max(pad_y, pad_x))
            style_prefix = f"\\3c{grp_bg_ass}\\bord{grp_bg_pad}{shad_tag}\\4c{grp_shadow_ass}"
        else:
            style_prefix = f"\\3c{grp_outline_ass}\\bord{grp_outline_w}{shad_tag}\\4c{grp_shadow_ass}"

        scale_tag = f"\\fscx{base_scale}\\fscy{base_scale}" if enter_type != "pop" else ""
        pos_prefix = "{" + f"{pos_tag}{an_tag}{scale_tag}{font_tags}{anim_tags}{style_prefix}" + "}"

        # Glow layer parameters with matching font tags
        glow_enabled = bool(getattr(grp_style, "glow_enabled", False))
        if getattr(grp_style, "glow_opacity", None) is not None and grp_style.glow_opacity < 0.999:
            glow_alpha = float_to_ass_alpha(float(grp_style.glow_opacity) * (1.0 - 0x35 / 255.0))
        else:
            glow_alpha = "35"
        glow_ass = hex_to_ass_color(getattr(grp_style, "glow_color", "#00F0FF") or "#00F0FF")
        glow_blur = int(round(float(getattr(grp_style, "glow_blur", 12.0) or 12.0)))
        glow_spread = max(3, int(round(glow_blur * 0.45)))
        glow_prefix = "{" + f"{pos_tag}{an_tag}{scale_tag}{font_tags}{anim_tags}\\c{glow_ass}\\3c{glow_ass}\\bord{glow_spread}\\blur{glow_blur}\\shad0\\alpha&H{glow_alpha}&" + "}"

        # Determine visual line breaks matching frontend linePartition.ts
        line_break_indices = set()
        running_w = 0.0
        eff_font_size = float(grp_style.font_size or 54.0)
        eff_letter_spacing = float(grp_style.letter_spacing or 0.0)
        space_w = eff_font_size * 0.35
        max_content_w = float(grp_layout.max_width or 0.88) * TARGET_WIDTH

        for w_idx, w_item in enumerate(words):
            raw_w_text = w_item.text or ""
            if "\n" in raw_w_text and w_idx > 0:
                line_break_indices.add(w_idx)
                running_w = 0.0
            
            clean_token = raw_w_text.replace("\n", " ").strip()
            w_w = len(clean_token) * eff_font_size * 0.55 + eff_letter_spacing * len(clean_token)
            if running_w > 0 and (running_w + w_w > max_content_w):
                line_break_indices.add(w_idx)
                running_w = w_w + space_w
            else:
                running_w += w_w + space_w

        grp_type = getattr(grp, "type", "caption") or "caption"
        is_caption = grp_type == "caption"
        highlight_enabled = is_caption and bool(getattr(grp_style, "highlight_enabled", True))

        has_word_timing = is_caption and getattr(grp, "has_word_timing", True) and getattr(grp, "timing_mode", "word") == "word"
        if has_word_timing:
            if not any(getattr(w, "is_exact", True) for w in words):
                has_word_timing = False
            elif len(words) > 1 and all(w.start == words[0].start and w.end == words[0].end for w in words):
                has_word_timing = False

        if not has_word_timing:
            # Phrase-level dialogue: render entire phrase across card's start to end without fake karaoke
            phrase_start = format_ass_time(grp.start)
            phrase_end = format_ass_time(grp.end)
            phrase_tokens = []
            for w_k, w in enumerate(words):
                if w_k in line_break_indices and w_k > 0:
                    phrase_tokens.append(r"\N")
                w_text = apply_case_transform(w.text, getattr(grp_style, "text_transform", "none"))
                if w_text:
                    phrase_tokens.append(w_text)
            card_text = " ".join(phrase_tokens).replace(r" \N ", r"\N").replace(r" \N", r"\N").replace(r"\N ", r"\N")

            if glow_enabled:
                # Layer 0: Background Gaussian Glow
                events.append(f"Dialogue: 0,{phrase_start},{phrase_end},{style_name},,0,0,0,,{glow_prefix}{card_text}")
                # Layer 1: Foreground Crisp Text
                events.append(f"Dialogue: 1,{phrase_start},{phrase_end},{style_name},,0,0,0,,{pos_prefix}{{\\c{grp_primary_ass}}}{card_text}")
            else:
                events.append(f"Dialogue: 0,{phrase_start},{phrase_end},{style_name},,0,0,0,,{pos_prefix}{{\\c{grp_primary_ass}}}{card_text}")
            continue

        # Generate word-by-word karaoke illumination within this card preserving authentic word boundaries
        for j, target_word in enumerate(words):
            w_start = target_word.start
            w_end = target_word.end
            if w_end <= w_start:
                if j < len(words) - 1:
                    w_end = max(w_start + 0.10, words[j + 1].start)
                else:
                    w_end = max(w_start + 0.10, grp.end)

            if w_end <= w_start:
                continue

            rel_start = format_ass_time(w_start)
            rel_end = format_ass_time(w_end)

            card_tokens = []
            glow_tokens = []
            for k, w in enumerate(words):
                if k in line_break_indices and k > 0:
                    card_tokens.append(r"\N")
                    glow_tokens.append(r"\N")
                w_text = apply_case_transform(w.text, getattr(grp_style, "text_transform", "none"))
                if not w_text:
                    continue

                glow_tokens.append(w_text)
                override = getattr(w, "style_override", None)
                w_primary_ass = hex_to_ass_color(override.color, grp_text_alpha) if (override and override.color) else grp_primary_ass
                if highlight_enabled:
                    w_highlight_ass = hex_to_ass_color(override.highlight_color, grp_text_alpha) if (override and override.highlight_color) else grp_highlight_ass
                else:
                    w_highlight_ass = w_primary_ass
                w_scale_mult = float(override.scale) if (override and override.scale) else 1.0
                effective_word_scale = int(round(base_scale * w_scale_mult))
                w_weight_tag = get_ass_weight_tag(override.font_weight) if (override and override.font_weight) else ""
                w_u_tag = (r"\u1" if override.underline else r"\u0") if (override and override.underline is not None) else ""

                word_anim_type = "pop"
                active_scale_mult = 1.16
                active_dur = 100
                if hasattr(grp_anim, "active_word") and grp_anim.active_word:
                    if isinstance(grp_anim.active_word, dict):
                        word_anim_type = grp_anim.active_word.get("type", "pop")
                        active_scale_mult = float(grp_anim.active_word.get("active_scale", 1.16))
                        active_dur = int(grp_anim.active_word.get("duration_ms", 100))
                    else:
                        word_anim_type = getattr(grp_anim.active_word, "type", "pop")
                        active_scale_mult = float(getattr(grp_anim.active_word, "active_scale", 1.16))
                        active_dur = int(getattr(grp_anim.active_word, "duration_ms", 100))
                elif hasattr(grp_anim, "type") and grp_anim.type:
                    word_anim_type = grp_anim.type
                    active_scale_mult = float(getattr(grp_anim, "active_scale", 1.16))
                    active_dur = int(getattr(grp_anim, "duration_ms", 100))

                if k == j:
                    # Active spoken word with scale pop bounce or color highlight
                    if word_anim_type in ("pop", "bounce"):
                        punch_pct = int(round(effective_word_scale * active_scale_mult))
                        anim_tag = f"{{\\c{w_highlight_ass}{w_weight_tag}{w_u_tag}\\fscx{punch_pct}\\fscy{punch_pct}\\t(0,{active_dur},\\fscx{effective_word_scale}\\fscy{effective_word_scale})}}"
                    else:
                        anim_tag = f"{{\\c{w_highlight_ass}{w_weight_tag}{w_u_tag}\\fscx{effective_word_scale}\\fscy{effective_word_scale}}}"

                    card_tokens.append(f"{anim_tag}{w_text}{{\\c{w_primary_ass}{font_b}{font_u}\\fscx{base_scale}\\fscy{base_scale}}}")
                else:
                    if w_primary_ass != grp_primary_ass or effective_word_scale != base_scale or w_weight_tag or w_u_tag:
                        card_tokens.append(f"{{\\c{w_primary_ass}{w_weight_tag}{w_u_tag}\\fscx{effective_word_scale}\\fscy{effective_word_scale}}}{w_text}{{\\c{grp_primary_ass}{font_b}{font_u}\\fscx{base_scale}\\fscy{base_scale}}}")
                    else:
                        card_tokens.append(w_text)

            card_str = " ".join(card_tokens).replace(r" \N ", r"\N").replace(r" \N", r"\N").replace(r"\N ", r"\N")
            glow_str = " ".join(glow_tokens).replace(r" \N ", r"\N").replace(r" \N", r"\N").replace(r"\N ", r"\N")

            if glow_enabled:
                # Layer 0: Background Gaussian Glow
                glow_line = glow_prefix + glow_str
                events.append(f"Dialogue: 0,{rel_start},{rel_end},{style_name},,0,0,0,,{glow_line}")
                # Layer 1: Foreground Illuminated Text
                line_text = pos_prefix + card_str
                events.append(f"Dialogue: 1,{rel_start},{rel_end},{style_name},,0,0,0,,{line_text}")
            else:
                line_text = pos_prefix + card_str
                events.append(f"Dialogue: 0,{rel_start},{rel_end},{style_name},,0,0,0,,{line_text}")

    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, "w", encoding="utf-8") as f:
        f.write(header)
        f.write("\n".join(events))
        f.write("\n")

    return output_path


def export_to_srt(track: CaptionTrack, output_path: Path) -> Path:
    """Export canonical CaptionTrack to standard SRT."""
    lines = []
    counter = 1
    for grp in track.groups:
        grp_type = getattr(grp, "type", "caption") or "caption"
        if grp_type in ("audio", "voiceover", "overlay") or getattr(grp, "is_hidden", False):
            continue
        start_ts = format_srt_time(grp.start)
        end_ts = format_srt_time(grp.end)
        text = " ".join(w.text for w in grp.words)
        lines.append(f"{counter}")
        lines.append(f"{start_ts} --> {end_ts}")
        lines.append(text)
        lines.append("")
        counter += 1

    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, "w", encoding="utf-8") as f:
        f.write("\n".join(lines))
    return output_path

"""Deterministic Word-to-Caption Grouping Engine.
Translates raw word-level speech transcripts into short-form, high-retention CaptionGroups.
Enforces monotonic timings, natural pause boundaries, and word limits.
"""
from typing import List, Dict, Any, Optional
from dispatch.captions.model import Word, CaptionGroup, CaptionTrack, StyleConfig, LayoutConfig, AnimationConfig
from dispatch.captions.presets import get_preset


def group_words_into_captions(
    raw_segments: List[Dict[str, Any]],
    clip_start: float = 0.0,
    clip_end: float = float("inf"),
    clip_id: str = "",
    words_per_group: int = 3,
    max_chars_per_line: int = 24,
    max_pause_threshold: float = 0.45,  # Seconds of silence that forces a new card
    preset_id: str = "yellow_pop"
) -> CaptionTrack:
    """Deterministically partition word timestamps into rapid reading CaptionGroups.

    Args:
        raw_segments: List of segments with word timestamps [{"word": "Hi", "start": 0.0, "end": 0.3}, ...]
        clip_start: Start boundary relative to source video in seconds.
        clip_end: End boundary relative to source video in seconds.
        clip_id: ID of the clip.
        words_per_group: Maximum words per visual card (2-4 recommended for mobile).
        max_chars_per_line: Character wrapping budget.
        max_pause_threshold: Silence gap that triggers a new card boundary.
        preset_id: Styling preset to attach by default.

    Returns:
        Canonical CaptionTrack object.
    """
    preset_data = get_preset(preset_id)
    default_style: StyleConfig = preset_data["style"]
    default_layout: LayoutConfig = preset_data["layout"]
    default_anim: AnimationConfig = preset_data["animation"]

    # 1. Process segments: preserve real word timestamps if available; otherwise maintain phrase-level truth
    groups: List[CaptionGroup] = []
    current_words: List[Word] = []
    current_chars = 0
    prev_end = 0.0

    def flush_current_words():
        nonlocal current_words, current_chars, prev_end
        if not current_words:
            return
        grp = CaptionGroup(
            start=current_words[0].start,
            end=current_words[-1].end,
            words=list(current_words),
            layout=default_layout,
            style=default_style,
            animation=default_anim,
            has_word_timing=True,
            timing_mode="word"
        )
        groups.append(grp)
        current_words = []
        current_chars = 0

    for seg in raw_segments:
        seg_start = float(seg.get("start", 0.0))
        seg_end = float(seg.get("end", 0.0))

        if seg_end <= clip_start or seg_start >= clip_end:
            continue

        words = seg.get("words", [])
        has_exact_words = bool(words) and any(w.get("is_exact", True) for w in words)

        if has_exact_words:
            for w in words:
                w_start = float(w.get("start", 0.0))
                w_end = float(w.get("end", 0.0))
                text = str(w.get("word", "")).strip()
                if not text:
                    continue
                if w_end <= clip_start or w_start >= clip_end:
                    continue

                # Clamp to clip range and normalize relative to clip start exactly once
                rel_start = max(0.0, w_start - clip_start)
                rel_end = max(rel_start + 0.05, min(clip_end - clip_start, w_end - clip_start))

                w_obj = Word(
                    text=text,
                    start=round(rel_start, 3),
                    end=round(rel_end, 3),
                    confidence=float(w.get("confidence", 1.0)),
                    is_exact=bool(w.get("is_exact", True))
                )

                pause = w_obj.start - prev_end if prev_end > 0 else 0.0
                should_split = False

                # Split condition 1: Exceeded words per group (default 2-4 words)
                if len(current_words) >= words_per_group:
                    should_split = True
                # Split condition 2: Previous word ended with punctuation (. ? !)
                elif current_words and current_words[-1].text.endswith((".", "?", "!")):
                    should_split = True
                # Split condition 3: Previous word ended with comma/semicolon and group has >= 2 words
                elif current_words and current_words[-1].text.endswith((",", ";", ":", "—")) and len(current_words) >= 2:
                    should_split = True
                # Split condition 4: Exceeded character budget
                elif current_chars + len(w_obj.text) + 1 > max_chars_per_line and len(current_words) >= 2:
                    should_split = True
                # Split condition 5: Natural pause in speech
                elif pause >= max_pause_threshold and len(current_words) > 0:
                    should_split = True

                if should_split:
                    flush_current_words()

                current_words.append(w_obj)
                current_chars += len(w_obj.text) + 1
                prev_end = w_obj.end
        else:
            # Phrase-level segment (no inline word timestamps):
            # Flush any pending word-timed cards first
            flush_current_words()

            seg_text = seg.get("text", "").strip()
            if not seg_text:
                continue

            rel_seg_start = max(0.0, seg_start - clip_start)
            rel_seg_end = max(rel_seg_start + 0.2, min(clip_end - clip_start, seg_end - clip_start))

            # Render phrase using its authentic start and end times without fabricated linear subdivisions
            tokens = seg_text.split()
            phrase_words = [
                Word(
                    text=tok,
                    start=round(rel_seg_start, 3),
                    end=round(rel_seg_end, 3),
                    confidence=float(seg.get("confidence", 0.95)),
                    is_exact=False
                )
                for tok in tokens
            ]

            grp = CaptionGroup(
                start=round(rel_seg_start, 3),
                end=round(rel_seg_end, 3),
                words=phrase_words,
                layout=default_layout,
                style=default_style,
                animation=default_anim,
                has_word_timing=False,
                timing_mode="phrase"
            )
            groups.append(grp)
            prev_end = rel_seg_end

    # Flush any remaining word-timed group
    flush_current_words()

    # Sort groups by start time
    groups.sort(key=lambda g: g.start)

    # 3. Post-process to eliminate micro-gaps for smooth card transitions
    for idx, grp in enumerate(groups):
        if idx < len(groups) - 1:
            next_start = groups[idx + 1].start
            # If gap between cards is small (<0.25s), stretch card display end so text doesn't flicker out.
            # IMPORTANT: Keep source word timestamps strictly intact! Do NOT alter grp.words[-1].end.
            if next_start - grp.end < 0.25 and next_start > grp.end:
                grp.end = next_start

    return CaptionTrack(
        clip_id=clip_id,
        language="en",
        groups=groups,
        default_style=default_style,
        default_layout=default_layout,
        default_animation=default_anim
    )

"""Canonical Caption Domain Model for Dispatch.
Standardized JSON-serializable domain representation following CapForge and OpenCut architecture.
Serves as the single source of truth across preview, editing, persistence, and export.
"""
from dataclasses import dataclass, field, asdict, fields
from typing import List, Dict, Any, Optional
import uuid


def _filter_known_fields(cls, data: Dict[str, Any]) -> Dict[str, Any]:
    """Extract only declared dataclass fields to guarantee backwards/forwards compatibility without crashing."""
    if not isinstance(data, dict):
        return {}
    known = {f.name for f in fields(cls)}
    return {k: v for k, v in data.items() if k in known and v is not None}


@dataclass
class WordStyleOverride:
    color: Optional[str] = None          # Hex e.g. '#FFD700'
    highlight_color: Optional[str] = None
    scale: Optional[float] = None        # Scale multiplier e.g. 1.15
    font_weight: Optional[str] = None   # 'bold', '900', etc.
    underline: Optional[bool] = None
    background_color: Optional[str] = None

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "WordStyleOverride":
        if not isinstance(data, dict):
            return cls()
        return cls(**_filter_known_fields(cls, data))


@dataclass
class Word:
    id: str = field(default_factory=lambda: str(uuid.uuid4())[:8])
    text: str = ""
    start: float = 0.0                   # Millisecond precision in seconds (e.g. 1.240)
    end: float = 0.0
    confidence: float = 1.0
    is_exact: bool = True
    style_override: Optional[WordStyleOverride] = None

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        if not self.style_override:
            d.pop("style_override", None)
        return d

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "Word":
        if not isinstance(data, dict):
            return cls()
        override = None
        if "style_override" in data and data["style_override"]:
            override = WordStyleOverride.from_dict(data["style_override"])
        try:
            start_val = float(data.get("start", 0.0))
        except (ValueError, TypeError):
            start_val = 0.0
        try:
            end_val = float(data.get("end", 0.0))
        except (ValueError, TypeError):
            end_val = 0.0
        try:
            conf_val = float(data.get("confidence", 1.0))
        except (ValueError, TypeError):
            conf_val = 1.0

        return cls(
            id=str(data.get("id", str(uuid.uuid4())[:8])),
            text=str(data.get("text", "")),
            start=start_val,
            end=end_val,
            confidence=conf_val,
            is_exact=bool(data.get("is_exact", True)),
            style_override=override
        )

"""Keep saved clip-relative captions aligned when a source selection changes."""

from __future__ import annotations

from copy import deepcopy
from typing import Any, Dict, List, Optional, Tuple


def rebase_caption_track(
    saved_track: Dict[str, Any],
    saved_source_start: float,
    selected_start: float,
    selected_end: float,
    transcript_track: Optional[Dict[str, Any]] = None,
) -> Tuple[Dict[str, Any], bool]:
    """Re-anchor saved clip-relative words once and fill newly selected source time.

    Existing caption content and styling are retained. New transcript words are
    added only when their exact text/time identity is not already in the saved
    track. Phrase-level groups are preserved only when fully inside the range.
    """
    result = deepcopy(saved_track)
    groups: List[Dict[str, Any]] = []
    exact_keys = set()
    timing_warning = False

    def word_key(word: Dict[str, Any], source_start: float) -> tuple:
        return (
            round(source_start + float(word.get("start", 0.0)), 3),
            round(source_start + float(word.get("end", 0.0)), 3),
        )

    for original in saved_track.get("groups", []):
        group = deepcopy(original)
        phrase_timing = not original.get("has_word_timing", True) or original.get("timing_mode") == "phrase"
        words = []
        for word in original.get("words", []):
            start = saved_source_start + float(word.get("start", 0.0))
            end = saved_source_start + float(word.get("end", 0.0))
            if end <= selected_start or start >= selected_end:
                continue
            adjusted = deepcopy(word)
            adjusted["start"] = round(max(start, selected_start) - selected_start, 3)
            adjusted["end"] = round(min(end, selected_end) - selected_start, 3)
            words.append(adjusted)
            exact_keys.add(word_key(adjusted, selected_start))

        if words:
            if phrase_timing and original.get("type", "caption") == "caption":
                timing_warning = True
            group["words"] = words
            group["start"] = min(word["start"] for word in words)
            group["end"] = max(word["end"] for word in words)
            groups.append(group)
        elif not original.get("words"):
            start = saved_source_start + float(original.get("start", 0.0))
            end = saved_source_start + float(original.get("end", 0.0))
            # Non-word layers can be clipped to selection bounds; phrase captions
            # cannot be subdivided without inventing word timings.
            if start >= selected_start and end <= selected_end:
                group["start"] = round(start - selected_start, 3)
                group["end"] = round(end - selected_start, 3)
                groups.append(group)
            elif original.get("type", "caption") == "caption":
                timing_warning = True

    if transcript_track:
        for original in transcript_track.get("groups", []):
            if not original.get("has_word_timing", True) or original.get("timing_mode") == "phrase":
                missing_phrase_words = [
                    word for word in original.get("words", [])
                    if word_key(word, selected_start) not in exact_keys
                ]
                if missing_phrase_words:
                    timing_warning = True
                continue
            new_word_runs: List[List[Dict[str, Any]]] = []
            current_run: List[Dict[str, Any]] = []
            for word in original.get("words", []):
                key = word_key(word, selected_start)
                if key in exact_keys:
                    if current_run:
                        new_word_runs.append(current_run)
                        current_run = []
                    continue
                exact_keys.add(key)
                current_run.append(deepcopy(word))
            if current_run:
                new_word_runs.append(current_run)
            for run_index, new_words in enumerate(new_word_runs):
                group = deepcopy(original)
                if len(new_word_runs) > 1:
                    group["id"] = f"{original.get('id', 'transcript')}-range-{run_index + 1}"
                group["words"] = new_words
                group["start"] = min(word["start"] for word in new_words)
                group["end"] = max(word["end"] for word in new_words)
                groups.append(group)

    groups.sort(key=lambda group: float(group.get("start", 0.0)))
    result["groups"] = groups
    return result, timing_warning

    return CloudConfig(**values)

"""Abstract media-store contract (plan doc D5/D8).

The PC, the Worker, and tests all speak this interface; concrete implementations are
the Worker-backed client (Phase D/E) and mocked stores in unit tests. Implementations
must verify ownership server-side, treat client-supplied sizes as unverified claims,
and never return permanent public URLs for private media.
"""

from __future__ import annotations

from abc import ABC, abstractmethod
from dataclasses import dataclass
from datetime import datetime
from typing import Optional


class MediaStoreError(Exception):
    """Base class for cloud media-store failures."""


class CloudNotConfiguredError(MediaStoreError):
    """Cloud env is not configured; callers should stay on the local workflow."""


class QuotaExceededError(MediaStoreError):
    """User or global storage budget exhausted; the upload was refused."""


class ReservationError(MediaStoreError):
    """Reservation missing, expired, or in a state that forbids this operation."""


class OwnershipError(MediaStoreError):
    """The authenticated user does not own the requested resource."""


class ValidationError(MediaStoreError):
    """Upload payload violates the media contract (size, MIME, or state)."""


CLIP_VIDEO = "clip_video"
THUMBNAIL = "thumbnail"
OBJECT_KINDS = (CLIP_VIDEO, THUMBNAIL)


@dataclass(frozen=True)
class UploadReservation:
    reservation_id: str
    user_id: str
    object_kind: str
    local_clip_id: Optional[str]
    staging_key: str
    expected_bytes: int
    mime_type: str
    expires_at: datetime


@dataclass(frozen=True)
class StoredObject:
    object_id: str
    user_id: str
    object_kind: str
    final_key: str
    byte_count: int  # actual, server-verified size
    mime_type: str
    status: str
    expires_at: Optional[datetime]


class MediaStore(ABC):
    """Control-plane operations for private clip media (plan doc D5)."""

    @abstractmethod
    def reserve_upload(
        self,
        user_id: str,
        object_kind: str,
        mime_type: str,
        expected_bytes: int,
        local_clip_id: Optional[str] = None,
        client_token: Optional[str] = None,
    ) -> UploadReservation:
        """Atomically check ownership + quotas and issue a short-lived staging PUT URL.

        The client token makes repeated reserve calls idempotent. The expected byte
        count is a claim, never proof; the reservation only bounds the attempt.
        """

    @abstractmethod
    def complete_upload(self, user_id: str, reservation_id: str) -> StoredObject:
        """Verify the actual R2 object (real size/MIME), promote staging -> immutable
        final key, record the verified byte count, release the reservation.

        Must be idempotent: repeating a completion returns the same StoredObject.
        """

    @abstractmethod
    def get_download_url(self, user_id: str, object_id: str) -> str:
        """Short-lived signed GET URL after verifying ownership and active status."""

    @abstractmethod
    def delete_media(self, user_id: str, object_id: str) -> None:
        """Idempotent delete for a valid application state.

        On R2 failure the object enters a deletion-pending state for scheduled retry;
        quota is not released until deletion is confirmed. Never report success
        before deletion is confirmed.
        """

"""Pure reservation decision kernel (plan doc D6).

This module holds the quota/MIME/size rules as a side-effect-free function so they
are unit-testable without any cloud dependency. The authoritative enforcement is the
Postgres transaction + Worker (plan doc D4/D5); this kernel is the shared definition
of the policy both must implement. Keep the two in sync deliberately.
"""

from __future__ import annotations

from dataclasses import dataclass

from dispatch.cloud.media_store import CLIP_VIDEO, THUMBNAIL

ALLOWED_MIME_TYPES = {
    CLIP_VIDEO: frozenset({"video/mp4"}),
    THUMBNAIL: frozenset({"image/jpeg"}),
}


@dataclass(frozen=True)
class UsageSummary:
    """Server-computed storage usage for quota decisions.

    user_active_bytes/user_pending_bytes are summed from media_objects with status
    active/pending; outstanding reservations count pending uploads not yet completed.
    """

    user_active_bytes: int
    user_pending_bytes: int
    user_outstanding_reservations: int
    global_active_bytes: int
    global_pending_bytes: int


@dataclass(frozen=True)
class QuotaDecision:
    allowed: bool
    reason: str


def evaluate_upload_reservation(
    usage: UsageSummary,
    object_kind: str,
    mime_type: str,
    expected_bytes: int,
    *,
    user_quota_bytes: int,
    global_budget_bytes: int,
    max_clip_object_bytes: int,
    max_thumbnail_bytes: int,
    max_outstanding_uploads_per_user: int,
) -> QuotaDecision:
    """Decide whether one upload reservation may be issued.

    Expected bytes bound the attempt only; the actual object size is re-verified at
    completion (plan doc D5). Concurrent requests are serialized by the database
    transaction that computes `usage` and inserts the reservation row together.
    """
    if object_kind not in ALLOWED_MIME_TYPES:
        return QuotaDecision(False, f"Unsupported object kind: {object_kind!r}")
    if mime_type not in ALLOWED_MIME_TYPES[object_kind]:
        return QuotaDecision(False, f"Unsupported MIME type for {object_kind}: {mime_type!r}")
    if expected_bytes <= 0:
        return QuotaDecision(False, "Expected byte count must be positive")

    kind_cap = max_clip_object_bytes if object_kind == CLIP_VIDEO else max_thumbnail_bytes
    if expected_bytes > kind_cap:
        return QuotaDecision(
            False,
            f"Expected {expected_bytes} bytes exceeds the {object_kind} limit of {kind_cap} bytes",
        )

    user_used = usage.user_active_bytes + usage.user_pending_bytes
    if user_used + expected_bytes > user_quota_bytes:
        return QuotaDecision(
            False,
            f"User storage quota exceeded: {user_used} used, {expected_bytes} requested, "
            f"quota {user_quota_bytes}",
        )
    if usage.user_outstanding_reservations >= max_outstanding_uploads_per_user:
        return QuotaDecision(
            False,
            f"Too many outstanding uploads: {usage.user_outstanding_reservations} "
            f"(limit {max_outstanding_uploads_per_user})",
        )

    global_used = usage.global_active_bytes + usage.global_pending_bytes
    if global_used + expected_bytes > global_budget_bytes:
        return QuotaDecision(
            False,
            f"Global storage budget exceeded: {global_used} used, {expected_bytes} requested, "
            f"budget {global_budget_bytes}",
        )
    return QuotaDecision(True, "ok")
