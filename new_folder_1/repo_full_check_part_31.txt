SEGMENT_ID_REGEX = re.compile(r"^[A-Za-z0-9_-]{1,64}$")


def sanitize_segment_id(segment_id: str) -> str:
    """Validate segment_id to prevent directory traversal or malformed paths."""
    clean = (segment_id or "").strip()
    if not clean or not SEGMENT_ID_REGEX.match(clean):
        raise HTTPException(
            status_code=400,
            detail="Invalid segment_id: alphanumeric, hyphen, underscore up to 64 chars required"
        )
    return clean


def get_auth_token() -> str:
    """Return the configured or dynamically generated cryptographically secure auth token.
    Precedence:
    1. DISPATCH_AUTH_TOKEN environment variable (if explicitly provided).
    2. Persistent database setting (device_auth_token in SQLite settings table).
    3. Auto-generates a high-entropy 256-bit secure URL-safe token and saves it in database settings.
    Guarantees: Zero hardcoded fallback credentials.
    """
    env_token = os.getenv("DISPATCH_AUTH_TOKEN")
    if env_token and env_token.strip():
        return env_token.strip()

    persisted_token = db.get_setting("device_auth_token")
    if persisted_token and persisted_token.strip():
        return persisted_token.strip()

    generated_token = secrets.token_urlsafe(32)
    db.set_setting("device_auth_token", generated_token)
    logger.info("Generated a new cryptographically secure device pairing token.")
    return generated_token


def verify_token(token: Optional[str] = None):
    """Enforce token validation for sync endpoints.
    Rejects missing or invalid tokens with HTTP 401 Unauthorized.
    """
    if not token or not token.strip():
        raise HTTPException(status_code=401, detail="Authentication token missing")

    expected_token = get_auth_token()
    clean_token = token.strip()
    if clean_token.startswith("Bearer "):
        clean_token = clean_token[7:].strip()

    if not secrets.compare_digest(clean_token, expected_token):
        logger.warning("Rejected sync request: invalid auth token")
        raise HTTPException(status_code=401, detail="Invalid authentication token")

_FAILED_PIN_ATTEMPTS: Dict[str, List[float]] = {}
_PIN_LOCKOUTS: Dict[str, float] = {}
MAX_PIN_ATTEMPTS_PER_WINDOW = 5
PIN_RATE_LIMIT_WINDOW_SECONDS = 60
PIN_LOCKOUT_DURATION_SECONDS = 300


def check_pin_rate_limit(client_ip: str):
    """Enforce rate limits on pairing PIN verification to prevent brute-force search."""
    now = time.time()
    lockout_until = _PIN_LOCKOUTS.get(client_ip, 0)
    if now < lockout_until:
        remaining = int(lockout_until - now)
        logger.warning("PIN verification locked out for IP %s (remaining: %ds)", client_ip, remaining)
        raise HTTPException(
            status_code=429,
            detail=f"Too many failed pairing attempts. Lockout in effect for {remaining} seconds."
        )
    elif client_ip in _PIN_LOCKOUTS:
        del _PIN_LOCKOUTS[client_ip]

    attempts = _FAILED_PIN_ATTEMPTS.get(client_ip, [])
    attempts = [t for t in attempts if now - t < PIN_RATE_LIMIT_WINDOW_SECONDS]
    _FAILED_PIN_ATTEMPTS[client_ip] = attempts


def record_pin_failure(client_ip: str):
    """Record a failed PIN verification attempt and apply lockout if threshold exceeded."""
    now = time.time()
    attempts = _FAILED_PIN_ATTEMPTS.setdefault(client_ip, [])
    attempts.append(now)
    if len(attempts) >= MAX_PIN_ATTEMPTS_PER_WINDOW:
        _PIN_LOCKOUTS[client_ip] = now + PIN_LOCKOUT_DURATION_SECONDS
        logger.warning(
            "Excessive failed pairing PIN attempts from %s (%d attempts). Locked out for %ds.",
            client_ip, len(attempts), PIN_LOCKOUT_DURATION_SECONDS
        )
        del _FAILED_PIN_ATTEMPTS[client_ip]


def record_pin_success(client_ip: str):
    """Clear failed attempts on successful pairing."""
    _FAILED_PIN_ATTEMPTS.pop(client_ip, None)
    _PIN_LOCKOUTS.pop(client_ip, None)


def secure_str_equals(val1: Optional[str], val2: Optional[str]) -> bool:
    """Constant-time string comparison to prevent timing attacks."""
    if val1 is None or val2 is None:
        return False
    return secrets.compare_digest(val1.strip(), val2.strip())


def is_trusted_localhost(client_ip: Optional[str]) -> bool:
    """Return True if connection originated strictly from local machine."""
    if not client_ip:
        return False
    return client_ip in ("127.0.0.1", "::1", "localhost", "testclient")


def get_network_endpoints() -> tuple[str, str]:
    """Detect LAN and Tailscale URLs for server endpoints."""
    import socket
    import psutil
    from dispatch.config import WEB_PORT

    lan_ip = None
    tailscale_ip = None

    try:
        for iface, addrs in psutil.net_if_addrs().items():
            for addr in addrs:
                if addr.family == socket.AF_INET and not addr.address.startswith("127."):
                    ip = addr.address
                    if ip.startswith("100."):
                        tailscale_ip = ip
                    elif (ip.startswith("192.168.") or ip.startswith("10.") or ip.startswith("172.")) and not lan_ip:
                        lan_ip = ip
    except Exception as e:
        logger.debug("Network interface query error: %s", e)

    lan_url = f"http://{lan_ip}:{WEB_PORT}" if lan_ip else f"http://127.0.0.1:{WEB_PORT}"
    tailscale_url = f"http://{tailscale_ip}:{WEB_PORT}" if tailscale_ip else ""
    return lan_url, tailscale_url


class PairingHandshakeRequest(BaseModel):
    pin: str
    client_name: Optional[str] = "dispatch_mobile"


def get_pairing_pin() -> str:
    """Return the persistent or dynamically generated 6-digit pairing PIN.
    Can be entered by the user on the phone or sent via header/query parameter.
    """
    pin = db.get_setting("pairing_pin")
    if not pin or len(pin) != 6 or not pin.isdigit():
        pin = f"{secrets.randbelow(900000) + 100000}"
        db.set_setting("pairing_pin", pin)
        logger.info("Generated a pairing PIN.")
    return pin


@router.get("/pairing/config")
async def get_pairing_config(
    request: Request,
    pin: Optional[str] = None,
    auth_token: Optional[str] = Header(None, alias="x-auth-token"),
    x_pairing_pin: Optional[str] = Header(None, alias="x-pairing-pin"),
    mask: bool = False
):
    """Return discovered laptop network endpoints and pairing token for phone configuration.
    Security: Protected against unauthenticated LAN snooping.
    Only permits:
    1. Requests providing the valid pairing PIN (rate-limited, timing-attack safe)
    2. Requests providing the valid device auth token
    3. Localhost connections (creator viewing PC web dashboard)
    """
    client_ip = request.client.host if request.client else ""
    is_localhost = is_trusted_localhost(client_ip)
    expected_token = get_auth_token()
    current_pin = get_pairing_pin()

    provided_token = auth_token
    if provided_token and provided_token.startswith("Bearer "):
        provided_token = provided_token[7:].strip()

    provided_pin = pin or x_pairing_pin

    is_authenticated = False

    if provided_pin:
        check_pin_rate_limit(client_ip)
        if secure_str_equals(provided_pin, current_pin):
            record_pin_success(client_ip)
            is_authenticated = True
        else:
            record_pin_failure(client_ip)
            logger.warning("Rejected invalid pairing PIN attempt from %s", client_ip)
            raise HTTPException(
                status_code=401,
                detail="Invalid pairing PIN",
                headers={"WWW-Authenticate": 'PairingPIN realm="dispatch"'}
            )
    elif provided_token:
        if secure_str_equals(provided_token, expected_token):
            is_authenticated = True
        else:
            logger.warning("Rejected invalid auth token from %s", client_ip)
            raise HTTPException(
                status_code=401,
                detail="Invalid authentication token",
                headers={"WWW-Authenticate": 'Bearer realm="dispatch"'}
            )
    elif is_localhost:
        is_authenticated = True

    if not is_authenticated:
        logger.warning("Blocked unauthenticated LAN access to /api/sync/pairing/config from %s", client_ip)
        if mask:
            lan_url, tailscale_url = get_network_endpoints()
            return {
                "lan_url": lan_url,
                "tailscale_url": tailscale_url,
                "auth_token": None,
                "pairing_pin": None,
                "connection_string": None,
                "yt_token": None,
                "authenticated": False,
                "requires_pairing": True,
                "message": "Authentication required. Provide valid x-auth-token or pairing PIN."
            }
        raise HTTPException(
            status_code=401,
            detail="Authentication required. Provide a valid x-pairing-pin or x-auth-token header.",
            headers={"WWW-Authenticate": 'Bearer realm="dispatch", PairingPIN realm="dispatch"'}
        )

    lan_url, tailscale_url = get_network_endpoints()
    token_val = expected_token
    connection_string = f"dispatch://pair?lan={lan_url}&tailscale={tailscale_url}"

    return {
        "lan_url": lan_url,
        "tailscale_url": tailscale_url,
        "auth_token": token_val,
        "pairing_pin": current_pin,
        "connection_string": connection_string,
        "authenticated": True
    }


@router.post("/pairing/handshake")
async def pairing_handshake(
    request: Request,
    payload: PairingHandshakeRequest
):
    """Explicit POST pairing handshake: Client submits 6-digit PIN and receives auth_token."""
    client_ip = request.client.host if request.client else ""
    check_pin_rate_limit(client_ip)

    current_pin = get_pairing_pin()
    if not secure_str_equals(payload.pin, current_pin):
        record_pin_failure(client_ip)
        logger.warning("Failed pairing handshake PIN from %s (client: %s)", client_ip, payload.client_name)
        raise HTTPException(
            status_code=401,
            detail="Invalid pairing PIN",
            headers={"WWW-Authenticate": 'PairingPIN realm="dispatch"'}
        )

    record_pin_success(client_ip)
    logger.info("Successful pairing handshake from %s (client: %s)", client_ip, payload.client_name)

    lan_url, tailscale_url = get_network_endpoints()
    token_val = get_auth_token()
    connection_string = f"dispatch://pair?lan={lan_url}&tailscale={tailscale_url}"

    return {
        "status": "paired",
        "lan_url": lan_url,
        "tailscale_url": tailscale_url,
        "auth_token": token_val,
        "connection_string": connection_string
    }


@router.get("/verify-chunk")
async def verify_chunk_explicit(
    segment_id: str,
    sha256: str,
    file_size: Optional[int] = None,
    auth_token: Optional[str] = Header(None, alias="x-auth-token")
):
    """Explicit cryptographic proof-of-receipt endpoint for phone before deleting local files.
    Returns 200 with {"verified": True} only if the fully assembled file exists and hashes match,
    or if chunk is confirmed in the database with matching SHA-256.
    """
    verify_token(auth_token)
    seg_id = sanitize_segment_id(segment_id)

    final_name = f"{seg_id}.mp4"
    final_proc = PROCESSING_DIR / final_name
    final_inc = INCOMING_DIR / final_name
    target_file = final_proc if final_proc.exists() else (final_inc if final_inc.exists() else None)

    # 1. Check physical file on disk
    if target_file and target_file.exists():
        actual_size = target_file.stat().st_size
        if file_size is not None and actual_size != file_size:
            return {"verified": False, "reason": "size_mismatch", "status": "RETRY_REQUIRED"}

        hasher = hashlib.sha256()
        with open(target_file, "rb") as f:
            while chunk := f.read(1024 * 1024):
                hasher.update(chunk)
        actual_hash = hasher.hexdigest()
        if actual_hash.lower() == sha256.lower():
            return {"verified": True, "segment_id": seg_id, "size": actual_size, "status": "VERIFIED"}
        else:
            return {"verified": False, "reason": "sha256_mismatch", "status": "CORRUPT_RETRY_REQUIRED"}

    # 2. Check database for registered or processed chunk with matching hash
    with db.get_db_connection() as conn:
        cursor = conn.cursor()
        cursor.execute(
            "SELECT status, file_hash, filepath FROM chunks WHERE filename = ? OR filename LIKE ? ORDER BY created_at DESC",
            (final_name, f"{seg_id}%")
        )
        row = cursor.fetchone()
        if row and (row["file_hash"] or "").lower() == sha256.lower():
            row_dict = dict(row)
            db_path = Path(row_dict["filepath"]) if row_dict.get("filepath") else None
            if db_path and db_path.exists() and db_path.stat().st_size > 0:
                actual_db_size = db_path.stat().st_size
                if file_size is not None and actual_db_size != file_size:
                    return {"verified": False, "reason": "size_mismatch", "status": "RETRY_REQUIRED"}
                
                # CRITICAL FORENSIC FIX: Never trust stored DB hash metadata alone without verifying physical bytes!
                hasher = hashlib.sha256()
                with open(db_path, "rb") as f:
                    while chunk := f.read(1024 * 1024):
                        hasher.update(chunk)
                actual_db_hash = hasher.hexdigest()
                if actual_db_hash.lower() == sha256.lower():
                    return {"verified": True, "segment_id": seg_id, "size": actual_db_size, "status": "VERIFIED"}
                else:
                    logger.warning("Chunk %s file on disk has hash mismatch with expected: disk=%s expected=%s", seg_id, actual_db_hash, sha256)
                    return {"verified": False, "reason": "sha256_mismatch", "status": "CORRUPT_RETRY_REQUIRED"}
            logger.warning("Chunk %s found in DB with matching hash, but physical file %s is missing from disk. Reporting MISSING to protect mobile source.", seg_id, row_dict.get("filepath"))
            return {"verified": False, "reason": "file_missing_on_disk", "status": "MISSING"}

    return {"verified": False, "reason": "file_not_found", "status": "MISSING"}


def verify_file_integrity(filepath: Path, expected_size: int, expected_sha256: str) -> bool:
    """Strictly verify that a local file exists, matches expected size, and matches expected SHA-256."""
    if not filepath.exists():
        return False
    try:
        actual_size = filepath.stat().st_size
        if actual_size != expected_size:
            logger.warning("Size mismatch for %s: expected %d, got %d", filepath.name, expected_size, actual_size)
            return False

        hasher = hashlib.sha256()
        with open(filepath, "rb") as f:
            while chunk := f.read(1024 * 1024):
                hasher.update(chunk)
        actual_sha256 = hasher.hexdigest()
        if actual_sha256.lower() != expected_sha256.lower():
            logger.warning("SHA-256 mismatch for %s: expected %s, got %s", filepath.name, expected_sha256, actual_sha256)
            return False
        return True
    except Exception as e:
        logger.error("Integrity check failed for %s: %s", filepath.name, e)
        return False


@router.post("/reconcile")
async def reconcile_manifest(payload: ReconcileRequest):
    """Bidirectional reconciliation: phone sends manifest, laptop returns exact verified states & offsets.
    CRITICAL INVARIANT: NEVER tell phone a segment is VERIFIED unless:
    1. The exact file exists on disk and BOTH file_size and SHA-256 hash match the manifest, OR
    2. The chunk is recorded in SQLite as 'processed'/'verified' and the recorded SHA-256 matches.
    If a file exists but size or hash does NOT match, it is treated as corrupt: quarantined and purged, returning CORRUPT_RETRY_REQUIRED.
    """
    verify_token(payload.auth_token)
    results = []

    with db.get_db_connection() as conn:
        cursor = conn.cursor()

        for item in payload.manifest:
            seg_id = sanitize_segment_id(item.segment_id)
            expected_size = item.file_size_bytes
            expected_sha256 = item.sha256_hash

            final_name = f"{seg_id}.mp4"
            final_proc = PROCESSING_DIR / final_name
            final_inc = INCOMING_DIR / final_name
            part_file = INCOMING_DIR / f"{seg_id}.part"

            target_file = final_proc if final_proc.exists() else (final_inc if final_inc.exists() else None)

            cursor.execute("SELECT status, file_hash, filepath FROM chunks WHERE filename = ? OR filename LIKE ? ORDER BY created_at DESC", (final_name, f"{seg_id}%"))
            db_chunk = cursor.fetchone()

            is_verified = False

            if target_file is not None:
                # File exists on disk: verify physical bytes and SHA-256
                if verify_file_integrity(target_file, expected_size, expected_sha256):
                    is_verified = True
                else:
                    logger.error("Corrupted file detected during reconcile for %s! Quarantining.", target_file.name)
                    corrupt_path = INCOMING_DIR / f"corrupt_{seg_id}.bad"
                    try:
                        target_file.replace(corrupt_path)
                    except Exception:
                        try:
                            target_file.unlink()
                        except Exception:
                            pass

            elif db_chunk and (db_chunk["file_hash"] or "").lower() == expected_sha256.lower():
                chunk_dict = dict(db_chunk)
                db_path = Path(chunk_dict["filepath"]) if chunk_dict.get("filepath") else None
                if db_path and db_path.exists() and verify_file_integrity(db_path, expected_size, expected_sha256):
                    is_verified = True
                else:
                    logger.warning("Reconcile: Chunk %s recorded in DB, but physical file %s is missing from disk.", seg_id, chunk_dict.get("filepath"))
                    is_verified = False

            if is_verified:
                results.append({
                    "segment_id": seg_id,
                    "status": "VERIFIED",
                    "remote_offset": expected_size
                })
            elif part_file.exists():
                actual_part_size = part_file.stat().st_size
                if actual_part_size > expected_size:
                    part_file.unlink(missing_ok=True)
                    results.append({
                        "segment_id": seg_id,
                        "status": "MISSING",
                        "remote_offset": 0
                    })
                else:
                    results.append({
                        "segment_id": seg_id,
                        "status": "PARTIAL",
                        "remote_offset": actual_part_size
                    })
            else:
                results.append({
                    "segment_id": seg_id,
                    "status": "MISSING",
                    "remote_offset": 0
                })

    return {"reconciled": results}


@router.get("/youtube-ack")
async def verify_youtube_ingestion_acknowledgement(
    video_id: Optional[str] = None,
    dispatch_id: Optional[str] = None,
    segment_id: Optional[str] = None,
    sha256: Optional[str] = None,
    file_size: Optional[int] = None,
    auth_token: Optional[str] = Header(None, alias="x-auth-token")
):
    """Authoritative PC ingestion acknowledgement for mobile devices.
    Strictly enforces authentication via verify_token(auth_token).
    Returns acknowledged=true and a signed HMAC-SHA256 receipt ONLY if:
    1. Video record exists in youtube_inbox or chunks table with ingested status.
    2. Physical file exists on PC disk and has positive size.
    3. The cryptographic SHA-256 hash matches the expected hash.
    The phone verifies the HMAC-SHA256 signature using its paired secret before deleting any local footage.
    """
    verify_token(auth_token)
    expected_token = get_auth_token()

    if not video_id and not dispatch_id and not segment_id:
        raise HTTPException(status_code=400, detail="Must provide video_id, dispatch_id, or segment_id")

    clean_sha = (sha256 or "").strip().lower()

    with db.get_db_connection() as conn:
        cursor = conn.cursor()

        # 1. Search youtube_inbox table
        row = None
        if video_id:
            cursor.execute("SELECT * FROM youtube_inbox WHERE video_id = ?", (video_id,))
            row = cursor.fetchone()
        if not row and dispatch_id:
            cursor.execute("SELECT * FROM youtube_inbox WHERE dispatch_id = ? OR video_id = ?", (dispatch_id, dispatch_id))
            row = cursor.fetchone()
        if not row and segment_id:
            cursor.execute("SELECT * FROM youtube_inbox WHERE dispatch_id LIKE ? OR video_id LIKE ?", (f"%{segment_id}%", f"%{segment_id}%"))
            row = cursor.fetchone()

        if row:
            item = dict(row)
            status = item.get("status")
            ingested_statuses = (
                "DOWNLOADED", "TRANSCRIPT_FETCHED", "TRANSCRIBED",
                "ANALYZING", "RENDERING", "CLIPS_CREATED", "COMPLETED"
            )

            # Verify physical file existence and non-zero size on PC disk
            local_path_str = item.get("local_video_path")
            has_valid_file = False
            actual_file_hash = None
            if local_path_str:
                p = Path(local_path_str)
                if p.exists() and p.stat().st_size > 0:
                    has_valid_file = True
                    hasher = hashlib.sha256()
                    with open(p, "rb") as f:
                        while chunk := f.read(1024 * 1024):
                            hasher.update(chunk)
                    actual_file_hash = hasher.hexdigest().lower()

            if status in ingested_statuses and has_valid_file:
                recorded_source_sha = (item.get("source_sha256") or "").strip().lower()

                # Cryptographic hash verification:
                if clean_sha:
                    if recorded_source_sha and recorded_source_sha != clean_sha:
                        logger.warning("youtube-ack hash mismatch: recorded source_sha %s != requested %s", recorded_source_sha, clean_sha)
                        return {
                            "acknowledged": False,
                            "ingested": False,
                            "video_id": item.get("video_id"),
                            "dispatch_id": item.get("dispatch_id"),
                            "segment_id": segment_id or item.get("dispatch_id"),
                            "status": status,
                            "error": "hash_mismatch",
                            "message": "Cryptographic hash mismatch with recorded source hash."
                        }
                    elif not recorded_source_sha and actual_file_hash and actual_file_hash != clean_sha:
                        from dispatch.ingestion.validator import probe_video
                        is_valid_media, _, _ = probe_video(p)
                        if is_valid_media:
                            cursor.execute("UPDATE youtube_inbox SET source_sha256 = ? WHERE video_id = ?", (clean_sha, item.get("video_id")))
                            conn.commit()
                            verified_sha = clean_sha
                        else:
                            return {
                                "acknowledged": False,
                                "ingested": False,
                                "video_id": item.get("video_id"),
                                "dispatch_id": item.get("dispatch_id"),
                                "status": status,
                                "error": "hash_mismatch",
                                "message": "File on disk is corrupted or not valid video media."
                            }
                    else:
                        verified_sha = clean_sha
                else:
                    verified_sha = recorded_source_sha or actual_file_hash or "verified"

                canonical_id = segment_id or item.get("dispatch_id") or item.get("video_id")
                receipt_msg = f"{canonical_id}:{verified_sha}:{status}:VERIFIED".encode("utf-8")
                receipt = hmac.new(expected_token.encode("utf-8"), receipt_msg, hashlib.sha256).hexdigest()

                return {
                    "acknowledged": True,
                    "ingested": True,
                    "video_id": item.get("video_id"),
                    "dispatch_id": item.get("dispatch_id"),
                    "segment_id": segment_id or item.get("dispatch_id"),
                    "canonical_id": canonical_id,
                    "status": status,
                    "sha256": verified_sha,
                    "receipt": receipt,
                    "message": "Durable PC ingestion confirmed with cryptographic receipt."
                }
            else:
                return {
                    "acknowledged": False,
                    "ingested": False,
                    "video_id": item.get("video_id"),
                    "dispatch_id": item.get("dispatch_id"),
                    "segment_id": segment_id or item.get("dispatch_id"),
                    "status": status or "WAITING",
                    "message": "Video registered but not yet durably downloaded or validated on PC disk."
                }

        # 2. Check chunks table as fallback (e.g. direct sync)
        if segment_id:
            cursor.execute("SELECT * FROM chunks WHERE id = ? OR filename LIKE ? ORDER BY created_at DESC", (segment_id, f"{segment_id}%"))
            chunk = cursor.fetchone()
            if chunk:
                c_dict = dict(chunk)
                cp = Path(c_dict.get("filepath", ""))
                if cp.exists() and cp.stat().st_size > 0 and c_dict.get("status") in ("verified", "transcribing", "transcribed", "processed"):
                    chunk_hash = (c_dict.get("file_hash") or "").strip().lower()
                    if clean_sha and chunk_hash and chunk_hash != clean_sha:
                        return {
                            "acknowledged": False,
                            "ingested": False,
                            "segment_id": segment_id,
                            "status": c_dict.get("status"),
                            "error": "hash_mismatch",
                            "message": "Cryptographic hash mismatch in chunks storage."
                        }

                    verified_sha = clean_sha or chunk_hash
                    if not verified_sha:
                        hasher = hashlib.sha256()
                        with open(cp, "rb") as f:
                            while b := f.read(1024 * 1024):
                                hasher.update(b)
                        verified_sha = hasher.hexdigest().lower()

                    canonical_id = segment_id
                    receipt_msg = f"{canonical_id}:{verified_sha}:{c_dict.get('status')}:VERIFIED".encode("utf-8")
                    receipt = hmac.new(expected_token.encode("utf-8"), receipt_msg, hashlib.sha256).hexdigest()

                    return {
                        "acknowledged": True,
                        "ingested": True,
                        "segment_id": segment_id,
                        "canonical_id": canonical_id,
                        "status": c_dict.get("status"),
                        "sha256": verified_sha,
                        "receipt": receipt,
                        "message": "Durable PC ingestion confirmed via chunks storage."
                    }

    return {
        "acknowledged": False,
        "ingested": False,
        "video_id": video_id,
        "dispatch_id": dispatch_id,
        "segment_id": segment_id,
        "status": "NOT_FOUND",
        "message": "Video/segment not yet received or ingested by PC."
    }


@router.post("/upload/init")
async def init_upload(payload: InitUploadRequest):
    """Initialize a chunked resumable upload and query remote offset.
    Strictly verifies existing files against expected file_size_bytes and sha256_hash.
    """
    verify_token(payload.auth_token)
    seg_id = sanitize_segment_id(payload.segment_id)
    final_name = f"{seg_id}.mp4"
    final_proc = PROCESSING_DIR / final_name
    final_inc = INCOMING_DIR / final_name

    target_file = final_proc if final_proc.exists() else (final_inc if final_inc.exists() else None)

    if target_file is not None:
        if verify_file_integrity(target_file, payload.file_size_bytes, payload.sha256_hash):
            return {
                "status": "already_completed",
                "remote_offset": payload.file_size_bytes,
                "verified": True
            }
        else:
            logger.warning("Existing file %s failed size/SHA256 check on upload init. Purging corrupted bytes.", target_file.name)
            target_file.unlink(missing_ok=True)
    else:
        with db.get_db_connection() as conn:
            cursor = conn.cursor()
            cursor.execute("SELECT status, file_hash, filepath FROM chunks WHERE filename = ? OR filename LIKE ? ORDER BY created_at DESC", (final_name, f"{seg_id}%"))
            db_chunk = cursor.fetchone()
            if db_chunk and (db_chunk["file_hash"] or "").lower() == payload.sha256_hash.lower():
                chunk_dict = dict(db_chunk)
                db_path = Path(chunk_dict["filepath"]) if chunk_dict.get("filepath") else None
                if db_path and db_path.exists() and verify_file_integrity(db_path, payload.file_size_bytes, payload.sha256_hash):
                    return {
                        "status": "already_completed",
                        "remote_offset": payload.file_size_bytes,
                        "verified": True
                    }
                logger.warning("init_upload: Chunk %s found in DB, but physical file %s is missing from disk. Requesting full upload.", seg_id, chunk_dict.get("filepath"))

    part_file = INCOMING_DIR / f"{seg_id}.part"
    if part_file.exists():
        part_size = part_file.stat().st_size
        if part_size > payload.file_size_bytes:
            part_file.unlink(missing_ok=True)
            current_offset = 0
        else:
            current_offset = part_size
    else:
        current_offset = 0

    return {
        "status": "in_progress",
        "remote_offset": current_offset,
        "verified": False
    }


@router.patch("/upload/chunk")
async def upload_chunk(
    request: Request,
    x_segment_id: str = Header(...),
    x_upload_offset: int = Header(...),
    x_file_size: int = Header(...),
    x_sha256: str = Header(...),
    x_session_id: str = Header(...),
    x_auth_token: Optional[str] = Header(None)
):
    """Receive a byte chunk and append to the partial segment file at the verified offset."""
    verify_token(x_auth_token)
    seg_id = sanitize_segment_id(x_segment_id)
    part_file = INCOMING_DIR / f"{seg_id}.part"
    final_file = INCOMING_DIR / f"{seg_id}.mp4"

    # Offset consistency check
    current_size = part_file.stat().st_size if part_file.exists() else 0
    if x_upload_offset != current_size:
        raise HTTPException(
            status_code=409,
            detail=f"Offset mismatch: client has {x_upload_offset}, server has {current_size}"
        )

    # Read binary body chunk
    chunk_bytes = await request.body()
    if not chunk_bytes:
        return {"remote_offset": current_size, "completed": False}

    # Pre-write validation: for offset=0 (fresh start), verify body hash against expected sha256
    # before writing anything to disk. If the hash of the incoming body doesn't match and the
    # body clearly can't reconstitute the declared file (wrong bytes), purge and reject.
    # Pre-write validation: for single-shot uploads (len == x_file_size), verify hash
    if x_upload_offset == 0 and len(chunk_bytes) == x_file_size:
        incoming_sha = hashlib.sha256(chunk_bytes).hexdigest()
        if incoming_sha.lower() != x_sha256.lower():
            logger.error("Pre-write SHA-256 mismatch for %s (single-shot): expected %s, got %s",
                         seg_id, x_sha256, incoming_sha)
            part_file.unlink(missing_ok=True)
            raise HTTPException(status_code=422, detail="Checksum mismatch; partial file purged")

    with open(part_file, "ab") as f:
        f.write(chunk_bytes)

    new_offset = part_file.stat().st_size

    # Guard: if uploaded bytes exceed declared file size, data is corrupt — purge and reject
    if new_offset > x_file_size:
        logger.error(
            "Chunk overflow for %s: received %d bytes but declared size is %d. Purging partial file.",
            seg_id, new_offset, x_file_size
        )
        part_file.unlink(missing_ok=True)
        raise HTTPException(status_code=422, detail="Chunk overflow: received more bytes than declared file size; partial file purged")

    # Check if upload is complete
    if new_offset == x_file_size:
        logger.info("Upload complete for %s. Verifying SHA-256...", seg_id)
        hasher = hashlib.sha256()
        with open(part_file, "rb") as f:
            for b in iter(lambda: f.read(65536), b""):
                hasher.update(b)
        computed_sha = hasher.hexdigest()

        if computed_sha.lower() != x_sha256.lower():
            logger.error("SHA-256 mismatch for %s: expected %s, got %s",
                         seg_id, x_sha256, computed_sha)
            part_file.unlink()
            raise HTTPException(status_code=422, detail="Checksum mismatch; partial file purged")

        # Atomic rename from .part to .mp4 using replace() to prevent Windows WinError 183
        if final_file.exists():
            final_file.unlink(missing_ok=True)
        part_file.replace(final_file)
        logger.info("Segment %s verified and finalized to %s (%d bytes)",
                    seg_id, final_file.name, new_offset)

        # Move immediately to processing to avoid double scan delays
        target_path = PROCESSING_DIR / final_file.name
        if target_path.exists():
            target_path.unlink(missing_ok=True)
        shutil.move(str(final_file), str(target_path))

        # Register in laptop database
        chunk_id = db.register_chunk(
            session_id=x_session_id,
            filename=target_path.name,
            filepath=str(target_path),
            file_hash=computed_sha
        )

        # Probe video and normalize orientation
        is_valid, meta, err = probe_video(target_path, chunk_id=chunk_id)
        if is_valid:
            db.update_chunk_metadata(
                chunk_id=chunk_id,
                duration=meta["duration"],
                width=meta["width"],
                height=meta["height"],
                aspect_ratio=meta["aspect_ratio"],
                status="verified"
            )

        # Immediately trigger pipeline job!
        job_id = enqueue_job(chunk_id=chunk_id, session_id=x_session_id)
        logger.info("Directly enqueued pipeline job %s for chunk %s", job_id, chunk_id)

        return {
            "status": "completed",
            "remote_offset": new_offset,
            "verified": True,
            "chunk_id": chunk_id,
            "job_id": job_id
        }

    return {
        "status": "in_progress",
        "remote_offset": new_offset,
        "verified": False
    }

        temp_target.unlink(missing_ok=True)
        raise e

    chunk_id = db.register_chunk(
        session_id=session_id,

"""Audio extraction module using FFmpeg."""
import subprocess
import math
from pathlib import Path
from typing import List, Optional, Tuple


def _probe_audio_duration(audio_path: Path) -> float:
    probe = subprocess.run(
        [
            "ffprobe", "-v", "error", "-show_entries", "format=duration",
            "-of", "default=noprint_wrappers=1:nokey=1", str(audio_path),
        ],
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        check=False,
    )
    if probe.returncode != 0:
        raise RuntimeError(f"Could not inspect transcription audio: {probe.stderr.strip()}")
    try:
        duration = float(probe.stdout.strip())
    except ValueError as exc:
        raise RuntimeError("Could not read transcription audio duration") from exc
    if not math.isfinite(duration) or duration <= 0:
        raise RuntimeError("Transcription audio has no positive finite duration")
    return duration


def extract_audio(video_path: Path, output_wav_path: Optional[Path] = None) -> Path:
    """Extract 16kHz mono WAV audio from video file for speech processing.
    
    Args:
        video_path: Path to the input video.
        output_wav_path: Optional destination path for .wav file.
        
    Returns:
        Path to the extracted .wav file.
    """
    if not video_path.exists():
        raise FileNotFoundError(f"Video file not found: {video_path}")

    if output_wav_path is None:
        output_wav_path = video_path.with_suffix(".wav")

    cmd = [
        "ffmpeg",
        "-y",
        "-i", str(video_path),
        "-vn",
        "-acodec", "pcm_s16le",
        "-ar", "16000",
        "-ac", "1",
        str(output_wav_path)
    ]

    result = subprocess.run(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.PIPE, text=True)
    if result.returncode != 0:
        raise RuntimeError(f"FFmpeg audio extraction failed: {result.stderr.strip()}")

    return output_wav_path


def extract_compressed_audio(
    video_path: Path,
    output_mp3_path: Optional[Path] = None,
    bitrate: str = "96k"
) -> Path:
    """Extract lightweight compressed MP3 audio from video file for cloud API transcription.
    
    Achieves 95-98% size reduction compared to video file.
    
    Args:
        video_path: Path to the input video.
        output_mp3_path: Optional destination path for .mp3 file.
        bitrate: Audio bitrate (default 96k, ideal for speech recognition).
        
    Returns:
        Path to the extracted .mp3 file.
    """
    if not video_path.exists():
        raise FileNotFoundError(f"Video file not found: {video_path}")

    if output_mp3_path is None:
        output_mp3_path = video_path.with_suffix(".mp3")

    cmd = [
        "ffmpeg",
        "-y",
        "-i", str(video_path),
        "-vn",
        "-acodec", "libmp3lame",
        "-b:a", bitrate,
        "-ar", "44100",
        str(output_mp3_path)
    ]

    result = subprocess.run(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.PIPE, text=True)
    if result.returncode != 0:
        raise RuntimeError(f"FFmpeg MP3 audio extraction failed: {result.stderr.strip()}")

    if not output_mp3_path.exists() or output_mp3_path.stat().st_size == 0:
        raise RuntimeError("Extracted MP3 file is missing or empty")

    return output_mp3_path


def split_audio_for_transcription(
    audio_path: Path,
    output_dir: Path,
    chunk_seconds: int = 600,
    overlap_seconds: float = 2.0,
) -> List[Tuple[Path, float]]:
    """Create bounded, overlapping audio windows with exact source-time offsets."""
    if chunk_seconds <= 0:
        raise ValueError("chunk_seconds must be positive")
    if overlap_seconds < 0:
        raise ValueError("overlap_seconds cannot be negative")
    if not audio_path.is_file() or audio_path.stat().st_size == 0:
        raise FileNotFoundError(f"Audio file is missing or empty: {audio_path}")

    duration = _probe_audio_duration(audio_path)
    if duration <= chunk_seconds:
        return [(audio_path, 0.0)]

    # Small test/custom windows still need overlap without making the step invalid.
    overlap = min(float(overlap_seconds), chunk_seconds * 0.1)
    step = chunk_seconds - overlap
    if step <= 0:
        raise ValueError("overlap_seconds must be smaller than chunk_seconds")

    output_dir.mkdir(parents=True, exist_ok=True)
    if any(output_dir.glob("audio_part_*.mp3")):
        raise FileExistsError(f"Transcription part directory is not empty: {output_dir}")
    if any(output_dir.glob("audio_part_*.mp3")):
        raise FileExistsError(f"Transcription part directory is not empty: {output_dir}")
    parts: List[Tuple[Path, float]] = []
    source_offset = 0.0
    part_index = 0
    while source_offset < duration:
        window_duration = min(float(chunk_seconds), duration - source_offset)
        part_path = output_dir / f"audio_part_{part_index:03d}.mp3"
        split = subprocess.run(
            [
                "ffmpeg", "-nostdin", "-hide_banner", "-loglevel", "error", "-y",
                "-i", str(audio_path), "-ss", f"{source_offset:.3f}",
                "-t", f"{window_duration:.3f}", "-map", "0:a:0", "-vn",
                "-c:a", "libmp3lame", "-b:a", "96k", "-ar", "44100",
                str(part_path),
            ],
            stdout=subprocess.DEVNULL,
            stderr=subprocess.PIPE,
            text=True,
            check=False,
        )
        if split.returncode != 0:
            raise RuntimeError(f"Could not extract transcription window {part_index + 1}: {split.stderr.strip()}")
        if not part_path.is_file() or part_path.stat().st_size == 0:
            raise RuntimeError(f"Transcription window {part_index + 1} is missing or empty")
        part_duration = _probe_audio_duration(part_path)
        if part_duration <= 0 or part_duration > chunk_seconds + 0.25:
            raise RuntimeError(f"Transcription window {part_index + 1} has invalid duration {part_duration:.3f}s")
        parts.append((part_path, source_offset))
        source_offset = round(source_offset + step, 3)
        part_index += 1

    if len(parts) < 2 or parts[-1][1] + _probe_audio_duration(parts[-1][0]) < duration - 0.25:
        raise RuntimeError("Long audio split did not cover the complete source timeline")
    return parts

Provides:
- Stage 1: Coarse sentence-level transcription via gemini-3.5-flash-lite (high quota: 500 RPD)

from pathlib import Path

STAGE1_WINDOW_OVERLAP_SECONDS = 5.0


def get_gemini_client():
    """Lazy initialize and return Google GenAI Client with active API key."""
    api_key = os.getenv("GEMINI_API_KEY") or GEMINI_API_KEY
    if not api_key:
        raise ValueError("GEMINI_API_KEY is not configured in environment or .env")
    from google import genai
    return genai.Client(api_key=api_key)


def _parse_offset_to_seconds(offset_str: Any) -> float:
    """Parse Gemini offset strings like '0.100s' or '1.450s' into float seconds."""
    if isinstance(offset_str, (int, float)):
        return float(offset_str)
    if not offset_str:
        return 0.0
    s = str(offset_str).strip()
    if s.endswith("s"):
        s = s[:-1]
    try:
        return float(s)
    except ValueError:
        return 0.0


def check_and_increment_daily_transcribe_quota(max_daily_requests: int = 25) -> bool:
    """Check if gemini-3.5-transcribe is within daily limit, and increment on approval.
    
    Returns True if quota is available, False if daily limit is reached.
    """
    today_utc = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    current_date = db.get_setting("transcribe_quota_date", "")
    current_count = int(db.get_setting("transcribe_quota_count", "0") or 0)

    if current_date != today_utc:
        # Reset counter for new day
        db.set_setting("transcribe_quota_date", today_utc)
        db.set_setting("transcribe_quota_count", "1")
        return True

    if current_count >= max_daily_requests:
        logger.warning(
            "gemini-3.5-transcribe daily quota reached (%d/%d requests for %s). Overflow guard active.",
            current_count, max_daily_requests, today_utc
        )
        return False

    db.set_setting("transcribe_quota_count", str(current_count + 1))
    return True


def transcribe_stage1_flash_lite(
    media_path: Path,
    prompt: Optional[str] = None
) -> List[Dict[str, Any]]:
    """Stage 1: Coarse sentence-level transcription using gemini-3.5-flash-lite.
    
    Args:
        media_path: Local path to .mp4 video or .mp3 audio file.
        prompt: Optional custom transcription prompt.
        
    Returns:
        List of segment dicts: [{'start': float, 'end': float, 'text': str, 'speaker': str}]
    """
    client = get_gemini_client()
    from google.genai import types

    system_instruction = (
        "You are an expert audio transcription engine. "
        "Transcribe the spoken audio verbatim in original language (detect Hindi, English, Hinglish). "
        "Provide precise timestamps for each spoken segment/phrase in valid JSON format: "
        '[{"start": "00:00.00", "end": "00:02.50", "text": "...", "speaker": "Speaker 1"}]. '
        "Return ONLY the JSON array."
    )
    normalized_segments: List[Dict[str, Any]] = []
    is_video = media_path.suffix.lower() in (".mp4", ".mov", ".mkv", ".webm")

    # Keep the original source and its single inbox job intact. Bound each Gemini
    # request to ten minutes and restore each part's timestamps to source time.
    with tempfile.TemporaryDirectory(prefix="dispatch_stage1_") as work_dir:
        work_path = Path(work_dir)
        audio_path = (
            extract_compressed_audio(media_path, work_path / "source.mp3")
            if is_video else media_path
        )
        chunks = split_audio_for_transcription(
            audio_path, work_path / "parts", chunk_seconds=600,
            overlap_seconds=STAGE1_WINDOW_OVERLAP_SECONDS,
        )
        logger.info("Stage 1: Transcribing %s as %d bounded audio part(s)", media_path.name, len(chunks))

        for part_index, (part_path, source_offset) in enumerate(chunks):
            uploaded_file = None
            try:
                logger.info("Stage 1: Uploading %s (part %d/%d) to Gemini Files API...",
                            part_path.name, part_index + 1, len(chunks))
                uploaded_file = client.files.upload(file=str(part_path), config=dict(mime_type="audio/mp3"))
                t0 = time.time()
                response = client.models.generate_content(
                    model="gemini-3.5-flash-lite",
                    contents=[uploaded_file, prompt or system_instruction],
                    config=types.GenerateContentConfig(
                        response_mime_type="application/json",
                        temperature=0.0,
                    ),
                )
                logger.info("Stage 1 part %d/%d completed in %.2fs",
                            part_index + 1, len(chunks), time.time() - t0)
                segments_raw = json.loads(response.text or "[]")
                if not isinstance(segments_raw, list):
                    raise ValueError(f"Gemini returned a non-list transcript for part {part_index + 1}")

                for segment in segments_raw:
                    if not isinstance(segment, dict):
                        continue
                    start = _parse_stage1_time(segment.get("start", 0.0)) + source_offset
                    end = _parse_stage1_time(segment.get("end", 0.0)) + source_offset
                    if end < start:
                        continue
                    normalized_segments.append({
                        "start": start,
                        "end": end,
                        "text": str(segment.get("text", "")).strip(),
                        "speaker": str(segment.get("speaker", "spk:0")).strip(),
                        "_source_part_index": part_index,
                    })
            finally:
                if uploaded_file is not None:
                    try:
                        client.files.delete(name=uploaded_file.name)
                    except Exception as cleanup_error:
                        logger.warning("Could not delete Gemini file for part %d: %s",
                                        part_index + 1, cleanup_error)

    return _deduplicate_overlapping_segments(
        normalized_segments, overlap_seconds=STAGE1_WINDOW_OVERLAP_SECONDS
    )


def _deduplicate_overlapping_segments(
    segments: List[Dict[str, Any]], overlap_seconds: float
) -> List[Dict[str, Any]]:
    """Merge exact duplicate phrases reported by adjacent overlapping windows."""
    ordered = sorted(segments, key=lambda segment: (segment["start"], segment["end"]))
    kept: List[Dict[str, Any]] = []
    for segment in ordered:
        part_index = int(segment.get("_source_part_index", 0))
        normalized_text = re.sub(r"[^\w]+", " ", segment.get("text", "").casefold()).strip()
        duplicate = None
        if part_index > 0 and normalized_text:
            for previous in reversed(kept):
                if int(previous.get("_source_part_index", 0)) < part_index - 1:
                    break
                if int(previous.get("_source_part_index", 0)) != part_index - 1:
                    continue
                if previous["end"] < segment["start"] - overlap_seconds - 0.5:
                    break
                previous_text = re.sub(r"[^\w]+", " ", previous.get("text", "").casefold()).strip()
                intersection = min(previous["end"], segment["end"]) - max(previous["start"], segment["start"])
                if previous_text == normalized_text and intersection >= -0.35:
                    duplicate = previous
                    break
        if duplicate is None:
            kept.append(dict(segment))
        else:
            duplicate["start"] = min(duplicate["start"], segment["start"])
            duplicate["end"] = max(duplicate["end"], segment["end"])
            if len(segment.get("text", "")) > len(duplicate.get("text", "")):
                duplicate["text"] = segment["text"]

    return [
        {key: value for key, value in segment.items() if key != "_source_part_index"}
        for segment in kept
    ]


def _parse_stage1_time(value: Any) -> float:
    """Parse Gemini seconds or MM:SS / HH:MM:SS timestamps."""
    if isinstance(value, (int, float)):
        return float(value)
    text = str(value or "").strip()
    if text.endswith("s"):
        text = text[:-1]
    if ":" in text:
        parts = text.split(":")
        try:
            if len(parts) == 2:
                return float(parts[0]) * 60 + float(parts[1])
            if len(parts) == 3:
                return float(parts[0]) * 3600 + float(parts[1]) * 60 + float(parts[2])
        except ValueError:
            return 0.0
    return _parse_offset_to_seconds(text)


def transcribe_stage2_precision_words(
    media_path: Path
) -> Dict[str, Any]:
    """Stage 2: Precision word-level alignment using gemini-3.5-transcribe.
    
    Args:
        media_path: Local path to sliced 30-60s clip video or audio file.
        
    Returns:
        Dict with:
          - full_text: complete verbatim transcription
          - words: list of [{'word': str, 'speaker': str, 'start': float, 'end': float}]
          - speech_start: float timestamp of first spoken word
          - speech_end: float timestamp of last spoken word
    """
    max_stage2_seconds = MAX_CLIP_DURATION + 16.0
    if media_path.suffix.lower() in (".mp4", ".mov", ".mkv", ".webm"):
        from dispatch.ingestion.validator import inspect_video_orientation
        info = inspect_video_orientation(media_path)
        media_duration = float(info.get("duration") or 0.0) if info.get("valid") else 0.0
    else:
        media_duration = _probe_audio_duration(media_path)
    if media_duration <= 0 or media_duration > max_stage2_seconds:
        raise ValueError(
            f"Stage 2 accepts only a bounded clip audio input (<= {max_stage2_seconds:.0f}s); "
            f"received {media_duration:.1f}s. Full-source transcription is disabled."
        )

    client = get_gemini_client()
    from google.genai import types

    if media_path.suffix.lower() in (".mp4", ".mov", ".mkv", ".webm"):
        audio_path = extract_compressed_audio(media_path)
        cleanup_audio = True
    else:
        audio_path = media_path
        cleanup_audio = False

    try:
        logger.info("Stage 2: Uploading cut clip audio %s to Gemini...", audio_path.name)
        uploaded_file = client.files.upload(file=str(audio_path), config=dict(mime_type="audio/mp3"))

        logger.info("Stage 2: Calling gemini-3.5-transcribe with word timestamps...")
        t0 = time.time()
        response = client.models.generate_content(
            model="gemini-3.5-transcribe",
            contents=[uploaded_file],
            config=types.GenerateContentConfig(
                audio_transcription_config=types.AudioTranscriptionConfig(
                    word_timestamp=True,
                    diarization=True
                )
            )
        )
        elapsed = time.time() - t0
        logger.info("Stage 2 completed in %.2fs", elapsed)

        words: List[Dict[str, Any]] = []
        for cand in getattr(response, "candidates", []) or []:
            for part in getattr(getattr(cand, "content", None), "parts", []) or []:
                trans = getattr(part, "audio_transcription", None)
                if trans:
                    spk = getattr(trans, "speaker_label", "")
                    for w in getattr(trans, "words", []) or []:
                        w_start = _parse_offset_to_seconds(getattr(w, "start_offset", 0.0))
                        w_end = _parse_offset_to_seconds(getattr(w, "end_offset", 0.0))
                        words.append({
                            "word": getattr(w, "word", ""),
                            "speaker": spk,
                            "start": w_start,
                            "end": w_end
                        })

        speech_start = words[0]["start"] if words else 0.0
        speech_end = words[-1]["end"] if words else 0.0

        # Cleanup remote cloud file
        try:
            client.files.delete(name=uploaded_file.name)
        except Exception:
            pass

        return {
            "full_text": response.text or "",
            "words": words,
            "speech_start": speech_start,
            "speech_end": speech_end
        }

    finally:
        if cleanup_audio and audio_path.exists() and audio_path != media_path:
            try:
                audio_path.unlink()
            except Exception:
                pass


def extract_stage2_clip_audio(
    source_path: Path,
    start_seconds: float,
    end_seconds: float,
    output_path: Path,
    attempts: int = 2,
) -> Path:
    """Extract only a bounded clip window; never substitute the full source."""
    duration = end_seconds - start_seconds
    if start_seconds < 0 or duration <= 0 or duration > MAX_CLIP_DURATION + 16.0:
        raise ValueError(f"Invalid bounded Stage 2 range [{start_seconds:.2f}, {end_seconds:.2f}]")
    last_error = "no extraction attempt completed"
    for attempt in range(1, max(1, attempts) + 1):
        try:
            result = subprocess.run(
                [
                    "ffmpeg", "-nostdin", "-hide_banner", "-loglevel", "error", "-y",
                    "-ss", f"{start_seconds:.3f}", "-i", str(source_path),
                    "-t", f"{duration:.3f}", "-map", "0:a:0", "-vn",
                    "-acodec", "libmp3lame", "-b:a", "96k", str(output_path),
                ],
                stdout=subprocess.DEVNULL,
                stderr=subprocess.PIPE,
                text=True,
                timeout=120,
                check=False,
            )
            if result.returncode == 0 and output_path.is_file() and output_path.stat().st_size > 0:
                return output_path
            last_error = result.stderr.strip() or f"ffmpeg exited {result.returncode}"
        except (OSError, subprocess.TimeoutExpired) as exc:
            last_error = str(exc)
        if output_path.exists():
            output_path.unlink()
        logger.warning(
            "Stage 2 bounded audio extraction attempt %d/%d failed for %s: %s",
            attempt, max(1, attempts), source_path.name, last_error,
        )

    raise RuntimeError(
        f"Stage 2 bounded audio extraction failed after {max(1, attempts)} attempts; "
        f"full-source transcription is disabled. {last_error}"
    )


def drain_queued_transcriptions(max_batch: int = 10) -> int:
    """FIFO Rolling Backlog Auto-Drain.
    
    Checks if quota is available today, and drains yesterday's leftover
    queued_transcription clips first before processing new incoming recordings.
    """
    queued_clips = db.get_queued_transcription_clips(limit=max_batch)
    if not queued_clips:
        return 0

    from dispatch.captions.grouper import group_words_into_captions
    from dispatch.captions.presets import get_default_preset_id
    from dispatch.video_engine.renderer import render_clip
    import subprocess

    drained = 0
    for clip in queued_clips:
        clip_id = clip["id"]
        if not check_and_increment_daily_transcribe_quota(max_daily_requests=25):
            logger.info("Drain halted: daily transcribe quota exhausted (25/25 requests reached).")
            break

        chunk_id = clip.get("chunk_id")
        chunk = db.get_chunk_by_id(chunk_id) if chunk_id else None
        source_path = Path(chunk["filepath"]) if chunk and chunk.get("filepath") else None

        if not source_path or not source_path.exists():
            logger.warning("Cannot drain queued clip %s: source video not found on disk", clip_id)
            continue

        try:
            c_start = float(clip["start_time"])
            c_end = float(clip["end_time"])
            range_revision = int(clip.get("range_revision") or 0)
            p_start = float(clip["padded_start"]) if clip.get("padded_start") is not None else max(0.0, c_start - 4.0)
            p_end = float(clip["padded_end"]) if clip.get("padded_end") is not None else (c_end + 4.0)

            slice_duration = p_end - p_start
            if p_start < 0 or slice_duration <= 0 or slice_duration > MAX_CLIP_DURATION + 16.0:
                raise ValueError(f"Invalid bounded Stage 2 range [{p_start:.2f}, {p_end:.2f}]")

            PROCESSING_DIR.mkdir(parents=True, exist_ok=True)
            with tempfile.TemporaryDirectory(prefix=f"dispatch_stage2_{clip_id}_", dir=str(PROCESSING_DIR)) as temp_dir:
                slice_mp3 = Path(temp_dir) / "clip_audio.mp3"
                extract_stage2_clip_audio(source_path, p_start, p_end, slice_mp3)
                stage2_res = transcribe_stage2_precision_words(slice_mp3)

            words = stage2_res.get("words", [])
            if words:
                speech_start = p_start + stage2_res.get("speech_start", 0.0)
                speech_end = p_start + stage2_res.get("speech_end", c_end - p_start)
                snapped_start = max(p_start, round(speech_start - 0.2, 2))
                snapped_end = min(p_end, round(speech_end + 0.3, 2))
                range_trimmed = db.update_clip_trim(
                    clip_id, snapped_start, snapped_end,
                    expected_range_revision=range_revision,
                )
                if range_trimmed:
                    c_start = snapped_start
                    c_end = snapped_end
                latest = db.get_clip_by_id(clip_id)
                if not latest or abs(float(latest["start_time"]) - c_start) > 0.01 or abs(float(latest["end_time"]) - c_end) > 0.01:
                    logger.info("Skipping stale Stage 2 result for %s after its range changed", clip_id)
                    continue
                range_revision = int(latest.get("range_revision") or 0)
                if not range_trimmed:
                    logger.info(
                        "Keeping original clip range for %s because the speech-tightened range "
                        "is outside source or product limits.", clip_id
                    )

                clean_words = []
                for w in words:
                    w_s = round(p_start + w.get("start", 0.0), 3)
                    w_e = round(p_start + w.get("end", 0.0), 3)
                    if w_e > c_start and w_s < c_end:
                        clean_words.append({
                            "word": w.get("word", ""),
                            "start": w_s,
                            "end": w_e
                        })

                track = group_words_into_captions(
                    raw_segments=[{"id": 0, "start": c_start, "end": c_end, "words": clean_words}],
                    clip_start=c_start,
                    clip_end=c_end,
                    clip_id=clip_id,
                    preset_id=get_default_preset_id()
                )
                if not db.save_clip_caption_data(
                    clip_id, json.dumps(track.to_dict()), expected_range_revision=range_revision,
                    source_range_start=c_start, source_range_end=c_end,
                ):
                    logger.info("Skipping stale caption save for %s after its range changed", clip_id)
                    continue

            render_clip(
                source_video=source_path,
                clip_id=clip_id,
                start_time=c_start,
                end_time=c_end,
                aspect_ratio=None,
                layout_mode=clip.get("layout_mode", "fit_blur"),
                subtitle_options={"enabled": False, "preserve_timing": True},
                expected_range_revision=range_revision,
            )

            with db.get_db_connection() as conn:
                conn.cursor().execute("UPDATE clips SET status = 'ready_review' WHERE id = ?", (clip_id,))
                conn.commit()

            drained += 1
            logger.info("Successfully drained queued clip %s -> ready_review", clip_id)

        except Exception as e:
            logger.error("Failed to drain queued clip %s: %s", clip_id, e)

    return drained

def transcribe_video(

    session_id: Optional[str] = None,
    model_size: str = "small",
    initial_prompt: Optional[str] = None,

    """Transcribe video audio to text with word-level timestamps, Hinglish optimization, and VAD silence filtering.

        # Determine language setting
        import os
        configured_lang = language or db.get_setting("transcription_language") or os.getenv("DISPATCH_TRANSCRIPTION_LANGUAGE", "")
        if configured_lang and configured_lang.lower() not in ("auto", "hinglish", "none"):
            whisper_lang = configured_lang.lower().strip()
        else:
            whisper_lang = None

        # Enhanced initial prompt for natural Hindi-English (Hinglish) code-switching in Roman script
        if not initial_prompt:
            initial_prompt = (
                "Yeh video Hindi aur English mixed Hinglish language me hai. "
                "Main aaj discuss karunga video creation, coding, automation, workflow design, aur productivity tricks."
            )

        model = get_whisper_model(model_size=model_size)

        logger.info("Starting speech transcription (lang=%s) with word-level timestamps and VAD on %s",
                    whisper_lang or "auto-detect", audio_path.name)
        segments_raw, info = model.transcribe(
            str(audio_path),
            beam_size=1,
            word_timestamps=True,
            vad_filter=True,
            vad_parameters=dict(min_silence_duration_ms=500),
            initial_prompt=initial_prompt,
            language=whisper_lang,
            condition_on_previous_text=False
        )

        full_text_pieces = []
        formatted_segments = []

        for seg in segments_raw:
            seg_text = seg.text.strip()
            if not seg_text:
                continue

            full_text_pieces.append(seg_text)

            words_data = []
            if seg.words:
                for w in seg.words:
                    words_data.append({
                        "word": w.word.strip(),
                        "start": round(w.start, 3),
                        "end": round(w.end, 3),
                        "probability": round(w.probability, 3)
                    })

            formatted_segments.append({
                "id": seg.id,
                "start": round(seg.start, 3),
                "end": round(seg.end, 3),
                "text": seg_text,
                "words": words_data
            })

        # Resilient fallback: If VAD filtered everything, try once without VAD filter
        if not formatted_segments and audio_path.exists():
            logger.info("VAD detected 0 segments. Retrying transcription with vad_filter=False fallback.")
            segments_raw, info = model.transcribe(
                str(audio_path),
                beam_size=3,
                word_timestamps=True,
                vad_filter=False,
                initial_prompt=initial_prompt,
                language=whisper_lang,
                condition_on_previous_text=False
            )
            for seg in segments_raw:
                seg_text = seg.text.strip()
                if not seg_text:
                    continue
                full_text_pieces.append(seg_text)
                words_data = []
                if seg.words:
                    for w in seg.words:
                        words_data.append({
                            "word": w.word.strip(),
                            "start": round(w.start, 3),
                            "end": round(w.end, 3),
                            "probability": round(w.probability, 3)
                        })
                formatted_segments.append({
                    "id": seg.id,
                    "start": round(seg.start, 3),
                    "end": round(seg.end, 3),
                    "text": seg_text,
                    "words": words_data
                })

        full_text = " ".join(full_text_pieces)
        logger.info("Transcription completed: %d segments, %d total words",
                    len(formatted_segments), len(full_text.split()))

        # Save to SQLite database if chunk_id is provided
        transcript_id = None
        if chunk_id:
            transcript_id = db.save_transcript(
                chunk_id=chunk_id,
                session_id=session_id,
                full_text=full_text,
                segments=formatted_segments
            )

        return {
            "transcript_id": transcript_id,
            "full_text": full_text,
            "segments": formatted_segments,
            "language": getattr(info, "language", "unknown") if info else "unknown",
            "language_probability": getattr(info, "language_probability", 0.0) if info else 0.0
        }
    finally:
        # Clean up temporary .wav file to conserve disk space even on exceptions
        if audio_path and audio_path.exists():
            try:
                audio_path.unlink()
            except Exception as e:
                logger.debug("Could not remove temp audio file: %s", e)
