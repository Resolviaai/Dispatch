const clamp = (val: number, min: number, max: number) => Math.min(max, Math.max(min, val));

function formatTimeRuler(sec: number): string {
  const safe = Math.max(0, Number.isFinite(sec) ? sec : 0);
  const m = Math.floor(safe / 60);
  const s = Math.floor(safe % 60);
  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
}

export const CaptionTimeline: React.FC<CaptionTimelineProps> = ({
  track,
  currentTime,
  duration = 40,
  selectedGroupId,
  isVideoSelected = false,
  isPlaying,
  filmstripFrames = [],
  mediaElementRef,
  clipStart = 0,
  clipEnd,
  paddedStart,

  sourceDuration,
  sourceTimebase = true,
  onTrimChange,
  onPreviewContext,
  onSeek,
  onSelectGroup,
  onSelectVideo,
  onUpdateGroup,
}) => {
  const viewportRef = useRef<HTMLDivElement>(null);
  const scrollContainerRef = useRef<HTMLDivElement>(null);
  const rulerContentRef = useRef<HTMLDivElement>(null);
  const [viewportWidth, setViewportWidth] = useState<number>(340);
  const [pxPerSec, setPxPerSec] = useState<number>(55);
  const [isScrubbing, setIsScrubbing] = useState<boolean>(false);
  const [isSnapEnabled, setIsSnapEnabled] = useState<boolean>(false);
  const [snapGuideTime, setSnapGuideTime] = useState<number | null>(null);
  const [dragSession, setDragSession] = useState<DragSession | null>(null);
  const [hiddenTracks, setHiddenTracks] = useState<Record<number, boolean>>({});
  const canvasDragRef = useRef<{
    startX: number;
    startY: number;
    startScrollLeft: number;
    startScrollTop: number;
    isIntentLocked: boolean;
    isScrubbingHorizontal: boolean;
  } | null>(null);
  const leftRailRef = useRef<HTMLDivElement>(null);
  const latestBlockPatchRef = useRef<{ groupId: string; patch: Partial<CaptionGroup> } | null>(null);
  const isProgrammaticScrollRef = useRef<boolean>(false);
  const pendingSeekTimeRef = useRef<number | null>(null);
  const rafSeekIdRef = useRef<number | null>(null);

  const timelineDuration = Math.max(1, sourceDuration || clipEnd || (clipStart + duration), duration);
  const safeClipDuration = Math.max(1, duration);
  const timelineCurrentTime = clipTimeToSourceTime(currentTime, clipStart);

  // Trim boundaries and the filmstrip use source time. Caption words stay clip-relative.
  const [activeTrim, setActiveTrim] = useState<{ start: number; end: number }>({
    start: clipStart,
    end: clipEnd ?? (clipStart + duration),
  });
  const activeTrimRef = useRef(activeTrim);
  activeTrimRef.current = activeTrim;

  const [trimDrag, setTrimDrag] = useState<TrimPointerState | null>(null);
  const trimDragRef = useRef<TrimPointerState | null>(null);
  const [trimSaveState, setTrimSaveState] = useState<'idle' | 'saving' | 'saved' | 'error'>('idle');

  useLayoutEffect(() => {
    const nextRange = {
      start: Math.max(0, clipStart),
      end: Math.min(timelineDuration, clipEnd ?? (clipStart + duration)),
    };
    activeTrimRef.current = nextRange;
    setActiveTrim(nextRange);
  }, [clipStart, clipEnd, duration, timelineDuration]);

  const currentTimeRef = useRef(currentTime);
  currentTimeRef.current = currentTime;
  const pxPerSecRef = useRef(pxPerSec);
  pxPerSecRef.current = pxPerSec;

  const safeDuration = timelineDuration;
  const totalTimelineWidth = Math.max(340, Math.round(safeDuration * pxPerSec));
  const selectionDuration = Math.max(0, activeTrim.end - activeTrim.start);
  const canEditSelection = Boolean(sourceDuration && sourceDuration > 0 && onTrimChange);
  const beforeContext = sourceContextRange(activeTrim, 'before', sourceDuration || 0);
  const afterContext = sourceContextRange(activeTrim, 'after', sourceDuration || 0);

  // Measure timeline viewport width to anchor fixed center playhead (50% horizontal center)
  useEffect(() => {
    const el = viewportRef.current;
    if (!el) return;
    const updateWidth = () => {
      if (el.clientWidth > 0) {
        setViewportWidth(el.clientWidth);
      }
    };
    updateWidth();
    const observer = new ResizeObserver(updateWidth);
    observer.observe(el);
    return () => observer.disconnect();
  }, []);

  const halfWidth = Math.max(80, Math.round(viewportWidth / 2));

  // =========================================================================
  // UNIFORM DYNAMIC MULTI-TRACK ALLOCATION
  // Maps any media type to uniform tracks: Main, L1, L2, L3, L4...
  // =========================================================================
  const dynamicTracks = useMemo(() => {
    if (!track || !track.groups || track.groups.length === 0) {
      return [];
    }

    const sorted = [...track.groups].sort((a, b) => a.start - b.start);
    const buckets: CaptionGroup[][] = [];

    sorted.forEach((grp) => {
      if (grp.trackIndex !== undefined && grp.trackIndex >= 0) {
        while (buckets.length <= grp.trackIndex) {
          buckets.push([]);
        }
        buckets[grp.trackIndex].push(grp);
        return;
      }

      let placed = false;
      for (let i = 0; i < buckets.length; i++) {
        const bucket = buckets[i];
        const hasOverlap = bucket.some(
          (existing) => Math.max(grp.start, existing.start) < Math.min(grp.end, existing.end)
        );
        if (!hasOverlap) {
          bucket.push(grp);
          placed = true;
          break;
        }
      }

      if (!placed) {
        buckets.push([grp]);
      }
    });

    return buckets;
  }, [track]);

  // Dynamic ruler ticks based on zoom level
  const rulerTicks = useMemo(() => {
    let step = 5;
    if (pxPerSec >= 80) step = 1;
    else if (pxPerSec >= 45) step = 2;
    else if (pxPerSec >= 25) step = 5;
    else step = 10;

    const count = Math.ceil(safeDuration / step);
    const ticks: number[] = [];
    for (let i = 0; i <= count; i++) {
      const t = i * step;
      if (t <= safeDuration) ticks.push(t);
    }
    return ticks;
  }, [safeDuration, pxPerSec]);

  // Coalesced seeking to prevent video playback thrashing and seek queuing
  const scheduleSeek = (timeSec: number) => {
    pendingSeekTimeRef.current = timeSec;
    if (!rafSeekIdRef.current) {
      rafSeekIdRef.current = requestAnimationFrame(() => {
        rafSeekIdRef.current = null;
        if (pendingSeekTimeRef.current !== null) {
          onSeek(pendingSeekTimeRef.current);
        }
      });
    }
  };

  const flushSeek = (timeSec?: number) => {
    if (rafSeekIdRef.current) {
      cancelAnimationFrame(rafSeekIdRef.current);
      rafSeekIdRef.current = null;
    }
    const finalTime = timeSec !== undefined ? timeSec : pendingSeekTimeRef.current;
    if (finalTime !== null && finalTime !== undefined) {
      onSeek(finalTime);
      pendingSeekTimeRef.current = null;
    }
  };

  const cancelSeek = () => {
    if (rafSeekIdRef.current) {
      cancelAnimationFrame(rafSeekIdRef.current);
      rafSeekIdRef.current = null;
    }
    pendingSeekTimeRef.current = null;
  };

  // Zoom adjustments: keep the current time anchored at fixed center playhead
  const handleZoom = (delta: number) => {
    const nextPx = clamp(pxPerSec + delta, 20, 140);
    if (nextPx === pxPerSec) return;
    setPxPerSec(nextPx);
    requestAnimationFrame(() => {
      if (scrollContainerRef.current) {
        isProgrammaticScrollRef.current = true;
        scrollContainerRef.current.scrollLeft = timelineCurrentTime * nextPx;
        if (rulerContentRef.current) {
          rulerContentRef.current.style.transform = `translateX(${-timelineCurrentTime * nextPx + halfWidth}px)`;
        }
      }
    });
  };

  const handleZoomFit = () => {
    const scrollEl = scrollContainerRef.current;
    if (!scrollEl) return;
    const availableWidth = Math.max(160, viewportWidth - 40);
    const fitPx = clamp(Math.floor(availableWidth / safeDuration), 15, 60);
    setPxPerSec(fitPx);
    requestAnimationFrame(() => {
      if (scrollContainerRef.current) {
        isProgrammaticScrollRef.current = true;
        scrollContainerRef.current.scrollLeft = timelineCurrentTime * fitPx;
        if (rulerContentRef.current) {
          rulerContentRef.current.style.transform = `translateX(${-timelineCurrentTime * fitPx + halfWidth}px)`;
        }
      }
    });
  };

  // Follow playhead smoothly during playback using decoupled media-frame scheduling
  // Drives timeline movement directly from the media clock (requestVideoFrameCallback where supported)
  // Does NOT restart/teardown on every frame update of currentTime
  useEffect(() => {
    if (!isPlaying || isScrubbing || dragSession) return;
    let animId: number;
    let rvfcId: number;
    let cancelled = false;

    const v = mediaElementRef?.current;

    const tick = () => {
      if (cancelled) return;
      const scrollEl = scrollContainerRef.current;
      if (scrollEl) {
        let cur = clipStart + currentTimeRef.current;
        if (v && !v.paused) {
          const raw = v.currentTime;
          cur = sourceTimebase ? raw : clipStart + raw;
        }
        const targetScroll = cur * pxPerSecRef.current;
        if (Math.abs(scrollEl.scrollLeft - targetScroll) > 0.5) {
          isProgrammaticScrollRef.current = true;
          scrollEl.scrollLeft = targetScroll;
          if (rulerContentRef.current) {
            rulerContentRef.current.style.transform = `translateX(${-targetScroll + halfWidth}px)`;
          }
        }
      }

      if (v && 'requestVideoFrameCallback' in v) {
        rvfcId = (v as any).requestVideoFrameCallback(tick);
      } else {
        animId = requestAnimationFrame(tick);
      }
    };

    if (v && 'requestVideoFrameCallback' in v) {
      rvfcId = (v as any).requestVideoFrameCallback(tick);
    } else {
      animId = requestAnimationFrame(tick);
    }

    return () => {
      cancelled = true;
      if (v && 'cancelVideoFrameCallback' in v && rvfcId) {
        (v as any).cancelVideoFrameCallback(rvfcId);
      }
      if (animId) cancelAnimationFrame(animId);
    };
  }, [isPlaying, isScrubbing, !!dragSession, mediaElementRef, clipStart, sourceTimebase, halfWidth]);

  // When paused and external seek happens (e.g. transport scrub), center timeline
  useEffect(() => {
    if (isPlaying || isScrubbing || dragSession) return;
    const scrollEl = scrollContainerRef.current;
    if (!scrollEl) return;

    const targetScroll = timelineCurrentTime * pxPerSec;
    if (Math.abs(scrollEl.scrollLeft - targetScroll) > 1) {
      isProgrammaticScrollRef.current = true;
      scrollEl.scrollLeft = targetScroll;
      if (rulerContentRef.current) {
        rulerContentRef.current.style.transform = `translateX(${-targetScroll + halfWidth}px)`;
      }
    }
  }, [timelineCurrentTime, pxPerSec, isPlaying, isScrubbing, !!dragSession, halfWidth]);

  // Synchronize ruler on mount or viewport resize
  useEffect(() => {
    if (rulerContentRef.current && scrollContainerRef.current) {
      rulerContentRef.current.style.transform = `translateX(${-scrollContainerRef.current.scrollLeft + halfWidth}px)`;
    }
  }, [halfWidth, pxPerSec]);

  // Scroll listener: updates ruler and optionally seeks during manual touch/trackpad scrub
  // Programmatic scroll (playhead follower) never seeks or triggers recursive loops
  const handleScroll = (e: React.UIEvent<HTMLDivElement>) => {
    const sl = e.currentTarget.scrollLeft;
    if (rulerContentRef.current) {
      rulerContentRef.current.style.transform = `translateX(${-sl + halfWidth}px)`;
    }
    if (leftRailRef.current) {
      leftRailRef.current.scrollTop = e.currentTarget.scrollTop;
    }

    if (isProgrammaticScrollRef.current) {
      isProgrammaticScrollRef.current = false;
      return;
    }

    if (isScrubbing) {
      const scrubTime = clamp(sl / pxPerSec, 0, safeDuration);
      scheduleSeek(Number(sourceTimeToClipTime(scrubTime, clipStart, safeClipDuration).toFixed(2)));
    }
  };

  const handleLeftRailWheel = (e: React.WheelEvent) => {
    e.preventDefault();
    if (scrollContainerRef.current) {
      scrollContainerRef.current.scrollTop += e.deltaY;
    }
  };

  const leftRailTouchRef = useRef<{ startY: number; startScrollTop: number } | null>(null);
  const handleLeftRailTouchStart = (e: React.TouchEvent) => {
    if (e.touches.length > 0) {
      leftRailTouchRef.current = {
        startY: e.touches[0].clientY,
        startScrollTop: scrollContainerRef.current?.scrollTop || 0,
      };
    }
  };
  const handleLeftRailTouchMove = (e: React.TouchEvent) => {
    if (leftRailTouchRef.current && scrollContainerRef.current && e.touches.length > 0) {
      const dy = e.touches[0].clientY - leftRailTouchRef.current.startY;
      scrollContainerRef.current.scrollTop = leftRailTouchRef.current.startScrollTop - dy;
    }
  };
  const handleLeftRailTouchEnd = () => {
    leftRailTouchRef.current = null;
  };

  // Direct swipe/scrub on canvas or ruler (distinguishes horizontal scrub from vertical scrolling)
  const handleCanvasPointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
    if (dragSession) return;
    if ((e.target as HTMLElement).closest('[data-layer-block="true"], [data-video-track="true"], button, input, [data-handle]')) return;
    canvasDragRef.current = {
      startX: e.clientX,
      startY: e.clientY,
      startScrollLeft: scrollContainerRef.current?.scrollLeft || 0,
      startScrollTop: scrollContainerRef.current?.scrollTop || 0,
      isIntentLocked: false,
      isScrubbingHorizontal: false,
    };
    (e.currentTarget as HTMLElement).setPointerCapture?.(e.pointerId);
  };

  const handleCanvasPointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
    if (!canvasDragRef.current || !scrollContainerRef.current) return;
    const dx = e.clientX - canvasDragRef.current.startX;
    const dy = e.clientY - canvasDragRef.current.startY;

    if (!canvasDragRef.current.isIntentLocked) {
      if (Math.abs(dx) > 6 || Math.abs(dy) > 6) {
        canvasDragRef.current.isIntentLocked = true;
        canvasDragRef.current.isScrubbingHorizontal = Math.abs(dx) >= Math.abs(dy);
        if (canvasDragRef.current.isScrubbingHorizontal) {
          setIsScrubbing(true);
        }
      }
      return;
    }

    if (canvasDragRef.current.isScrubbingHorizontal) {
      const targetScroll = Math.max(0, canvasDragRef.current.startScrollLeft - dx);
      isProgrammaticScrollRef.current = true;
      scrollContainerRef.current.scrollLeft = targetScroll;
      if (rulerContentRef.current) {
        rulerContentRef.current.style.transform = `translateX(${-targetScroll + halfWidth}px)`;
      }
      const scrubTime = clamp(targetScroll / pxPerSec, 0, safeDuration);
      scheduleSeek(Number(sourceTimeToClipTime(scrubTime, clipStart, safeClipDuration).toFixed(2)));
    } else {
      // Smooth vertical scrolling
      scrollContainerRef.current.scrollTop = canvasDragRef.current.startScrollTop - dy;
    }
  };

  const handleCanvasPointerUp = (e: React.PointerEvent<HTMLDivElement>) => {
    if (canvasDragRef.current) {
      const moved = Math.hypot(e.clientX - canvasDragRef.current.startX, e.clientY - canvasDragRef.current.startY);
      const wasHorizontal = canvasDragRef.current.isScrubbingHorizontal;
      canvasDragRef.current = null;
      setIsScrubbing(false);
      try {
        (e.currentTarget as HTMLElement).releasePointerCapture?.(e.pointerId);
      } catch { }

      if (wasHorizontal) {
        flushSeek();
      }

      // Genuinely empty timeline area clicked/tapped (motion < 5px): deselect active layer
      if (moved < 5) {
        const isInteractive = (e.target as HTMLElement).closest('[data-layer-block="true"], [data-video-track="true"], button, input, [data-handle]');
        if (!isInteractive) {
          onSelectGroup(null);
        }
      }
    }
  };

  const handleCanvasPointerCancel = (e: React.PointerEvent<HTMLDivElement>) => {
    canvasDragRef.current = null;
    setIsScrubbing(false);
    cancelSeek();
    try {
      (e.currentTarget as HTMLElement).releasePointerCapture?.(e.pointerId);
    } catch { }
  };

  // Draggable / Trimmable Layer Block Pointer Handlers
  const handleBlockPointerDown = (
    e: React.PointerEvent,
    grp: CaptionGroup,
    type: 'move' | 'trim-start' | 'trim-end'
  ) => {
    e.stopPropagation();
    onSelectGroup(grp.id);

    // Capture stable anchors for the entire gesture (never move with playhead)
    const otherEdges: number[] = [];
    if (track && track.groups) {
      for (const g of track.groups) {
        if (g.id !== grp.id) {
          otherEdges.push(g.start);
          otherEdges.push(g.end);
        }
      }
    }
    const stableAnchors = [0, safeClipDuration, ...otherEdges];

    setDragSession({
      type,
      groupId: grp.id,
      startX: e.clientX,
      startY: e.clientY,
      lastY: e.clientY,
      origStart: grp.start,
      origEnd: grp.end,
      origWords: grp.words ? [...grp.words] : [],
      stableAnchors,
      playheadTime: currentTimeRef.current,
      isIntentLocked: type !== 'move',
      isVerticalScroll: false,
    });

    if (type !== 'move') {
      (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
    }
  };

  const handleBlockPointerMove = (e: React.PointerEvent) => {
    if (!dragSession || !onUpdateGroup) return;
    e.stopPropagation();

    // Intent arbitration: distinguish vertical scrolling across multiple layer tracks from horizontal dragging
    if (!dragSession.isIntentLocked && dragSession.type === 'move') {
      const dx = Math.abs(e.clientX - dragSession.startX);
      const dy = Math.abs(e.clientY - dragSession.startY);
      if (dx > 5 || dy > 5) {
        if (dy > dx) {
          dragSession.isIntentLocked = true;
          dragSession.isVerticalScroll = true;
        } else {
          dragSession.isIntentLocked = true;
          dragSession.isVerticalScroll = false;
          (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
        }
      } else {
        return;
      }
    }

    if (dragSession.isVerticalScroll) {
      if (scrollContainerRef.current) {
        const deltaY = e.clientY - dragSession.lastY;
        scrollContainerRef.current.scrollTop -= deltaY;
        dragSession.lastY = e.clientY;
      }
      return;
    }

    const deltaSec = (e.clientX - dragSession.startX) / pxPerSec;
    const origDuration = Math.max(0.1, dragSession.origEnd - dragSession.origStart);
    let activeSnap: number | null = null;
    const SNAP_PX = 12;
    const snapThreshold = SNAP_PX / pxPerSec;

    if (dragSession.type === 'move') {
      let proposedStart = dragSession.origStart + deltaSec;
      let proposedEnd = proposedStart + origDuration;

      // Bound within media duration
      if (proposedStart < 0) {
        proposedStart = 0;
        proposedEnd = origDuration;
      } else if (proposedEnd > safeClipDuration) {
        proposedEnd = safeClipDuration;
        proposedStart = Math.max(0, safeClipDuration - origDuration);
      }

      let finalStart = proposedStart;
      let finalEnd = proposedEnd;

      // Snapping is only active when explicitly enabled via Magnet toggle
      if (isSnapEnabled) {
        const candidateAnchors = Array.from(
          new Set([dragSession.playheadTime, ...dragSession.stableAnchors])
        );

        let bestCandidate: { edge: 'start' | 'end'; anchor: number; diff: number } | null = null;

        for (const anchor of candidateAnchors) {
          const startDiff = Math.abs(proposedStart - anchor);
          const endDiff = Math.abs(proposedEnd - anchor);

          // Directional playhead & boundary snapping:
          // If block is positioned primarily to the right of anchor, snap start edge
          if (startDiff <= snapThreshold && proposedStart >= anchor - snapThreshold) {
            if (!bestCandidate || startDiff < bestCandidate.diff) {
              bestCandidate = { edge: 'start', anchor, diff: startDiff };
            }
          }
          // If block is positioned primarily to the left of anchor, snap end edge
          if (endDiff <= snapThreshold && proposedEnd <= anchor + snapThreshold) {
            if (!bestCandidate || endDiff < bestCandidate.diff) {
              bestCandidate = { edge: 'end', anchor, diff: endDiff };
            }
          }
        }

        // Fallback to nearest edge within threshold
        if (!bestCandidate) {
          for (const anchor of candidateAnchors) {
            const startDiff = Math.abs(proposedStart - anchor);
            const endDiff = Math.abs(proposedEnd - anchor);
            if (startDiff <= snapThreshold && (!bestCandidate || startDiff < bestCandidate.diff)) {
              bestCandidate = { edge: 'start', anchor, diff: startDiff };
            }
            if (endDiff <= snapThreshold && (!bestCandidate || endDiff < bestCandidate.diff)) {
              bestCandidate = { edge: 'end', anchor, diff: endDiff };
            }
          }
        }

        if (bestCandidate) {
          activeSnap = bestCandidate.anchor;
          if (bestCandidate.edge === 'start') {
            finalStart = bestCandidate.anchor;
            finalEnd = finalStart + origDuration;
          } else {
            finalEnd = bestCandidate.anchor;
            finalStart = Math.max(0, finalEnd - origDuration);
          }
        }
      }

      setSnapGuideTime(activeSnap);

      const timeShift = finalStart - dragSession.origStart;
      const nextWords: Word[] = dragSession.origWords.map((w) => ({
        ...w,
        start: Number((w.start + timeShift).toFixed(3)),
        end: Number((w.end + timeShift).toFixed(3)),
      }));

      const patch: Partial<CaptionGroup> = {
        start: Number(finalStart.toFixed(3)),
        end: Number(finalEnd.toFixed(3)),
        words: nextWords,
      };
      latestBlockPatchRef.current = { groupId: dragSession.groupId, patch };
      onUpdateGroup(dragSession.groupId, patch, false);
      // NOTE: Stationary playhead during drag: never call onSeek here!
    } else if (dragSession.type === 'trim-start') {
      let proposedStart = dragSession.origStart + deltaSec;
      proposedStart = clamp(proposedStart, 0, dragSession.origEnd - 0.2);

      let finalStart = proposedStart;
      if (isSnapEnabled) {
        const candidateAnchors = Array.from(
          new Set([dragSession.playheadTime, ...dragSession.stableAnchors])
        );
        for (const anchor of candidateAnchors) {
          if (anchor < dragSession.origEnd - 0.2 && Math.abs(proposedStart - anchor) <= snapThreshold) {
            finalStart = anchor;
            activeSnap = anchor;
            break;
          }
        }
      }
      setSnapGuideTime(activeSnap);

      // Preserve authentic speech word timestamps without proportional stretching or fabricating words
      const remainingWords = dragSession.origWords.filter((w) => w.end > finalStart);
      const nextWords: Word[] = remainingWords.map((w) => ({
        ...w,
        start: Math.max(finalStart, w.start),
        end: Math.max(finalStart, w.end),
      }));

      const patch: Partial<CaptionGroup> = {
        start: Number(finalStart.toFixed(3)),
        words: nextWords,
      };
      latestBlockPatchRef.current = { groupId: dragSession.groupId, patch };
      onUpdateGroup(dragSession.groupId, patch, false);
    } else if (dragSession.type === 'trim-end') {
      let proposedEnd = dragSession.origEnd + deltaSec;
      proposedEnd = clamp(proposedEnd, dragSession.origStart + 0.2, safeClipDuration);

      let finalEnd = proposedEnd;
      if (isSnapEnabled) {
        const candidateAnchors = Array.from(
          new Set([dragSession.playheadTime, ...dragSession.stableAnchors])
        );
        for (const anchor of candidateAnchors) {
          if (anchor > dragSession.origStart + 0.2 && Math.abs(proposedEnd - anchor) <= snapThreshold) {
            finalEnd = anchor;
            activeSnap = anchor;
            break;
          }
        }
      }
      setSnapGuideTime(activeSnap);

      // Preserve authentic speech word timestamps without proportional stretching or fabricating words
      const remainingWords = dragSession.origWords.filter((w) => w.start < finalEnd);
      const nextWords: Word[] = remainingWords.map((w) => ({
        ...w,
        start: Math.min(finalEnd, w.start),
        end: Math.min(finalEnd, w.end),
      }));

      const patch: Partial<CaptionGroup> = {
        end: Number(finalEnd.toFixed(3)),
        words: nextWords,
      };
      latestBlockPatchRef.current = { groupId: dragSession.groupId, patch };
      onUpdateGroup(dragSession.groupId, patch, false);
    }
  };

  const handleBlockPointerUp = (e: React.PointerEvent) => {
    setSnapGuideTime(null);
    if (!dragSession?.isVerticalScroll && latestBlockPatchRef.current && onUpdateGroup) {
      onUpdateGroup(latestBlockPatchRef.current.groupId, latestBlockPatchRef.current.patch, true);
    }
    latestBlockPatchRef.current = null;
    if (dragSession) {
      setDragSession(null);
      try {
        (e.target as HTMLElement).releasePointerCapture?.(e.pointerId);
      } catch { }
    }
  };

  const handleBlockPointerCancel = (e: React.PointerEvent) => {
    setSnapGuideTime(null);
    if (dragSession && !dragSession.isVerticalScroll && onUpdateGroup) {
      // Safely revert transient changes on cancellation
      onUpdateGroup(dragSession.groupId, {
        start: dragSession.origStart,
        end: dragSession.origEnd,
        words: dragSession.origWords,
      }, false);
    }
    latestBlockPatchRef.current = null;
    if (dragSession) {
      setDragSession(null);
      try {
        (e.target as HTMLElement).releasePointerCapture?.(e.pointerId);
      } catch { }
    }
  };

  const handleTrimPointerDown = (side: 'start' | 'end', e: React.PointerEvent) => {
    if (!(sourceDuration && sourceDuration > 0) || !onTrimChange || trimSaveState === 'saving') return;
    e.stopPropagation();
    const nextDrag = beginTrimPointer(side, e.pointerId, e.clientX, activeTrimRef.current);
    trimDragRef.current = nextDrag;
    setTrimDrag(nextDrag);
    try { (e.target as HTMLElement).setPointerCapture?.(e.pointerId); } catch { /* pointer can end before capture */ }
  };

  const handleTrimPointerMove = (e: React.PointerEvent) => {
    const drag = trimDragRef.current;
    if (!drag || drag.pointerId !== e.pointerId) return;
    e.stopPropagation();
    const nextRange = moveTrimPointer(drag, e.pointerId, e.clientX, pxPerSec, safeDuration);
    nextRange.start = Number(nextRange.start.toFixed(2));
    nextRange.end = Number(nextRange.end.toFixed(2));
    activeTrimRef.current = nextRange;
    setActiveTrim(nextRange);
    const sourceSeek = drag.side === 'start' ? nextRange.start : nextRange.end;
    scheduleSeek(sourceTimeToClipTime(sourceSeek, clipStart, safeClipDuration));
  };

  const releaseTrimPointer = (e: React.PointerEvent) => {
    try {
      (e.target as HTMLElement).releasePointerCapture?.(e.pointerId);
    } catch { }
  };

  const persistTrimRange = async (range: { start: number; end: number }) => {
    if (!onTrimChange || !(sourceDuration && sourceDuration > 0)) return false;
    const previous = { start: Math.max(0, clipStart), end: clipEnd ?? (clipStart + duration) };
    setTrimSaveState('saving');
    try {
      const saved = await onTrimChange(range.start, range.end);
      if (saved === false) {
        activeTrimRef.current = previous;
        setActiveTrim(previous);
        setTrimSaveState('error');
        return false;
      }
      setTrimSaveState('saved');
      return true;
    } catch {
      activeTrimRef.current = previous;
      setActiveTrim(previous);
      setTrimSaveState('error');
      return false;
    }
  };

  const applyRangeExtension = (side: 'start' | 'end', delta: number) => {
    const previous = activeTrimRef.current;
    const next = extendSourceRange(previous, side, delta, sourceDuration || 0);
    if (next.start === previous.start && next.end === previous.end) return;
    activeTrimRef.current = next;
    setActiveTrim(next);
    void persistTrimRange(next);
  };

  const handleTrimPointerUp = (e: React.PointerEvent) => {
    const drag = trimDragRef.current;
    if (drag) {
      if (drag.pointerId !== e.pointerId) return;
      const finalRange = activeTrimRef.current;
      const sourceSeek = drag.side === 'start' ? finalRange.start : finalRange.end;
      flushSeek(sourceTimeToClipTime(sourceSeek, clipStart, safeClipDuration));
      trimDragRef.current = null;
      setTrimDrag(null);
      releaseTrimPointer(e);
      if (onTrimChange) void persistTrimRange(finalRange);
    }
  };

  const handleTrimPointerCancel = (e: React.PointerEvent) => {
    const drag = trimDragRef.current;
    if (!drag || drag.pointerId !== e.pointerId) return;
    cancelSeek();
    activeTrimRef.current = drag.initialRange;
    setActiveTrim(drag.initialRange);
    trimDragRef.current = null;
    setTrimDrag(null);
    releaseTrimPointer(e);
  };

  const toggleTrackVisibility = (idx: number) => {
    const isNowHidden = !hiddenTracks[idx];
    setHiddenTracks((prev) => ({ ...prev, [idx]: isNowHidden }));
    const rowGroups = dynamicTracks[idx] || [];
    rowGroups.forEach((grp) => {
      onUpdateGroup?.(grp.id, { is_hidden: isNowHidden }, true);
    });
  };

  // Palette and Icon based on layer type
  const getLayerMeta = (grp: CaptionGroup, idx: number) => {
    if (grp.type === 'audio') {
      return {
        icon: <Music size={11} className="mr-1 shrink-0" />,
        bg: 'bg-[#059669]/90',
        border: 'border-[#10B981]',
        text: 'text-emerald-100',
        name: grp.label || 'Audio',
      };
    }
    if (grp.type === 'voiceover') {
      return {
        icon: <Mic size={11} className="mr-1 shrink-0" />,
        bg: 'bg-[#D97706]/90',
        border: 'border-[#F59E0B]',
        text: 'text-amber-100',
        name: grp.label || 'Voiceover',
      };
    }
    if (grp.type === 'overlay') {
      const isVideo = grp.overlay_type === 'video';
      return {
        icon: isVideo ? <Film size={11} className="mr-1 shrink-0" /> : <ImageIcon size={11} className="mr-1 shrink-0" />,
        bg: isVideo ? 'bg-[#4338CA]/90' : 'bg-[#7C3AED]/90',
        border: isVideo ? 'border-[#6366F1]' : 'border-[#8B5CF6]',
        text: 'text-purple-100',
        name: grp.label || (isVideo ? 'Video' : 'Image'),
      };
    }
    if (grp.type === 'text') {
      return {
        icon: <Type size={11} className="mr-1 shrink-0 text-cyan-200" />,
        bg: 'bg-[#0284C7]/90',
        border: 'border-[#38BDF8]',
        text: 'text-sky-100',
        badge: 'Text',
        name: (grp.words || []).map((w) => w.text || '').join(' ') || 'Text',
      };
    }
    // Default caption (type === 'caption' or undefined)
    return {
      icon: <Sparkles size={11} className="mr-1 shrink-0 text-amber-200" />,
      bg: 'bg-[#2563EB]/90',
      border: 'border-[#3B82F6]',
      text: 'text-blue-100',
      badge: 'Caption',
      name: (grp.words || []).map((w) => w.text || '').join(' ') || 'Caption',
    };
  };

  return (
    <section
      className="w-full flex-1 min-h-[220px] max-h-full bg-[#13161B] overflow-hidden select-none flex flex-col min-h-0"
    >
      {/* TIMELINE UTILITY BAR: Magnet Snapping + Clean compact zoom controls */}
      <div className="h-6 px-2 bg-[#16191E] border-b border-[#232832] flex items-center justify-between text-text-secondary text-[10px] shrink-0">
        {/* Left side: Snap / Magnet toggle */}
        <div className="flex items-center gap-1.5">
          <button
            type="button"
            onClick={() => setIsSnapEnabled(!isSnapEnabled)}
            className={`px-1.5 h-4.5 rounded flex items-center gap-1 text-[10px] font-semibold transition-all active:scale-95 ${
              isSnapEnabled
                ? 'bg-primary/20 text-primary border border-primary/40 shadow-hero-sm font-bold'
                : 'text-text-muted hover:text-text-secondary hover:bg-[#232832] border border-transparent'
            }`}
            title={isSnapEnabled ? 'Magnetic Snapping: ON (Click to disable)' : 'Magnetic Snapping: OFF (Click to enable)'}
            aria-label="Toggle Magnet Snapping"
          >
            <Magnet size={11} className={isSnapEnabled ? 'text-primary' : 'text-text-muted'} />
            <span>Snap {isSnapEnabled ? 'ON' : 'OFF'}</span>
          </button>
        </div>

        {/* Right side: Zoom controls */}
        <div className="flex items-center gap-1">
          <button
            type="button"
            onClick={() => handleZoom(-15)}
            className="w-5 h-5 rounded flex items-center justify-center text-text-secondary hover:text-text-main hover:bg-[#232832] active:scale-95 transition-all"
            title="Zoom Out"
            aria-label="Zoom Out"
          >
            <ZoomOut size={12} />
          </button>

          <button
            type="button"
            onClick={() => handleZoom(15)}
            className="w-5 h-5 rounded flex items-center justify-center text-text-secondary hover:text-text-main hover:bg-[#232832] active:scale-95 transition-all"
            title="Zoom In"
            aria-label="Zoom In"
          >
            <ZoomIn size={12} />
          </button>

          <button
            type="button"
            onClick={handleZoomFit}
            className="w-5 h-5 rounded flex items-center justify-center text-text-secondary hover:text-text-main hover:bg-[#232832] active:scale-95 transition-all"
            title="Fit to Timeline"
            aria-label="Fit to Timeline"
          >
            <Maximize2 size={11} />
          </button>
        </div>
      </div>

      <div
        aria-label="Selected source range controls"
        className="h-11 min-h-11 px-2 bg-[#16191E] border-b border-[#232832] flex items-center gap-2 overflow-x-auto whitespace-nowrap text-[10px]"
      >
        <span className="text-text-secondary font-mono tabular-nums" aria-live="polite">
          Start {formatTimeRuler(activeTrim.start)} · End {formatTimeRuler(activeTrim.end)} · {selectionDuration.toFixed(1)} s
        </span>
        <button type="button" disabled={!canEditSelection || trimSaveState === 'saving'}
          onClick={() => applyRangeExtension('start', -RANGE_CONTROL_STEP_SECONDS)}
          className="h-11 min-w-[44px] px-2 rounded-md border border-border bg-surface-200 text-text-main disabled:opacity-40 active:scale-95"
          aria-label={`Extend selection start earlier by ${RANGE_CONTROL_STEP_SECONDS} seconds`}>
          Start −0.5 s
        </button>
        <button type="button" disabled={!canEditSelection || trimSaveState === 'saving'}
          onClick={() => applyRangeExtension('end', RANGE_CONTROL_STEP_SECONDS)}
          className="h-11 min-w-[44px] px-2 rounded-md border border-border bg-surface-200 text-text-main disabled:opacity-40 active:scale-95"
          aria-label={`Extend selection end later by ${RANGE_CONTROL_STEP_SECONDS} seconds`}>
          End +0.5 s
        </button>
        <button type="button" disabled={!beforeContext || !onPreviewContext || !sourceTimebase}
          onClick={() => beforeContext && onPreviewContext?.(beforeContext)}
          className="h-11 min-w-[44px] px-2 rounded-md border border-border bg-surface-200 text-text-main disabled:opacity-40 active:scale-95"
          aria-label="Preview four seconds before the selected range">
          Preview before
        </button>
        <button type="button" disabled={!afterContext || !onPreviewContext || !sourceTimebase}
          onClick={() => afterContext && onPreviewContext?.(afterContext)}
          className="h-11 min-w-[44px] px-2 rounded-md border border-border bg-surface-200 text-text-main disabled:opacity-40 active:scale-95"
          aria-label="Preview four seconds after the selected range">
          Preview after
        </button>
        <span role="status" className={trimSaveState === 'error' ? 'text-red-400' : 'text-text-muted'}>
          {trimSaveState === 'saving' ? 'Saving range…' : trimSaveState === 'saved' ? 'Range saved' : trimSaveState === 'error' ? 'Range save failed' : ''}
        </span>
        {track?.timing_warning && <span role="alert" className="text-amber-400">{track.timing_warning}</span>}
        {!sourceDuration && <span className="text-text-muted">Source duration unavailable; range editing is disabled.</span>}
      </div>

      {/* TIMELINE MAIN BODY: Sticky Left Rail + Canvas Viewport with Fixed Center Playhead */}
      <div className="flex-1 min-h-0 flex relative overflow-hidden">
        {/* STICKY LEFT RAIL: Main + L1, L2, L3... */}
        <div className="w-12 shrink-0 bg-[#16191E] border-r border-[#232832] flex flex-col z-20">
          {/* Seamless Ruler Header Cell */}
          <div className="h-6 border-b border-[#232832] bg-[#16191E]" />

          {/* Track Headers (scrolls vertically with tracks) */}
          <div
            ref={leftRailRef}
            onWheel={handleLeftRailWheel}
            onTouchStart={handleLeftRailTouchStart}
            onTouchMove={handleLeftRailTouchMove}
            onTouchEnd={handleLeftRailTouchEnd}
            className="flex-1 min-h-0 overflow-hidden flex flex-col touch-none"
          >
            {/* Header: Main Video Track */}
            <button
              type="button"
              onClick={onSelectVideo}
              className={`h-10 shrink-0 border-b border-[#232832] flex items-center justify-center gap-1 transition-colors ${isVideoSelected ? 'bg-primary/20 text-primary font-bold' : 'text-text-muted hover:text-text-secondary'
                }`}
              title="Select Main Video Track"
            >
              <Film size={11} />
              <span className="text-[9px] font-mono font-bold">Main</span>
            </button>

            {/* Headers: Uniform Layers (L1, L2, L3...) */}
            {dynamicTracks.map((_, idx) => (
              <div
                key={`hdr_${idx}`}
                className="h-10 shrink-0 border-b border-[#232832] flex items-center justify-between px-1.5 bg-[#16191E]"
              >
                <span className="text-[10px] font-bold text-text-secondary font-mono">
                  L{idx + 1}
                </span>
                <button
                  type="button"
                  onClick={() => toggleTrackVisibility(idx)}
                  className="text-text-muted hover:text-text-secondary active:scale-95"
                  title={`Toggle Track L${idx + 1}`}
                >
                  {!hiddenTracks[idx] ? <Eye size={10} /> : <EyeOff size={10} className="text-red-400" />}
                </button>
              </div>
            ))}
          </div>
        </div>

        {/* TIMELINE CANVAS VIEWPORT: Fixed Playhead + Ruler + Synchronized Scroll Canvas */}
        <div
          ref={viewportRef}
          className="flex-1 min-h-0 flex flex-col relative overflow-hidden"
        >
          {/* PINNED TIME RULER: Synchronized horizontally with scrollContainer */}
          <div
            className="h-6 shrink-0 border-b border-[#232832] bg-[#13161B] overflow-hidden relative cursor-ew-resize select-none"
            onPointerDown={handleCanvasPointerDown}
            onPointerMove={handleCanvasPointerMove}
            onPointerUp={handleCanvasPointerUp}
            onPointerCancel={handleCanvasPointerUp}
            onClick={(e) => {
              const isInteractive = (e.target as HTMLElement).closest('[data-layer-block="true"], [data-video-track="true"], button, input, [data-handle]');
              if (!isInteractive) {
                onSelectGroup(null);
              }
            }}
          >
            <div
              ref={rulerContentRef}
              className="relative h-full flex items-center pointer-events-none"
              style={{
                transform: `translateX(${-timelineCurrentTime * pxPerSec + halfWidth}px)`,
                width: `${totalTimelineWidth}px`,
                willChange: 'transform',
              }}
            >
              {rulerTicks.map((t) => (
                <div
                  key={t}
                  className="absolute top-0 bottom-0 flex flex-col items-center -translate-x-1/2"
                  style={{ left: `${t * pxPerSec}px` }}
                >
                  <span className="text-[9px] font-mono tabular-nums text-text-secondary font-medium mt-0.5">
                    {formatTimeRuler(t)}
                  </span>
                  <div className="w-[1px] h-1.5 bg-[#3A4350]" />
                </div>
              ))}
            </div>
          </div>

          {/* HORIZONTALLY & VERTICALLY SCROLLABLE TRACK CANVAS */}
          <div
            ref={scrollContainerRef}
            onScroll={handleScroll}
            onTouchStart={() => setIsScrubbing(true)}
            onTouchEnd={() => setIsScrubbing(false)}
            onPointerDown={handleCanvasPointerDown}
            onPointerMove={handleCanvasPointerMove}
            onPointerUp={handleCanvasPointerUp}
            onPointerCancel={handleCanvasPointerUp}
            onClick={(e) => {
              const isInteractive = (e.target as HTMLElement).closest('[data-layer-block="true"], [data-video-track="true"], button, input, [data-handle]');
              if (!isInteractive) {
                onSelectGroup(null);
              }
            }}
            className="flex-1 overflow-x-auto overflow-y-auto relative bg-[#0D1014] timeline-scroll-canvas select-none cursor-ew-resize"
            style={{ touchAction: 'pan-x pan-y' }}
          >
            <div
              className="relative min-h-full"
              style={{
                width: `${totalTimelineWidth + halfWidth * 2}px`,
                paddingLeft: `${halfWidth}px`,
                paddingRight: `${halfWidth}px`,
              }}
            >
              {/* BACKGROUND VERTICAL GRID LINES */}
              <div
                className="absolute inset-y-0 pointer-events-none"
                style={{
                  left: `${halfWidth}px`,
                  width: `${totalTimelineWidth}px`,
                }}
              >
                {rulerTicks.map((t) => (
                  <div
                    key={`grid_${t}`}
                    className="absolute top-0 bottom-0 w-[1px] bg-[#1A1F26]"
                    style={{ left: `${t * pxPerSec}px` }}
                  />
                ))}
              </div>

              {/* 1. MAIN VIDEO TRACK WITH CAPCUT-STYLE SAFETY HANDLES & DUAL DRAG TRIMMING */}
              {(() => {
                const trimLeftPx = Math.max(0, activeTrim.start * pxPerSec);
                const trimRightPx = Math.max(trimLeftPx + 20, activeTrim.end * pxPerSec);
                const trimWidthPx = Math.max(20, trimRightPx - trimLeftPx);

                return (
                  <div
                    data-video-track="true"
                    onClick={(e) => {
                      e.stopPropagation();
                      onSelectVideo?.();
                    }}
                    className={`h-10 shrink-0 border-b border-[#232832] relative p-0.5 overflow-visible cursor-pointer transition-colors ${isVideoSelected ? 'bg-primary/10 ring-1 ring-primary' : ''
                      }`}
                    style={{ width: `${totalTimelineWidth}px` }}
                  >
                    <div className="w-full h-full rounded bg-[#090C0F] border border-[#232832]/60 flex items-center overflow-hidden relative">
                      {filmstripFrames.length > 0 ? (
                        filmstripFrames.map((frameUrl, idx) => (
                          <div
                            key={idx}
                            className="flex-1 h-full border-r border-black/50 overflow-hidden bg-cover bg-center"
                            style={{ backgroundImage: `url(${frameUrl})` }}
                          />
                        ))
                      ) : (
                        <div className="w-full h-full flex items-center justify-around opacity-30 px-2">
                          {Array.from({ length: 8 }).map((_, i) => (
                            <div key={i} className="w-10 h-7 rounded bg-[#2A313C] border border-[#3A4350]" />
                          ))}
                        </div>
                      )}

                      {/* Head Margin Dimmed Overlay (Safety Handle) */}
                      {trimLeftPx > 0 && (
                        <div
                          className="absolute left-0 top-0 bottom-0 bg-black/75 backdrop-blur-[1px] border-r border-dashed border-amber-400/60 pointer-events-none z-10"
                          style={{ width: `${trimLeftPx}px` }}
                        />
                      )}

                      {/* Tail Margin Dimmed Overlay (Safety Handle) */}
                      {trimRightPx < totalTimelineWidth && (
                        <div
                          className="absolute top-0 bottom-0 right-0 bg-black/75 backdrop-blur-[1px] border-l border-dashed border-amber-400/60 pointer-events-none z-10"
                          style={{ left: `${trimRightPx}px` }}
                        />
                      )}

                      {/* Active Video Bounding Box with Dual Drag Handles (CapCut Style) */}
                      <div
                        className="absolute top-0 bottom-0 rounded border-2 border-amber-400 shadow-[0_0_12px_rgba(251,191,36,0.35)] pointer-events-none z-20"
                        style={{ left: `${trimLeftPx}px`, width: `${trimWidthPx}px` }}
                      >
                        {/* Left Trim Handle (Head Padding Expansion) */}
                        <div
                          onPointerDown={(e) => handleTrimPointerDown('start', e)}
                          onPointerMove={handleTrimPointerMove}
                          onPointerUp={handleTrimPointerUp}
                          onPointerCancel={handleTrimPointerCancel}
                          onLostPointerCapture={handleTrimPointerCancel}
                          className="absolute left-0 top-0 bottom-0 w-3.5 -translate-x-1/2 bg-amber-400 hover:bg-amber-300 rounded-l cursor-ew-resize pointer-events-auto flex items-center justify-center shadow-md active:scale-105 transition-transform touch-none"
                          title="Drag to trim start (reveals earlier speech in safety handle)"
                        >
                          <div className="w-[1px] h-3 bg-black/80 rounded-full mx-[0.5px]" />
                          <div className="w-[1px] h-3 bg-black/80 rounded-full mx-[0.5px]" />
                        </div>

                        {/* Right Trim Handle (Tail Padding Expansion) */}
                        <div
                          onPointerDown={(e) => handleTrimPointerDown('end', e)}
                          onPointerMove={handleTrimPointerMove}
                          onPointerUp={handleTrimPointerUp}
                          onPointerCancel={handleTrimPointerCancel}
                          onLostPointerCapture={handleTrimPointerCancel}
                          className="absolute right-0 top-0 bottom-0 w-3.5 translate-x-1/2 bg-amber-400 hover:bg-amber-300 rounded-r cursor-ew-resize pointer-events-auto flex items-center justify-center shadow-md active:scale-105 transition-transform touch-none"
                          title="Drag to trim end (extends ending speech into safety handle)"
                        >
                          <div className="w-[1px] h-3 bg-black/80 rounded-full mx-[0.5px]" />
                          <div className="w-[1px] h-3 bg-black/80 rounded-full mx-[0.5px]" />
                        </div>
                      </div>
                    </div>
                  </div>
                );
              })()}

              {/* 2. UNIFORM LAYER TRACKS (L1, L2, L3...) */}
              {dynamicTracks.map((rowGroups, trackIdx) => {
                if (hiddenTracks[trackIdx]) {
                  return (
                    <div
                      key={`track_${trackIdx}`}
                      className="h-10 shrink-0 border-b border-[#232832] relative bg-black/20"
                      style={{ width: `${totalTimelineWidth}px` }}
                    />
                  );
                }

                return (
                  <div
                    key={`track_${trackIdx}`}
                    className="h-10 shrink-0 border-b border-[#232832] relative p-1"
                    style={{ width: `${totalTimelineWidth}px` }}
                  >
                    {rowGroups.map((grp) => {
                      const left = (clipStart + grp.start) * pxPerSec;
                      const rawDur = Math.max(0.04, grp.end - grp.start);
                      const trueWidth = Math.max(4, rawDur * pxPerSec);
                      const isSelected = selectedGroupId === grp.id;
                      const isCaption = !grp.type || grp.type === 'caption';
                      const isSpoken = isCaption && captionTimeIsActive(currentTime, grp.start, grp.end);
                      const isGroupHidden = grp.is_hidden || hiddenTracks[trackIdx];
                      const meta = getLayerMeta(grp, trackIdx);

                      const isThisBlockDragging = dragSession?.groupId === grp.id;
                      const showHandles = isSelected && trueWidth >= 32;
                      const showDuration = isSelected && trueWidth >= 96;
                      const showLabel = trueWidth >= 40;
                      const showIcon = trueWidth >= 20;

                      return (
                        <div
                          key={grp.id}
                          data-layer-block="true"
                          className={`absolute top-1 bottom-1 rounded-md flex items-center overflow-hidden select-none ${
                            isThisBlockDragging
                              ? 'transition-none z-30 shadow-2xl scale-[1.01]'
                              : isSelected
                              ? 'transition-colors duration-150 bg-[#2563EB] border-2 border-white text-white font-bold shadow-[0_0_12px_rgba(37,99,235,0.6)] z-20'
                              : isGroupHidden
                              ? 'transition-colors duration-150 bg-black/40 border border-white/20 text-white/40 opacity-40 grayscale'
                              : isSpoken
                              ? `transition-colors duration-150 ${meta.bg} border-2 border-white/80 text-white font-semibold z-10`

export function applyColorOpacity(color: string, opacity: number = 1.0): string {
  if (!color) return color;
  const clamped = Math.max(0, Math.min(1, opacity));
  if (clamped >= 0.999 && !color.startsWith('#') && !color.startsWith('rgb')) return color;
  if (color.startsWith('#')) {
    let hex = color.slice(1);
    if (hex.length === 3) hex = hex.split('').map((c) => c + c).join('');
    if (hex.length >= 6) {
      const r = parseInt(hex.substring(0, 2), 16) || 0;
      const g = parseInt(hex.substring(2, 4), 16) || 0;
      const b = parseInt(hex.substring(4, 6), 16) || 0;
      let a = 1.0;
      if (hex.length === 8) a = (parseInt(hex.substring(6, 8), 16) || 255) / 255;
      return `rgba(${r}, ${g}, ${b}, ${Number((a * clamped).toFixed(3))})`;
    }
  }
  const rgbaMatch = color.match(/rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)(?:\s*,\s*([\d.]+))?\s*\)/);
  if (rgbaMatch) {
    const r = rgbaMatch[1];
    const g = rgbaMatch[2];
    const b = rgbaMatch[3];
    const a = rgbaMatch[4] !== undefined ? parseFloat(rgbaMatch[4]) : 1.0;
    return `rgba(${r}, ${g}, ${b}, ${Number((a * clamped).toFixed(3))})`;
  }
  return color;
}

export function applyCaseTransform(text: string, mode?: string): string {
  if (!text) return '';
  if (mode === 'uppercase') return text.toUpperCase();
  if (mode === 'lowercase') return text.toLowerCase();
  if (mode === 'title') {
    return text.split(' ').map((w) => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase()).join(' ');
  }
  return text;
}

  showBoundingBox,

  const layout = group.layout || defaultLayout;
  const anim = group.animation || defaultAnimation;

  const isCaption = (group.type || 'caption') === 'caption';
  const highlightEnabled = isCaption && (style.highlight_enabled !== false);

  // Determine vertical placement percentage based on layout mode
  let posY = layout.position_y ?? 0.78;
  if (isCaption) {
    if (layoutMode === 'fit_black' || layoutMode === 'fit_blur') {
      posY = Math.max(0.74, posY);
    } else if (layoutMode === 'landscape') {
      posY = Math.min(0.86, Math.max(0.70, posY));
    }
  }

  // Entrance, exit, and loop animation calculations
  let enterOpacity = 1.0;
  let enterTransform = '';

  const enterElapsed = currentTime - group.start;
  if (enterType !== 'none' && enterElapsed >= 0 && enterElapsed < enterDur) {
    const p = Math.max(0, Math.min(1, enterElapsed / enterDur));
    if (enterType === 'fade') enterOpacity = p;

      enterTransform = `translateY(${(1 - p) * 25}px)`;
    }

  const exitDur = (anim.exit?.duration_ms || (anim as any)?.exit_duration_ms || 150) / 1000;

  if (exitType !== 'none' && exitRemaining >= 0 && exitRemaining < exitDur) {
    const p = Math.max(0, Math.min(1, exitRemaining / exitDur));
    if (exitType === 'fade') exitOpacity = p;
    else if (exitType === 'scale_down') {
      exitOpacity = p;
      exitTransform = `scale(${0.5 + 0.5 * p})`;
    } else if (exitType === 'slide_down') {
      exitOpacity = p;
      exitTransform = `translateY(${(1 - p) * 25}px)`;

    const loopTime = (currentTime - group.start) / speed;
    loopTransform = `scale(${1.0 + 0.04 * Math.sin(loopTime * 2 * Math.PI)})`;
  } else if (loopType === 'wiggle') {
    const speed = ((anim as any)?.loop?.speed_ms || 1000) / 1000;

    loopTransform = `rotate(${2.0 * Math.sin(loopTime * 2 * Math.PI)}deg)`;

  }
  const filterStr = filters.length > 0 ? filters.join(' ') : undefined;

  // Background styling mode: 'none' | 'line' / 'per_line' | 'block' / 'full_box'
  const bgEnabled = style.background_enabled !== false && style.background_mode !== 'none';
  const rawBgMode = bgEnabled ? (style.background_mode || 'block') : 'none';
  const isBgBlock = rawBgMode === 'block' || (rawBgMode as string) === 'full_box';
  const isBgLine = rawBgMode === 'line' || (rawBgMode as string) === 'per_line';
  const effectiveBgColor = applyColorOpacity(style.background_color || '#000000', style.background_opacity ?? 1.0);

  const bgOffsetX = (isBgBlock || isBgLine) ? (style.background_offset_x || 0) : 0;
  const bgOffsetY = (isBgBlock || isBgLine) ? (style.background_offset_y || 0) : 0;

  const fontFam = style.font_family || 'Montserrat';
  const fontSize = Math.max(10, Math.round((style.font_size || 44) * 0.44));
  const fontWeight = style.font_weight || 'bold';
  const letterSpacing = style.letter_spacing || 0;

  // Derive visual lines deterministically using font metrics and container width
  const visualLines = useMemo(() => {
    const stageWidth = containerWidth || (typeof window !== 'undefined' ? Math.min(window.innerWidth, 480) : 360);
    const maxContentWidth = stageWidth * (layout.max_width ?? 0.88);
    return partitionWordsIntoVisualLines(

        left: `${(layout.position_x ?? 0.5) * 100}%`,
        top: `${posY * 100}%`,
        transform: bgOffsetTransform ? `${combinedTransform} ${bgOffsetTransform}` : combinedTransform,
        maxWidth: `${(layout.max_width ?? 0.88) * 100}%`,
        width: 'max-content',
        transformOrigin: 'center center',
        opacity: effectiveOpacity,
        ...(isBgBlock && {
          backgroundColor: effectiveBgColor,
          padding: `${style.background_padding_y ?? 8}px ${style.background_padding_x ?? 14}px`,
          borderRadius: `${style.background_corner_radius ?? 12}px`,
        })

          const hasWordTiming = isCaption && (group as any).has_word_timing !== false && (group as any).timing_mode !== 'phrase' && (w as any).is_exact !== false;
          const isSpoken = hasWordTiming && currentTime >= w.start && currentTime <= w.end;

          const animType = anim?.active_word?.type || anim?.type || (anim as any)?.active_word_effect;
          if (isCaption && isSpoken && highlightEnabled && animType && animType !== 'none' && (animType === 'pop' || animType === 'bounce')) {
            const durationSec = (anim?.duration_ms || 100) / 1000;
            const elapsed = Math.max(0, currentTime - w.start);
            if (elapsed < durationSec) {
              const progress = elapsed / durationSec;
              const punch = (anim?.active_scale || 1.16) - 1.0;
              scale = 1.0 + punch * Math.cos(progress * (Math.PI / 2));
            }
          }

          let textColor = style.primary_color || (style as any).color || '#FFFF00';
          if (isCaption && isSpoken && highlightEnabled) {
            textColor = wordOverride?.highlight_color || style.highlight_color || '#FFFFFF';
          } else if (wordOverride?.color) {

          }

                  fontFamily: fontFam,

  currentTime,
  containerWidth,
  containerHeight,
  layoutMode = 'fit_black',
  showBoundingBox = true
}) => {

    return null;

        <CaptionGroupItem
          key={group.id}
          group={group}
          defaultStyle={track.default_style}
          defaultLayout={track.default_layout}
          defaultAnimation={track.default_animation}
          currentTime={currentTime}
          containerWidth={containerWidth}
          layoutMode={layoutMode}
          showBoundingBox={showBoundingBox}
        />
      ))}

  );

// In-memory cache for dynamically loaded fonts
const loadedFontFaceFamilies = new Set<string>();

/**
 * Ensure a font asset is registered in document.fonts via the Web FontFace API.
 */
export async function loadFontFace(family: string, fontUrl: string): Promise<boolean> {
  if (loadedFontFaceFamilies.has(family)) return true;
  if (typeof window === 'undefined' || !('FontFace' in window)) return false;

  try {
    const font = new FontFace(family, `url(${fontUrl})`);
    const loaded = await font.load();
    document.fonts.add(loaded);
    loadedFontFaceFamilies.add(family);
    return true;
  } catch (err) {
    console.warn(`[FontLoader] Could not load @font-face for ${family}:`, err);
    return false;
  }
}

/**
 * Fetch all available fonts from backend API, merging custom uploaded fonts with curated defaults.
 */
export async function fetchFontCatalog(apiBase: string = ''): Promise<FontEntry[]> {
  try {
    const res = await fetch(`${apiBase}/api/fonts`);
    if (!res.ok) return CURATED_FONTS;
    const data = await res.json();
    const serverFonts: any[] = data.fonts || [];

    const customEntries: FontEntry[] = serverFonts
      .filter((sf) => sf.is_custom)
      .map((sf) => ({
        family: sf.family,
        category: 'Custom',
        categoryKey: 'custom',
        weights: Array.isArray(sf.weights) && sf.weights.length > 0 ? sf.weights : ['400'],
        isCustom: true,
        filename: sf.filename,
        url: sf.url ? (sf.url.startsWith('http') ? sf.url : `${apiBase}${sf.url}`) : undefined,
        license: sf.license || 'Custom Imported',
      }));

    // Auto-load all custom fonts into document.fonts
    for (const cf of customEntries) {
      if (cf.url) {
        loadFontFace(cf.family, cf.url).catch(() => {});
      }
    }

    // Merge: Put custom fonts first, then curated
    return [...customEntries, ...CURATED_FONTS];
  } catch (err) {
    console.warn('[FontCatalog] Failed to fetch server fonts, falling back to curated list:', err);
    return CURATED_FONTS;
  }
}

/**
 * Upload a custom TTF/OTF font to the backend and register it in document.fonts.
 */
export async function importCustomFontFile(file: File, apiBase: string = ''): Promise<FontEntry> {
  const formData = new FormData();
  formData.append('file', file);

  const res = await fetch(`${apiBase}/api/fonts/import`, {
    method: 'POST',
    body: formData,
  });

  if (!res.ok) {
    const errData = await res.json().catch(() => ({ detail: 'Failed to import font' }));
    throw new Error(errData.detail || 'Failed to import font file');
  }

  const result = await res.json();
  const fontData = result.font;
  const entry: FontEntry = {
    family: fontData.family,
    category: 'Custom',
    categoryKey: 'custom',
    weights: fontData.weights || ['400'],
    isCustom: true,
    filename: fontData.filename,
    url: fontData.url ? (fontData.url.startsWith('http') ? fontData.url : `${apiBase}${fontData.url}`) : undefined,
    license: fontData.license || 'Custom Imported',
  };

  // Register in browser document.fonts immediately
  if (entry.url) {
    await loadFontFace(entry.family, entry.url);
  }

  return entry;
}

/**
 * Get real supported weights for a font family.
 */
export function getSupportedWeights(family: string, catalog: FontEntry[] = CURATED_FONTS): string[] {
  const found = catalog.find((f) => f.family.toLowerCase() === family.toLowerCase());
  return found?.weights && found.weights.length > 0 ? found.weights : ['400', 'bold'];
}

/**
 * Return closest valid weight for a chosen font.
 */
export function getClosestSupportedWeight(family: string, currentWeight: string = 'bold', catalog: FontEntry[] = CURATED_FONTS): string {
  const weights = getSupportedWeights(family, catalog);
  if (weights.includes(currentWeight)) return currentWeight;

  // Map common strings to numbers
  const norm = (w: string): number => {
    if (w === 'bold') return 700;
    if (w === 'black' || w === 'extrabold') return 900;
    if (w === 'semibold') return 600;
    if (w === 'medium') return 500;
    if (w === 'normal' || w === 'regular') return 400;
    const num = parseInt(w, 10);
    return isNaN(num) ? 400 : num;
  };

  const targetNum = norm(currentWeight);
  let best = weights[0];
  let bestDiff = 9999;
  for (const w of weights) {
    const diff = Math.abs(norm(w) - targetNum);
    if (diff < bestDiff) {
      bestDiff = diff;
      best = w;
    }
  }
  return best;
}

const RECENT_KEY = 'dispatch_recent_fonts';
const FAVORITES_KEY = 'dispatch_favorite_fonts';

export function getRecentFonts(): string[] {
  try {
    const raw = localStorage.getItem(RECENT_KEY);
    return raw ? JSON.parse(raw) : ['Impact', 'Montserrat', 'Anton'];
  } catch {
    return ['Impact', 'Montserrat', 'Anton'];
  }
}

export function addRecentFont(family: string): void {
  try {
    const recents = getRecentFonts().filter((f) => f.toLowerCase() !== family.toLowerCase());
    recents.unshift(family);
    localStorage.setItem(RECENT_KEY, JSON.stringify(recents.slice(0, 8)));
  } catch {}
}

export function getFavoriteFonts(): string[] {
  try {
    const raw = localStorage.getItem(FAVORITES_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
}

export function toggleFavoriteFont(family: string): boolean {
  try {
    const favs = getFavoriteFonts();
    const exists = favs.some((f) => f.toLowerCase() === family.toLowerCase());
    let next: string[];
    if (exists) {
      next = favs.filter((f) => f.toLowerCase() !== family.toLowerCase());
    } else {
      next = [...favs, family];
    }
    localStorage.setItem(FAVORITES_KEY, JSON.stringify(next));
    return !exists;
  } catch {
    return false;
  }
}
