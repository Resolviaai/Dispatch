"""Pillar 3 Unit & Integration Tests: Gemini Semantic Highlight Selection.

Tests all required Pillar 3 properties:
  A. Happy-path discovery & processing: TRANSCRIBED -> ANALYZING -> CLIPS_DEFINED.
  B. Valid clip schema: title, hook, description, hashtags, virality_score, layout_mode, duration.
  C. Malformed JSON & schema violations gracefully skipped/filtered.
  D. Word boundary snapping: timestamps snapped to closest spoken word.
  E. Strict 20.0s <= duration <= 90.0s constraint: invalid durations rejected, never manufactured.
  F. Overlap deduplication: overlapping candidates (IoU > 0.5) suppressed, retaining higher virality.
  G. Negative feedback conditioning: rejections injected into prompt.
  H. Rate-limit (429/503/timeout) retry backoff: scheduled via next_ai_attempt_at.
  I. Missing or invalid GEMINI_API_KEY: transitions to permanent failure.
  J. Strict failure: zero fallback to local heuristic.
  K. Restart idempotency: resuming or re-running does not duplicate clips.
  L. Transaction atomicity: partial failures roll back cleanly.
  M. Silence / zero speech: marks CLIPS_DEFINED with 0 clips without calling Gemini.
  N. Invariant check: clips table rows have video_path IS NULL, thumbnail_path IS NULL, status='ready_review'.
  O. Publishing outbox empty: Pillar 3 does not queue or touch social outbox.
"""

    def setUp(self):
        init_db()
        init_job_queue_schema()
        with get_db_connection() as conn:
            conn.execute("DELETE FROM pipeline_jobs")
        self.retry_engine = RetryEngine(base_delay_seconds=10, max_delay_seconds=60)

    def test_retry_engine_backoff_and_classification(self):
        # 1. Backoff delays
        delay1 = self.retry_engine.compute_backoff_seconds(1)
        delay3 = self.retry_engine.compute_backoff_seconds(3)
        self.assertGreaterEqual(delay1, 5)
        self.assertGreaterEqual(delay3, delay1)

        # 2. AI 429 Rate limit classification
        status, perm, reason = RetryEngine.classify_error(Exception("HTTP 429 Too Many Requests: quota exceeded"))
        self.assertEqual(status, JobStatus.WAITING_FOR_AI)
        self.assertFalse(perm)

        # 3. Disk space error classification
        status, perm, reason = RetryEngine.classify_error(IOError("[Errno 28] No space left on device"))
        self.assertEqual(status, JobStatus.WAITING_FOR_RESOURCES)
        self.assertFalse(perm)

        # 4. Permanent corrupt file classification
        status, perm, reason = RetryEngine.classify_error(ValueError("moov atom not found in video"))
        self.assertEqual(status, JobStatus.FAILED_PERMANENT)
        self.assertTrue(perm)

        try:
            # 1. Unauthenticated request returns 401
            unauth_res = client.get(f"/api/sync/youtube-ack?video_id={video_id}")
            self.assertEqual(unauth_res.status_code, 401)

            bad_auth_res = client.get(f"/api/sync/youtube-ack?video_id={video_id}", headers={"x-auth-token": "wrong_secret"})
            self.assertEqual(bad_auth_res.status_code, 401)

            # 2. Unregistered video returns acknowledged: False
            res = client.get(f"/api/sync/youtube-ack?video_id={video_id}", headers=headers)
            self.assertEqual(res.status_code, 200)
            self.assertFalse(res.json()["acknowledged"])
            self.assertFalse(res.json()["ingested"])

            # 3. Registered in youtube_inbox with status='WAITING' returns acknowledged: False
            with get_db_connection() as conn:
                conn.execute("""

                    VALUES (?, ?, 'WAITING', 'Test YouTube Video', ?)
                """, (video_id, dispatch_id, dummy_sha256))
                conn.commit()

            res = client.get(f"/api/sync/youtube-ack?video_id={video_id}&sha256={dummy_sha256}", headers=headers)
            self.assertEqual(res.status_code, 200)
            self.assertFalse(res.json()["acknowledged"])
            self.assertFalse(res.json()["ingested"])

            # 4. Status='DOWNLOADED' but physical file missing returns acknowledged: False
            with get_db_connection() as conn:
                conn.execute("""

                conn.commit()

            self.assertEqual(res.status_code, 200)
            self.assertFalse(res.json()["acknowledged"], "Must not acknowledge without physical file on PC disk")
            self.assertFalse(res.json()["ingested"])

            self.assertEqual(receipt, expected_receipt)

            self.assertEqual(res_disp.status_code, 200)
            self.assertTrue(res_disp.json()["acknowledged"])
            self.assertTrue(res_disp.json()["ingested"])
            self.assertIn("receipt", res_disp.json())

        finally:
            dummy_file.unlink(missing_ok=True)
            with get_db_connection() as conn:
                conn.execute("DELETE FROM youtube_inbox WHERE video_id = ?", (video_id,))

"""Vault round-trip (Worker-compatible AES-GCM format) and remote-job claim/finish
semantics with mocked Supabase RPCs."""

import base64
import os
import unittest
from unittest.mock import patch

from cryptography.hazmat.primitives.ciphers.aead import AESGCM

from dispatch import youtube_connections as yc


def b64url(bytes_: bytes) -> str:
    import base64 as b
    return b.urlsafe_b64encode(bytes_).decode().rstrip("=")


class TestVaultRoundTrip(unittest.TestCase):
    def setUp(self):
        self.key = base64.b64encode(os.urandom(32)).decode()
        self._env = patch.dict(os.environ, {"DISPATCH_TOKEN_VAULT_KEY": self.key})
        self._env.start()
        self.addCleanup(self._env.stop)

    def _encrypt_like_worker(self, plaintext: str) -> str:
        iv = os.urandom(12)
        ct = AESGCM(base64.b64decode(self.key)).encrypt(iv, plaintext.encode(), None)
        return b64url(iv + ct)

    def test_worker_encrypted_token_decrypts_on_pc(self):
        token = "1//0gRefreshTokenFromGoogle"
        enc = self._encrypt_like_worker(token)
        self.assertEqual(yc.decrypt_refresh_token(enc), token)

    def test_tampered_ciphertext_fails_closed(self):
        # Deterministic corruption: flip one ciphertext byte in the middle - the GCM
        # tag MUST reject it. (Base64-tail edits can decode to identical bytes: flaky.)
        import base64 as b64
        enc = self._encrypt_like_worker("secret")
        raw = bytearray(b64.urlsafe_b64decode(enc + "=" * (-len(enc) % 4)))
        raw[len(raw) // 2] ^= 0x01
        tampered = b64url(bytes(raw))
        with self.assertRaises(Exception):
            yc.decrypt_refresh_token(tampered)


    def test_missing_vault_key_raises_unavailable(self):
        os.environ.pop("DISPATCH_TOKEN_VAULT_KEY", None)
        with self.assertRaises(yc.VaultUnavailable):
            yc.vault_key()




class TestRemoteJobs(unittest.TestCase):
    def test_claim_returns_job_and_finish_records_status(self):
        calls = []

        def fake_rpc(action, params):
            calls.append((action, params))
            if action == "cloud_claim_job":
                return [{"job_id": "j1", "user_id": "u1", "youtube_url": "https://youtu.be/x"}]
            return None

        with patch.object(yc, "_supabase_rpc", side_effect=fake_rpc):
            job = yc.claim_remote_job("owner-pc-1")
            self.assertEqual(job.job_id, "j1")
            yc.finish_remote_job("j1", "done", "vid", "")

        self.assertEqual(calls[1][0], "cloud_finish_job")

    def test_empty_queue_returns_none(self):
        with patch.object(yc, "_supabase_rpc", return_value=[]):
            self.assertIsNone(yc.claim_remote_job("w"))

    def test_poller_processes_claimed_job_through_ingest(self):
        processed = []
        with patch.object(yc, "_supabase_rpc", return_value=[]), \
             patch.object(yc, "claim_remote_job", return_value=yc.RemoteJob("j9", "u9", "https://youtu.be/ok")):
            poller = yc.RemoteJobPoller(ingest_video=lambda url: processed.append(url) or {"status": "DOWNLOADED", "video_id": "v9"})
            poller._process(poller and yc.claim_remote_job("w"))
        self.assertEqual(processed, ["https://youtu.be/ok"])

    def test_ingest_failure_marks_job_failed(self):
        with patch.object(yc, "_supabase_rpc", return_value=[]) as rpc, \
             patch.object(yc, "claim_remote_job", return_value=yc.RemoteJob("j8", "u8", "https://youtu.be/bad")):
            poller = yc.RemoteJobPoller(ingest_video=lambda url: {"status": "FAILED", "error": "private"})
            poller._process(yc.claim_remote_job("w"))
        finish_calls = [c for c in rpc.call_args_list if c.args[0] == "cloud_finish_job"]
        self.assertEqual(finish_calls[0].args[1]["p_status"], "failed")
