function applyColorOpacity(color: string, opacity: number = 1.0): string {
  if (opacity >= 1.0) return color;
  if (color.startsWith('#')) {
    const hex = color.slice(1);
    let r = 0, g = 0, b = 0;
    if (hex.length === 3) {
      r = parseInt(hex[0] + hex[0], 16);
      g = parseInt(hex[1] + hex[1], 16);
      b = parseInt(hex[2] + hex[2], 16);
    } else if (hex.length >= 6) {
      r = parseInt(hex.slice(0, 2), 16);
      g = parseInt(hex.slice(2, 4), 16);
      b = parseInt(hex.slice(4, 6), 16);
    }
    return `rgba(${r}, ${g}, ${b}, ${opacity})`;
  }
  return color;
}

function getAaPreviewStyle(style: Partial<StyleConfig>) {
  const color = applyColorOpacity(style.primary_color || '#FFFFFF', style.text_opacity ?? 1.0);
  const strokeColor = applyColorOpacity(style.outline_color || '#000000', style.outline_opacity ?? 1.0);
  const stroke = style.outline_enabled
    ? `${Math.max(0.8, Math.min(1.5, (style.outline_width || 2) * 0.35))}px ${strokeColor}`
    : 'none';
  const shadows: string[] = [];
  if (style.glow_enabled && style.glow_color) {
    const glowColor = applyColorOpacity(style.glow_color, style.glow_opacity ?? 1.0);
    const blur = Math.min(14, style.glow_blur || 10);
    shadows.push(`0 0 ${Math.max(2, Math.round(blur * 0.35))}px ${glowColor}`);
    shadows.push(`0 0 ${blur}px ${glowColor}`);
  }
  if (style.shadow_enabled) {
    const dist = Math.min(3, style.shadow_distance || 2);
    const rad = ((style.shadow_angle || 45) * Math.PI) / 180;
    const x = Math.round(dist * Math.cos(rad));
    const y = Math.round(dist * Math.sin(rad));
    const shadowColor = applyColorOpacity(style.shadow_color || 'rgba(0,0,0,0.85)', style.shadow_opacity ?? 0.85);
    shadows.push(`${x}px ${y}px ${Math.min(4, style.shadow_blur || 3)}px ${shadowColor}`);
  }
  const textShadow = shadows.length > 0 ? shadows.join(', ') : 'none';

  return {
    color,
    fontFamily: style.font_family || 'Impact, "Arial Black", Montserrat, sans-serif',
    fontWeight: (style.font_weight as any) || '900',
    WebkitTextStroke: stroke,
    paintOrder: 'stroke fill',
    textShadow,
  };
}

function formatTime(sec: number): string {
  const m = Math.floor(sec / 60);
  const s = Math.floor(sec % 60);
  const ms = Math.floor((sec % 1) * 10);
  return `${m}:${s.toString().padStart(2, '0')}.${ms}`;
}

function ValueSlider({
  label,

  label: string;

}) {
  return (

      <div className="flex items-center justify-between text-[11px]">
        <span className="text-text-secondary font-medium">{label}</span>
        <span className="font-mono tabular-nums text-text-main font-semibold">

      <input

  );
}

}: {
  icon: React.ReactNode;
  label: string;
  onClick?: () => void;
  active?: boolean;
  disabled?: boolean;
  title?: string;
}) {
  return (
    <button
      type="button"
      onClick={onClick}
      disabled={disabled}
      aria-disabled={disabled}
      title={title || label}
      className={`flex flex-col items-center justify-center min-w-[54px] h-12 px-1 text-center select-none transition-colors active:scale-95 shrink-0 ${
        disabled
          ? 'opacity-30 cursor-not-allowed pointer-events-none'
          : active
          ? 'text-primary font-semibold'
          : 'text-text-secondary hover:text-text-main'
      }`}
    >
      <div className="h-5 flex items-center justify-center mb-0.5">
        {icon}
      </div>
      <span className="text-[10px] leading-tight font-medium truncate max-w-[52px]">
        {label}
      </span>
    </button>
  );
}

// CapCut-style dual-ring color row with independent selection indicator and custom color preview
function CapCutColorRow({
  label,
  value,
  defaultValue = '#FFFFFF',
  swatches = COLOR_SWATCHES,
  onChange,
  onOpenCustom,
  customTitle,
}: {
  label: string;
  value: string | undefined;
  defaultValue?: string;
  swatches?: string[];
  onChange: (color: string) => void;
  onOpenCustom: () => void;
  customTitle?: string;
}) {
  const effectiveVal = (value || defaultValue).trim();
  const normalizedVal = effectiveVal.toUpperCase();
  const matchedSwatch = swatches.find((s) => s.toUpperCase() === normalizedVal);
  const isCustom = !matchedSwatch;

  return (
    <div className="space-y-1">
      <div className="flex items-center justify-between">
        <span className="text-[11px] font-medium text-text-secondary">{label}</span>
        <span className="text-[10px] font-mono tabular-nums text-text-muted">{effectiveVal}</span>
      </div>
      <div className="flex items-center gap-2 overflow-x-auto no-scrollbar py-1.5 px-0.5">
        {/* Custom color button with live preview indicator */}
        <button
          type="button"
          onClick={onOpenCustom}
          aria-pressed={isCustom}
          title={customTitle || `Custom ${label}`}
          className={`h-7 px-2.5 rounded-full border text-[10px] font-semibold flex items-center gap-1.5 shrink-0 active:scale-95 transition-all ${
            isCustom
              ? 'border-primary bg-primary/20 text-white ring-2 ring-primary ring-offset-2 ring-offset-[#16191E] shadow-sm'
              : 'border-dashed border-border bg-surface-200 hover:bg-surface-300 text-text-secondary hover:text-text-main'
          }`}
        >
          {isCustom ? (
            <span
              className="w-3.5 h-3.5 rounded-full border border-white/50 shrink-0"
              style={{ backgroundColor: effectiveVal }}
            />
          ) : (
            <Plus size={11} className="text-primary shrink-0" />
          )}
          <span>Custom</span>
        </button>

        {/* Built-in swatches with CapCut dual-ring selection */}
        {swatches.map((c) => {
          const isSelected = !isCustom && matchedSwatch?.toUpperCase() === c.toUpperCase();
          return (
            <button
              key={c}
              type="button"
              onClick={() => onChange(c)}
              aria-pressed={isSelected}
              className={`w-7 h-7 rounded-full shrink-0 transition-transform active:scale-90 ${
                isSelected
                  ? 'ring-2 ring-primary ring-offset-2 ring-offset-[#16191E] scale-110 shadow-md'
                  : 'border border-white/10 hover:scale-105'
              }`}
              style={{ backgroundColor: c }}
              title={c}
              aria-label={`${label} ${c}`}
            />
          );
        })}
      </div>
    </div>
  );
}

const CaptionStudioComponent: React.FC<CaptionStudioProps> = ({

    color: string;
    enableAlpha?: boolean;
  } | null>(null);
  const [overlayAnimTab, setOverlayAnimTab] = useState<'enter' | 'exit'>('enter');
  const isTranscriptOpen = activePanel === 'transcript';
  const setIsTranscriptOpen = (open: boolean) => setActivePanel(open ? 'transcript' : 'none');
  const fileInputRef = useRef<HTMLInputElement | null>(null);
  const [fontCatalog, setFontCatalog] = useState<FontEntry[]>(CURATED_FONTS);
  const [fontSearch, setFontSearch] = useState<string>('');
  const [selectedFontCategory, setSelectedFontCategory] = useState<FontCategoryKey>('all');
  const [recentFonts, setRecentFonts] = useState<string[]>([]);
  const [favoriteFonts, setFavoriteFonts] = useState<string[]>([]);
  const [fontImportError, setFontImportError] = useState<string | null>(null);
  const [isImportingFont, setIsImportingFont] = useState<boolean>(false);
  const fontFileInputRef = useRef<HTMLInputElement | null>(null);

  React.useEffect(() => {

    else if (activePanel === 'animation') setTextMainTab('animations');

    else if (activePanel === 'effects') setTextMainTab('effects');
    else if (activePanel === 'style') setTextMainTab('styles');
  }, [activePanel]);

  // Global Undo / Redo keyboard shortcuts
  React.useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      const tag = (e.target as HTMLElement)?.tagName?.toLowerCase();
      if (tag === 'input' || tag === 'textarea' || (e.target as HTMLElement)?.isContentEditable) return;
      if ((e.ctrlKey || e.metaKey) && !e.altKey) {
        if (e.key === 'z' || e.key === 'Z') {
          if (e.shiftKey) {
            if (_canRedo && _onRedo) {
              e.preventDefault();
              _onRedo();
            }
          } else {
            if (_canUndo && _onUndo) {
              e.preventDefault();
              _onUndo();
            }
          }
        } else if (e.key === 'y' || e.key === 'Y') {
          if (_canRedo && _onRedo) {
            e.preventDefault();
            _onRedo();
          }
        }
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [_canUndo, _canRedo, _onUndo, _onRedo]);

  // Active selected group
  const currentGroup = useMemo(() => {
    if (!track || !track.groups || track.groups.length === 0) return null;
    if (selectedGroupId) {
      return track.groups.find((g) => g.id === selectedGroupId) || null;
    }
    return null;
  }, [track, selectedGroupId]);

  // Enforce valid text layer selection: must be an active group of type caption or text

    currentGroup && (!currentGroup.type || currentGroup.type === 'caption' || currentGroup.type === 'text')
  );
  const isCaption = Boolean(currentGroup && (!currentGroup.type || currentGroup.type === 'caption'));
  const isNormalText = Boolean(currentGroup && currentGroup.type === 'text');

  React.useEffect(() => {
    if (isNormalText && animCategory === 'word') {
      setAnimCategory('in');
    }
  }, [isNormalText, animCategory]);

  // Auto-close styling/font/animation/transform panels if layer was deselected
  React.useEffect(() => {
    if (!isTextLayerSelected && [
      'style',
      'fonts',
      'effects',
      'animation',
      'templates',
      'overlay_transform',
      'overlay_animation',
      'layer_opacity',
      'layer_volume',
    ].includes(activePanel)) {
      setActivePanel('none');
    }
  }, [isTextLayerSelected, activePanel]);

  // Active text layer content synchronized draft
  const currentGroupText = useMemo(() => {
    if (!currentGroup || !currentGroup.words) return '';
    return currentGroup.words.map((w) => w.text || (w as any).word || '').join(' ');
  }, [currentGroup?.id, currentGroup?.words]);

  const [textInputDraft, setTextInputDraft] = useState('');
  const [isInputFocused, setIsInputFocused] = useState(false);
  const [isKeyboardVisible, setIsKeyboardVisible] = useState(false);

  React.useEffect(() => {
    if (typeof window === 'undefined' || !window.visualViewport) return;
    const vv = window.visualViewport;
    const handleResize = () => {
      const isSmaller = vv.height < window.innerHeight * 0.75;
      setIsKeyboardVisible(isSmaller);
    };
    vv.addEventListener('resize', handleResize);
    return () => vv.removeEventListener('resize', handleResize);
  }, []);

  const hideTabsForKeyboard = isInputFocused || isKeyboardVisible;

  React.useEffect(() => {
    setTextInputDraft(currentGroupText);
  }, [currentGroup?.id, currentGroupText]);

  const handleTextDraftChange = (newVal: string) => {
    setTextInputDraft(newVal);
    if (!currentGroup || !track) return;
    const nextWords = preserveWordTimingsOnEdit(
      currentGroup.words || [],
      newVal,
      currentGroup.start,
      currentGroup.end,
      currentGroup.id
    );
    const updatedGroup = { ...currentGroup, words: nextWords };
    onUpdateTrack({
      ...track,
      groups: track.groups.map(g => g.id === currentGroup.id ? updatedGroup : g)
    }, false);
  };

  const handleTextDraftCommit = () => {
    if (!currentGroup || !track) return;
    const nextWords = preserveWordTimingsOnEdit(
      currentGroup.words || [],
      textInputDraft,
      currentGroup.start,
      currentGroup.end,
      currentGroup.id
    );
    const updatedGroup = { ...currentGroup, words: nextWords };
    onUpdateTrack({
      ...track,
      groups: track.groups.map(g => g.id === currentGroup.id ? updatedGroup : g)
    }, true);
  };

  React.useEffect(() => {
    setRecentFonts(getRecentFonts());
    setFavoriteFonts(getFavoriteFonts());
    fetchFontCatalog().then((catalog) => {
      if (catalog && catalog.length > 0) {
        setFontCatalog(catalog);
        catalog.forEach((f) => {
          if (f.url) loadFontFace(f.family, f.url).catch(() => {});
        });
      }
    });
  }, []);

  // Derive current effective category:
  // If layer selected, it automatically determines category (overlay / audio / text)
  // If video selected, it's 'edit'
  // Else uses activeCategory
  const effectiveCategory: ToolbarCategory = useMemo(() => {
    if (currentGroup) {
      if (currentGroup.type === 'overlay') return 'overlay';
      if (currentGroup.type === 'audio' || currentGroup.type === 'voiceover') return 'audio';

    }
    if (isVideoSelected) return 'edit';

  }, [currentGroup, isVideoSelected, activeCategory]);

  const currentStyle = currentGroup?.style || track?.default_style;

  const currentAnimation = currentGroup?.animation || track?.default_animation;

  // Patch function for captions/text: edits affect ONLY the selected group
  // Never fall back to track.groups[0] and never silently mutate default_style
  const patch = (
    stylePatch?: Partial<StyleConfig>,
    layoutPatch?: Partial<LayoutConfig>,
    animationPatch?: Partial<AnimationConfig>,
    commitToHistory: boolean = false
  ) => {
    if (!track || !currentGroup || !isTextLayerSelected) return;

    const nextTrack: CaptionTrack = {
      ...track,
      groups: track.groups.map((grp) => {
        if (grp.id !== currentGroup.id) return grp;
        return {
          ...grp,
          style: stylePatch ? { ...(grp.style || track.default_style || {}), ...stylePatch } : grp.style,
          layout: layoutPatch ? { ...(grp.layout || track.default_layout || {}), ...layoutPatch } : grp.layout,
          animation: animationPatch
            ? { ...(grp.animation || track.default_animation || {}), ...animationPatch }
            : grp.animation,
        };
      }),
    };

    onUpdateTrack(nextTrack, commitToHistory);
  };

  const handleSelectTextMainTab = (tab: TextMainTab) => {
    // Non-template tabs require an active text layer
    if (!isTextLayerSelected && tab !== 'templates') return;
    setTextMainTab(tab);
    if (tab === 'templates') setActivePanel('templates');
    else if (tab === 'animations') setActivePanel('animation');
    else if (tab === 'fonts') setActivePanel('fonts');
    else if (tab === 'effects') setActivePanel('effects');
    else setActivePanel('style');
  };

  // One-time action: Applies ONLY the properties belonging to the specified section to all eligible text/caption elements.
  const handleApplySectionToAll = (section: StyleSection): number => {

      track,
      currentGroup.id,
      section
    );
    // Explicitly commit to undo/redo history as an intentional user action
    onUpdateTrack(nextTrack, true);

  const handleSelectFont = async (f: FontEntry) => {
    addRecentFont(f.family);
    setRecentFonts(getRecentFonts());
    if (f.url) {
      setLoadingFontFamily(f.family);

        await loadFontFace(f.family, f.url);

        // ignore
      } finally {
        setLoadingFontFamily(null);
      }
    }
    const nextWeight = getClosestSupportedWeight(f.family, currentStyle?.font_weight, fontCatalog);
    patch({ font_family: f.family, font_weight: nextWeight }, undefined, undefined, true);
  };

    e.stopPropagation();
    toggleFavoriteFont(family);
    setFavoriteFonts(getFavoriteFonts());
  };

  const handleImportFont = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    setFontImportError(null);
    setIsImportingFont(true);

      const newEntry = await importCustomFontFile(file);

      setFontImportError(err.message || 'Failed to import font');

        fontFileInputRef.current.value = '';

  const filteredFonts = useMemo(() => {
    const list = fontCatalog.filter((f) => {

        const q = fontSearch.toLowerCase().trim();
        return f.family.toLowerCase().includes(q) || f.category.toLowerCase().includes(q);
      }
      return true;
    });

    // Starred fonts automatically appear first in the Fonts grid, followed by unstarred fonts
    return [...list].sort((a, b) => {
      const aFav = favoriteFonts.some((fav) => fav.toLowerCase() === a.family.toLowerCase());
      const bFav = favoriteFonts.some((fav) => fav.toLowerCase() === b.family.toLowerCase());
      if (aFav && !bFav) return -1;
      if (!aFav && bFav) return 1;
      return 0;
    });
  }, [fontCatalog, selectedFontCategory, fontSearch, favoriteFonts]);

  const selectedFontEntry = useMemo(() => {
    const curFam = currentStyle?.font_family || 'Montserrat';
    return fontCatalog.find((f) => f.family.toLowerCase() === curFam.toLowerCase()) || CURATED_FONTS[0];
  }, [fontCatalog, currentStyle?.font_family]);

  const supportedWeights = useMemo(() => {
    return selectedFontEntry?.weights && selectedFontEntry.weights.length > 0
      ? selectedFontEntry.weights
      : ['400'];
  }, [selectedFontEntry]);

  // Return to primary toolbar: deselects layer/video and restores primary toolbar
  const handleBackToPrimary = () => {
    setActiveCategory('none');
    setActivePanel('none');
    onSelectGroup?.(null);
    onSelectVideo?.(false);
  };

  // Add new text layer
  const handleAddText = () => {
    if (!track) return;
    const newId = `grp_${Date.now()}`;
    const start = Number(currentTimeRef.current.toFixed(2));
    const end = Number(Math.min(duration, start + 3.0).toFixed(2));
    const newGroup: CaptionGroup = {
      id: newId,
      start,
      end,
      type: 'text',
      words: [
        { id: `w_${Date.now()}_0`, text: 'New', start, end: start + 1.0 },

      layout: { ...(track.default_layout || { position_x: 0.5, position_y: 0.5, alignment: 'center', max_width: 0.88, max_lines: 2 }) },

          font_family: 'Montserrat',
          font_size: 44,
          font_weight: '900',
          text_transform: 'none',
          letter_spacing: 0,
          line_height: 1.2,
          primary_color: '#FFFFFF',
          outline_enabled: true,
          outline_color: '#000000',
          outline_width: 2,
          shadow_enabled: false,
          background_enabled: false,
        }),
        highlight_enabled: false,
      },
      animation: {
        type: 'none',
        duration_ms: 150,
        active_scale: 1.0,
        easing: 'ease_out',
        enter: { type: 'none', duration_ms: 150 },
        exit: { type: 'none', duration_ms: 150 },
        loop: { type: 'none', speed_ms: 1000 },

      ...track,
      groups: [...track.groups, newGroup],

      const res = await fetch('/api/media/upload', {
        method: 'POST',
        body: formData,
      });
      if (res.ok) {
        const data = await res.json();
        if (data.url) return data.url;
      }
    } catch (e) {
      console.warn('Upload to /api/media/upload failed, falling back to blob URL:', e);
    }
    return URL.createObjectURL(file);
  };

  // Native file picker selection for Add Overlay
  const handleOverlayFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file || !track) return;
    const isVideo = file.type.startsWith('video');
    const mediaUrl = await uploadMediaFile(file, file.name);
    const newId = `ovl_${Date.now()}`;
    const start = Number(currentTimeRef.current.toFixed(2));
    const end = Number(Math.min(duration, start + (isVideo ? 8.0 : 4.0)).toFixed(2));
    const cleanName = file.name.replace(/\.[^/.]+$/, '').slice(0, 16);

      id: newId,
      start,
      end,
      type: 'overlay',
      overlay_type: isVideo ? 'video' : 'image',
      overlay_url: mediaUrl,
      label: cleanName || (isVideo ? 'Video Clip' : 'Image Overlay'),
      opacity: 100,
      volume: isVideo ? 100 : 0,
      is_muted: false,
      layout: {
        position_x: 0.5,
        position_y: 0.5,
        scale: 0.8,
        rotation: 0,

        max_width: 0.8,
        max_lines: 1,

      id: newId,
      start,
      end,
      type: 'audio',
      audio_url: mediaUrl,
      label: cleanName || 'BGM Track',
      volume: 100,
      is_muted: false,
      words: [], // Honest non-text media layer

      ...track,
      groups: [...track.groups, newAudio],

    e.target.value = '';
  };

  const handleImportAudio = () => {
    if (audioFileInputRef.current) {
      audioFileInputRef.current.click();
    } else if (track) {
      // Direct fallback if file input not mounted
      const newId = `aud_${Date.now()}`;
      const start = Number(currentTimeRef.current.toFixed(2));
      const end = Number(Math.min(duration, start + 15.0).toFixed(2));
      const newAudio: CaptionGroup = {
        id: newId,
        start,
        end,
        type: 'audio',
        label: 'Audio Track',
        volume: 100,
        is_muted: false,
        words: [],
      };
      onUpdateTrack({ ...track, groups: [...track.groups, newAudio] }, true);
      onSelectGroup?.(newId);
      setActiveCategory('audio');
    }
  };

  // Add Voiceover audio layer with real microphone recorder
  const [isRecordingVoiceover, setIsRecordingVoiceover] = useState(false);
  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
  const audioChunksRef = useRef<Blob[]>([]);

  const handleRecordVoiceover = async () => {
    if (isRecordingVoiceover) {
      if (mediaRecorderRef.current) {
        mediaRecorderRef.current.stop();
        setIsRecordingVoiceover(false);
      }
      return;
    }

    if (!navigator.mediaDevices?.getUserMedia) {
      alert('Microphone recording is not supported in this browser.');
      return;
    }

    try {
      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
      const recorder = new MediaRecorder(stream);
      audioChunksRef.current = [];
      recorder.ondataavailable = (e) => {
        if (e.data && e.data.size > 0) {
          audioChunksRef.current.push(e.data);
        }
      };
      recorder.onstop = async () => {
        stream.getTracks().forEach((t) => t.stop());
        const blob = new Blob(audioChunksRef.current, { type: 'audio/webm' });
        if (blob.size === 0 || !track) return;
        const filename = `voiceover_${Date.now()}.webm`;
        const mediaUrl = await uploadMediaFile(blob, filename);
        const newId = `voc_${Date.now()}`;
        const start = Number(currentTimeRef.current.toFixed(2));
        const end = Number(Math.min(duration, start + Math.max(1, audioChunksRef.current.length * 0.5)).toFixed(2));

        const newVoice: CaptionGroup = {
          id: newId,
          start,
          end,
          type: 'voiceover',
          audio_url: mediaUrl,
          label: 'Voiceover',
          volume: 100,
          is_muted: false,
          words: [], // Honest non-text media layer
        };

        onUpdateTrack({
          ...track,
          groups: [...track.groups, newVoice],
        }, true);
        onSelectGroup?.(newId);
        setActiveCategory('audio');
      };

      recorder.start(500);
      mediaRecorderRef.current = recorder;
      setIsRecordingVoiceover(true);
    } catch (err) {
      console.warn('Microphone permission denied or error:', err);
      alert('Microphone access unavailable or denied.');
    }
  };

  // Split layer at playhead preserving authentic word timestamps
  const handleSplit = () => {
    if (!track || !currentGroup) return;
    const splitTime = Number(currentTimeRef.current.toFixed(2));
    const splitResult = splitCaptionGroupAtTime(currentGroup, splitTime);
    if (!splitResult) return;

    const [firstGroup, secondGroup] = splitResult;
    onUpdateTrack({
      ...track,
      groups: track.groups.flatMap((g) => (g.id === currentGroup.id ? [firstGroup, secondGroup] : [g])),
    }, true);
    onSelectGroup?.(secondGroup.id);
  };

  // Merge selected group with adjacent segment of compatible type
  const handleMergeNext = () => {
    if (!track || !currentGroup) return;
    const groupType = currentGroup.type || 'caption';
    const sorted = [...track.groups]
      .filter((g) => (g.type || 'caption') === groupType)
      .sort((a, b) => a.start - b.start);
    const currIdx = sorted.findIndex((g) => g.id === currentGroup.id);
    if (currIdx < 0 || currIdx >= sorted.length - 1) return;

    const nextGroup = sorted[currIdx + 1];
    // Adjacency check: within 3 seconds
    if (nextGroup.start > currentGroup.end + 3.0) return;

    const merged = mergeAdjacentCaptionGroups(currentGroup, nextGroup);

    onUpdateTrack({
      ...track,
      groups: track.groups
        .filter((g) => g.id !== nextGroup.id)
        .map((g) => (g.id === currentGroup.id ? merged : g)),
    }, true);
    onSelectGroup?.(merged.id);
  };

  // Duplicate layer with time-shifted authentic word timestamps
  const handleDuplicate = () => {
    if (!track || !currentGroup) return;
    const isOverlay = currentGroup.type === 'overlay';
    const newId = `${isOverlay ? 'ovl' : 'grp'}_${Date.now()}`;
    const nextStart = Number(Math.min(duration - 0.5, currentGroup.start + 0.3).toFixed(2));
    const nextEnd = Number(Math.min(duration, currentGroup.end + 0.3).toFixed(2));
    const timeDelta = nextStart - currentGroup.start;

    const dup: CaptionGroup = {
      ...currentGroup,
      id: newId,
      start: nextStart,
      end: nextEnd,
      words: (currentGroup.words || []).map((w, idx) => ({
        ...w,
        id: `w_${Date.now()}_${idx}`,
        start: Number((w.start + timeDelta).toFixed(3)),
        end: Number((w.end + timeDelta).toFixed(3)),
      })),
    };

    onUpdateTrack({
      ...track,
      groups: [...track.groups, dup],
    }, true);
    onSelectGroup?.(newId);
  };

  // Delete layer
  const handleDelete = () => {
    if (!track || !currentGroup) return;
    onUpdateTrack({
      ...track,
      groups: track.groups.filter((g) => g.id !== currentGroup.id),
    }, true);
    onSelectGroup?.(null);
  };

  // Span layer across entire video duration
  const handleLastingText = () => {
    if (!track || !currentGroup) return;
    onUpdateTrack({
      ...track,
      groups: track.groups.map((g) =>
        g.id === currentGroup.id ? { ...g, start: 0, end: Number(duration.toFixed(2)) } : g
      ),
    }, true);
  };

  // Export SRT download
  const handleExportSrt = () => {
    if (track) {
      downloadSrtFile(track, track.clip_id ? `${track.clip_id}.srt` : 'captions.srt');
    }
  };

  // Reorder stacking layers
  const handleMoveLayer = (direction: 'up' | 'down') => {
    if (!track || !currentGroup) return;
    const idx = track.groups.findIndex((g) => g.id === currentGroup.id);
    if (idx === -1) return;
    const targetIdx = direction === 'up' ? idx + 1 : idx - 1;
    if (targetIdx < 0 || targetIdx >= track.groups.length) return;

    const copy = [...track.groups];
    const [removed] = copy.splice(idx, 1);
    copy.splice(targetIdx, 0, removed);
    onUpdateTrack({ ...track, groups: copy });
  };

  // Overlay Transform Actions
  const handleRotate90 = () => {
    if (!currentGroup || !track) return;
    const curRot = currentGroup.layout?.rotation || 0;
    const nextRot = (curRot + 90) % 360;
    onUpdateTrack({
      ...track,
      groups: track.groups.map((g) =>
        g.id === currentGroup.id
          ? { ...g, layout: { ...(g.layout || track.default_layout), rotation: nextRot } }
          : g
      ),
    });
  };

  const handleToggleFlipH = () => {
    if (!currentGroup || !track) return;
    const nextFlipH = !currentGroup.flip_h;
    onUpdateTrack({
      ...track,
      groups: track.groups.map((g) => (g.id === currentGroup.id ? { ...g, flip_h: nextFlipH } : g)),
    });
  };

  const handleToggleFlipV = () => {
    if (!currentGroup || !track) return;
    const nextFlipV = !currentGroup.flip_v;
    onUpdateTrack({
      ...track,
      groups: track.groups.map((g) => (g.id === currentGroup.id ? { ...g, flip_v: nextFlipV } : g)),
    });
  };

  const handleResetTransform = () => {
    if (!currentGroup || !track) return;
    onUpdateTrack({
      ...track,
      groups: track.groups.map((g) =>
        g.id === currentGroup.id
          ? {
              ...g,
              flip_h: false,
              flip_v: false,
              layout: {
                ...(g.layout || track.default_layout),
                scale: 1.0,
                rotation: 0,
                position_x: 0.5,
                position_y: 0.5,
              },
            }
          : g
      ),
    });
  };

  const handleUpdateOverlayAnimation = (patchAnim: Partial<OverlayAnimationConfig>) => {
    if (!currentGroup || !track) return;
    const curAnim = currentGroup.overlay_animation || {};
    onUpdateTrack({
      ...track,
      groups: track.groups.map((g) =>
        g.id === currentGroup.id ? { ...g, overlay_animation: { ...curAnim, ...patchAnim } } : g
      ),
    });
  };

  // =========================================================================
  // EXPANDED WORKSPACE PANELS (Edge-to-Edge, Continuous Dark Surface)
  // =========================================================================

  // 1. OVERLAY TRANSFORM PANEL
  if (activePanel === 'overlay_transform' && currentGroup) {
    const curLayout = currentGroup.layout || track?.default_layout || { scale: 1.0, rotation: 0 };
    return (
      <div className="w-full h-full bg-[#16191E] flex flex-col p-3 space-y-2 select-none overflow-hidden animate-in fade-in duration-150">
        <div className="flex items-center justify-between pb-1.5 border-b border-[#232832] shrink-0">
          <button
            type="button"
            onClick={() => setActivePanel('none')}
            className="flex items-center gap-1 text-[12px] font-semibold text-text-secondary hover:text-text-main"
          >
            <ChevronLeft size={16} />
            <span>Tools</span>
          </button>
          <span className="text-[12px] font-semibold text-text-main">Overlay Transform</span>
          <button
            type="button"
            onClick={() => setActivePanel('none')}
            className="w-7 h-7 rounded-full bg-primary flex items-center justify-center text-white active:scale-95"
            title="Done"
          >
            <Check size={14} />
          </button>
        </div>

        <div className="flex-1 min-h-0 overflow-y-auto no-scrollbar space-y-2.5 py-1">
          {/* Quick Buttons: Rotate 90, Flip H, Flip V, Reset */}
          <div className="grid grid-cols-4 gap-1.5">
            <button
              type="button"
              onClick={handleRotate90}
              className="h-9 rounded-lg bg-surface-200 border border-border flex flex-col items-center justify-center text-[11px] font-medium text-text-main hover:bg-surface-300 active:scale-95"
            >
              <RotateCw size={13} className="text-primary mb-0.5" />
              <span>Rotate 90°</span>
            </button>
            <button
              type="button"
              onClick={handleToggleFlipH}
              className={`h-9 rounded-lg border flex flex-col items-center justify-center text-[11px] font-medium transition-colors active:scale-95 ${
                currentGroup.flip_h ? 'bg-primary/20 border-primary text-white' : 'bg-surface-200 border-border text-text-main'
              }`}
            >
              <FlipHorizontal size={13} className="mb-0.5" />
              <span>Flip H</span>
            </button>
            <button
              type="button"
              onClick={handleToggleFlipV}
              className={`h-9 rounded-lg border flex flex-col items-center justify-center text-[11px] font-medium transition-colors active:scale-95 ${
                currentGroup.flip_v ? 'bg-primary/20 border-primary text-white' : 'bg-surface-200 border-border text-text-main'
              }`}
            >
              <FlipVertical size={13} className="mb-0.5" />
              <span>Flip V</span>
            </button>
            <button
              type="button"
              onClick={handleResetTransform}
              className="h-9 rounded-lg bg-surface-200 border border-border flex flex-col items-center justify-center text-[11px] font-medium text-text-secondary hover:text-text-main active:scale-95"
            >
              <RotateCcw size={13} className="mb-0.5" />
              <span>Reset</span>
            </button>
          </div>

          <ValueSlider
            label="Scale / Size"
            value={curLayout.scale ?? 1.0}
            min={0.2}
            max={2.5}
            step={0.05}
            unit="x"
            onChange={(val) => {
              if (!track) return;
              onUpdateTrack({
                ...track,
                groups: track.groups.map((g) =>
                  g.id === currentGroup.id ? { ...g, layout: { ...(g.layout || track.default_layout), scale: val } } : g
                ),
              });
            }}
          />

          <ValueSlider
            label="Rotation"
            value={curLayout.rotation ?? 0}
            min={-180}
            max={180}
            step={1}
            unit="°"
            onChange={(val) => {
              if (!track) return;
              onUpdateTrack({
                ...track,
                groups: track.groups.map((g) =>
                  g.id === currentGroup.id ? { ...g, layout: { ...(g.layout || track.default_layout), rotation: val } } : g
                ),
              });
            }}
          />
        </div>
      </div>
    );
  }

  // 2. OVERLAY ANIMATION PANEL
  if (activePanel === 'overlay_animation' && currentGroup) {
    const curAnim = currentGroup.overlay_animation || {};
    const enterChoices = [
      { id: 'none', label: 'None' },
      { id: 'fade', label: 'Fade In' },
      { id: 'pop', label: 'Pop / Zoom' },
      { id: 'scale', label: 'Scale Up' },
      { id: 'slide_up', label: 'Slide Up' },
      { id: 'slide_down', label: 'Slide Down' },
      { id: 'slide_left', label: 'Slide Left' },
      { id: 'slide_right', label: 'Slide Right' },
    ];
    const exitChoices = [
      { id: 'none', label: 'None' },
      { id: 'fade', label: 'Fade Out' },
      { id: 'scale_down', label: 'Scale Down' },
      { id: 'slide_up', label: 'Slide Up' },
      { id: 'slide_down', label: 'Slide Down' },
      { id: 'slide_left', label: 'Slide Left' },
      { id: 'slide_right', label: 'Slide Right' },
    ];

    return (
      <div className="w-full h-full bg-[#16191E] flex flex-col p-3 space-y-2 select-none overflow-hidden animate-in fade-in duration-150">
        <div className="flex items-center justify-between pb-1.5 border-b border-[#232832] shrink-0">
          <button
            type="button"
            onClick={() => setActivePanel('none')}
            className="flex items-center gap-1 text-[12px] font-semibold text-text-secondary hover:text-text-main"
          >
            <ChevronLeft size={16} />
            <span>Tools</span>
          </button>
          
          <div className="flex items-center gap-1">
            <button
              type="button"
              onClick={() => setOverlayAnimTab('enter')}
              className={`px-2.5 py-1 rounded text-[11px] font-semibold transition-colors ${
                overlayAnimTab === 'enter' ? 'bg-primary text-white' : 'text-text-muted hover:text-text-secondary'
              }`}
            >
              In (Entrance)
            </button>
            <button
              type="button"
              onClick={() => setOverlayAnimTab('exit')}
              className={`px-2.5 py-1 rounded text-[11px] font-semibold transition-colors ${
                overlayAnimTab === 'exit' ? 'bg-primary text-white' : 'text-text-muted hover:text-text-secondary'
              }`}
            >
              Out (Exit)
            </button>
          </div>

          <button
            type="button"
            onClick={() => setActivePanel('none')}
            className="w-7 h-7 rounded-full bg-primary flex items-center justify-center text-white active:scale-95"
            title="Done"
          >
            <Check size={14} />
          </button>
        </div>

        <div className="flex-1 min-h-0 overflow-y-auto no-scrollbar space-y-2 py-1">
          {overlayAnimTab === 'enter' ? (
            <>
              <div className="grid grid-cols-4 gap-1.5">
                {enterChoices.map((c) => (
                  <button
                    key={c.id}
                    type="button"
                    onClick={() => handleUpdateOverlayAnimation({ enter: c.id as any })}
                    className={`h-9 rounded-lg border text-[11px] font-medium transition-colors ${
                      (curAnim.enter || 'none') === c.id
                        ? 'bg-primary/20 border-primary text-white font-bold'
                        : 'bg-surface-200 border-border text-text-secondary hover:text-text-main'
                    }`}
                  >
                    {c.label}
                  </button>
                ))}
              </div>
              <ValueSlider
                label="Entrance Duration"
                value={(curAnim.enter_duration_ms || 400) / 1000}
                min={0.1}
                max={2.0}
                step={0.1}
                unit="s"
                onChange={(val) => handleUpdateOverlayAnimation({ enter_duration_ms: Math.round(val * 1000) })}
              />
            </>
          ) : (
            <>
              <div className="grid grid-cols-4 gap-1.5">
                {exitChoices.map((c) => (
                  <button
                    key={c.id}
                    type="button"
                    onClick={() => handleUpdateOverlayAnimation({ exit: c.id as any })}
                    className={`h-9 rounded-lg border text-[11px] font-medium transition-colors ${
                      (curAnim.exit || 'none') === c.id
                        ? 'bg-primary/20 border-primary text-white font-bold'
                        : 'bg-surface-200 border-border text-text-secondary hover:text-text-main'
                    }`}
                  >
                    {c.label}
                  </button>
                ))}
              </div>
              <ValueSlider
                label="Exit Duration"
                value={(curAnim.exit_duration_ms || 400) / 1000}
                min={0.1}
                max={2.0}
                step={0.1}
                unit="s"
                onChange={(val) => handleUpdateOverlayAnimation({ exit_duration_ms: Math.round(val * 1000) })}
              />
            </>
          )}
        </div>
      </div>
    );
  }

  // 3. LAYER OPACITY PANEL
  if (activePanel === 'layer_opacity' && currentGroup) {
    const curOpacity = currentGroup.opacity ?? 100;
    return (
      <div className="w-full h-full bg-[#16191E] flex flex-col p-3 space-y-2 select-none overflow-hidden animate-in fade-in duration-150">
        <div className="flex items-center justify-between pb-1.5 border-b border-[#232832] shrink-0">
          <button
            type="button"
            onClick={() => setActivePanel('none')}
            className="flex items-center gap-1 text-[12px] font-semibold text-text-secondary hover:text-text-main"
          >
            <ChevronLeft size={16} />
            <span>Tools</span>
          </button>
          <span className="text-[12px] font-semibold text-text-main">Overlay Opacity</span>
          <button
            type="button"
            onClick={() => setActivePanel('none')}
            className="w-7 h-7 rounded-full bg-primary flex items-center justify-center text-white active:scale-95"
            title="Done"
          >
            <Check size={14} />
          </button>
        </div>

        <div className="flex-1 min-h-0 flex flex-col justify-center py-4">
          <ValueSlider
            label="Opacity"
            value={curOpacity}
            min={0}
            max={100}
            step={1}
            unit="%"
            onChange={(val) => {
              if (!track) return;
              onUpdateTrack({
                ...track,
                groups: track.groups.map((g) => (g.id === currentGroup.id ? { ...g, opacity: val } : g)),
              });
            }}
          />
        </div>
      </div>
    );
  }

  // 4. LAYER VOLUME PANEL (For Video Overlay or Audio Layers)
  if (activePanel === 'layer_volume' && currentGroup) {
    const layerVol = currentGroup.volume ?? 100;
    const layerMuted = currentGroup.is_muted ?? false;

    return (
      <div className="w-full h-full bg-[#16191E] flex flex-col p-3 space-y-2 select-none overflow-hidden animate-in fade-in duration-150">
        <div className="flex items-center justify-between pb-1.5 border-b border-[#232832] shrink-0">
          <button
            type="button"
            onClick={() => setActivePanel('none')}
            className="flex items-center gap-1 text-[12px] font-semibold text-text-secondary hover:text-text-main"
          >
            <ChevronLeft size={16} />
            <span>Tools</span>
          </button>
          <span className="text-[12px] font-semibold text-text-main">Layer Audio</span>
          <button
            type="button"
            onClick={() => setActivePanel('none')}
            className="w-7 h-7 rounded-full bg-primary flex items-center justify-center text-white active:scale-95"
            title="Done"
          >
            <Check size={14} />
          </button>
        </div>

        <div className="flex-1 min-h-0 overflow-y-auto no-scrollbar space-y-2 py-1">
          <div className="flex items-center justify-between py-1">
            <button
              type="button"
              onClick={() => {
                if (!track) return;
                onUpdateTrack({
                  ...track,
                  groups: track.groups.map((g) =>
                    g.id === currentGroup.id ? { ...g, is_muted: !layerMuted } : g
                  ),
                });
              }}
              className={`p-2 rounded-lg border flex items-center gap-1.5 text-[11px] font-semibold ${
                layerMuted ? 'bg-red-500/20 border-red-500 text-red-300' : 'bg-surface-200 border-border text-text-main'
              }`}
            >
              {layerMuted ? <VolumeX size={15} /> : <Volume2 size={15} />}
              <span>{layerMuted ? 'Muted' : 'Unmuted'}</span>
            </button>
          </div>

          <ValueSlider
            label="Layer Volume"
            value={layerVol}
            min={0}
            max={200}
            unit="%"
            onChange={(val) => {
              if (!track) return;
              onUpdateTrack({
                ...track,
                groups: track.groups.map((g) =>
                  g.id === currentGroup.id ? { ...g, volume: val } : g
                ),
              });
            }}
          />
        </div>
      </div>
    );
  }

  // 5. MAIN VIDEO VOLUME PANEL
  if (activePanel === 'main_volume') {
    return (
      <div className="w-full h-full bg-[#16191E] flex flex-col p-3 space-y-2 select-none overflow-hidden animate-in fade-in duration-150">
        <div className="flex items-center justify-between pb-1.5 border-b border-[#232832] shrink-0">
          <button
            type="button"
            onClick={() => setActivePanel('none')}
            className="flex items-center gap-1 text-[12px] font-semibold text-text-secondary hover:text-text-main"
          >
            <ChevronLeft size={16} />
            <span>Tools</span>
          </button>
          <span className="text-[12px] font-semibold text-text-main">Main Video Audio</span>
          <button
            type="button"
            onClick={() => setActivePanel('none')}
            className="w-7 h-7 rounded-full bg-primary flex items-center justify-center text-white active:scale-95"
            title="Done"
          >
            <Check size={14} />
          </button>
        </div>

        <div className="flex-1 min-h-0 overflow-y-auto no-scrollbar space-y-2 py-1">
          <div className="flex items-center justify-between py-1">
            <button
              type="button"
              onClick={() => onUpdateMainVideoAudio?.(mainVideoVolume, !isMainVideoMuted)}
              className={`p-2 rounded-lg border flex items-center gap-1.5 text-[11px] font-semibold ${
                isMainVideoMuted ? 'bg-red-500/20 border-red-500 text-red-300' : 'bg-surface-200 border-border text-text-main'
              }`}
            >
              {isMainVideoMuted ? <VolumeX size={15} /> : <Volume2 size={15} />}
              <span>{isMainVideoMuted ? 'Muted' : 'Unmuted'}</span>
            </button>
          </div>

          <ValueSlider
            label="Video Volume"
            value={mainVideoVolume}
            min={0}
            max={200}
            unit="%"
            onChange={(val) => onUpdateMainVideoAudio?.(val, isMainVideoMuted)}
          />
        </div>
      </div>
    );
  }

  // 6. UNIFIED TEXT & STYLE DRAWER (Templates | Fonts | Styles | Effects | Animations)
  const isTextDrawerOpen =
    activePanel === 'style' ||
    activePanel === 'fonts' ||
    activePanel === 'effects' ||
    activePanel === 'templates' ||
    activePanel === 'animation';

  if (isTextDrawerOpen) {
    const enterEffects = [
      { id: 'none', label: 'None', desc: 'Static' },
      { id: 'fade', label: 'Fade In', desc: 'Smooth opacity ramp' },
      { id: 'pop', label: 'Pop In', desc: 'Spring scale bounce' },
      { id: 'slide_up', label: 'Slide Up', desc: 'Rise from below' },
    ];

    const exitEffects = [
      { id: 'none', label: 'None', desc: 'Static' },
      { id: 'fade', label: 'Fade Out', desc: 'Smooth opacity exit' },
      { id: 'scale_down', label: 'Scale Down', desc: 'Shrink away' },
      { id: 'slide_down', label: 'Slide Down', desc: 'Drop down' },
    ];

    const loopEffects = [
      { id: 'none', label: 'None', desc: 'Static' },
      { id: 'pulse', label: 'Pulse', desc: 'Rhythmic breathing' },
      { id: 'wiggle', label: 'Wiggle', desc: 'Playful subtle tilt' },
    ];

    const wordEffects = [
      { id: 'none', label: 'None', desc: 'No word pop' },
      { id: 'pop', label: 'Pop Scale', desc: 'Active word punch' },
      { id: 'bounce', label: 'Punch Jump', desc: 'Active word bounce' },
      { id: 'highlight', label: 'Highlight', desc: 'Color only' },
    ];

    const curEnter = currentAnimation?.enter?.type || (currentAnimation as any)?.enter_type || 'none';
    const curExit = currentAnimation?.exit?.type || (currentAnimation as any)?.exit_type || 'none';
    const curLoop = (currentAnimation as any)?.loop?.type || (currentAnimation as any)?.loop_type || 'none';
    const curWord = currentAnimation?.active_word?.type || currentAnimation?.type || 'pop';

    return (
      <div className={`w-full ${hideTabsForKeyboard ? 'h-auto' : 'h-full'} bg-[#121418] flex flex-col px-3 pt-2 pb-1 space-y-1.5 select-none overflow-hidden animate-in fade-in duration-150`}>
        {/* CapCut-Style Synchronized Text Input Box (Docked flush above keyboard) */}
        {isTextLayerSelected && (
          <div className="w-full bg-[#1A1D24] border border-[#2A2F3A] focus-within:border-primary/80 rounded-xl px-3 py-1.5 flex items-center justify-between gap-2 shadow-inner shrink-0">
            <textarea
              value={textInputDraft}
              onChange={(e) => handleTextDraftChange(e.target.value)}
              onFocus={() => setIsInputFocused(true)}
              onBlur={() => {
                setIsInputFocused(false);
                handleTextDraftCommit();
              }}
              onKeyDown={(e) => {
                if (e.key === 'Enter' && !e.shiftKey) {
                  e.preventDefault();
                  handleTextDraftCommit();
                  (document.activeElement as HTMLElement)?.blur();
                }
              }}
              rows={hideTabsForKeyboard ? 2 : 1}
              className="w-full bg-transparent text-white text-[13px] font-medium outline-none resize-none leading-normal placeholder-text-muted select-text"
              placeholder="Type caption or text here…"
            />
            {hideTabsForKeyboard && (
              <button
                type="button"
                onPointerDown={(e) => {
                  e.preventDefault();
                  handleTextDraftCommit();
                  (document.activeElement as HTMLElement)?.blur();
                }}
                onClick={(e) => {
                  e.preventDefault();
                  handleTextDraftCommit();
                  (document.activeElement as HTMLElement)?.blur();
                }}
                className="w-8 h-8 rounded-full bg-primary hover:bg-primary-hover text-white flex items-center justify-center shrink-0 active:scale-95 shadow-hero-sm transition-all"
                title="Done typing"
                aria-label="Done typing"
              >
                <Check size={16} />
              </button>
            )}
          </div>
        )}

        {/* Hide tabs and panel controls when keyboard is on to eliminate visual clutter */}
        {!hideTabsForKeyboard && (
          <>
            {/* Primary Navigation Tab Bar: Flat typographic tabs inspired by CapCut */}
            <div className="flex items-center justify-between border-b border-[#222630] shrink-0 pb-0.5">
              <div className="flex items-center gap-4 sm:gap-6 overflow-x-auto no-scrollbar flex-1 min-w-0">
                {([
                  { id: 'templates', label: 'Templates' },
                  { id: 'fonts', label: 'Fonts' },
                  { id: 'styles', label: 'Styles' },
                  { id: 'effects', label: 'Effects' },
                  { id: 'animations', label: 'Animations' },
                ] as { id: TextMainTab; label: string }[]).map((tab) => {
                  const isActive = textMainTab === tab.id;
                  const isDisabled = !isTextLayerSelected && tab.id !== 'templates';
                  return (
                    <button
                      key={tab.id}
                      type="button"
                      disabled={isDisabled}
                      onClick={() => handleSelectTextMainTab(tab.id)}
                      className={`py-2 px-1 text-[13px] font-semibold whitespace-nowrap transition-all shrink-0 relative ${
                        isDisabled
                          ? 'opacity-30 cursor-not-allowed text-text-muted pointer-events-none'
                          : isActive
                          ? 'text-white'
                          : 'text-[#8E95A2] hover:text-white'
                      }`}
                    >
                      <span>{tab.label}</span>
                      {isActive && (
                        <div className="absolute bottom-0 left-0 right-0 h-[2.5px] bg-primary rounded-full shadow-[0_0_8px_rgba(37,99,235,0.7)]" />
                      )}
                    </button>
                  );
                })}
              </div>

              <div className="flex items-center gap-1 shrink-0 ml-2">
                <button
                  type="button"
                  onClick={() => setActivePanel('none')}
                  className="w-7 h-7 rounded-full bg-surface-200 hover:bg-surface-300 text-text-secondary hover:text-white flex items-center justify-center active:scale-95 transition-all"
                  title="Done"
                  aria-label="Done editing"
                >
                  <Check size={14} className="text-primary" />
                </button>
              </div>
            </div>

            {/* TAB 1: TEMPLATES */}
            {textMainTab === 'templates' && (
              <div className="flex-1 min-h-0 flex flex-col space-y-2 py-1">
                <div className="flex items-center justify-between">
                  <span className="text-[11px] font-medium text-text-secondary">Full Caption Presets</span>
                  <span className="text-[10px] text-text-muted">{CAPTION_TEMPLATES.length} templates</span>
                </div>
                <div className="flex-1 min-h-0 overflow-y-auto no-scrollbar py-1 space-y-2">
                  <div className="grid grid-cols-2 sm:grid-cols-3 gap-1.5">
                    {CAPTION_TEMPLATES.map((tmpl) => (
                      <button
                        key={tmpl.id}
                        type="button"
                        onClick={() => {
                          patch(tmpl.style, undefined, tmpl.animation, true);
                        }}
                        className="p-2 rounded-lg border border-[#262B35] bg-[#1C1F26] hover:bg-[#232832] hover:border-text-muted flex flex-col items-center justify-center gap-1 transition-all active:scale-95 min-h-[52px]"
                      >
                        <span
                          className="text-[13px] font-bold"
                          style={{
                            fontFamily: tmpl.style.font_family || 'Impact',
                            color: tmpl.style.primary_color || '#FFF',
                          }}
                        >
                          {tmpl.name}
                        </span>
                        <span className="text-[10px] text-text-muted">{tmpl.category}</span>
                      </button>
                    ))}
                  </div>
                  <ApplyToAllButton section="templates" sectionLabel="Preset Template" onApply={handleApplySectionToAll} />
                </div>
              </div>
            )}

            {/* TAB 2: FONTS */}
            {textMainTab === 'fonts' && (
              <div className="flex-1 min-h-0 flex flex-col space-y-1.5 py-1">
                {/* Search & Import Bar */}
                <div className="flex items-center gap-1.5 shrink-0">
                  <div className="relative flex-1">
                    <Search size={13} className="absolute left-2.5 top-1/2 -translate-y-1/2 text-text-muted pointer-events-none" />
                    <input
                      type="text"
                      value={fontSearch}
                      onChange={(e) => setFontSearch(e.target.value)}
                      placeholder="Search fonts…"
                      className="w-full h-7 pl-7 pr-6 rounded-lg bg-[#1C1F26] border border-[#262B35] text-[11px] text-white placeholder-text-muted focus:outline-none focus:border-primary transition-colors"
                    />
                    {fontSearch && (
                      <button
                        type="button"
                        onClick={() => setFontSearch('')}
                        className="absolute right-2 top-1/2 -translate-y-1/2 text-text-muted hover:text-white p-0.5"
                      >
                        <X size={11} />
                      </button>
                    )}
                  </div>

                  <input
                    ref={fontFileInputRef}
                    type="file"
                    accept=".ttf,.otf,.woff,.woff2"
                    className="hidden"
                    onChange={handleImportFont}
                  />
                  <button
                    type="button"
                    onClick={() => fontFileInputRef.current?.click()}
                    disabled={isImportingFont}
                    className="h-7 px-2.5 rounded-lg bg-[#1C1F26] hover:bg-[#252A34] border border-[#262B35] text-primary hover:text-white text-[11px] font-medium flex items-center gap-1 shrink-0 active:scale-95 transition-all"
                    title="Import custom TTF/OTF font file"
                  >
                    {isImportingFont ? (
                      <Loader2 size={12} className="animate-spin text-primary" />
                    ) : (
                      <Upload size={12} className="text-primary" />
                    )}
                    <span className="hidden sm:inline">Import</span>
                  </button>
                </div>

                {fontImportError && (
                  <div className="px-2 py-1 rounded-md bg-red-500/15 border border-red-500/30 text-[10px] text-red-400 flex items-center justify-between shrink-0">
                    <span>{fontImportError}</span>
                    <button type="button" onClick={() => setFontImportError(null)} className="text-red-400 hover:text-white">
                      <X size={11} />
                    </button>
                  </div>
                )}

                {/* Category Filters (Horizontal Scroll) */}
                <div className="flex items-center gap-1 overflow-x-auto no-scrollbar py-0.5 shrink-0">
                  {FONT_CATEGORIES.map((cat) => (
                    <button
                      key={cat.key}
                      type="button"
                      onClick={() => setSelectedFontCategory(cat.key)}
                      className={`h-6 px-2.5 rounded-full text-[10px] font-medium whitespace-nowrap transition-all shrink-0 active:scale-95 ${
                        selectedFontCategory === cat.key
                          ? 'bg-primary text-white font-semibold'
                          : 'bg-[#1C1F26] text-[#8E95A2] hover:text-white'
                      }`}
                    >
                      {cat.label}
                      {cat.key === 'favorites' && favoriteFonts.length > 0 && ` (${favoriteFonts.length})`}
                    </button>
                  ))}
                </div>

                {/* Font Grid: Clean 3 columns on desktop, 2-3 columns on mobile, unified tiles */}
                <div className="flex-1 min-h-0 overflow-y-auto no-scrollbar pt-1 pb-4">
                  {filteredFonts.length === 0 ? (
                    <div className="h-32 flex flex-col items-center justify-center text-center p-4 text-text-muted">
                      <Type size={24} className="mb-2 opacity-40" />
                      <p className="text-xs font-medium">No fonts found</p>
                      <p className="text-[10px] mt-0.5 text-text-muted">
                        {selectedFontCategory === 'favorites'
                          ? 'Star a font to add it to your favorites'
                          : 'Try a different search term or category'}
                      </p>
                    </div>
                  ) : (
                    <div className="grid grid-cols-2 sm:grid-cols-3 gap-1.5 pb-2">
                      {filteredFonts.map((f) => {
                        const isSelected = (currentStyle?.font_family || 'Montserrat').toLowerCase() === f.family.toLowerCase();
                        const isFav = favoriteFonts.some((fav) => fav.toLowerCase() === f.family.toLowerCase());
                        const isLoading = loadingFontFamily?.toLowerCase() === f.family.toLowerCase();

                        return (
                          <button
                            key={f.family}
                            type="button"
                            onClick={() => handleSelectFont(f)}
                            className={`group flex items-center justify-between px-2.5 py-1.5 rounded-lg border text-left transition-all min-h-[42px] active:scale-95 ${
                              isSelected
                                ? 'bg-primary/20 border-primary ring-1 ring-primary text-white shadow-sm'
                                : 'bg-[#1C1F26] border-[#262B35] hover:bg-[#232832] hover:border-text-muted/60'
                            }`}
                            title={f.family}
                          >
                            <div className="flex-1 min-w-0 pr-1 flex items-center">
                              {isLoading ? (
                                <Loader2 size={13} className="animate-spin text-primary" />
                              ) : (
                                <span
                                  className="leading-tight text-white truncate select-none text-[12px] font-semibold group-hover:text-primary transition-colors"
                                  style={{ fontFamily: f.family }}
                                >
                                  {f.family}
                                </span>
                              )}
                            </div>

                            <button
                              type="button"
                              onClick={(e) => handleToggleFavorite(e, f.family)}
                              className={`p-1 rounded transition-colors shrink-0 ${
                                isFav
                                  ? 'text-amber-400'
                                  : 'text-text-muted/30 hover:text-text-secondary opacity-0 group-hover:opacity-100'
                              }`}
                              title={isFav ? 'Remove from favorites' : 'Add to favorites'}
                            >
                              <Star size={11} fill={isFav ? 'currentColor' : 'none'} />
                            </button>
                          </button>
                        );
                      })}
                    </div>
                  )}
                </div>
              </div>
            )}

        {/* TAB 3: STYLES (Property Navigation & Controls: Text | Stroke | Glow | Background | Shadow | Spacing) */}
        {textMainTab === 'styles' && (
          <div className="flex-1 min-h-0 flex flex-col space-y-1.5 py-0.5">
            {/* Style-Property Sub-Tabs: Text | Stroke | Glow | Background | Shadow | Spacing (Moved directly to top) */}
            <div 
              className="flex items-center gap-1 sm:gap-2 overflow-x-auto no-scrollbar py-1.5 shrink-0 border-b border-[#20242D]"
              style={{ WebkitOverflowScrolling: 'touch', touchAction: 'pan-x' }}
            >
              {([
                { id: 'text', label: 'Text' },
                { id: 'stroke', label: 'Stroke' },
                { id: 'glow', label: 'Glow' },
                { id: 'background', label: 'Background' },
                { id: 'shadow', label: 'Shadow' },
                { id: 'spacing', label: 'Spacing' },
              ] as { id: StyleTab; label: string }[]).map((tab) => {
                const isTabActive = styleTab === tab.id;
                return (
                  <button
                    key={tab.id}
                    type="button"
                    onClick={() => setStyleTab(tab.id)}
                    className={`px-3 py-1 rounded-md text-[12px] font-medium whitespace-nowrap transition-all shrink-0 active:scale-95 ${
                      isTabActive
                        ? 'bg-white/10 text-white font-semibold'
                        : 'text-[#8E95A2] hover:text-white'
                    }`}
                  >
                    {tab.label}
                  </button>
                );
              })}
            </div>

            {/* Property Controls Container: Maximized vertical space for parameters */}
            <div className="flex-1 min-h-0 overflow-y-auto no-scrollbar pb-6 pt-0.5">
          {styleTab === 'text' && (
            <div className="space-y-3 py-1">
              {/* Primary Text Fill Color */}
              <CapCutColorRow
                label="Text Fill Color"
                value={currentStyle?.primary_color}
                defaultValue="#FFFFFF"
                onChange={(c) => patch({ primary_color: c })}
                onOpenCustom={() => setActiveColorPicker({
                  isOpen: true,
                  title: 'Custom Text Fill Color',
                  property: 'primary_color',
                  color: currentStyle?.primary_color || '#FFFFFF',
                })}
              />

              {/* Spoken Word Active Highlight Color: Caption Layers ONLY */}
              {isCaption && (
                <div className="pt-2 border-t border-[#232832]">
                  <CapCutColorRow
                    label="Highlight Color"
                    value={currentStyle?.highlight_color}
                    defaultValue="#FFE600"
                    onChange={(c) => patch({ highlight_color: c })}
                    onOpenCustom={() => setActiveColorPicker({
                      isOpen: true,
                      title: 'Custom Spoken Highlight Color',
                      property: 'highlight_color',
                      color: currentStyle?.highlight_color || '#FFE600',
                    })}
                  />
                </div>
              )}

              {/* Text Opacity Slider */}
              <ValueSlider
                label="Text Opacity"
                value={Math.round((currentStyle?.text_opacity ?? 1.0) * 100)}
                min={0}
                max={100}
                step={1}
                unit="%"
                onChange={(val) => patch({ text_opacity: Number((val / 100).toFixed(2)) })}
              />

              {/* Font Size Slider */}
              <ValueSlider
                label="Font Size"
                value={currentStyle?.font_size || 44}
                min={16}
                max={90}
                unit="px"
                onChange={(val) => patch({ font_size: val })}
              />

              {/* Text Style: Bold, Italic, Underline Quick Action Bar */}
              <div className="space-y-1 pt-1 border-t border-[#232832]">
                <span className="text-[11px] font-medium text-text-secondary">Text Style</span>
                <div className="grid grid-cols-3 gap-2">
                  {/* B (Bold) */}
                  {(() => {
                    const currentWeight = String(currentStyle?.font_weight || '700').toLowerCase();
                    const isBold = currentWeight === 'bold' || currentWeight === '700' || currentWeight === '800' || currentWeight === '900';
                    return (
                      <button
                        type="button"
                        onClick={() => {
                          const targetWeight = isBold
                            ? (supportedWeights.includes('400') ? '400' : supportedWeights[0] || 'normal')
                            : (supportedWeights.includes('700') ? '700' : supportedWeights[supportedWeights.length - 1] || 'bold');
                          patch({ font_weight: targetWeight });
                        }}
                        className={`h-8 rounded-lg border text-xs font-bold flex items-center justify-center gap-1.5 transition-all active:scale-95 ${
                          isBold
                            ? 'bg-primary border-primary text-white shadow-hero-sm'
                            : 'bg-[#1C1F26] border-[#262B35] text-text-secondary hover:text-white'
                        }`}
                        title="Toggle Bold"
                      >
                        <Bold size={14} strokeWidth={2.5} />
                        <span>Bold</span>
                      </button>
                    );
                  })()}

                  {/* I (Italic) */}
                  {(() => {
                    const isItalic = currentStyle?.font_style === 'italic';
                    return (
                      <button
                        type="button"
                        onClick={() => patch({ font_style: isItalic ? 'normal' : 'italic' })}
                        className={`h-8 rounded-lg border text-xs font-semibold italic flex items-center justify-center gap-1.5 transition-all active:scale-95 ${
                          isItalic
                            ? 'bg-primary border-primary text-white shadow-hero-sm'
                            : 'bg-[#1C1F26] border-[#262B35] text-text-secondary hover:text-white'
                        }`}
                        title="Toggle Italic"
                      >
                        <Italic size={14} strokeWidth={2.5} />
                        <span>Italic</span>
                      </button>
                    );
                  })()}

                  {/* U (Underline) */}
                  {(() => {
                    const isUnderline = Boolean(currentStyle?.underline);
                    return (
                      <button
                        type="button"
                        onClick={() => patch({ underline: !isUnderline })}
                        className={`h-8 rounded-lg border text-xs font-semibold underline flex items-center justify-center gap-1.5 transition-all active:scale-95 ${
                          isUnderline
                            ? 'bg-primary border-primary text-white shadow-hero-sm'
                            : 'bg-[#1C1F26] border-[#262B35] text-text-secondary hover:text-white'
                        }`}
                        title="Toggle Underline"
                      >
                        <Underline size={14} strokeWidth={2.5} />
                        <span>Underline</span>
                      </button>
                    );
                  })()}
                </div>
              </div>

              {/* Font Weight */}
              {supportedWeights.length > 1 && (
                <div className="space-y-1 pt-1 border-t border-[#20242D]">
                  <div className="flex items-center justify-between text-[11px]">
                    <span className="font-medium text-text-secondary">Font Weight</span>
                  </div>

                  <div className="flex flex-wrap gap-1 py-0.5">
                    {(['400', '500', '600', '700', '800', '900'] as const).map((wt) => {
                      const isAvailable = supportedWeights.includes(wt);
                      if (!isAvailable) return null;
                      const currentWeight = String(currentStyle?.font_weight || '700').toLowerCase();
                      const isSelected = currentWeight === wt || (wt === '700' && currentWeight === 'bold') || (wt === '400' && currentWeight === 'normal');
                      const label =
                        wt === '400' ? 'Regular' :
                        wt === '500' ? 'Medium' :
                        wt === '600' ? 'SemiBold' :
                        wt === '700' ? 'Bold' :
                        wt === '800' ? 'ExtraBold' : 'Black (900)';

                      return (
                        <button
                          key={wt}
                          type="button"
                          onClick={() => patch({ font_weight: wt })}
                          className={`h-6 px-2.5 rounded-md text-[10px] font-medium transition-all ${
                            isSelected
                              ? 'bg-primary text-white font-bold'
                              : 'bg-[#1C1F26] border border-[#262B35] text-[#8E95A2] hover:text-white'
                          }`}
                        >
                          {label}
                        </button>
                      );
                    })}
                  </div>
                </div>
              )}

              {/* Text Capitalization (Case) */}
              <div className="space-y-1 pt-1 border-t border-[#20242D]">
                <span className="text-[11px] font-medium text-text-secondary">Text Capitalization</span>
                <div className="grid grid-cols-2 sm:grid-cols-4 gap-1.5">
                  {[
                    { id: 'none', label: 'Original', preview: 'Aa' },
                    { id: 'uppercase', label: 'UPPERCASE', preview: 'AA' },
                    { id: 'lowercase', label: 'lowercase', preview: 'aa' },
                    { id: 'title', label: 'Title Case', preview: 'Aa Bb' },
                  ].map((mode) => {
                    const isSelected = (currentStyle?.text_transform || 'none') === mode.id;
                    return (
                      <button
                        key={mode.id}
                        type="button"
                        onClick={() => patch({ text_transform: mode.id as any })}
                        className={`h-8 px-2 rounded-lg border text-left transition-all active:scale-95 flex items-center justify-between ${
                          isSelected
                            ? 'bg-primary/20 border-primary text-white font-bold shadow-sm'
                            : 'bg-[#1C1F26] border-[#262B35] text-text-secondary hover:text-white'
                        }`}
                      >
                        <span className="text-[11px] font-medium">{mode.label}</span>
                        <span className="text-[10px] font-mono text-primary bg-primary/10 px-1 py-0.2 rounded">{mode.preview}</span>
                      </button>
                    );
                  })}
                </div>
              </div>

              {/* Apply To All Text Button */}
              <ApplyToAllButton section="text" sectionLabel="Text (Font, Colors, Style, Case & Size)" onApply={handleApplySectionToAll} />
            </div>
          )}

          {styleTab === 'stroke' && (
            <div className="space-y-2 py-1">
              <div className="flex items-center justify-between text-[11px]">
                <span className="text-text-secondary">Enable Text Outline</span>
                <input
                  type="checkbox"
                  checked={currentStyle?.outline_enabled ?? true}
                  onChange={(e) => patch({ outline_enabled: e.target.checked })}
                  className="w-4 h-4 accent-primary cursor-pointer"
                />
              </div>
              {currentStyle?.outline_enabled && (
                <>
                  <ValueSlider
                    label="Stroke Width"
                    value={currentStyle?.outline_width || 3}
                    min={0.5}
                    max={12}
                    step={0.5}
                    unit="px"
                    onChange={(val) => patch({ outline_width: val })}
                  />
                  <ValueSlider
                    label="Stroke Opacity"
                    value={Math.round((currentStyle?.outline_opacity ?? 1.0) * 100)}
                    min={0}
                    max={100}
                    step={1}
                    unit="%"
                    onChange={(val) => patch({ outline_opacity: Number((val / 100).toFixed(2)) })}
                  />
                  <CapCutColorRow
                    label="Stroke Color"
                    value={currentStyle?.outline_color}
                    defaultValue="#000000"
                    onChange={(c) => patch({ outline_color: c })}
                    onOpenCustom={() => setActiveColorPicker({
                      isOpen: true,
                      title: 'Custom Outline Stroke Color',
                      property: 'outline_color',
                      color: currentStyle?.outline_color || '#000000',
                    })}
                  />
                </>
              )}
              <ApplyToAllButton section="stroke" sectionLabel="Stroke" onApply={handleApplySectionToAll} />
            </div>
          )}

          {styleTab === 'glow' && (
            <div className="space-y-2 py-1">
              <div className="flex items-center justify-between text-[11px]">
                <span className="text-text-secondary">Enable Neon Glow</span>
                <input
                  type="checkbox"
                  checked={currentStyle?.glow_enabled ?? false}
                  onChange={(e) => patch({ glow_enabled: e.target.checked })}
                  className="w-4 h-4 accent-primary cursor-pointer"
                />
              </div>
              {currentStyle?.glow_enabled && (
                <>
                  <ValueSlider
                    label="Glow Radius"
                    value={currentStyle?.glow_blur || 12}
                    min={4}
                    max={40}
                    step={1}
                    unit="px"
                    onChange={(val) => patch({ glow_blur: val })}
                  />
                  <ValueSlider
                    label="Glow Opacity"
                    value={Math.round((currentStyle?.glow_opacity ?? 1.0) * 100)}
                    min={0}
                    max={100}
                    step={1}
                    unit="%"
                    onChange={(val) => patch({ glow_opacity: Number((val / 100).toFixed(2)) })}
                  />
                  <CapCutColorRow
                    label="Glow Color"
                    value={currentStyle?.glow_color}
                    defaultValue="#00F0FF"
                    onChange={(c) => patch({ glow_color: c })}
                    onOpenCustom={() => setActiveColorPicker({
                      isOpen: true,
                      title: 'Custom Neon Glow Color',
                      property: 'glow_color',
                      color: currentStyle?.glow_color || '#00F0FF',
                    })}
                  />
                </>
              )}
              <ApplyToAllButton section="glow" sectionLabel="Glow" onApply={handleApplySectionToAll} />
            </div>
          )}

          {styleTab === 'background' && (
            <div className="space-y-3 py-1">
              {/* Three Visual Modes */}
              <div className="space-y-1">
                <span className="text-[11px] font-medium text-text-secondary">Background Mode</span>
                <div className="grid grid-cols-3 gap-1.5">
                  {(['none', 'line', 'block'] as const).map((mode) => {
                    const currentMode = (!currentStyle?.background_enabled || currentStyle?.background_mode === 'none')
                      ? 'none'
                      : (currentStyle?.background_mode || 'line');
                    const isSelected = currentMode === mode;
                    const label = mode === 'none' ? 'None' : mode === 'line' ? 'Per-Line' : 'Full Box';
                    return (
                      <button
                        key={mode}
                        type="button"
                        onClick={() => patch({
                          background_mode: mode,
                          background_enabled: mode !== 'none',
                        })}
                        className={`h-8 rounded-lg text-[11px] font-semibold transition-all active:scale-95 ${
                          isSelected
                            ? 'bg-primary text-white shadow-hero-sm font-bold'
                            : 'bg-[#1C1F26] border border-[#262B35] text-text-secondary hover:text-white'
                        }`}
                      >
                        {label}
                      </button>
                    );
                  })}
                </div>
              </div>

              {Boolean(currentStyle?.background_enabled && currentStyle?.background_mode !== 'none') && (
                <>
                  <CapCutColorRow
                    label="Background Color"
                    value={currentStyle?.background_color}
                    defaultValue="#000000"
                    onChange={(c) => patch({ background_color: c })}
                    onOpenCustom={() => setActiveColorPicker({
                      isOpen: true,
                      title: 'Custom Background Color',
                      property: 'background_color',
                      color: currentStyle?.background_color || '#000000',
                      enableAlpha: true,
                    })}
                  />

                  {/* Background Opacity */}
                  <ValueSlider
                    label="Background Opacity"
                    value={Math.round((currentStyle?.background_opacity ?? 1.0) * 100)}
                    min={0}
                    max={100}
                    step={1}
                    unit="%"
                    onChange={(val) => patch({ background_opacity: Number((val / 100).toFixed(2)) })}
                  />

                  {/* Corner Radius */}
                  <ValueSlider
                    label="Corner Radius"
                    value={currentStyle?.background_corner_radius ?? 8}
                    min={0}
                    max={30}
                    step={1}
                    unit="px"
                    onChange={(val) => patch({ background_corner_radius: val })}
                  />

                  {/* Width Padding */}
                  <ValueSlider
                    label="Width Padding (X)"
                    value={currentStyle?.background_padding_x ?? 14}
                    min={0}
                    max={40}
                    step={1}
                    unit="px"
                    onChange={(val) => patch({ background_padding_x: val })}
                  />

                  {/* Height Padding */}
                  <ValueSlider
                    label="Height Padding (Y)"
                    value={currentStyle?.background_padding_y ?? 8}
                    min={0}
                    max={30}
                    step={1}
                    unit="px"
                    onChange={(val) => patch({ background_padding_y: val })}
                  />

                  {/* X Offset */}
                  <ValueSlider
                    label="Offset X"
                    value={currentStyle?.background_offset_x || 0}
                    min={-50}
                    max={50}
                    step={1}
                    unit="px"
                    onChange={(val) => patch({ background_offset_x: val })}
                  />

                  {/* Y Offset */}
                  <ValueSlider
                    label="Offset Y"
                    value={currentStyle?.background_offset_y || 0}
                    min={-50}
                    max={50}
                    step={1}
                    unit="px"
                    onChange={(val) => patch({ background_offset_y: val })}
                  />
                </>
              )}

              <ApplyToAllButton section="background" sectionLabel="Background" onApply={handleApplySectionToAll} />
            </div>
          )}

          {styleTab === 'shadow' && (
            <div className="space-y-2 py-1">
              <div className="flex items-center justify-between text-[11px]">
                <span className="text-text-secondary">Enable Drop Shadow</span>
                <input
                  type="checkbox"
                  checked={currentStyle?.shadow_enabled ?? false}
                  onChange={(e) => patch({ shadow_enabled: e.target.checked })}
                  className="w-4 h-4 accent-primary cursor-pointer"
                />
              </div>
              {currentStyle?.shadow_enabled && (
                <>
                  <CapCutColorRow
                    label="Shadow Color"
                    value={currentStyle?.shadow_color}
                    defaultValue="#000000"
                    onChange={(c) => patch({ shadow_color: c })}
                    onOpenCustom={() => setActiveColorPicker({
                      isOpen: true,
                      title: 'Custom Shadow Color',
                      property: 'shadow_color',
                      color: currentStyle?.shadow_color || '#000000',
                    })}
                  />

                  {/* Shadow Opacity */}
                  <ValueSlider
                    label="Shadow Opacity"
                    value={Math.round((currentStyle?.shadow_opacity ?? 1.0) * 100)}
                    min={0}
                    max={100}
                    step={1}
                    unit="%"
                    onChange={(val) => patch({ shadow_opacity: Number((val / 100).toFixed(2)) })}
                  />

                  {/* Blur */}
                  <ValueSlider
                    label="Blur"
                    value={currentStyle?.shadow_blur || 6}
                    min={0}
                    max={25}
                    step={1}
                    unit="px"
                    onChange={(val) => patch({ shadow_blur: val })}
                  />

                  {/* Distance */}
                  <ValueSlider
                    label="Distance"
                    value={currentStyle?.shadow_distance ?? 2.8}
                    min={0}
                    max={30}
                    step={0.5}
                    unit="px"
                    onChange={(dist) => {
                      const angle = currentStyle?.shadow_angle ?? 45.0;
                      const rad = (angle * Math.PI) / 180;
                      const offX = Number((dist * Math.cos(rad)).toFixed(1));
                      const offY = Number((dist * Math.sin(rad)).toFixed(1));
                      patch({ shadow_distance: dist, shadow_offset_x: offX, shadow_offset_y: offY });
                    }}
                  />

                  {/* Angle */}
                  <ValueSlider
                    label="Angle"
                    value={Math.round(currentStyle?.shadow_angle ?? 45)}
                    min={0}
                    max={360}
                    step={5}
                    unit="°"
                    onChange={(angle) => {
                      const dist = currentStyle?.shadow_distance ?? 2.8;
                      const rad = (angle * Math.PI) / 180;
                      const offX = Number((dist * Math.cos(rad)).toFixed(1));
                      const offY = Number((dist * Math.sin(rad)).toFixed(1));
                      patch({ shadow_angle: angle, shadow_offset_x: offX, shadow_offset_y: offY });
                    }}
                  />
                </>
              )}
              <ApplyToAllButton section="shadow" sectionLabel="Shadow" onApply={handleApplySectionToAll} />
            </div>
          )}

          {styleTab === 'spacing' && (
            <div className="space-y-3 py-1">
              {/* Horizontal Alignment */}
              <div className="space-y-1">
                <span className="text-[11px] font-medium text-text-secondary">Text Alignment</span>
                <div className="grid grid-cols-3 gap-1.5">
                  {[
                    { align: 'left', icon: <AlignLeft size={15} />, label: 'Left' },
                    { align: 'center', icon: <AlignCenter size={15} />, label: 'Center' },
                    { align: 'right', icon: <AlignRight size={15} />, label: 'Right' },
                  ].map(({ align, icon, label }) => {
                    const isSelected = (currentLayout?.alignment || 'center') === align;
                    return (
                      <button
                        key={align}
                        type="button"
                        onClick={() => patch(undefined, { alignment: align as any })}
                        className={`h-8 rounded-lg border text-[11px] font-semibold flex items-center justify-center gap-1.5 transition-all active:scale-95 ${
                          isSelected
                            ? 'bg-primary border-primary text-white shadow-hero-sm font-bold'
                            : 'bg-[#1C1F26] border-[#262B35] text-text-secondary hover:text-white'
                        }`}
                      >
                        {icon}
                        <span>{label}</span>
                      </button>
                    );
                  })}
                </div>
              </div>

              {/* Layer Scale */}
              <ValueSlider
                label="Layer Scale"
                value={currentLayout?.scale ?? 1.0}
                min={0.5}
                max={2.0}
                step={0.05}
                unit="x"
                onChange={(val) => patch(undefined, { scale: Number(val.toFixed(2)) })}
              />

              {/* Character Spacing */}
              <ValueSlider
                label="Character Spacing"
                value={currentStyle?.letter_spacing ?? 0}
                min={-2}
                max={15}
                step={0.5}
                unit="px"
                onChange={(val) => patch({ letter_spacing: Number(val.toFixed(1)) })}
              />

              {/* Line Spacing */}
              <ValueSlider
                label="Line Spacing"
                value={currentStyle?.line_height ?? 1.25}
                min={0.8}
                max={2.2}
                step={0.05}
                unit=""
                onChange={(val) => patch({ line_height: Number(val.toFixed(2)) })}
              />

              {/* Position Presets & Offsets */}
              <div className="pt-2 border-t border-[#20242D] space-y-2">
                <div className="flex items-center justify-between text-[11px]">
                  <span className="font-medium text-text-secondary">Position Presets</span>
                  <div className="flex items-center gap-1">
                    <button
                      type="button"
                      onClick={() => patch(undefined, { position_y: 0.22, position_x: 0.5 })}
                      className="px-2 py-0.5 rounded bg-[#1C1F26] border border-[#262B35] text-[10px] text-text-secondary hover:text-white"
                    >
                      Top
                    </button>
                    <button
                      type="button"
                      onClick={() => patch(undefined, { position_y: 0.50, position_x: 0.5 })}
                      className="px-2 py-0.5 rounded bg-[#1C1F26] border border-[#262B35] text-[10px] text-text-secondary hover:text-white"
                    >
                      Center
                    </button>
                    <button
                      type="button"
                      onClick={() => patch(undefined, { position_y: 0.78, position_x: 0.5 })}
                      className="px-2 py-0.5 rounded bg-[#1C1F26] border border-[#262B35] text-[10px] text-text-secondary hover:text-white"
                    >
                      Bottom
                    </button>
                  </div>
                </div>

                <ValueSlider
                  label="Vertical Position (Y)"
                  value={Math.round((currentLayout?.position_y ?? 0.78) * 100)}
                  min={15}
                  max={90}
                  step={1}
                  unit="%"
                  onChange={(val) => patch(undefined, { position_y: Number((val / 100).toFixed(2)) })}
                />

                <ValueSlider
                  label="Horizontal Position (X)"
                  value={Math.round((currentLayout?.position_x ?? 0.5) * 100)}
                  min={10}
                  max={90}
                  step={1}
                  unit="%"
                  onChange={(val) => patch(undefined, { position_x: Number((val / 100).toFixed(2)) })}
                />
              </div>

              <ApplyToAllButton section="spacing" sectionLabel="Spacing & Alignment" onApply={handleApplySectionToAll} />
            </div>
          )}
        </div>
      </div>
    )}

    {/* TAB 4: EFFECTS */}
    {textMainTab === 'effects' && (
      <div className="flex-1 min-h-0 flex flex-col space-y-2 py-0.5">
        <div className="flex-1 min-h-0 overflow-y-auto no-scrollbar space-y-3 pb-8 pt-0.5">
          {/* 1. Style Effects Presets */}
          <div className="space-y-1.5">
            <span className="text-[11px] font-medium text-text-secondary">Style Effects</span>
            <div className="grid grid-cols-3 sm:grid-cols-4 gap-1.5">
              {/* None / Reset Style Effects */}
              {(() => {
                const isNone = !currentStyle?.outline_enabled &&
                  !currentStyle?.glow_enabled &&
                  !currentStyle?.shadow_enabled &&
                  (!currentStyle?.background_enabled || currentStyle?.background_mode === 'none');
                return (
                  <button
                    type="button"
                    onClick={() => {
                      patch({
                        primary_color: '#FFFFFF',
                        text_opacity: 1.0,
                        outline_enabled: false,
                        outline_width: 0,
                        glow_enabled: false,
                        shadow_enabled: false,
                        background_enabled: false,
                        background_mode: 'none',
                      });
                    }}
                    className={`flex flex-col items-center justify-between p-2 rounded-lg border min-h-[64px] transition-all active:scale-95 ${
                      isNone
                        ? 'border-2 border-primary bg-primary/15 ring-1 ring-primary/60 text-white shadow-sm'
                        : 'border-[#262B35] bg-[#1C1F26] hover:bg-[#232832] hover:border-text-muted/60 text-text-muted hover:text-white'
                    }`}
                    title="None / Reset Effects"
                    aria-label="None / Reset Effects"
                  >
                    <div className="w-full h-8 flex items-center justify-center">
                      <Ban size={20} className={isNone ? 'text-primary' : 'text-text-muted'} />
                    </div>
                    <span className="text-[10px] text-text-muted truncate max-w-full text-center">
                      None
                    </span>
                  </button>
                );
              })()}
              {INLINE_STYLE_PREVIEWS.map((prev) => {
                const isSelected = (() => {
                  if (!currentStyle) return false;
                  const c1 = (currentStyle.primary_color || '').toUpperCase();
                  const c2 = (prev.style.primary_color || '').toUpperCase();
                  if (c1 !== c2) return false;

                  const curOutline = Boolean(currentStyle.outline_enabled);
                  const prevOutline = Boolean(prev.style.outline_enabled);
                  if (curOutline !== prevOutline) return false;
                  if (curOutline && (currentStyle.outline_color || '').toUpperCase() !== (prev.style.outline_color || '').toUpperCase()) {
                    return false;
                  }

                  const curGlow = Boolean(currentStyle.glow_enabled);
                  const prevGlow = Boolean(prev.style.glow_enabled);
                  if (curGlow !== prevGlow) return false;

                  const curShadow = Boolean(currentStyle.shadow_enabled);
                  const prevShadow = Boolean(prev.style.shadow_enabled);
                  if (curShadow !== prevShadow) return false;

                  const curBg = Boolean(currentStyle.background_enabled && currentStyle.background_mode !== 'none');
                  const prevBg = Boolean(prev.style.background_enabled && prev.style.background_mode !== 'none');
                  if (curBg !== prevBg) return false;
                  if (curBg && (currentStyle.background_color || '').toUpperCase() !== (prev.style.background_color || '').toUpperCase()) {
                    return false;
                  }

                  return true;
                })();

                return (
                  <button
                    key={prev.id}
                    type="button"
                    onClick={() => patch(prev.style, undefined, undefined, true)}
                    className={`flex flex-col items-center justify-between p-2 rounded-lg border min-h-[64px] transition-all active:scale-95 ${
                      isSelected
                        ? 'border-2 border-primary bg-primary/15 ring-1 ring-primary/60 text-white shadow-sm'
                        : 'border-[#262B35] bg-[#1C1F26] hover:bg-[#232832] hover:border-text-muted/60'
                    }`}
                    title={prev.name}
                    aria-label={prev.name}
                  >
                    <div className="w-full h-8 flex items-center justify-center overflow-visible relative">
                      {prev.style.background_enabled && prev.style.background_mode !== 'none' ? (
                        <span
                          className="inline-flex items-center justify-center px-1.5 py-0.5 rounded-[4px]"
                          style={{
                            backgroundColor: prev.style.background_color || '#000000',
                            opacity: prev.style.background_opacity ?? 0.9,
                          }}
                        >
                          <span
                            className="text-[17px] font-black leading-none tracking-tight select-none"
                            style={getAaPreviewStyle(prev.style)}
                          >
                            Aa
                          </span>
                        </span>
                      ) : (
                        <span
                          className="text-[17px] font-black leading-none tracking-tight select-none"
                          style={getAaPreviewStyle(prev.style)}
                        >
                          Aa
                        </span>
                      )}
                    </div>
                    <span className="text-[10px] font-medium text-text-muted truncate w-full text-center mt-1">
                      {prev.name}
                    </span>
                  </button>
                );
              })}
            </div>
          </div>

          {/* 2. Glow & Neon Effects */}
          <div className="space-y-1.5 pt-2 border-t border-[#20242D]">
            <span className="text-[11px] font-medium text-text-secondary">Neon & Glow Effects</span>
            <div className="grid grid-cols-3 sm:grid-cols-4 gap-1.5">
            {EFFECT_PRESETS.map((eff) => {
              const isSelected = (() => {
                if (!currentStyle) return false;
                const c1 = (currentStyle.primary_color || '').toUpperCase();
                const c2 = (eff.style.primary_color || '').toUpperCase();
                if (c1 !== c2) return false;
                const curOutline = Boolean(currentStyle.outline_enabled);
                const effOutline = Boolean(eff.style.outline_enabled);
                if (curOutline !== effOutline) return false;
                const curGlow = Boolean(currentStyle.glow_enabled);
                const effGlow = Boolean(eff.style.glow_enabled);
                if (curGlow !== effGlow) return false;
                return true;
              })();

              return (
                <button
                  key={eff.id}
                  type="button"
                  onClick={() => patch(eff.style, undefined, undefined, true)}
                  className={`p-2 rounded-lg border flex flex-col items-center justify-between min-h-[64px] transition-all active:scale-95 group ${
                    isSelected
                      ? 'border-2 border-primary bg-primary/15 ring-1 ring-primary/60 text-white shadow-sm'
                      : 'border-[#262B35] bg-[#1C1F26] hover:bg-[#232832] hover:border-text-muted/60'
                  }`}
                  title={eff.name}
                >
                  <div className="h-8 w-full flex items-center justify-center overflow-visible relative">
                    {eff.style.background_enabled && eff.style.background_mode !== 'none' ? (
                      <span
                        className="inline-flex items-center justify-center px-1.5 py-0.5 rounded-[4px]"
                        style={{
                          backgroundColor: eff.style.background_color || '#000000',
                          opacity: eff.style.background_opacity ?? 0.9,
                        }}
                      >
                        <span
                          className="text-[18px] font-black leading-none tracking-tight select-none"
                          style={getAaPreviewStyle({ ...eff.style, font_family: eff.style.font_family || 'Impact' })}
                        >
                          Aa
                        </span>
                      </span>
                    ) : (
                      <span
                        className="text-[18px] font-black leading-none tracking-tight select-none"
                        style={getAaPreviewStyle({ ...eff.style, font_family: eff.style.font_family || 'Impact' })}
                      >
                        Aa
                      </span>
                    )}
                  </div>
                  <span className="text-[10px] font-medium text-text-muted group-hover:text-primary transition-colors truncate w-full text-center mt-1">
                    {eff.name}
                  </span>
                </button>
              );
            })}
          </div>
        </div>
        <ApplyToAllButton section="glow" sectionLabel="Effect" onApply={handleApplySectionToAll} />
      </div>
    </div>
  )}

    {/* TAB 5: ANIMATIONS */}
    {textMainTab === 'animations' && (
      <div className="flex-1 min-h-0 flex flex-col space-y-2 py-1">
        <style>{`
          @keyframes cs-in-fade {
            0% { opacity: 0; }
            35%, 75% { opacity: 1; }
            90%, 100% { opacity: 0; }
          }
          @keyframes cs-in-pop {
            0% { transform: scale(0.35); opacity: 0; }
            30% { transform: scale(1.22); opacity: 1; }
            45%, 75% { transform: scale(1); opacity: 1; }
            90%, 100% { transform: scale(0.35); opacity: 0; }
          }
          @keyframes cs-in-slide-up {
            0% { transform: translateY(16px); opacity: 0; }
            35%, 75% { transform: translateY(0); opacity: 1; }
            90%, 100% { transform: translateY(16px); opacity: 0; }
          }
          @keyframes cs-out-fade {
            0%, 45% { opacity: 1; }
            75%, 90% { opacity: 0; }
            100% { opacity: 1; }
          }
          @keyframes cs-out-scale-down {
            0%, 45% { transform: scale(1); opacity: 1; }
            75%, 90% { transform: scale(0.35); opacity: 0; }
            100% { transform: scale(1); opacity: 1; }
          }
          @keyframes cs-out-slide-down {
            0%, 45% { transform: translateY(0); opacity: 1; }
            75%, 90% { transform: translateY(16px); opacity: 0; }
            100% { transform: translateY(0); opacity: 1; }
          }
          @keyframes cs-loop-pulse {
            0%, 100% { transform: scale(1); }
            50% { transform: scale(1.24); }
          }
          @keyframes cs-loop-wiggle {
            0%, 100% { transform: rotate(0deg); }
            25% { transform: rotate(-10deg); }
            75% { transform: rotate(10deg); }
          }
          @keyframes cs-word-pop {
            0%, 25%, 85%, 100% { transform: scale(1); color: #9CA3AF; }
            45%, 65% { transform: scale(1.28); color: #FFE600; }
          }
          @keyframes cs-word-bounce {
            0%, 25%, 85%, 100% { transform: translateY(0); color: #9CA3AF; }
            45%, 65% { transform: translateY(-5px) scale(1.15); color: #FFE600; }
          }
          @keyframes cs-word-highlight {
            0%, 25%, 85%, 100% { color: #9CA3AF; text-shadow: none; }
            45%, 65% { color: #FFE600; text-shadow: 0 0 8px rgba(255,230,0,0.75); }
          }
        `}</style>

        {/* Category Tabs: In · Out · Loop · Captions (Captions shown for caption layers only) */}
        <div className={`grid ${isCaption ? 'grid-cols-4' : 'grid-cols-3'} gap-1 p-0.5 bg-[#1C1F26] rounded-lg shrink-0 border border-[#262B35]`}>
          {(isCaption ? (['in', 'out', 'loop', 'word'] as const) : (['in', 'out', 'loop'] as const)).map((cat) => (
            <button
              key={cat}
              type="button"
              onClick={() => setAnimCategory(cat)}
              className={`h-7 rounded-md text-[11px] font-semibold capitalize transition-all ${
                animCategory === cat
                  ? 'bg-primary text-white shadow-hero-sm font-bold'
                  : 'text-text-muted hover:text-text-secondary'
              }`}
            >
              {cat === 'word' ? 'Captions' : cat}
            </button>
          ))}
        </div>

        {/* Animation Category Body */}
        <div className="flex-1 min-h-0 overflow-y-auto no-scrollbar space-y-2 py-1">
          {(() => {
            const triggerPlayPreview = () => {
              if (currentGroup) {
                setTimeout(() => {
                  if (onPreviewAnimation) {
                    onPreviewAnimation(currentGroup.start, currentGroup.end);
                  } else {
                    onSeek(currentGroup.start);
                  }
                }, 30);
              }
            };

            const renderAnimThumbnail = (category: 'in' | 'out' | 'loop' | 'word', id: string) => {
              if (category === 'in') {
                if (id === 'none') return <span className="text-xs font-bold text-white">Aa</span>;
                if (id === 'fade') return <span className="text-xs font-bold text-white" style={{ animation: 'cs-in-fade 2s infinite ease-in-out' }}>Aa</span>;
                if (id === 'pop') return <span className="text-xs font-bold text-white inline-block" style={{ animation: 'cs-in-pop 2s infinite cubic-bezier(0.175, 0.885, 0.32, 1.275)' }}>Aa</span>;
                if (id === 'slide_up') return <span className="text-xs font-bold text-white inline-block" style={{ animation: 'cs-in-slide-up 2s infinite ease-in-out' }}>Aa</span>;
              }
              if (category === 'out') {
                if (id === 'none') return <span className="text-xs font-bold text-white">Aa</span>;
                if (id === 'fade') return <span className="text-xs font-bold text-white" style={{ animation: 'cs-out-fade 2s infinite ease-in-out' }}>Aa</span>;
                if (id === 'scale_down') return <span className="text-xs font-bold text-white inline-block" style={{ animation: 'cs-out-scale-down 2s infinite ease-in-out' }}>Aa</span>;
                if (id === 'slide_down') return <span className="text-xs font-bold text-white inline-block" style={{ animation: 'cs-out-slide-down 2s infinite ease-in-out' }}>Aa</span>;
              }
              if (category === 'loop') {
                if (id === 'none') return <span className="text-xs font-bold text-white">Aa</span>;
                if (id === 'pulse') return <span className="text-xs font-bold text-white inline-block" style={{ animation: 'cs-loop-pulse 1.4s infinite ease-in-out' }}>Aa</span>;
                if (id === 'wiggle') return <span className="text-xs font-bold text-white inline-block" style={{ animation: 'cs-loop-wiggle 1.2s infinite ease-in-out' }}>Aa</span>;
              }
              if (category === 'word') {
                if (id === 'none') return <span className="text-[11px] font-bold text-text-muted">Aa Bb</span>;
                if (id === 'pop') {
                  return (
                    <span className="text-[11px] font-bold flex items-center gap-1 select-none">
                      <span className="text-text-muted">Aa</span>
                      <span className="inline-block" style={{ animation: 'cs-word-pop 1.8s infinite ease-in-out' }}>Bb</span>
                      <span className="text-text-muted">Cc</span>
                    </span>
                  );
                }
                if (id === 'bounce') {
                  return (
                    <span className="text-[11px] font-bold flex items-center gap-1 select-none">
                      <span className="text-text-muted">Aa</span>
                      <span className="inline-block" style={{ animation: 'cs-word-bounce 1.8s infinite ease-in-out' }}>Bb</span>
                      <span className="text-text-muted">Cc</span>
                    </span>
                  );
                }
                if (id === 'highlight') {
                  return (
                    <span className="text-[11px] font-bold flex items-center gap-1 select-none">
                      <span className="text-text-muted">Aa</span>
                      <span className="inline-block" style={{ animation: 'cs-word-highlight 1.8s infinite ease-in-out' }}>Bb</span>
                      <span className="text-text-muted">Cc</span>
                    </span>
                  );
                }
              }
              return <span className="text-xs font-bold text-white">Aa</span>;
            };

            return (
              <>
                {animCategory === 'in' && (
                  <div className="space-y-2.5">
                    <span className="text-[11px] font-medium text-text-secondary">Entrance Transition</span>
                    <div className="grid grid-cols-2 gap-1.5">
                      {enterEffects.map((eff) => {
                        const isSelected = curEnter === eff.id;
                        return (
                          <button
                            key={eff.id}
                            type="button"
                            onClick={() => {
                              patch(undefined, undefined, {
                                enter: {
                                  type: eff.id as any,
                                  duration_ms: currentAnimation?.enter?.duration_ms || 200,
                                },
                              }, true);
                              triggerPlayPreview();
                            }}
                            className={`p-2 rounded-xl border flex flex-col items-stretch text-left transition-all active:scale-[0.98] ${
                              isSelected
                                ? 'bg-primary/20 border-primary shadow-hero-sm ring-1 ring-primary/40'
                                : 'bg-surface-200 border-border hover:bg-surface-300'
                            }`}
                          >
                            <div className="w-full h-11 rounded-lg bg-surface-100 flex items-center justify-center overflow-hidden mb-1.5 border border-white/[0.04]">
                              {renderAnimThumbnail('in', eff.id)}
                            </div>

      )}
    </div>
  );
};

function areCaptionStudioPropsEqual(prev: CaptionStudioProps, next: CaptionStudioProps): boolean {
  // If transcript is open, allow live playback re-renders for word highlighting
  if (next.activePanel === 'transcript') {
    if (prev.currentTime !== next.currentTime) return false;
  }

  // Functional state checks
  if (prev.track !== next.track) return false;
  if (prev.selectedGroupId !== next.selectedGroupId) return false;
  if (prev.isVideoSelected !== next.isVideoSelected) return false;
  if (prev.layoutMode !== next.layoutMode) return false;
  if (prev.mainVideoVolume !== next.mainVideoVolume) return false;
  if (prev.isMainVideoMuted !== next.isMainVideoMuted) return false;
  if (prev.activePanel !== next.activePanel) return false;
  if (prev.duration !== next.duration) return false;
  if (prev.canUndo !== next.canUndo) return false;
  if (prev.canRedo !== next.canRedo) return false;

  return true;
}

export const CaptionStudio = React.memo(CaptionStudioComponent, areCaptionStudioPropsEqual);
